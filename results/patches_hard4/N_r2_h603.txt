import calendar
from datetime import datetime, timedelta


def _occurrence(start, rule, k):
    if rule == "daily":
        return start + timedelta(days=k)
    if rule == "weekly":
        return start + timedelta(days=7 * k)
    idx = start.month - 1 + k
    year = start.year + idx // 12
    month = idx % 12 + 1
    day = min(start.day, calendar.monthrange(year, month)[1])
    return start.replace(year=year, month=month, day=day)


def next_occurrence(start, rule, after):
    if rule not in ("daily", "weekly", "monthly"):
        raise ValueError(rule)
    k = 0
    if after > start:
        if rule == "daily":
            k = max(0, (after - start).days - 1)
        elif rule == "weekly":
            k = max(0, (after - start).days // 7 - 1)
        else:
            k = max(0, (after.year - start.year) * 12 + after.month - start.month - 1)
    cur = _occurrence(start, rule, k)
    while cur <= after:
        k += 1
        cur = _occurrence(start, rule, k)
    return cur
