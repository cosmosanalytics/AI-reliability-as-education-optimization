import re

_NUMBER = re.compile(r"^[+-]?(\d+\.?\d*|\.\d+)([eE][+-]?\d+)?$")


def _format(field):
    if field is None:
        return ""
    text = str(field)
    if isinstance(field, str) and text and text[0] in "=+-@" and not _NUMBER.match(text):
        text = "'" + text
    if text and (
        any(c in text for c in ',"\n\r') or text[0] == " " or text[-1] == " "
    ):
        text = '"' + text.replace('"', '""') + '"'
    return text


def write_row(fields):
    return ",".join(_format(f) for f in fields)


def read_rows(text):
    rows = []
    row = []
    field = []
    in_quotes = False
    field_start = True
    pending = False
    i = 0
    n = len(text)
    while i < n:
        ch = text[i]
        pending = True
        if in_quotes:
            if ch == '"':
                if i + 1 < n and text[i + 1] == '"':
                    field.append('"')
                    i += 1
                else:
                    in_quotes = False
            else:
                field.append(ch)
        elif ch == '"' and field_start:
            in_quotes = True
            field_start = False
        elif ch == ",":
            row.append("".join(field))
            field = []
            field_start = True
        elif ch == "\n" or ch == "\r":
            if ch == "\r" and i + 1 < n and text[i + 1] == "\n":
                i += 1
            row.append("".join(field))
            rows.append(row)
            row = []
            field = []
            field_start = True
            pending = False
        else:
            field.append(ch)
            field_start = False
        i += 1
    if in_quotes:
        raise ValueError("unterminated quote")
    if pending:
        row.append("".join(field))
        rows.append(row)
    return rows
