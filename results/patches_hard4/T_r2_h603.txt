import calendar
from datetime import datetime, timedelta


def _monthly(start, k):
    index = start.year * 12 + (start.month - 1) + k
    year, month = divmod(index, 12)
    month += 1
    day = min(start.day, calendar.monthrange(year, month)[1])
    return start.replace(year=year, month=month, day=day)


def next_occurrence(start, rule, after):
    if rule not in ("daily", "weekly", "monthly"):
        raise ValueError(rule)
    k = 0
    while True:
        if rule == "daily":
            cur = start + timedelta(days=k)
        elif rule == "weekly":
            cur = start + timedelta(days=7 * k)
        else:
            cur = _monthly(start, k)
        if cur > after:
            return cur
        k += 1
