from datetime import datetime


def month_bounds(year, month):
    if not 1 <= month <= 12:
        raise ValueError("month must be in 1..12")
    start = datetime(year, month, 1)
    end = datetime(year + 1, 1, 1) if month == 12 else datetime(year, month + 1, 1)
    return start, end

from datetime import datetime


class Ledger:

    def __init__(self):
        self.entries = []

    def record(self, customer, ts, units):
        if not isinstance(customer, str) or not customer:
            raise ValueError("customer must be a non-empty string")
        if not isinstance(ts, datetime):
            raise ValueError("ts must be a datetime")
        if isinstance(units, bool) or not isinstance(units, int) or units == 0:
            raise ValueError("units must be a non-zero integer")
        self.entries.append((customer, ts, units))

    def customers(self):
        return sorted({c for c, _, _ in self.entries})

from period import month_bounds


def monthly_report(ledger, year, month, customers=None):
    start, end = month_bounds(year, month)
    names = ledger.customers() if customers is None else customers
    out = {c: 0 for c in names}
    for c, ts, u in ledger.entries:
        if c in out and start <= ts < end:
            out[c] += u
    return out
