import calendar
from datetime import datetime, timedelta


def _nth(start, rule, k):
    if rule == "daily":
        naive = start.replace(tzinfo=None) + timedelta(days=k)
        return naive.replace(tzinfo=start.tzinfo)
    if rule == "weekly":
        naive = start.replace(tzinfo=None) + timedelta(days=7 * k)
        return naive.replace(tzinfo=start.tzinfo)
    months = start.month - 1 + k
    year = start.year + months // 12
    month = months % 12 + 1
    day = min(start.day, calendar.monthrange(year, month)[1])
    return start.replace(year=year, month=month, day=day)


def next_occurrence(start, rule, after):
    if rule not in ("daily", "weekly", "monthly"):
        raise ValueError(rule)
    k = 0
    cur = start
    while cur <= after:
        k += 1
        cur = _nth(start, rule, k)
    return cur
