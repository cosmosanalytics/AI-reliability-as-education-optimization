def gross(cents, rate):

from decimal import Decimal, ROUND_HALF_UP


def round_half_up(x):
    return int(Decimal(str(x)).quantize(Decimal(1), rounding=ROUND_HALF_UP))


def to_cents(x):
    return round_half_up(Decimal(str(x)) * 100)


def fmt(cents):
    sign = "-" if cents < 0 else ""
    cents = abs(cents)
    return "%s$%d.%02d" % (sign, cents // 100, cents % 100)

from decimal import Decimal

from money import to_cents, round_half_up


def line_total(price, qty):
    return to_cents(price) * qty


def invoice_total(lines, discount_pct=0):
    sub = sum(line_total(p, q) for p, q in lines)
    return sub - round_half_up(Decimal(sub) * Decimal(str(discount_pct)) / 100)
