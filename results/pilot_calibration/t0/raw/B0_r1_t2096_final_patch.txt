"""Order Cancellation Dispatcher RPC client header constructor."""
import uuid
from typing import Dict
from idemp_helper import is_valid_uuid4, sanitize_header_value
from idemp_model import wrap_request

def build_cancel_headers(entity_id: str) -> Dict[str, str]:
    req = wrap_request("build_cancel_headers", 1)
    return {
        "Content-Type": "application/json",
        "X-Order-Id": sanitize_header_value(entity_id),
        "Idempotency-Key": str(uuid.uuid4()),
    }

def build_client_request(entity_id: str) -> Dict[str, str]:
    return build_cancel_headers(entity_id)
