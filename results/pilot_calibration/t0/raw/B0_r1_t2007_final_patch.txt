"""Default step parameters."""
import copy
from typing import Dict, Any

DEFAULT_PIPELINE_OPTS = {
    "engine": "v2",
    "retries": {"max": 3, "backoff": 1.5},
    "monitoring": {"enabled": True, "sample_rate": 0.1},
    "concurrency": {"max_workers": 4, "queue_depth": 100}
}

def get_base_options() -> Dict[str, Any]:
    # Return a deep copy to ensure callers cannot mutate the global default options.
    return copy.deepcopy(DEFAULT_PIPELINE_OPTS)

def is_feature_enabled(feature_name: str) -> bool:
    return DEFAULT_PIPELINE_OPTS.get(feature_name, {}).get("enabled", False)

=== FILE: step_builder.py ===
"""Low-level step configuration compiler."""
import copy
from typing import Dict, Any

def compile_step_config(name: str, base_opts: Dict[str, Any], overrides: Dict[str, Any]) -> Dict[str, Any]:
    # Perform a deep copy of base_opts to ensure that nested dictionaries are also copied
    # and subsequent modifications do not affect the original base_opts.
    res = copy.deepcopy(base_opts)
    res["name"] = name
    
    # Recursively merge overrides into res.
    # This ensures that nested dictionaries in overrides are merged with,
    # rather than completely replacing, nested dictionaries in res.
    for key, value in overrides.items():
        if key in res and isinstance(res[key], dict) and isinstance(value, dict):
            res[key].update(value)
        else:
            res[key] = value
            
    return res

def format_step_manifest(spec: Dict[str, Any]) -> str:
    name = spec.get("name", "unnamed")
    engine = spec.get("engine", "default")
    return f"StageManifest(name={name}, engine={engine})"

"""Pipeline stage configurator."""
from typing import Dict, Any, List
from step_builder import compile_step_config, format_step_manifest
from defaults import get_base_options # Import the function to get a safe copy

def build_stage_spec(stage_name: str, step_overrides: Dict[str, Any]) -> Dict[str, Any]:
    """Compile pipeline stage specification.
    Inputs must remain immutable and nested options must be defensively copied.
    """
    # Use get_base_options() to retrieve a deep copy of the default options,
    # ensuring that the global DEFAULT_PIPELINE_OPTS remains immutable.
    spec = compile_step_config(stage_name, get_base_options(), step_overrides)
    _ = format_step_manifest(spec)
    return spec

def build_multi_stage_pipeline(stage_configs: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
    return [build_stage_spec(f"stage-{i}", cfg) for i, cfg in enumerate(stage_configs)]
