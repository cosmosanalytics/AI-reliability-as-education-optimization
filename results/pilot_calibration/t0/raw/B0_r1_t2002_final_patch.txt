"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    This function ensures that the total allocated discount matches total_discount_cents
    by distributing any remainder (pennies) to the first lines.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")

    total = sum(line_totals)
    num_lines = len(line_totals)

    if total == 0:
        # If total value of lines is zero, no proportional allocation is possible.
        # Return zeros for all lines. The total_allocated will be 0,
        # and is_fully_allocated will be False if total_discount_cents > 0.
        return [0] * num_lines

    # Calculate base allocations using floor division
    allocations = [cents * total_discount_cents // total for cents in line_totals]

    # Calculate the remainder that needs to be distributed
    current_total_allocated = sum(allocations)
    remainder = total_discount_cents - current_total_allocated

    # Distribute the remainder (pennies) to the first 'remainder' lines
    # This ensures the total allocated discount matches total_discount_cents
    # and resolves the unallocated penny mismatch.
    for i in range(remainder):
        # Ensure we don't try to add to a line that doesn't exist
        # (e.g., if remainder is larger than num_lines, which shouldn't happen
        # if total_discount_cents is positive and total is positive).
        if i < num_lines:
            allocations[i] += 1
        else:
            # This case indicates an unexpected scenario where remainder exceeds
            # the number of lines. For robustness, we could log a warning or
            # raise an error, but for now, we'll just stop distributing.
            # In typical penny-rounding scenarios, remainder < num_lines.
            break

    return allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
