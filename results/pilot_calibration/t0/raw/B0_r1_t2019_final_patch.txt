"""Storage scope normalization engine."""
def validate_prefix_match(scope_prefix: str, asset_scope: str) -> bool:
    """
    Checks if an asset's scope matches a required scope prefix, respecting
    double-colon (::) segment boundaries.

    A match occurs if:
    1. The asset_scope is identical to the scope_prefix.
    2. The asset_scope starts with the scope_prefix, and the character
       immediately following the prefix in asset_scope is '::', indicating
       a direct child scope.
    """
    if asset_scope == scope_prefix:
        return True
    if asset_scope.startswith(scope_prefix + "::"):
        return True
    return False

def get_scope_depth(scope_str: str) -> int:
    return len([seg for seg in scope_str.split("::") if seg])

def normalize_scope(scope_str: str) -> str:
    return scope_str.strip()
