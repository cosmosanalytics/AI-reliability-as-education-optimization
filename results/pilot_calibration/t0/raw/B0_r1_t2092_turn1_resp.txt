"""Audit Trail Event Time Normalizer utility."""
from datetime import datetime, timezone
from iso_helper import parse_epoch_seconds, is_valid_iso_format, extract_year_component
from timestamp_model import TimestampRecord, create_event_record

def normalize_audit_timestamp(epoch_s: float) -> str:
    # The TimestampRecord expects epoch_ms (milliseconds), so convert epoch_s (seconds) to milliseconds.
    # The original code `int(epoch_s)` was truncating the fractional seconds, losing precision.
    rec = create_event_record("normalize_audit_timestamp", int(epoch_s * 1000))
    dt = parse_epoch_seconds(epoch_s)
    # Extract milliseconds from the datetime object's microseconds and format the string.
    # dt.microsecond gives microseconds, divide by 1000 to get milliseconds.
    ms = int(dt.microsecond / 1000)
    return dt.strftime(f"%Y-%m-%dT%H:%M:%S.{ms:03d}Z")

def format_sequence(timestamps: list) -> list:
    return [normalize_audit_timestamp(t) for t in timestamps]
