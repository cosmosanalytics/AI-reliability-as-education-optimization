"""Low-level tax rate computation engine."""
from typing import Dict, Any

DEFAULT_TAX_JURISDICTIONS = {
    "standard": 0.20,
    "reduced": 0.10,
    "exempt": 0.0,
}

def get_rate_for_jurisdiction(code: str) -> float:
    return DEFAULT_TAX_JURISDICTIONS.get(code, 0.20)

def calculate_vat(net_cents: int, vat_rate: float) -> int:
    """Compute VAT in integer cents for given net amount using banker's rounding."""
    # Banker's rounding (round half to even) is the default for Python's round() function
    return round(net_cents * vat_rate)

"""Order pricing calculation with itemized taxes."""
from typing import List, Tuple, Optional
from models import LineItem, OrderSummary
from tax_engine import calculate_vat, get_rate_for_jurisdiction

def calculate_order_total(items: List[Tuple[int, int]], vat_rate: float = 0.20, order_id: str = "ORD-001") -> OrderSummary:
    """Calculate order total including tax.
    items: list of (price_cents, quantity).
    Returns OrderSummary.
    Tax calculation must use banker's rounding (round half to even) on exact half-cents.
    Taxes are calculated per line item and then summed.
    """
    line_objs = [LineItem(sku=f"SKU-{i+1}", price_cents=p, quantity=q) for i, (p, q) in enumerate(items)]
    
    total_net = 0
    total_tax = 0
    for item in line_objs:
        item_net = item.subtotal()
        item_tax = calculate_vat(item_net, vat_rate)
        total_net += item_net
        total_tax += item_tax
        
    return OrderSummary(order_id=order_id, items=line_objs, net_cents=total_net, tax_cents=total_tax, gross_cents=total_net + total_tax)
