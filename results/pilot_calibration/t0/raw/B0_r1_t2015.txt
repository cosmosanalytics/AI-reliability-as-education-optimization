"""TSV cell sanitizer."""
from typing import Optional

def sanitize_cell(text: Optional[str]) -> str:
    if text is None:
        return ""
    # Escape backslashes first to prevent double-escaping of newly introduced backslashes
    # Then escape double quotes with a backslash, as expected by the audit parser.
    return text.replace('\\', '\\\\').replace('"', '\\"')

def requires_sanitization(text: str) -> bool:
    # This function is currently not used by exporter.py for actual sanitization logic,
    # but it correctly identifies characters that would typically require escaping
    # or quoting in a TSV context.
    return any(c in text for c in ('\t', '\n', '"', '\\'))
