"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    pattern_segments = policy.allowed_pattern.split('::')
    request_segments = clean_req.split('::')

    i = 0 # index for pattern segments
    j = 0 # index for request segments

    while i < len(pattern_segments) and j < len(request_segments):
        pattern_segment = pattern_segments[i]
        request_segment = request_segments[j]

        if pattern_segment == "**":
            # '**' matches arbitrary multi-segment prefixes.
            # If '**' is the last pattern segment, it matches all remaining request segments (including none).
            return True
        elif pattern_segment == "*":
            # '*' matches exactly one segment.
            # If there are more pattern segments after '*', we need to ensure there's a request segment to match.
            if j < len(request_segments):
                i += 1
                j += 1
            else:
                return False # '*' requires a segment, but none left in request
        elif pattern_segment == request_segment:
            # Exact segment match
            i += 1
            j += 1
        else:
            # Mismatch
            return False

    # After loop, check if all pattern segments are consumed
    # and if any remaining pattern segments are '**'
    while i < len(pattern_segments):
        if pattern_segments[i] == "**":
            i += 1
        else:
            return False # Remaining pattern segments are not '**'
    
    # If we reached here, all pattern segments (or '**') matched.
    # The request is permitted if all its segments were consumed by the pattern.
    return j == len(request_segments)

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
