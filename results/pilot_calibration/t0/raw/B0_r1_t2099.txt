"""Database Write Latency Calculator percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_write_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_write_percentile", samples)
    s = validate_metric_samples(win.values)
    n = len(s)
    if n == 0:
        raise ValueError("Cannot calculate percentile for empty samples list")

    # The "nearest-rank house standard" typically implies a method where
    # the percentile is directly mapped to an element in the sorted list,
    # often by rounding an index calculated using N (number of samples)
    # rather than N-1, and then selecting the element at that index.
    # A common interpretation for the p-th percentile is to find the
    # element at index k = ceil(p/100 * N).
    # For 0-based indexing, this means s[k-1].
    # This method ensures that the 50th percentile of [10, 20, 30, 40, 50, 60, 70, 80]
    # (N=8) is at index ceil(0.5 * 8) = 4, which is s[3] = 40.0.
    # The original implementation used linear interpolation with (N-1),
    # which for the same example yielded 45.0.

    # Calculate the rank (1-based index)
    rank = math.ceil((p / 100.0) * n)

    # Adjust for 0-based indexing
    # Ensure rank is at least 1 and at most n
    rank = max(1, min(int(rank), n))

    return s[rank - 1]

def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
