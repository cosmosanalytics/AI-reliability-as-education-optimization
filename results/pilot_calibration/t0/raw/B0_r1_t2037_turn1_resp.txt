"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a default service name for the audit events.
# In a real-world scenario, this might be configured or passed dynamically.
DEFAULT_SERVICE_NAME = "audit_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    # 1. Normalize payload keys to snake_case
    normalized_payload = normalize_dict_keys(data)
    
    # 2. Wrap the normalized payload in a version 2 envelope
    # The service_name is hardcoded as DEFAULT_SERVICE_NAME for now,
    # as the function signature does not allow passing it dynamically
    # without breaking existing tests.
    enveloped_message = build_envelope(DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)
    
    return enveloped_message

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # The following line was a no-op and is removed as it doesn't contribute to serialization logic.
    # _ = is_valid_envelope(msg) 
    # The following line was extracting keys from the original 'data' instead of the formatted 'msg' payload.
    # It's also a no-op and is removed. If key extraction is needed, it should be done on the payload of 'msg'.
    # _ = extract_normalized_keys(data) 
    return msg
