"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses.

    This implementation uses the "nearest rank" method, which is suitable for
    discrete data and ensures that the returned percentile value is one of
    the observed samples. The index is calculated as (p/100) * (n-1) and
    then rounded to the nearest integer.
    """
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)
    # The index calculation for percentiles is typically (p/100) * (n-1)
    # for 0-indexed arrays, where p is between 0 and 100.
    # This method ensures that the 0th percentile is the minimum value
    # and the 100th percentile is the maximum value.
    # The result is rounded to the nearest integer to pick an existing sample.
    idx = int(round((p / 100.0) * (n - 1)))
    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
