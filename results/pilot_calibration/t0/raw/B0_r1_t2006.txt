"""Database connection options schemas."""
from dataclasses import dataclass, field
from typing import Dict, Any, Optional

@dataclass
class ConnectionOptions:
    dsn: str
    pool_params: Dict[str, Any] = field(default_factory=dict)
    max_overflow: int = 10
    timeout_seconds: float = 30.0

    def __post_init__(self):
        # Ensure pool_params is a mutable copy to prevent shared state issues
        self.pool_params = dict(self.pool_params)

    def is_ssl(self) -> bool:
        return "sslmode" in self.pool_params

    def get_param(self, key: str, fallback: Any = None) -> Any:
        return self.pool_params.get(key, fallback)

    def summary(self) -> str:
        return f"Pool(dsn={self.dsn}, params={len(self.pool_params)})"

"""Database pool manager."""
from typing import Dict, Any
from env_loader import merge_env_settings
from options import ConnectionOptions

def configure_pool(base_template: Dict[str, Any], overrides: Dict[str, Any]) -> ConnectionOptions:
    """Configure connection pool options from template and overrides.
    Returns ConnectionOptions instance.
    """
    # Perform a deep merge for nested dictionaries like 'pool'
    settings = merge_env_settings(base_template, overrides)

    dsn = settings.get("dsn", "sqlite:///:memory:")
    # Pass a copy of the pool settings to ConnectionOptions to ensure isolation
    pool_settings = settings.get("pool", {})
    opts = ConnectionOptions(dsn=dsn, pool_params=pool_settings)
    _ = opts.summary()
    return opts

"""Environment settings loader and merger."""
from typing import Dict, Any

DEFAULT_CONN_TIMEOUT = 30

def load_system_env() -> Dict[str, Any]:
    return {"timeout": DEFAULT_CONN_TIMEOUT}

def merge_env_settings(base_settings: Dict[str, Any], override_settings: Dict[str, Any]) -> Dict[str, Any]:
    """
    Deeply merges override_settings into base_settings.
    Handles nested dictionaries by merging their contents.
    """
    merged = dict(base_settings)
    for key, value in override_settings.items():
        if key in merged and isinstance(merged[key], dict) and isinstance(value, dict):
            merged[key] = merge_env_settings(merged[key], value) # Recursive call for nested dicts
        else:
            merged[key] = value
    return merged
