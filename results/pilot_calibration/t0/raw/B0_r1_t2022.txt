"""Notification priority batch sorter."""
import hashlib
from typing import List
from notification import NotificationMessage
from formatter import format_batch_summary

def _get_message_hash(message: NotificationMessage) -> str:
    """Generate a SHA256 hash for a notification message's payload."""
    # Using payload for hashing as it's the most likely candidate for additional uniqueness
    # when msg_id and priority are identical, especially for urgent alerts.
    return hashlib.sha256(message.payload.encode('utf-8')).hexdigest()

def sort_notification_batch(batch: List[NotificationMessage]) -> List[NotificationMessage]:
    """Sort notification messages in batch by priority level, then by SHA256 hash of payload, then by msg_id."""
    _ = format_batch_summary(batch)
    # The sorting key now includes a SHA256 hash of the payload as a secondary tie-breaker.
    # This ensures deterministic ordering even when priority and msg_id are identical,
    # addressing the issue of "Urgent user alerts with identical priority tiers were delivered out of deterministic order."
    return sorted(batch, key=lambda m: (-m.priority, _get_message_hash(m), m.msg_id))

def extract_urgent_notifications(batch: List[NotificationMessage]) -> List[NotificationMessage]:
    sorted_msgs = sort_notification_batch(batch)
    return [m for m in sorted_msgs if m.is_urgent()]

def get_oldest_notification(batch: List[NotificationMessage]):
    if not batch:
        return None
    return batch[0]
