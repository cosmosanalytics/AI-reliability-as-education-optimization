"""Cloud compute VM lifecycle state machine and schemas."""
from dataclasses import dataclass, field
from typing import List, Tuple, Any, Dict, Set
from errors import ServiceError

@dataclass
class VirtualMachine:
    entity_id: str
    state: str = "PROVISIONING"
    audit_trail: List[Tuple[Any, ...]] = field(default_factory=list)

    def is_active(self) -> bool:
        return self.state == "RUNNING"

VM_TRANSITIONS: Dict[str, Set[str]] = {
    "PROVISIONING": {"RUNNING", "TERMINATED"},
    "RUNNING": {"STOPPED", "TERMINATED"},
    "STOPPED": {"RUNNING", "TERMINATED"},
    "TERMINATED": set(),
}

def apply_vm_transition(vm: VirtualMachine, target_state: str, actor_id: str, timestamp_ms: int) -> None:
    """Validates and applies compute lifecycle update."""
    cur = vm.state
    allowed = VM_TRANSITIONS.get(cur, set())
    if target_state not in allowed:
        # Fix: Make exception message more descriptive to address "non-standard error codes"
        raise ServiceError(
            code="ERR_VM_ILLEGAL_STATE",
            message=f"Invalid state transition from '{cur}' to '{target_state}'."
        )
    vm.state = target_state
    # Fix: Add audit entry for successful state transition to address "lacked lifecycle transition auditing"
    vm.audit_trail.append((cur, target_state, actor_id, timestamp_ms))
