"""Low-level dictionary and list hierarchy merger."""
from typing import Dict, Any, List

def merge_spec_values(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """
    Recursively merges dictionary values.
    - If both base and override values are dictionaries, they are recursively merged.
    - If both base and override values are lists, their unique elements are combined (union).
    - In all other cases, the override value replaces the base value.
    """
    result = dict(base)
    for k, v in override.items():
        if k in result:
            if isinstance(result[k], dict) and isinstance(v, dict):
                result[k] = merge_spec_values(result[k], v)
            elif isinstance(result[k], list) and isinstance(v, list):
                # Merge lists by taking a union of unique elements
                # This ensures elements from both lists are present and duplicates are removed.
                # The order of elements in the resulting list is not guaranteed.
                merged_list = list(set(result[k]) | set(v))
                result[k] = merged_list
            else:
                # For other types or mixed types, override the base value
                result[k] = v
        else:
            # Key not in base, simply add it
            result[k] = v
    return result

def is_spec_empty(spec: Dict[str, Any]) -> bool:
    return len(spec) == 0
