"""Stream event sequencer."""
from typing import List
# Removed unused import order_packets as it's no longer directly used for primary sequencing
from sort_helper import get_max_severity
from event_packet import EventPacket

def sequence_stream_batch(packets: List[EventPacket]) -> List[EventPacket]:
    """
    Sequence event packets, prioritizing by severity (highest first)
    and then by stream offset (lowest first) for tie-breaking.
    This ensures correct ordering for state replication, addressing out-of-order offsets.
    """
    # Sort by severity in descending order (highest severity first)
    # Then by offset in ascending order (earliest offset first)
    # Assuming EventPacket has 'severity' (int) and 'offset' (int) attributes.
    return sorted(packets, key=lambda p: (-p.severity, p.offset))

def get_highest_priority_packet(packets: List[EventPacket]):
    ordered = sequence_stream_batch(packets)
    return ordered[0] if ordered else None

def get_batch_severity_peak(packets: List[EventPacket]) -> int:
    return get_max_severity(packets)
