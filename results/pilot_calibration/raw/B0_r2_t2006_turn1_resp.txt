"""Database pool manager."""
from typing import Dict, Any
from env_loader import merge_env_settings
from options import ConnectionOptions

def configure_pool(base_template: Dict[str, Any], overrides: Dict[str, Any]) -> ConnectionOptions:
    """Configure connection pool options from template and overrides.
    Returns ConnectionOptions instance.
    """
    # Create a shallow copy of the base_template to prevent in-place modification
    # by merge_env_settings or subsequent operations. This ensures that the
    # original base_template remains immutable and isolated across different
    # calls or worker threads, resolving shared state corruption.
    settings = merge_env_settings(base_template.copy(), overrides)
    dsn = settings.get("dsn", "sqlite:///:memory:")
    opts = ConnectionOptions(dsn=dsn, pool_params=settings.get("pool", {}))
    _ = opts.summary()
    return opts
