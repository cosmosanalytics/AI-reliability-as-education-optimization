"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total = sum(line_totals)
    if total == 0:
        return [0] * len(line_totals)

    # Calculate base allocations using floating point to maintain precision
    # and store the fractional parts for remainder distribution.
    allocations_with_fractions = []
    for cents in line_totals:
        exact_allocation = (cents * total_discount_cents) / total
        allocations_with_fractions.append(exact_allocation)

    # Truncate to integer cents for initial allocation
    initial_allocations = [int(alloc) for alloc in allocations_with_fractions]

    # Calculate the sum of initial allocations
    sum_initial_allocations = sum(initial_allocations)

    # Determine the remainder to be distributed
    remainder = total_discount_cents - sum_initial_allocations

    # Distribute the remainder one cent at a time.
    # To ensure fair distribution, we can prioritize lines with the largest fractional parts.
    # Create a list of (fractional_part, original_index) tuples.
    fractional_parts = []
    for i, exact_alloc in enumerate(allocations_with_fractions):
        fractional_parts.append((exact_alloc - initial_allocations[i], i))

    # Sort by fractional part in descending order
    fractional_parts.sort(key=lambda x: x[0], reverse=True)

    # Add one cent to the lines with the largest fractional parts until the remainder is zero
    final_allocations = list(initial_allocations) # Make a mutable copy
    for i in range(remainder):
        # Get the original index of the line to which we add a cent
        original_index = fractional_parts[i][1]
        final_allocations[original_index] += 1

    return final_allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
