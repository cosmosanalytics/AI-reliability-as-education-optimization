"""Order item CSV row serializer."""
from typing import List
from item import OrderItem
from catalog_validator import validate_catalog

def _quote_csv_field(field_value: str) -> str:
    """Applies CSV quoting rules to a single field.

    If the field contains a comma, a double quote, or a newline,
    the entire field is enclosed in double quotes. Any double quotes
    within the field are escaped by doubling them.
    """
    s_field = str(field_value)

    # Check if quoting is necessary
    # Quoting is necessary if the field contains a comma, a double quote, or a newline.
    needs_quoting = ',' in s_field or '"' in s_field or '\n' in s_field

    if needs_quoting:
        # Escape internal double quotes by doubling them
        escaped_field = s_field.replace('"', '""')
        # Enclose the entire field in double quotes
        return f'"{escaped_field}"'
    else:
        return s_field

def serialize_item_row(item: OrderItem) -> str:
    """Serialize OrderItem into a CSV formatted line."""
    # The item_id is assumed not to require quoting based on the failing test example.
    # The description field is the one that needs proper CSV quoting and escaping.
    quoted_description = _quote_csv_field(item.description)
    return f"{item.item_id},{quoted_description}"

def serialize_catalog(items: List[OrderItem]) -> str:
    if not validate_catalog(items):
        raise ValueError("Invalid catalog items")
    return "\n".join(serialize_item_row(i) for i in items)

def count_serialized_items(catalog_str: str) -> int:
    return len([line for line in catalog_str.splitlines() if line.strip()])
