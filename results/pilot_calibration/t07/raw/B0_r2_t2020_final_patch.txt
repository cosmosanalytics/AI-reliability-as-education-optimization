"""Topic path subscription matcher."""
def is_topic_subscribed(subscribed_pattern: str, event_topic: str) -> bool:
    """
    Determines if an event topic matches a subscribed pattern.

    A subscription pattern matches an event topic if:
    1. They are an exact match.
    2. The event topic is a sub-topic of the pattern (e.g., 'orders/created' matches 'orders').
       This is enforced by checking if the event topic starts with the pattern followed by '/'.
    This prevents 'orders_admin' from matching 'orders'.
    """
    if event_topic == subscribed_pattern:
        return True
    return event_topic.startswith(subscribed_pattern + "/")

def clean_topic_segment(topic: str) -> str:
    return topic.strip().lower()

def is_wildcard_subscription(pattern: str) -> bool:
    return pattern.strip() in ("*", "#", "/")
