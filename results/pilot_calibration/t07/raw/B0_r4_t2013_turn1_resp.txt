"""TSV serialization utilities."""
from typing import List, Optional
from schema import CustomerProfile
from io_helpers import validate_tsv_header

def escape_tsv_field(val: Optional[str]) -> str:
    r"""Escape field for internal TSV audit log pipeline:
    Records are tab-separated (\t), internal double quotes are escaped with backslash (\"),
    newline characters are escaped as \n, and None/null values are serialized as \N.
    Backslash characters are escaped as \\.
    Do NOT use RFC 4180 double-quoting.
    """
    if val is None:
        return "\\N" # None/null values are serialized as \N

    # Apply escapes in specific order to avoid double-escaping
    # Escape backslashes first to distinguish from escape sequences
    # Then escape specific characters that use backslash as prefix
    escaped_val = val.replace("\\", "\\\\") # Escape backslashes
    escaped_val = escaped_val.replace('"', '\\"') # Escape double quotes
    escaped_val = escaped_val.replace("\n", "\\n") # Escape newline characters
    escaped_val = escaped_val.replace("\t", "\\t") # Escape tab characters (as they are delimiters)

    return escaped_val

def export_profiles_tsv(profiles: List[CustomerProfile]) -> str:
    # The header and line joining use string literals for delimiters (e.g., "\\t", "\\n")
    # which implies the final output string will contain these literal sequences
    # to represent actual tabs and newlines when parsed by the audit pipeline.
    header = "cust_id\\tname\\tnotes"
    if not validate_tsv_header(header.replace("\\t", "\t"), ["cust_id", "name", "notes"]):
        raise ValueError("Invalid header")
    lines = [header]
    for p in profiles:
        f_name = escape_tsv_field(p.name)
        f_notes = escape_tsv_field(p.notes)
        lines.append(f"{p.cust_id}\\t{f_name}\\t{f_notes}")
    return "\\n".join(lines)

def count_records(payload: str) -> int:
    # This function needs to correctly interpret the payload given the escaping scheme
    # It assumes the payload string itself contains the literal \\n for newlines.
    lines = [l for l in payload.split("\\n") if l.strip()]
    return max(0, len(lines) - 1)
