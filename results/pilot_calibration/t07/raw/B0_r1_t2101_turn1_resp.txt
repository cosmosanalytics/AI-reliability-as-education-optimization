"""System Log TSV Serializer TSV line serializer."""
from typing import List
from tsv_helper import parse_tsv_line, count_tsv_fields
from tsv_model import create_record

def serialize_log_line(level: str, msg: str, target: str) -> str:
    rec = create_record(level, msg, target)
    
    # Apply specific escaping rules for TSV:
    # 1. Escape backslashes first to prevent double-escaping of new escapes.
    # 2. Escape double quotes with a backslash.
    # 3. Escape tab characters with a backslash and 't'.
    escaped_msg = rec.message.replace("\\", "\\\\")
    escaped_msg = escaped_msg.replace('"', '\\"')
    escaped_msg = escaped_msg.replace('\t', '\\t')
    
    return f"{rec.level}\t{escaped_msg}\t{rec.target}"

def validate_line(line: str) -> bool:
    return count_tsv_fields(line) == 3
