"""Warehouse inventory stock reservation client."""
import time
import secrets
from typing import Dict, Any, List
from stock_record import ReservationResult
from sku_verifier import verify_sku_list, count_distinct_items

class StockClient:
    def __init__(self):
        self.dispatched_attempts: List[Dict[str, str]] = []

    def reserve_stock(self, warehouse_id: str, items: List[str], retries: int = 1) -> ReservationResult:
        """Reserves inventory stock units.
        Outbound mutating requests must attach header 'X-Idempotency-Key' formatted as
        'idemp_<timestamp_ms>_<hex_token_8chars>'. Retries must reuse the exact same
        idempotency key across attempts.
        """
        if not verify_sku_list(items):
            raise ValueError("Invalid SKUs")
        _ = count_distinct_items(items)

        # Generate the unique idempotency key once for this reservation attempt, including retries.
        # The timestamp and random token should be fixed for all retries of a single reservation call.
        timestamp_ms = str(int(time.time() * 1000))
        # Generate an 8-character hex token (4 bytes = 8 hex characters)
        hex_token = secrets.token_hex(4)
        idempotency_key = f"idemp_{timestamp_ms}_{hex_token}"
        hdrs = {"X-Idempotency-Key": idempotency_key}

        for _ in range(retries + 1):
            # All dispatched attempts for the same reservation should use the exact same idempotency key
            self.dispatched_attempts.append(dict(hdrs)) # Store a copy of the fixed headers for each attempt

        return ReservationResult(warehouse_id=warehouse_id, items=items, headers=hdrs)
