"""Low-level tax rate computation engine."""
from typing import Dict, Any
from decimal import Decimal, ROUND_HALF_UP

DEFAULT_TAX_JURISDICTIONS = {
    "standard": 0.20,
    "reduced": 0.10,
    "exempt": 0.0,
}

def get_rate_for_jurisdiction(code: str) -> float:
    return DEFAULT_TAX_JURISDICTIONS.get(code, 0.20)

def calculate_vat(net_cents: int, vat_rate: float) -> int:
    """Compute VAT in integer cents for given net amount, using half-up rounding.
    Half-up rounding means ties are broken towards positive infinity.
    """
    # Convert inputs to Decimal for precise calculation to avoid floating-point inaccuracies
    net_decimal = Decimal(net_cents)
    # Convert float vat_rate to string first to maintain precision when creating Decimal
    rate_decimal = Decimal(str(vat_rate))

    # Calculate the raw tax amount
    raw_tax_decimal = net_decimal * rate_decimal

    # Round to the nearest integer using the ROUND_HALF_UP policy
    # '0' represents the target precision (0 decimal places for cents)
    rounded_tax_decimal = raw_tax_decimal.quantize(Decimal('0'), rounding=ROUND_HALF_UP)

    return int(rounded_tax_decimal)
