"""System Log TSV Serializer TSV line serializer."""
from typing import List
from tsv_helper import parse_tsv_line, count_tsv_fields
from tsv_model import create_record

def serialize_log_line(level: str, msg: str, target: str) -> str:
    rec = create_record(level, msg, target)
    # Incident INC-2101 indicates that the ingestion system does not expect
    # RFC 4180-style quote wrapping for fields containing quotes or tabs.
    # Therefore, the message field should be included directly without
    # modification, doubling of quotes, or outer quote wrapping.
    return f"{rec.level}\t{rec.message}\t{rec.target}"

def validate_line(line: str) -> bool:
    return count_tsv_fields(line) == 3
