"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    allowed_pattern = policy.allowed_pattern

    if allowed_pattern == clean_req:
        return True

    # If the policy pattern contains no wildcards, and it's not an exact match, it's not permitted.
    if '*' not in allowed_pattern and '**' not in allowed_pattern:
        return False

    policy_parts = allowed_pattern.split('::')
    request_parts = clean_req.split('::')

    policy_idx = 0
    request_idx = 0

    while policy_idx < len(policy_parts) and request_idx < len(request_parts):
        policy_part = policy_parts[policy_idx]
        request_part = request_parts[request_idx]

        if policy_part == '**':
            # If '**' is the last part of the policy, it matches all remaining request parts.
            return True
        elif policy_part == '*':
            # '*' matches exactly one segment.
            policy_idx += 1
            request_idx += 1
        elif policy_part == request_part:
            # Exact segment match.
            policy_idx += 1
            request_idx += 1
        else:
            # Mismatch, and not a wildcard that can resolve it.
            return False

    # After iterating, check if both pattern and request have been fully consumed.
    # If policy_idx == len(policy_parts), all policy parts were matched.
    # If request_idx == len(request_parts), all request parts were matched.
    # This ensures "a::b" does not match "a::b::c" unless policy ends with wildcard.

    # Special case: A policy like "a::b" should not match "a::b::c".
    # If policy ends without a wildcard, then request_parts must also end at the same point.
    if policy_idx == len(policy_parts) and request_idx == len(request_parts):
        return True
    
    # If policy ends with '**', it should already have returned True above.
    # If policy ends with '*', it should have consumed one segment, and if request also ended, it's a match.
    # All other cases where policy is exhausted but request is not, or vice versa, are false.
    return False
