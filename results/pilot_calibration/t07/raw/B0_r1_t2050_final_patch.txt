"""API gateway request journal writer."""
from datetime import datetime, timezone
from journal_entry import JournalEntry
from status_catalog import is_error_status, get_status_label

def record_request(request_id: str, status_code: int, dt: datetime) -> JournalEntry:
    """Records incoming HTTP request journal entry.
    Timestamps must be formatted strictly in ISO 8601 UTC with explicit millisecond
    precision and uppercase 'Z' suffix: YYYY-MM-DDTHH:MM:SS.sssZ.
    """
    _ = is_error_status(status_code)
    label = get_status_label(status_code)

    # Ensure the datetime object is in UTC for consistent logging.
    # If the datetime object is naive, assume it's UTC as per the logging requirement.
    # If it's timezone-aware but not UTC, convert it to UTC.
    if dt.tzinfo is None:
        dt_utc = dt.replace(tzinfo=timezone.utc)
    else:
        dt_utc = dt.astimezone(timezone.utc)

    # Format strictly in ISO 8601 UTC with explicit millisecond precision and 'Z' suffix.
    # datetime.isoformat(timespec='milliseconds') produces 'YYYY-MM-DDTHH:MM:SS.sss+00:00' for UTC.
    # We replace '+00:00' with 'Z' to match the specified format.
    ts = dt_utc.isoformat(timespec='milliseconds').replace('+00:00', 'Z')

    return JournalEntry(
        request_id=request_id,
        status_code=status_code,
        logged_at=ts,
        context={"label": label}
    )
