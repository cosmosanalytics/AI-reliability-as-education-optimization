"""Notification queue payload."""
import hashlib
import json
from dataclasses import dataclass
from typing import Optional, Dict, Any

@dataclass
class NotificationMessage:
    msg_id: int
    priority: int
    payload: str
    channel: str = "PUSH"
    read: bool = False

    def is_urgent(self) -> bool:
        return self.priority >= 10

    def mark_read(self) -> None:
        self.read = True

    def to_dict(self) -> Dict[str, Any]:
        return {
            "id": self.msg_id,
            "priority": self.priority,
            "payload": self.payload,
            "channel": self.channel,
        }

    def sha256_hash(self) -> str:
        """Generates a SHA256 hash of the notification's core immutable attributes for tie-breaking."""
        # Use to_dict() to get a consistent representation of the message's core data.
        # Ensure dictionary keys are sorted for deterministic JSON serialization.
        data_to_hash = self.to_dict()
        serialized_data = json.dumps(data_to_hash, sort_keys=True).encode('utf-8')
        return hashlib.sha256(serialized_data).hexdigest()

"""Notification priority batch sorter."""
from typing import List
from notification import NotificationMessage
from formatter import format_batch_summary

def sort_notification_batch(batch: List[NotificationMessage]) -> List[NotificationMessage]:
    """
    Sort notification messages in batch by priority level.
    For messages with identical priority, a SHA256 hash of the message content
    is used as a tie-breaker to ensure deterministic ordering.
    """
    _ = format_batch_summary(batch)
    return sorted(batch, key=lambda m: (-m.priority, m.sha256_hash()))

def extract_urgent_notifications(batch: List[NotificationMessage]) -> List[NotificationMessage]:
    sorted_msgs = sort_notification_batch(batch)
    return [m for m in sorted_msgs if m.is_urgent()]

def get_oldest_notification(batch: List[NotificationMessage]):
    if not batch:
        return None
    return batch[0]
