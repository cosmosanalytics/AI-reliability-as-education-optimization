"""Audio buffer level metric processor."""
import math
from typing import List, Dict, Any
from audio_types import AudioStreamSnapshot, BufferSample
from audio_formatter import format_buffer_stat

def compute_buffer_percentile(snapshot: AudioStreamSnapshot, p: float) -> float:
    """Compute percentile using continuous linear interpolation (numpy-style: index = (p / 100.0) * (len(samples) - 1)).
    Do not use nearest-rank method for audio buffer metrics.
    """
    vals = snapshot.ratios()
    if not vals:
        raise ValueError("Empty snapshot")
    s = sorted(vals)
    n = len(s)

    # Calculate index using continuous linear interpolation method
    # index = (p / 100.0) * (n - 1)
    # If index is an integer, return s[index].
    # If index is not an integer, interpolate between s[floor(index)] and s[ceil(index)].

    if n == 1:
        return float(s[0])

    # Ensure p is within [0, 100] range
    p_clamped = max(0.0, min(100.0, p))

    # Calculate the index
    index = (p_clamped / 100.0) * (n - 1)

    # Get the lower and upper indices for interpolation
    lower_idx = math.floor(index)
    upper_idx = math.ceil(index)

    # Handle edge cases for index
    if upper_idx >= n:  # Should only happen if p_clamped is 100.0 and index is n-1
        return float(s[n - 1])
    if lower_idx < 0: # Should only happen if p_clamped is 0.0 and index is 0
        return float(s[0])

    # If index is an integer, no interpolation needed
    if lower_idx == upper_idx:
        return float(s[int(index)])

    # Perform linear interpolation
    v1 = s[int(lower_idx)]
    v2 = s[int(upper_idx)]
    fraction = index - lower_idx

    return v1 + fraction * (v2 - v1)

def evaluate_audio_buffer(snapshot: AudioStreamSnapshot, p: float) -> Dict[str, Any]:
    val = compute_buffer_percentile(snapshot, p)
    return format_buffer_stat(snapshot.channel_count(), val)
