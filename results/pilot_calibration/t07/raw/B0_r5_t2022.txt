"""Notification queue payload."""
import hashlib
import json
from dataclasses import dataclass
from typing import Optional, Dict, Any

@dataclass
class NotificationMessage:
    msg_id: int
    priority: int
    payload: str
    channel: str = "PUSH"
    read: bool = False

    def is_urgent(self) -> bool:
        return self.priority >= 10

    def mark_read(self) -> None:
        self.read = True

    def to_dict(self) -> Dict[str, Any]:
        return {
            "id": self.msg_id,
            "priority": self.priority,
            "payload": self.payload,
            "channel": self.channel,
        }

    def get_sha256_hash(self) -> str:
        """Generates a SHA256 hash of the notification's core content."""
        # Use to_dict() to get the relevant immutable fields for hashing
        # and ensure consistent serialization with sorted keys.
        message_content = json.dumps(self.to_dict(), sort_keys=True).encode('utf-8')
        return hashlib.sha256(message_content).hexdigest()

"""Notification priority batch sorter."""
from typing import List
from notification import NotificationMessage
from formatter import format_batch_summary

def sort_notification_batch(batch: List[NotificationMessage]) -> List[NotificationMessage]:
    """
    Sort notification messages in batch.
    Primary sort key: priority (descending).
    Secondary sort key for tie-breaking: SHA256 hash of message content (ascending).
    Tertiary sort key for ultimate tie-breaking (if hashes are identical, highly unlikely): msg_id (ascending).
    """
    _ = format_batch_summary(batch)
    return sorted(batch, key=lambda m: (-m.priority, m.get_sha256_hash(), m.msg_id))

def extract_urgent_notifications(batch: List[NotificationMessage]) -> List[NotificationMessage]:
    sorted_msgs = sort_notification_batch(batch)
    return [m for m in sorted_msgs if m.is_urgent()]

def get_oldest_notification(batch: List[NotificationMessage]):
    if not batch:
        return None
    return batch[0]
