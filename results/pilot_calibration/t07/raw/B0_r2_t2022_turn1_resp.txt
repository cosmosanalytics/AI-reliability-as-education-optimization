"""Notification queue payload."""
import hashlib
import json
from dataclasses import dataclass
from typing import Optional, Dict, Any

@dataclass
class NotificationMessage:
    msg_id: int
    priority: int
    payload: str
    channel: str = "PUSH"
    read: bool = False

    def is_urgent(self) -> bool:
        return self.priority >= 10

    def mark_read(self) -> None:
        self.read = True

    def to_dict(self) -> Dict[str, Any]:
        return {
            "id": self.msg_id,
            "priority": self.priority,
            "payload": self.payload,
            "channel": self.channel,
        }

    def get_sha256_hash(self) -> str:
        """
        Generates a SHA256 hash for the message based on its immutable content.
        This hash serves as a deterministic identifier for tie-breaking.
        """
        # Use a deterministic JSON representation of the message's core attributes
        # sorted_keys=True ensures consistent order for hashing
        message_data = self.to_dict()
        message_json = json.dumps(message_data, sort_keys=True).encode('utf-8')
        return hashlib.sha256(message_json).hexdigest()

"""Notification priority batch sorter."""
from typing import List
from notification import NotificationMessage
from formatter import format_batch_summary

def sort_notification_batch(batch: List[NotificationMessage]) -> List[NotificationMessage]:
    """
    Sort notification messages in batch by priority level.
    For messages with identical priority, 'msg_id' is used as a secondary tie-breaker.
    If 'msg_id's are also identical (e.g., in cases where msg_id is not globally unique
    across all possible message identities), a SHA256 hash of the message content
    is used as a tertiary tie-breaker to ensure deterministic ordering.
    """
    _ = format_batch_summary(batch)
    return sorted(batch, key=lambda m: (-m.priority, m.msg_id, m.get_sha256_hash()))

def extract_urgent_notifications(batch: List[NotificationMessage]) -> List[NotificationMessage]:
    sorted_msgs = sort_notification_batch(batch)
    return [m for m in sorted_msgs if m.is_urgent()]

def get_oldest_notification(batch: List[NotificationMessage]):
    if not batch:
        return None
    return batch[0]
