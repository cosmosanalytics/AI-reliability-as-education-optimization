"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    Normalizes the keys of the input data.
    """
    return normalize_dict_keys(data)

def publish_stream_packet(service_name: str, data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepares and envelops a telemetry metric packet for streaming.

    The input data's keys are normalized, and then the normalized data
    is wrapped in a standard wire envelope.
    """
    normalized_payload = prepare_metric_packet(data, epoch_ms)
    enveloped_packet = build_envelope(service_name, epoch_ms, normalized_payload)
    return enveloped_packet
