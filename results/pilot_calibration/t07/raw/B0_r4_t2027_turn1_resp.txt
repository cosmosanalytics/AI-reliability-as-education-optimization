"""Lease validity checker."""
def is_lease_valid(valid_until: float, current_time: float) -> bool:
    # A lease is valid strictly *before* its valid_until timestamp.
    # At the exact valid_until time, the lease is considered expired.
    # This prevents split-brain scenarios where two nodes might simultaneously
    # consider the lease valid at the expiration boundary.
    return current_time < valid_until

def remaining_lease_time(valid_until: float, current_time: float) -> float:
    return max(0.0, valid_until - current_time)
