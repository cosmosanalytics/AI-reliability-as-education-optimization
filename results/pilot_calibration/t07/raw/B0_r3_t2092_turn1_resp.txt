"""Audit Trail Event Time Normalizer utility."""
from datetime import datetime, timezone
from iso_helper import parse_epoch_seconds, is_valid_iso_format, extract_year_component, format_epoch_millis
from timestamp_model import TimestampRecord, create_event_record

def normalize_audit_timestamp(epoch_s: float) -> str:
    # The TimestampRecord should store the full millisecond precision if available
    # Convert float seconds to integer milliseconds for the record
    rec = create_event_record("normalize_audit_timestamp", int(epoch_s * 1000))
    
    # Reuse the millisecond formatting utility from iso_helper.py
    # Convert epoch_s (float seconds) to epoch_ms (integer milliseconds) for the formatting function
    return format_epoch_millis(int(epoch_s * 1000))

def format_sequence(timestamps: list) -> list:
    return [normalize_audit_timestamp(t) for t in timestamps]
