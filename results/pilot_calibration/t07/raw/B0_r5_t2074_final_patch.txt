"""Low-level jitter percentile calculation engine."""
import math
from typing import List

def evaluate_jitter_percentile(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using the nearest-rank method.
    The k-th element (0-indexed) is chosen where k = math.ceil(p / 100 * n) - 1.
    For p=0, it returns the minimum value (index 0).
    For p=100, it returns the maximum value (index n-1).
    """
    if not values:
        raise ValueError("No jitter values")
    s = sorted(values)
    n = len(s)

    if p < 0.0 or p > 100.0:
        raise ValueError("Percentile p must be between 0 and 100 inclusive.")

    if p == 0.0:
        return s[0]
    if p == 100.0:
        return s[n-1]

    # Nearest-rank method: k = ceil(p/100 * n) - 1
    # The index k is 0-based.
    # math.ceil(x) returns a float, convert to int.
    k = int(math.ceil(p / 100.0 * n)) - 1

    # Ensure k is within valid bounds [0, n-1]
    k = max(0, min(k, n - 1))

    return float(s[k])
