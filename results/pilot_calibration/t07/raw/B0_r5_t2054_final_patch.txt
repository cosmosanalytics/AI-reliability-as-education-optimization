"""Low-level dictionary and list hierarchy merger."""
from typing import Dict, Any, List

def merge_spec_values(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """
    Recursively merges dictionary values.
    - If both base and override values for a key are dictionaries, they are merged recursively.
    - If both base and override values for a key are lists, the override list is appended to the base list.
      Duplicates are removed to maintain uniqueness, preserving order of first appearance from base then override.
    - For any other type, the override value replaces the base value.
    """
    result = dict(base)
    for k, v in override.items():
        if k in result and isinstance(result[k], dict) and isinstance(v, dict):
            result[k] = merge_spec_values(result[k], v)
        elif k in result and isinstance(result[k], list) and isinstance(v, list):
            # Merge lists by appending override to base and removing duplicates
            merged_list = list(result[k])
            for item in v:
                if item not in merged_list:
                    merged_list.append(item)
            result[k] = merged_list
        else:
            result[k] = v
    return result

def is_spec_empty(spec: Dict[str, Any]) -> bool:
    return len(spec) == 0
