"""Storage scope normalization engine."""
def validate_prefix_match(scope_prefix: str, asset_scope: str) -> bool:
    if not asset_scope.startswith(scope_prefix):
        return False
    
    # If the asset scope is exactly the prefix, it's a match.
    if asset_scope == scope_prefix:
        return True
    
    # If the asset scope is longer, the character immediately following the prefix
    # must be the scope segment separator "::" to ensure hierarchical matching.
    # For example, "assets::images" should match "assets::images::logo" but not "assets::images_private".
    if asset_scope[len(scope_prefix):].startswith("::"):
        return True
        
    return False

def get_scope_depth(scope_str: str) -> int:
    return len([seg for seg in scope_str.split("::") if seg])

def normalize_scope(scope_str: str) -> str:
    return scope_str.strip()
