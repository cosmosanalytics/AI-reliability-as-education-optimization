"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# TODO: The service_name should ideally be passed in or configured,
# but for now, we use a placeholder to enable envelope creation.
# This requires further refinement for a production system.
_DEFAULT_SERVICE_NAME = "unknown_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    # Use a placeholder service name as the function signature cannot be changed
    # due to test file modification constraints.
    enveloped_msg = build_envelope(_DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)
    return enveloped_msg

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # The following checks are for verification/side-effects but not directly
    # used in the return value path.
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data) # This extracts keys from the original data, not the enveloped payload
    return msg
