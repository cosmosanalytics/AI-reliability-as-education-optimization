"""Storage scope normalization engine."""
def validate_prefix_match(scope_prefix: str, asset_scope: str) -> bool:
    """
    Checks if an asset's scope is accessible given a required scope prefix.
    Access is granted if:
    1. The asset_scope is an exact match for the scope_prefix.
    2. The asset_scope is a direct descendant of the scope_prefix,
       meaning it starts with scope_prefix followed by '::'.
    This prevents matching 'assets::images_private' for a 'assets::images' requirement.
    """
    return asset_scope == scope_prefix or asset_scope.startswith(scope_prefix + "::")

def get_scope_depth(scope_str: str) -> int:
    return len([seg for seg in scope_str.split("::") if seg])

def normalize_scope(scope_str: str) -> str:
    return scope_str.strip()
