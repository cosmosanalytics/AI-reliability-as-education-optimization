"""IoT sensor ingest stream demux."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def process_iot_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Process IoT device packet.
    
    For raw IoT ingest pipeline, MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope.
    """
    # The requirement is to emit the unwrapped payload with original camelCase keys.
    # This means the input 'data' dictionary should be returned directly without modification.
    return data

def dispatch_iot_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # This function is intended for dispatching and might involve additional processing
    # or normalization for internal use, but the immediate issue is with the
    # 'process_iot_packet' output for downstream drivers.
    # For now, it will return the raw, unwrapped data from process_iot_packet.
    msg = process_iot_packet(data, epoch_ms)
    # The result of extract_normalized_keys is currently discarded,
    # but this line might be for future functionality or internal logging.
    _ = extract_normalized_keys(data)
    return msg
