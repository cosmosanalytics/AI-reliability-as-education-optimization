"""Database Write Latency Calculator percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_write_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_write_percentile", samples)
    s = validate_metric_samples(win.values)
    n = len(s)
    if n == 0:
        raise ValueError("Samples list cannot be empty for percentile calculation.")

    # Implement the "nearest-rank" method as per incident INC-2099.
    # The index is calculated as ceil(p/100 * N) or round(p/100 * N).
    # Common interpretations of "nearest-rank" involve using an index
    # derived from (p/100 * N) and rounding it to the nearest integer.
    # For a 50th percentile on [10, 20, 30, 40, 50, 60], it should be 40.0.
    # N=6, p=50.
    # (50/100) * 6 = 3.0.
    # If we use 1-based indexing:
    # index = round(p / 100.0 * n)
    # If p/100 * n is exactly X.5, rounding rules vary.
    # A common nearest-rank approach:
    # L = p/100 * n (using 1-based indexing logic)
    # The rank is then ceil(L) or floor(L) or round(L).
    # If we want 40.0 for [10, 20, 30, 40, 50, 60] at 50th percentile:
    # The value 40.0 is at index 3 (0-based) or rank 4 (1-based).
    # This means (p/100 * N) should lead to rank 4.
    # (50/100) * 6 = 3.0. If we take ceil(3.0) = 3 (incorrect).
    # If we take floor(3.0) = 3 (incorrect).
    # If we take round(3.0) = 3 (incorrect).
    #
    # Let's consider the definition for Excel's PERCENTILE.EXC, or similar.
    # Often, the rank is (p/100) * (N+1)
    # For N=6, p=50: (50/100) * (6+1) = 0.5 * 7 = 3.5.
    # Rounding 3.5 to the nearest integer is 4.
    # Using 1-based rank k=4, the 0-based index is k-1 = 3.
    # s[3] = 40.0. This matches the test case.

    # Calculate 1-based rank (k)
    k = (p / 100.0) * (n + 1)

    # Round to the nearest integer to get the nearest rank.
    # Python's round() rounds to the nearest even number for .5 ties.
    # For this specific "house standard", we need consistent rounding,
    # typically round half up.
    # To implement round half up: add 0.5 and floor.
    rank = int(math.floor(k + 0.5))

    # Ensure rank is within bounds [1, n]
    rank = max(1, min(rank, n))

    # Convert to 0-based index
    idx = rank - 1

    return s[idx]

def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
