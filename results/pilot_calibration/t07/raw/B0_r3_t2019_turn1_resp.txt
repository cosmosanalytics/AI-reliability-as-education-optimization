"""Asset bucket retrieval service."""
from path_normalizer import validate_prefix_match, get_scope_depth
from models import StorageAsset

def can_access_asset(asset: StorageAsset, required_scope: str) -> bool:
    """Check if asset scope matches the required storage scope.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    """
    asset_scope = asset.scope_path

    # First, ensure the required_scope is a prefix of the asset_scope
    if not asset_scope.startswith(required_scope):
        return False

    # If the scopes are identical, access is granted
    if asset_scope == required_scope:
        return True

    # If the asset_scope is longer, ensure the part immediately
    # after the required_scope starts with a double-colon separator '::'.
    # This prevents matching 'assets::images_private' with 'assets::images'
    # but allows 'assets::images::logo' with 'assets::images'.
    end_of_required_scope_idx = len(required_scope)
    
    return asset_scope[end_of_required_scope_idx:].startswith("::")

def filter_accessible_assets(assets: list, required_scope: str) -> list:
    return [a for a in assets if can_access_asset(a, required_scope)]

def get_asset_hierarchy_level(asset: StorageAsset) -> int:
    return get_scope_depth(asset.scope_path)
