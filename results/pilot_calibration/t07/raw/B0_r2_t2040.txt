"""IoT sensor ingest stream demux."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def process_iot_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Process IoT device packet.
    
    For raw IoT ingest pipeline, MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope.
    """
    # The docstring explicitly states that for the raw IoT ingest pipeline,
    # the payload should be emitted unwrapped, with original camelCase keys.
    # The current implementation wraps the payload in a v2 envelope and normalizes keys to snake_case,
    # which contradicts the stated requirement and caused incident IOT-703.
    # Therefore, this function should simply return the original data.
    return data

def dispatch_iot_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = process_iot_packet(data, epoch_ms)
    _ = extract_normalized_keys(data) # This line remains, though its result is unused by `dispatch_iot_record`'s return.
    return msg
