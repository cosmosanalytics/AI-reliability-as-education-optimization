"""Snapshot manifest timestamp formatting."""
from datetime import datetime, timezone

def format_snapshot_timestamp(dt: datetime) -> str:
    # Format as ISO 8601 with milliseconds and 'Z' for UTC.
    # %f gives microseconds, slicing [:-3] converts it to milliseconds.
    return dt.strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"

def parse_snapshot_epoch(dt: datetime) -> int:
    return int(dt.timestamp())
