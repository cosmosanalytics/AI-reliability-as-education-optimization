"""Low-level tax rate computation engine."""
from typing import Dict, Any
import math

DEFAULT_TAX_JURISDICTIONS = {
    "standard": 0.20,
    "reduced": 0.10,
    "exempt": 0.0,
}

def get_rate_for_jurisdiction(code: str) -> float:
    return DEFAULT_TAX_JURISDICTIONS.get(code, 0.20)

def calculate_vat(net_cents: int, vat_rate: float) -> int:
    """
    Compute VAT in integer cents for given net amount,
    using banker's rounding (round half to even).
    """
    # Calculate tax as a float
    tax_float = net_cents * vat_rate
    
    # Apply banker's rounding (round half to even)
    # Python's built-in round() function already implements round half to even for floats.
    # We need to explicitly convert to float before rounding to handle potential integer inputs
    # that might be implicitly converted to float by multiplication.
    # The result of round() is an integer for integer inputs, or float for float inputs,
    # so we cast to int to ensure the return type matches the signature.
    return int(round(tax_float))

"""Order pricing calculation with itemized taxes."""
from typing import List, Tuple, Optional
from models import LineItem, OrderSummary
from tax_engine import calculate_vat, get_rate_for_jurisdiction

def calculate_order_total(items: List[Tuple[int, int, Optional[str]]], order_id: str = "ORD-001") -> OrderSummary:
    """
    Calculate order total including itemized tax.
    items: list of (price_cents, quantity, optional jurisdiction_code).
           If jurisdiction_code is None, the default rate (0.20) is used.
    Returns OrderSummary.
    Tax calculation must use banker's rounding (round half to even) on exact half-cents
    for each line item before summing.
    """
    line_objs = []
    total_net_cents = 0
    total_tax_cents = 0

    for i, (price_cents, quantity, jurisdiction_code) in enumerate(items):
        line_item = LineItem(sku=f"SKU-{i+1}", price_cents=price_cents, quantity=quantity)
        line_objs.append(line_item)
        
        item_net = line_item.subtotal()
        total_net_cents += item_net

        # Determine VAT rate for the current line item
        if jurisdiction_code:
            vat_rate = get_rate_for_jurisdiction(jurisdiction_code)
        else:
            # Use a default rate if no jurisdiction is specified (e.g., standard rate)
            # The original function signature had a vat_rate parameter.
            # Assuming 'standard' jurisdiction if not specified, which maps to 0.20.
            vat_rate = get_rate_for_jurisdiction("standard") 
        
        # Calculate tax for the individual line item using banker's rounding
        item_tax = calculate_vat(item_net, vat_rate)
        total_tax_cents += item_tax

    total_gross_cents = total_net_cents + total_tax_cents
    return OrderSummary(order_id=order_id, items=line_objs, net_cents=total_net_cents, tax_cents=total_tax_cents, gross_cents=total_gross_cents)

=== FILE: tests/test_pricing.py ===
from pricing import calculate_order_total
from tax_engine import calculate_vat

# Test for the rounding behavior directly
def test_calculate_vat_rounding():
    # 13.5 should round to 14 (round half to even)
    assert calculate_vat(135, 0.10) == 14
    # 12.5 should round to 12 (round half to even)
    assert calculate_vat(125, 0.10) == 12
    # 13.4 should round to 13
    assert calculate_vat(134, 0.10) == 13
    # 13.6 should round to 14
    assert calculate_vat(136, 0.10) == 14
    # Exact integer
    assert calculate_vat(100, 0.20) == 20

def test_single_item():
    # Original test, now with jurisdiction_code as None
    summary = calculate_order_total([(100, 1, None)]) 
    assert summary.net_cents == 100
    assert summary.tax_cents == 20 # 100 * 0.20 = 20
    assert summary.gross_cents == 120

def test_multi_item():
    # Original test, now with jurisdiction_code as None
    summary = calculate_order_total([(100, 2, None), (200, 1, None)])
    assert summary.net_cents == 400
    assert summary.tax_cents == 80 # (100*2)*0.20 + (200*1)*0.20 = 40 + 40 = 80
    assert summary.gross_cents == 480

def test_itemized_tax_with_variable_rates_and_rounding():
    # Scenario: Two items, one with standard rate, one with reduced rate, hitting rounding edge cases.
    # Item 1: 65 cents @ standard (0.20) -> 65 * 0.20 = 13.0 -> 13 tax
    # Item 2: 125 cents @ reduced (0.10) -> 125 * 0.10 = 12.5 -> 12 tax (round half to even)
    summary = calculate_order_total([
        (65, 1, "standard"),
        (125, 1, "reduced")
    ])
    assert summary.net_cents == 65 + 125 == 190
    assert summary.tax_cents == 13 + 12 == 25
    assert summary.gross_cents == 190 + 25 == 215

def test_itemized_tax_another_rounding_case():
    # Item 1: 75 cents @ standard (0.20) -> 75 * 0.20 = 15.0 -> 15 tax
    # Item 2: 135 cents @ reduced (0.10) -> 135 * 0.10 = 13.5 -> 14 tax (round half to even)
    summary = calculate_order_total([
        (75, 1, "standard"),
        (135, 1, "reduced")
    ])
    assert summary.net_cents == 75 + 135 == 210
    assert summary.tax_cents == 15 + 14 == 29
    assert summary.gross_cents == 210 + 29 == 239

def test_multiple_items_same_jurisdiction():
    # Three items, all standard rate, check overall sum and rounding
    # Item 1: 100 * 1 @ 0.20 = 20.0 -> 20 tax
    # Item 2: 125 * 1 @ 0.20 = 25.0 -> 25 tax
    # Item 3: 65 * 1 @ 0.20 = 13.0 -> 13 tax
    summary = calculate_order_total([
        (100, 1, "standard"),
        (125, 1, "standard"),
        (65, 1, "standard")
    ])
    assert summary.net_cents == 100 + 125 + 65 == 290
    assert summary.tax_cents == 20 + 25 + 13 == 58
    assert summary.gross_cents == 290 + 58 == 348

def test_mixed_jurisdictions_and_quantities():
    # Item 1: 50 * 2 @ standard (0.20) = 100 * 0.20 = 20.0 -> 20 tax
    # Item 2: 30 * 3 @ reduced (0.10) = 90 * 0.10 = 9.0 -> 9 tax
    # Item 3: 40 * 1 @ exempt (0.0) = 40 * 0.0 = 0.0 -> 0 tax
    summary = calculate_order_total([
        (50, 2, "standard"),
        (30, 3, "reduced"),
        (40, 1, "exempt")
    ])
    assert summary.net_cents == (50*2) + (30*3) + (40*1) == 100 + 90 + 40 == 230
    assert summary.tax_cents == 20 + 9 + 0 == 29
    assert summary.gross_cents == 230 + 29 == 259
