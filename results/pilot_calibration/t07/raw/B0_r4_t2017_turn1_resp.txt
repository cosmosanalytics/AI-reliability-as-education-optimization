"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    policy_segments = policy.allowed_pattern.split('::')
    request_segments = clean_req.split('::')

    if not policy_segments:
        return False # A policy pattern must not be empty

    # Handle the special case where the pattern is just "**"
    if policy.allowed_pattern == "**":
        return True

    # Iterate through segments for comparison
    i = 0
    while i < len(policy_segments) and i < len(request_segments):
        policy_segment = policy_segments[i]
        request_segment = request_segments[i]

        if policy_segment == "**":
            # If "**" is encountered, it matches all remaining segments in the request.
            # This must be the last segment in the policy pattern.
            return True
        elif policy_segment == "*":
            # "*" matches exactly one segment.
            # Continue to the next segment if there is one.
            pass
        elif policy_segment != request_segment:
            # Segments don't match and it's not a wildcard.
            return False
        i += 1

    # After iterating, check remaining conditions based on pattern's last segment
    if i == len(policy_segments):
        # All policy segments have been matched.
        # If the policy's last segment was not "**", then the request must not have more segments.
        return i == len(request_segments)
    else: # i < len(policy_segments)
        # Policy pattern has remaining segments that were not matched by the request.
        # This can only be valid if the remaining policy segment is "**" and it was already handled above.
        # Otherwise, if we reached here, it means policy_segments is longer than request_segments,
        # and the unmatched policy segments are not "**" (or "*" at the end which would require equal length).
        return False


def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
