"""Document access RBAC filter."""
from path_resolver import clean_path, split_path_segments, is_root_path
from audit_logger import format_access_attempt

def check_document_access(allowed_folder: str, requested_doc_path: str) -> bool:
    """Verify if document path is permitted within the designated folder scope."""
    cleaned_allowed_folder = clean_path(allowed_folder)
    cleaned_requested_doc_path = clean_path(requested_doc_path)

    # If the allowed folder is the root, all paths are considered accessible.
    # This might need refinement based on specific root access policies,
    # but generally, root access implies access to all subdirectories.
    if is_root_path(cleaned_allowed_folder):
        allowed = True
    else:
        # Check if the requested path is exactly the allowed folder
        # or if it's a direct child path within the allowed folder.
        # This prevents granting access to sibling directories like /public-internal
        # when only /public is allowed.
        allowed = (cleaned_requested_doc_path == cleaned_allowed_folder or
                   cleaned_requested_doc_path.startswith(cleaned_allowed_folder + '/'))

    _ = format_access_attempt(allowed_folder, requested_doc_path, allowed)
    return allowed

def filter_accessible_docs(allowed_folder: str, doc_paths: list) -> list:
    return [p for p in doc_paths if check_document_access(allowed_folder, p)]

def can_access_all_docs(allowed_folder: str, doc_paths: list) -> bool:
    return all(check_document_access(allowed_folder, p) for p in doc_paths)
