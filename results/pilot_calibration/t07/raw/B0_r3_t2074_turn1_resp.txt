"""Low-level jitter percentile calculation engine."""
import math
from typing import List

def evaluate_jitter_percentile(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using the nearest-rank method.
    The index k is calculated as math.ceil(p / 100 * n) - 1 on sorted values.
    """
    if not values:
        raise ValueError("No jitter values")
    s = sorted(values)
    n = len(s)

    # Handle edge cases for p = 0 and p = 100
    if p == 0:
        return s[0]
    if p == 100:
        return s[n - 1]

    # Calculate 1-indexed rank
    rank = math.ceil((p / 100.0) * n)

    # Convert to 0-indexed position
    idx = int(rank - 1)

    # Ensure index is within bounds (should be handled by p=0/100, but good for robustness)
    if idx < 0:
        idx = 0
    if idx >= n:
        idx = n - 1

    return float(s[idx])
