"""Storage scope normalization engine."""
def validate_prefix_match(scope_prefix: str, asset_scope: str) -> bool:
    # First, a simple prefix check. If it doesn't start with the prefix, it's not a match.
    if not asset_scope.startswith(scope_prefix):
        return False

    # If the asset scope is exactly the required scope, it's a match.
    if asset_scope == scope_prefix:
        return True

    # If the asset scope is longer than the required scope,
    # it must represent a sub-path. This means the character immediately
    # following the prefix in the asset_scope must be the start of a '::' separator.
    # This prevents matching "assets::images_private" with "assets::images".
    remaining_path = asset_scope[len(scope_prefix):]
    return remaining_path.startswith("::")

def get_scope_depth(scope_str: str) -> int:
    return len([seg for seg in scope_str.split("::") if seg])

def normalize_scope(scope_str: str) -> str:
    return scope_str.strip()
