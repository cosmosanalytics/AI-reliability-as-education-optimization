"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total_invoice_amount = sum(line_totals)

    if total_invoice_amount == 0:
        return [0] * len(line_totals)

    # Calculate initial proportional allocations using floating point for precision
    proportional_allocations = [(cents * total_discount_cents) / total_invoice_amount for cents in line_totals]

    # Truncate to integer cents for initial allocation
    initial_allocations = [int(alloc) for alloc in proportional_allocations]

    # Calculate the sum of initial allocations
    current_allocated_sum = sum(initial_allocations)

    # Determine the remaining unallocated amount
    remainder = total_discount_cents - current_allocated_sum

    # Distribute the remainder one cent at a time to the lines that lost the most due to truncation
    # by sorting them based on their fractional part in descending order.
    if remainder > 0:
        # Create a list of (fractional_part, original_index) tuples
        fractional_parts_with_indices = []
        for i, prop_alloc in enumerate(proportional_allocations):
            fractional_parts_with_indices.append((prop_alloc - initial_allocations[i], i))

        # Sort in descending order of fractional part
        fractional_parts_with_indices.sort(key=lambda x: x[0], reverse=True)

        # Add 1 cent to the lines with the largest fractional parts until remainder is zero
        for i in range(remainder):
            # The index of the line to receive an extra cent is stored in the second element of the tuple
            line_index_to_add_penny = fractional_parts_with_indices[i][1]
            initial_allocations[line_index_to_add_penny] += 1

    return initial_allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
