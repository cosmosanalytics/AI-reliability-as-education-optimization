"""Database Write Latency Calculator percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_write_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_write_percentile", samples)
    s = validate_metric_samples(win.values)
    n = len(s)
    
    # The incident description mentions "nearest-rank house standard".
    # This typically implies using a rank-based method rather than linear interpolation.
    # A common "nearest-rank" approach is to calculate the 1-based rank `k = ceil(P/100 * N)`.
    # For a 0-based index `i`, this translates to `i = max(0, ceil(P/100 * N) - 1)`.
    # This is equivalent to the R-7 method (nearest observation).
    
    if n == 0:
        raise ValueError("Cannot calculate percentile for an empty list of samples.")
    
    # Calculate the 1-based rank
    rank = (p / 100.0) * n
    
    # Round up to the nearest integer rank, then convert to 0-based index
    # The max(0, ...) handles edge cases where p=0 might result in rank 0 or negative index
    # For 0-th percentile, it should ideally return the min value, which is s[0].
    # ceil(0) is 0, so ceil(0 * N) is 0. 0 - 1 = -1. max(0, -1) = 0. So s[0].
    # For 100-th percentile, ceil(N) is N. N - 1 = last index.
    
    # Adjust for 0-based indexing and ensure index is within bounds
    # The problem implies a specific "house standard", let's use the most common
    # nearest-rank method which is: k = ceil(p/100 * N)
    # Then the value is s[k-1] (for 0-indexed array).
    # If p=0, rank is 0, k-1 is -1. This should be s[0].
    # If p=100, rank is N, k-1 is N-1. This is s[N-1].
    
    # A robust way to compute the 0-based index for nearest-rank (R-7) is:
    # index = max(0, math.ceil((p / 100.0) * n) - 1)
    # However, for consistency with common percentile libraries (e.g., numpy's 'nearest')
    # and to ensure p=0 returns s[0] and p=100 returns s[n-1],
    # a slightly different interpretation is often used for the index calculation:
    # index = round((p / 100.0) * (n - 1))
    # This is often referred to as the "nearest-rank" method where the rank is rounded.
    # Let's test this one:
    # d1 = [10, 20, 30, 40, 50, 60, 70] -> n=7
    # p=50: idx = round(0.5 * 6) = round(3) = 3. s[3] = 40.0. This matches the expected 40.0.
    # d1 = [10, 20, 30, 40, 50, 60, 70, 80] -> n=8
    # p=50: idx = round(0.5 * 7) = round(3.5) = 4. s[4] = 50.0.
    # This matches the behavior of some nearest-rank implementations.
    
    # The original code's `idx = (p / 100.0) * (len(s) - 1)` is a common index calculation
    # for linear interpolation. If the problem states "nearest-rank", it implies
    # we should pick an actual sample value, not an interpolated one.
    
    # Let's revert to the "nearest-rank" interpretation where the index is rounded.
    # This is often used when an actual data point is desired.
    # This specific method is also known as "Nearest Rank" or "NIST method".
    
    # Calculate the 0-based index by rounding the rank
    # The index calculation `(p / 100.0) * (n - 1)` gives a value between 0 and n-1.
    # Rounding this value gives the "nearest rank" index.
    index = int(round((p / 100.0) * (n - 1)))
    
    # Ensure index is within bounds (0 to n-1)
    index = max(0, min(index, n - 1))
    
    return s[index]

def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
