"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total_line_value = sum(line_totals)
    if total_line_value == 0:
        return [0] * len(line_totals)

    # Calculate base allocations using float division to retain precision
    # and store the fractional part for remainder distribution.
    allocations_with_fractions = []
    for cents in line_totals:
        exact_allocation = (cents * total_discount_cents) / total_line_value
        allocations_with_fractions.append(exact_allocation)

    # Calculate initial integer allocations by truncating
    final_allocations = [int(alloc) for alloc in allocations_with_fractions]
    current_total_allocated = sum(final_allocations)

    # Distribute remaining cents (penny mismatch)
    remainder = total_discount_cents - current_total_allocated

    if remainder > 0:
        # Calculate the fractional part for each line
        fractional_parts = [alloc - int(alloc) for alloc in allocations_with_fractions]

        # Create a list of (fractional_part, original_index) tuples
        # Sort in descending order of fractional part to prioritize lines that lost more due to truncation
        sorted_fractions = sorted(
            enumerate(fractional_parts), key=lambda x: x[1], reverse=True
        )

        # Distribute the remainder cents one by one to the lines with the largest fractional parts
        for i in range(remainder):
            # The index is the original index of the line item
            original_index = sorted_fractions[i][0]
            final_allocations[original_index] += 1

    return final_allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
