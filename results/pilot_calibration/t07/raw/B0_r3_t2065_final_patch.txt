"""Idempotency key construction factory for refunds."""
import uuid
from typing import Optional

def create_refund_key(refund_id: str) -> str:
    """Creates refund idempotency key."""
    return f"ref_{refund_id}_{uuid.uuid4()}"

def is_valid_refund_key(key: str) -> bool:
    """Validates structure and length of refund key, ensuring standard prefix."""
    if not key or not isinstance(key, str):
        return False
    # Ensure the key starts with the standard prefix "ref_"
    if not key.startswith("ref_"):
        return False
    # Ensure the key is longer than just the prefix itself.
    # A key generated by create_refund_key will be significantly longer due to the UUID.
    return len(key) > len("ref_")
