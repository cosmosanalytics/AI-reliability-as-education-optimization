"""Low-level exchange rate computation."""
from typing import Tuple

def apply_fx_rate(cents: int, rate: float) -> int:
    """Compute converted currency amount in cents, rounded half up.
    This ensures proper handling of fractional cents for financial reconciliation.
    """
    # Use round half up for financial calculations.
    # This is equivalent to int(x + 0.5) for positive x, which is standard for commercial rounding.
    return int(cents * rate + 0.5)

def compute_spread(cents: int, base_rate: float, spread_bps: int = 10) -> Tuple[int, float]:
    effective_rate = base_rate * (1.0 + spread_bps / 10000.0)
    return apply_fx_rate(cents, effective_rate), effective_rate
