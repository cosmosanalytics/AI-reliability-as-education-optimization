"""Database pool manager."""
import copy
from typing import Dict, Any
from env_loader import merge_env_settings
from options import ConnectionOptions

def configure_pool(base_template: Dict[str, Any], overrides: Dict[str, Any]) -> ConnectionOptions:
    """Configure connection pool options from template and overrides.
    Returns ConnectionOptions instance.
    """
    # Perform a deep merge to ensure nested dictionaries (like 'pool')
    # are not shared references and can be safely modified by overrides
    # without affecting the original base_template or other configurations.
    # This prevents shared defaults from being corrupted across different
    # tenant or worker thread configurations.
    merged_settings = copy.deepcopy(base_template)
    for key, value in overrides.items():
        if isinstance(value, dict) and key in merged_settings and isinstance(merged_settings[key], dict):
            # If both base and override have a dictionary for the same key,
            # merge them recursively.
            merged_settings[key].update(value)
        else:
            # Otherwise, just update or add the value.
            merged_settings[key] = value

    # The merge_env_settings is no longer needed as deep_merge handles the combination.
    # The original env_loader.merge_env_settings performed a shallow merge.
    # For compatibility or if env_loader.merge_env_settings is used elsewhere,
    # it might still be called, but the deep copy and merge here is crucial
    # for resolving the shared state issue.
    # For now, let's assume the deep merge directly replaces the need for
    # env_loader.merge_env_settings for the specific problem of nested dicts.
    # If `env_loader.merge_env_settings` also handles environment-specific
    # settings that are not explicitly in `base_template` or `overrides`,
    # it should be called first, and then deep_merge should apply overrides.
    # Given the problem description about shared defaults, the issue is
    # in how `base_template` and `overrides` interact for nested structures.

    # Re-evaluating the role of env_loader.merge_env_settings:
    # It seems `merge_env_settings` is intended to load and merge actual
    # environment settings (like `load_system_env`). The current call pattern
    # suggests `base_template` and `overrides` are already the "settings"
    # to be merged.
    # The current `env_loader.merge_env_settings` does a shallow merge.
    # To correctly handle nested dictionaries and prevent shared state,
    # we need a deep merge.

    # Let's adapt the existing flow to use a deep merge.
    # First, get the initial settings (which might include env-loaded defaults),
    # then apply the `overrides` with a deep merge strategy.

    # The original `configure_pool` structure:
    # `settings = merge_env_settings(base_template, overrides)`
    # `dsn = settings.get("dsn", "sqlite:///:memory:")`
    # `opts = ConnectionOptions(dsn=dsn, pool_params=settings.get("pool", {}))`

    # The issue is that `merge_env_settings` performs a shallow copy.
    # If `base_template` contains `{"pool": default_pool_params_dict}`,
    # and `overrides` contains `{"pool": tenant_specific_pool_params_dict}`,
    # `merge_env_settings` will replace `default_pool_params_dict` with
    # `tenant_specific_pool_params_dict` if `overrides` has a "pool" key,
    # or if `overrides` *modifies* `default_pool_params_dict` if `overrides`
    # itself was a shallow modification of `base_template`.

    # The most robust solution for deep merging `base_template` and `overrides`
    # to prevent shared state corruption on nested dicts is to ensure that
    # any dictionary passed to ConnectionOptions for `pool_params` is a
    # completely independent copy.

    # The `merge_env_settings` in `env_loader.py` is a shallow merge.
    # We need a deep merge for configuration dictionaries.
    # Let's assume `base_template` is the default, and `overrides` are tenant-specific.
    # We must ensure that the `pool` dictionary within `base_template` is not
    # mutated by `overrides` if `overrides` provides specific pool parameters.

    # A simple deep copy of the base_template and then a deep update:
    settings = copy.deepcopy(base_template)
    for key, value in overrides.items():
        if key in settings and isinstance(settings[key], dict) and isinstance(value, dict):
            # If both are dictionaries, deep merge them.
            settings[key].update(value) # This performs a shallow merge of the nested dicts
                                        # which is what we need for pool params.
        else:
            settings[key] = value # Otherwise, replace or add.

    # Now, the 'settings' dictionary contains the deeply merged configuration.
    # The `pool_params` passed to `ConnectionOptions` will be a dictionary
    # that is a copy (from deepcopy) and potentially updated.
    # The original `base_template` remains untouched.

    dsn = settings.get("dsn", "sqlite:///:memory:")
    # Ensure pool_params is always a distinct dictionary, even if overrides was empty
    # or didn't specify 'pool'. `copy.deepcopy` already ensures this for `settings`.
    # `.get("pool", {})` handles the case where 'pool' key is missing.
    pool_params = copy.deepcopy(settings.get("pool", {}))
    opts = ConnectionOptions(dsn=dsn, pool_params=pool_params)
    _ = opts.summary()
    return opts

"""Environment settings loader and merger."""
import copy
from typing import Dict, Any

DEFAULT_CONN_TIMEOUT = 30

def load_system_env() -> Dict[str, Any]:
    return {"timeout": DEFAULT_CONN_TIMEOUT}

def merge_env_settings(base_settings: Dict[str, Any], override_settings: Dict[str, Any]) -> Dict[str, Any]:
    # This function is used by `pool.py` in its current form, but the issue
    # is with deep merging nested dictionaries.
    # To ensure nested dictionaries are also merged correctly and don't
    # share references, a deep copy and update is required for nested structures.
    # The `pool.py` file now handles its own deep merging for `base_template`
    # and `overrides` directly.
    # If this `merge_env_settings` function is intended for general use
    # and needs deep merge behavior for all callers, it should be updated.
    # For the specific ticket, the fix is applied in `pool.py` where the
    # `ConnectionOptions` are constructed.

    # If this function was to be the *sole* source of merging,
    # it would need to be changed to:
    # merged = copy.deepcopy(base_settings)
    # for key, value in override_settings.items():
    #     if key in merged and isinstance(merged[key], dict) and isinstance(value, dict):
    #         merged[key].update(value) # Shallow merge of nested dicts
    #     else:
    #         merged[key] = value
    # return merged

    # However, the ticket implies the problem is specifically within the
    # `configure_pool` logic where `pool_params` are handled.
    # The current `pool.py` fix ensures that the `pool_params` dictionary
    # passed to `ConnectionOptions` is always a distinct copy, preventing
    # shared state corruption.

    # Given the constraint "Do not modify files in tests/ or any test files."
    # and the specific traceback, the fix in `pool.py` is targeted.
    # This `merge_env_settings` function performs a shallow merge.
    # If other parts of the system rely on its shallow merge behavior,
    # changing it here might introduce regressions.
    # The fix for SEC-402 is specifically about `ConnectionOptions` and its `pool_params`.
    merged = dict(base_settings)
    merged.update(override_settings)
    return merged
