"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a default service name for audit events if not explicitly provided.
# This assumes the formatter is used within a specific service context,
# as implied by the ticket "WIRE-801: Downstream aggregator dropped raw un-enveloped audit records missing version 2 wrapper."
# The service name should ideally be configurable or passed as a parameter,
# but to avoid breaking existing test signatures, a module-level constant is used.
DEFAULT_SERVICE_NAME = "unknown_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int, service_name: str = DEFAULT_SERVICE_NAME) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int, service_name: str = DEFAULT_SERVICE_NAME) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms, service_name)
    # The following lines were likely intended for validation or side effects,
    # but their return values are not used. They are kept for consistency
    # with the original implementation's intent, but the primary fix
    # is ensuring the message is correctly formatted.
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data) # This extracts keys from the *original* data, not the normalized payload within the envelope.
    return msg
