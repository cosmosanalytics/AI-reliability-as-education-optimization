"""Document scope RBAC filter."""
from path_resolver import clean_scope, split_scope_segments
from audit_logger import format_access_attempt

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern.

    Access is granted if the requested scope is a direct descendant of the allowed scope.
    The allowed_scope can end with '::*' to signify any descendant.
    """
    cleaned_allowed = clean_scope(allowed_scope)
    cleaned_requested = clean_scope(requested_scope)

    # Handle wildcard for allowed_scope
    if cleaned_allowed.endswith("::*"):
        base_allowed_scope = cleaned_allowed[:-3] # Remove '::*'
        # If base_allowed_scope is empty, it means '::*' (root wildcard), allowing anything.
        # Otherwise, check if requested scope starts with the base, followed by a segment delimiter,
        # or if it's exactly the base scope itself (e.g., 'corp::public' allows 'corp::public' and 'corp::public::*')
        if not base_allowed_scope: # '::*' case, allows everything
            allowed = True
        else:
            # The requested scope must start with the base allowed scope
            # and either be identical or be a direct child (i.e., contain '::' after the base)
            allowed = cleaned_requested.startswith(base_allowed_scope) and \
                      (len(cleaned_requested) == len(base_allowed_scope) or \
                       cleaned_requested[len(base_allowed_scope):].startswith('::'))
    else:
        # No wildcard, exact prefix match on segments
        allowed_segments = split_scope_segments(cleaned_allowed)
        requested_segments = split_scope_segments(cleaned_requested)

        # The requested scope must have at least as many segments as the allowed scope
        # And all segments of the allowed scope must match the corresponding segments
        # of the requested scope exactly.
        # This prevents 'corp::public' from matching 'corp::public_internal'
        allowed = (len(requested_segments) >= len(allowed_segments) and
                   requested_segments[:len(allowed_segments)] == allowed_segments)

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
