"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# TODO: The service_name should ideally be configurable or passed as an argument.
# For now, a placeholder is used to resolve the immediate incident.
DEFAULT_SERVICE_NAME = "telemetry-service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.

    This function normalizes the payload keys and wraps the payload
    in a wire envelope with required metadata.
    """
    normalized_payload = normalize_dict_keys(data)
    # The service_name is a placeholder; it should be properly sourced.
    return build_envelope(DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following lines were likely intended for validation or logging,
    # but they don't modify the packet to ensure it's a valid envelope.
    # The packet is now built correctly by prepare_metric_packet.
    # We can keep them for potential future logging/debugging, but they are not
    # essential for the envelope creation itself anymore.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # Note: This extracts keys from original 'data', not the enveloped packet.
    return pkt
