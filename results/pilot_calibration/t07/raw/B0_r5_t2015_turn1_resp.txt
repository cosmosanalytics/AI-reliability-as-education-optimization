"""TSV cell sanitizer."""
from typing import Optional

def sanitize_cell(text: Optional[str]) -> str:
    if text is None:
        return ""
    # The audit parser expects double quotes to be escaped with a backslash.
    # The original implementation of replacing '"' with '""' is for CSV-like
    # formats where a double quote within a double-quoted field is doubled.
    # For the specific audit parser mentioned in the ticket, the requirement
    # is to escape with a backslash.
    return text.replace('"', '\\"')

def requires_sanitization(text: str) -> bool:
    # This function identifies characters that might need special handling.
    # Its logic remains valid even if the escaping mechanism for quotes changes.
    return any(c in text for c in ('\t', '\n', '"'))
