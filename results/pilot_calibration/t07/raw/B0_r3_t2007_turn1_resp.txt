"""Low-level step configuration compiler."""
import copy
from typing import Dict, Any

def compile_step_config(name: str, base_opts: Dict[str, Any], overrides: Dict[str, Any]) -> Dict[str, Any]:
    # Perform a deep copy of base_opts to ensure nested dictionaries are also independent.
    # This prevents mutations to stage-specific configurations from affecting the global defaults
    # or subsequent stages that might use the same base_opts.
    res = copy.deepcopy(base_opts)
    res["name"] = name
    
    # Merge overrides. For top-level keys that are dictionaries, perform a nested merge
    # to allow partial overrides of nested structures (e.g., {"retries": {"max": 5}}).
    for key, value in overrides.items():
        if key in res and isinstance(res[key], dict) and isinstance(value, dict):
            res[key].update(value)
        else:
            res[key] = value
            
    return res

def format_step_manifest(spec: Dict[str, Any]) -> str:
    name = spec.get("name", "unnamed")
    engine = spec.get("engine", "default")
    return f"StageManifest(name={name}, engine={engine})"
