"""Customer Notification Dispatcher wire constructor."""
from key_normalizer import normalize_dict_keys, count_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_envelope_version
from typing import Dict, Any

def dispatch_customer_notice(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    norm_payload = normalize_dict_keys(data)
    return build_envelope(src="customer_service", ts=epoch_ms, payload=norm_payload)

def validate_and_emit(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    res = dispatch_customer_notice(data, epoch_ms)
    _ = count_keys(data)
    return res
