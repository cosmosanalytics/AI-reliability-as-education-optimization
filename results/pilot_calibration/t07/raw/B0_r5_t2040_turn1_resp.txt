"""IoT sensor ingest stream demux."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def process_iot_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Process IoT device packet.
    
    For raw IoT ingest pipeline, MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope.
    """
    # As per the incident description and docstring, the raw IoT ingest pipeline
    # requires the original camelCase dictionary, unwrapped.
    # The previous implementation was normalizing keys and wrapping in a v2 envelope,
    # which caused crashes for the downstream edge driver.
    return data

def dispatch_iot_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # This function is intended for dispatching to a system that expects
    # a v2 envelope with normalized (snake_case) keys.
    # It now correctly handles the transformation and wrapping.
    norm = normalize_dict_keys(data)
    msg = build_envelope("iot_service", epoch_ms, norm)
    # The result of extract_normalized_keys was not used, removing the unnecessary call.
    return msg
