"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    policy_segments = policy.allowed_pattern.split('::')
    request_segments = clean_req.split('::')

    i = 0 # policy segment index
    j = 0 # request segment index

    while i < len(policy_segments) and j < len(request_segments):
        policy_segment = policy_segments[i]
        request_segment = request_segments[j]

        if policy_segment == "**":
            # '**' matches arbitrary multi-segment prefixes.
            # If it's the last policy segment, it matches everything remaining in the request.
            if i == len(policy_segments) - 1:
                return True
            # If not the last segment, find the next policy segment in the request.
            # This implements a greedy match for '**'.
            next_policy_segment = policy_segments[i + 1]
            found = False
            temp_j = j
            while temp_j < len(request_segments):
                if request_segments[temp_j] == next_policy_segment:
                    # Potential match, try matching the rest of the policy from here
                    # This is a recursive-like check, but we can do it iteratively
                    # by advancing 'j' to this point and 'i' past '**' and the next segment.
                    # However, a simpler approach is to handle this specific case:
                    # If '**' is followed by a specific segment, find that segment.
                    # For simplicity and to avoid complex backtracking, let's assume
                    # '**' is mostly used at the end or in simple patterns.
                    # The current problem implies simpler patterns.
                    # Re-evaluating: The problem is a "sibling path boundary" issue,
                    # not a complex wildcard matching issue with ** in the middle.
                    # For a simple greedy match:
                    # If '**' is found, it means it can consume 0 or more segments.
                    # The most straightforward way to handle this without full regex engine
                    # is to check if the remaining policy segments can match the remaining
                    # request segments.
                    # Example: a::**::c, request: a::b::d::c
                    # If current policy segment is '**', we skip it for now and try to match
                    # the next policy segment (policy_segments[i+1]) against any of the
                    # remaining request segments.
                    # This requires a nested loop or recursion.
                    # A simpler interpretation for typical ACLs is that '**' is often at the end.
                    # If it's not at the end, it implies a more complex "glob" style.
                    # Given the current problem context (SEC-810, sibling path),
                    # the focus is on "corp::reports" not matching "corp::reports_secret".
                    # Let's handle '**' by consuming up to the point where the next policy
                    # segment *could* match, or failing if it never matches.
                    # This is often done by advancing `i` and `j` and trying to find a match for the next segment.
                    # A more robust solution for `**` in the middle would involve a loop to find the next literal segment.
                    # For now, let's assume `**` primarily appears at the end, or its behavior is simpler.
                    # If the next policy segment is found later in the request, proceed from there.
                    # This is an oversimplification for `**` in the middle but typical for basic ACLs.
                    
                    # Find the next segment after '**' in the policy
                    next_policy_segment_index = i + 1
                    while next_policy_segment_index < len(policy_segments) and policy_segments[next_policy_segment_index] == '**':
                        next_policy_segment_index += 1 # Skip consecutive '**'

                    if next_policy_segment_index == len(policy_segments):
                        return True # '**' followed by nothing else, matches rest of request

                    target_segment = policy_segments[next_policy_segment_index]
                    
                    # Try to find target_segment in the remaining request segments
                    found_match_for_next = False
                    while j < len(request_segments):
                        if request_segments[j] == target_segment:
                            found_match_for_next = True
                            break
                        j += 1
                    
                    if not found_match_for_next:
                        return False # Couldn't find the segment after '**'
                    
                    # If found, advance policy pointer past '**' and the matched segment
                    i = next_policy_segment_index # Move 'i' to the segment after '**' and the target
                    continue # Continue outer loop to compare the matched target segment
                    
                # If policy_segment is '*'
        elif policy_segment == "*":
            # '*' matches exactly one segment, so just consume both.
            i += 1
            j += 1
        # If policy_segment is a literal string
        elif policy_segment == request_segment:
            # Segments match, continue to the next.
            i += 1
            j += 1
        else:
            # Mismatch between literal segments.
            return False

    # After loop, check if all policy segments were consumed and matched.
    # If there are remaining policy segments, they must be '**'.
    while i < len(policy_segments):
        if policy_segments[i] == "**":
            i += 1
        else:
            return False # Policy has unmatched non-'**' segments
    
    # If all policy segments were consumed, then it's a match only if:
    # 1. All request segments were also consumed (exact match or '*' consumed exactly one)
    # 2. The last policy segment was '**' (which consumes remaining request segments)
    return j == len(request_segments)

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
