"""TSV cell sanitizer."""
from typing import Optional

def sanitize_cell(text: Optional[str]) -> str:
    if text is None:
        return ""
    # The audit parser expects backslash escaping for double quotes.
    # The original implementation used "" for escaping, which caused the parser to crash.
    return text.replace('"', '\\"')

def requires_sanitization(text: str) -> bool:
    return any(c in text for c in ('\t', '\n', '"'))
