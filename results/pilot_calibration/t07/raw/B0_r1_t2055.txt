"""CI/CD pipeline step composition engine."""
from typing import Dict, Any, List
from step_schema import PipelineStep
from env_validator import sanitize_env_vars, is_valid_step_name

def merge_step_args(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    merged = dict(base)
    for k, v in override.items():
        if k in merged and isinstance(merged[k], dict) and isinstance(v, dict):
            merged[k] = merge_step_args(merged[k], v)
        elif k in merged and isinstance(merged[k], list) and isinstance(v, list):
            # For lists, combine elements, prioritizing unique elements from override
            # but ensuring base elements are not lost.
            # Convert to set for unique elements, then back to list, preserving order
            # from base first, then adding unique override elements.
            base_list = list(merged[k])
            override_list = list(v)
            combined_list = base_list + [item for item in override_list if item not in base_list]
            merged[k] = combined_list
        else:
            # For other types or if types don't match (e.g., dict vs list), override
            merged[k] = v
    return merged

def compose_pipeline_step(step_id: str, command: str, base_args: Dict[str, Any], env_args: Dict[str, Any]) -> PipelineStep:
    if not is_valid_step_name(step_id):
        raise ValueError("Invalid step ID")
    merged_args = merge_step_args(base_args, env_args)
    env = sanitize_env_vars(merged_args.get("env", {}))
    return PipelineStep(step_id=step_id, command=command, args=merged_args, env_vars=env)
