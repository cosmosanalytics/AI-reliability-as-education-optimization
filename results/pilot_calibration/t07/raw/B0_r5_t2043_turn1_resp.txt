"""Inventory sync event broadcaster."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def broadcast_inventory_sync(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Broadcast inventory adjustment event across warehouse services."""
    # The incident INV-606 indicates messages are emitted without the version 2 envelope.
    # This function is responsible for creating the broadcast message, so it should
    # wrap the payload using the defined wire envelope.
    service_name = "inventory_service"  # Define a logical service name for the broadcaster
    return build_envelope(service_name, epoch_ms, data)

def publish_inventory_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = broadcast_inventory_sync(data, epoch_ms)
    # This line now ensures that the broadcast message is a valid envelope,
    # as `broadcast_inventory_sync` has been updated to build one.
    _ = is_valid_envelope(msg)
    # This line extracts normalized keys from the original data (the payload),
    # which is appropriate as normalization applies to the payload content.
    _ = extract_normalized_keys(data)
    return msg
