"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.
    Uses the nearest-rank method, often suitable for discrete distributions
    or when a specific value from the sample is preferred.
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)
    # Calculate index using the formula (p/100) * n, and round up to get 1-based index.
    # Then convert to 0-based index.
    # This aligns with the common definition of "the smallest value in the dataset
    # such that at least p percent of the data is less than or equal to this value".
    if p == 0:
        return float(s[0])
    if p == 100:
        return float(s[n-1])
    
    # Using ceil((p/100)*n) for 1-based index, then -1 for 0-based.
    # This effectively finds the value at or above the p-th percentile.
    index = math.ceil((p / 100.0) * n) - 1
    
    # Ensure index is within bounds [0, n-1]
    index = max(0, min(int(index), n - 1))
    
    return float(s[index])
