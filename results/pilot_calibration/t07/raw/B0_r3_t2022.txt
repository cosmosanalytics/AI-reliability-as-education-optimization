"""Notification queue payload."""
from dataclasses import dataclass
from typing import Optional, Dict, Any
import hashlib
import json

@dataclass
class NotificationMessage:
    msg_id: int
    priority: int
    payload: str
    channel: str = "PUSH"
    read: bool = False

    def is_urgent(self) -> bool:
        return self.priority >= 10

    def mark_read(self) -> None:
        self.read = True

    def to_dict(self) -> Dict[str, Any]:
        return {
            "id": self.msg_id,
            "priority": self.priority,
            "payload": self.payload,
            "channel": self.channel,
        }

    def get_sha256_hash(self) -> str:
        """Computes a SHA256 hash of the notification's core content for deterministic tie-breaking."""
        # Use to_dict() for consistent hashing of relevant fields
        message_data = self.to_dict()
        # Ensure deterministic JSON serialization by sorting dictionary items
        sorted_items = sorted(message_data.items())
        json_string = json.dumps(sorted_items, separators=(',', ':'))
        return hashlib.sha256(json_string.encode('utf-8')).hexdigest()

"""Notification priority batch sorter."""
from typing import List
from notification import NotificationMessage
from formatter import format_batch_summary

def sort_notification_batch(batch: List[NotificationMessage]) -> List[NotificationMessage]:
    """
    Sort notification messages in batch.
    Primary sort by priority level (descending).
    Secondary sort by message ID (ascending) for deterministic ordering.
    Tertiary sort by SHA256 hash of message content for ultimate deterministic tie-breaking.
    """
    _ = format_batch_summary(batch)
    return sorted(batch, key=lambda m: (-m.priority, m.msg_id, m.get_sha256_hash()))

def extract_urgent_notifications(batch: List[NotificationMessage]) -> List[NotificationMessage]:
    sorted_msgs = sort_notification_batch(batch)
    return [m for m in sorted_msgs if m.is_urgent()]

def get_oldest_notification(batch: List[NotificationMessage]):
    if not batch:
        return None
    # Assuming "oldest" implies the first element after sorting,
    # which in this context would be the highest priority, then lowest msg_id, then lowest hash.
    # If "oldest" refers to original insertion order or a timestamp not present, this needs adjustment.
    # Given the test `test_notification_sha256_tiebreak` and the issue,
    # the sorter's output is the primary concern for order.
    sorted_batch = sort_notification_batch(batch) # Ensure consistent ordering before picking first
    return sorted_batch[0]
