"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import remove_query_string

def _normalize_path_for_matching(path: str) -> str:
    """Normalizes a path by removing trailing slashes, unless it's the root '/'."""
    if path == "/":
        return "/"
    return path.rstrip("/")

def is_route_permitted(policy: RoutePolicy, request_path: str) -> bool:
    """Check if request_path is authorized under policy.allowed_prefix.
    Path matching contract:
    - Normalizes trailing slashes before matching.
    - Matches only full path segments (e.g. '/api' covers '/api' and '/api/v1', but NOT '/api-v2').
    """
    clean_path = remove_query_string(request_path)

    # Handle wildcard policy: matches any path
    if policy.is_wildcard(): # policy.allowed_prefix == "/"
        return True

    normalized_policy_prefix = _normalize_path_for_matching(policy.allowed_prefix)
    normalized_request_path = _normalize_path_for_matching(clean_path)

    # Exact match: e.g., policy="/api", request="/api"
    if normalized_request_path == normalized_policy_prefix:
        return True

    # Prefix match with segment boundary: e.g., policy="/api", request="/api/v1"
    # Ensure the request path starts with the policy prefix
    # AND the character immediately following the prefix in the request path is a '/'
    if normalized_request_path.startswith(normalized_policy_prefix):
        prefix_len = len(normalized_policy_prefix)
        # Check if there's a character after the prefix and if it's a '/'
        if len(normalized_request_path) > prefix_len and normalized_request_path[prefix_len] == '/':
            return True

    return False

def evaluate_request(policies: list, role: str, path: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_route_permitted(p, path):
            return True
    return False

def count_permitted_routes(policy: RoutePolicy, paths: list) -> int:
    return sum(1 for p in paths if is_route_permitted(policy, p))

=== FILE: tests/test_guard.py ===
from policy import RoutePolicy
from guard import is_route_permitted

def test_nominal_match():
    p = RoutePolicy("admin", "/api/v1")
    assert is_route_permitted(p, "/api/v1/users") is True
    assert is_route_permitted(p, "/other") is False

def test_sibling_path_boundary():
    p = RoutePolicy("analyst", "/reports")
    assert is_route_permitted(p, "/reports-secret") is False
    assert is_route_permitted(p, "/reports-confidential") is False
    assert is_route_permitted(p, "/reports/summary") is True
    assert is_route_permitted(p, "/reports") is True

def test_trailing_slashes_normalization():
    p = RoutePolicy("admin", "/api/")
    assert is_route_permitted(p, "/api/v1") is True
    assert is_route_permitted(p, "/api") is True
    assert is_route_permitted(p, "/api/") is True

    p2 = RoutePolicy("admin", "/api")
    assert is_route_permitted(p2, "/api/v1") is True
    assert is_route_permitted(p2, "/api") is True
    assert is_route_permitted(p2, "/api/") is True

def test_wildcard_policy():
    p = RoutePolicy("admin", "/")
    assert is_route_permitted(p, "/any/path") is True
    assert is_route_permitted(p, "/") is True
    assert is_route_permitted(p, "/api/v1") is True
