"""TSV cell sanitizer."""
from typing import Optional

def sanitize_cell(text: Optional[str]) -> str:
    if text is None:
        return ""
    # The audit parser expects double quotes to be escaped with a backslash.
    return text.replace('"', '\\"')

def requires_sanitization(text: str) -> bool:
    # This function is not directly used for the current sanitization logic
    # but could be used for other types of sanitization or validation.
    # It checks for characters that might typically require escaping in TSV.
    return any(c in text for c in ('\t', '\n', '"'))
