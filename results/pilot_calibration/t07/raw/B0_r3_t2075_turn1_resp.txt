"""Disk IO wait time percentile calculation."""
import math
from typing import List, Dict, Any
from io_types import IoBatch, IoWaitSample
from io_summary import format_io_summary

def compute_io_percentile(batch: IoBatch, p: float) -> float:
    """Computes storage wait percentile using the nearest-rank method with ceiling for index calculation.
    This method ensures that for p=100, the maximum value is returned, and generally
    provides a more intuitive upper bound for high percentiles in small datasets.
    """
    times = batch.times()
    if not times:
        raise ValueError("Empty IO batch")
    s = sorted(times)
    n = len(s)

    # Handle edge cases for p=0 and p=100 explicitly for clarity, though the formula handles them.
    if p == 0:
        return float(s[0])
    if p == 100:
        return float(s[n - 1])

    # Calculate the rank (index) using the formula L = p/100 * n.
    # We use math.ceil to round up to the nearest integer rank.
    # The index for a list is 0-based, so we subtract 1 from the 1-based rank.
    # This aligns with common percentile definitions that round up to the next rank.
    # For example, with [100, 200, 300, 400, 500] (n=5):
    # - 90th percentile: (0.9 * 5) = 4.5. ceil(4.5) = 5. Index = 5 - 1 = 4. s[4] = 500.0
    # - 75th percentile: (0.75 * 5) = 3.75. ceil(3.75) = 4. Index = 4 - 1 = 3. s[3] = 400.0
    # - 50th percentile: (0.5 * 5) = 2.5. ceil(2.5) = 3. Index = 3 - 1 = 2. s[2] = 300.0
    # This approach is consistent with Excel's PERCENTILE.INC and R's type 6 method.
    rank = math.ceil((p / 100.0) * n)
    idx = int(rank) - 1
    
    # Ensure the index is within valid bounds [0, n-1]
    # This is implicitly handled by the rank calculation if p is between 0 and 100
    # and n is at least 1, but added as a safeguard.
    idx = max(0, min(idx, n - 1))

    return float(s[idx])

def summarize_io_batch(batch: IoBatch, p: float) -> Dict[str, Any]:
    val = compute_io_percentile(batch, p)
    return format_io_summary(batch.count(), val)
