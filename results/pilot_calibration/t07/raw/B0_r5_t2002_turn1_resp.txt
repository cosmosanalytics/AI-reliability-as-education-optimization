"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    Ensures that the total allocated discount matches total_discount_cents
    by distributing any remainder pennies to lines with the largest
    fractional parts.

    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")

    total_invoice_value = sum(line_totals)
    num_lines = len(line_totals)

    if total_invoice_value == 0:
        return [0] * num_lines

    if num_lines == 0:
        return []

    # Calculate initial allocations using floating point for precision
    # Store fractional parts to distribute remainder
    initial_allocations = []
    for cents in line_totals:
        exact_allocation = (cents * total_discount_cents) / total_invoice_value
        initial_allocations.append(exact_allocation)

    # Floor each allocation to get the base integer amount
    final_allocations = [int(alloc) for alloc in initial_allocations]

    # Calculate the total allocated so far and the remainder
    current_total_allocated = sum(final_allocations)
    remainder = total_discount_cents - current_total_allocated

    # Distribute the remainder pennies
    if remainder != 0:
        # Create a list of (fractional_part, original_index) tuples
        # Sort in descending order of fractional part
        # If fractional parts are equal, original index determines tie-breaking
        fractions_with_indices = []
        for i, exact_alloc in enumerate(initial_allocations):
            fractional_part = exact_alloc - final_allocations[i]
            fractions_with_indices.append((fractional_part, i))

        # Sort by fractional_part in descending order, then by original index
        # (The original index tie-breaking is consistent but not strictly necessary for correctness,
        # but ensures deterministic behavior for identical fractional parts)
        fractions_with_indices.sort(key=lambda x: (-x[0], x[1]))

        # Distribute remainder pennies to the lines with the largest fractional parts
        for i in range(remainder):
            # The remainder could be negative if total_discount_cents is less than the sum of floored parts
            # which can happen if total_discount_cents is negative, but we validate non-negative.
            # However, if total_discount_cents is very small positive and total_invoice_value is large,
            # all initial_allocations could be < 1, resulting in all 0s, and remainder == total_discount_cents.
            # If total_discount_cents is negative, this logic needs adjustment, but currently it's for discounts.
            # The problem description implies total_discount_cents >= 0.
            if i < len(fractions_with_indices): # Ensure we don't try to add to non-existent lines
                original_index = fractions_with_indices[i][1]
                final_allocations[original_index] += 1
            else:
                # This case should ideally not be reached if remainder <= num_lines and total_discount_cents is sane.
                # If remainder is larger than num_lines, it means all lines have received +1 and there are still pennies.
                # In such a case, we would continue distributing round-robin.
                # For typical "penny" mismatches (remainder < num_lines), the current logic is sufficient.
                # If remainder > num_lines, this implies a larger discrepancy not just "pennies".
                # For this specific problem context, we assume remainder will be <= num_lines.
                break # Stop if we run out of lines to add to

    return final_allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
