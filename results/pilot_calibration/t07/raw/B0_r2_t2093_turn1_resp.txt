"""Telemetry Metric Ingest Formatter utility."""
from datetime import datetime, timezone
from iso_helper import parse_epoch_seconds, is_valid_iso_format, extract_year_component, format_epoch_millis
from timestamp_model import TimestampRecord, create_event_record

def format_metric_timestamp(epoch_s: float) -> str:
    # Convert epoch seconds (float) to epoch milliseconds (int)
    epoch_ms_int = int(epoch_s * 1000)
    # Create the event record with millisecond precision
    rec = create_event_record("format_metric_timestamp", epoch_ms_int)
    # Use the helper function to format the timestamp with millisecond precision
    return format_epoch_millis(epoch_ms_int)

def format_sequence(timestamps: list) -> list:
    return [format_metric_timestamp(t) for t in timestamps]
