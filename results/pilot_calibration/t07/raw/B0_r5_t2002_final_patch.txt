"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value,
    distributing any remaining cents to ensure the total discount is fully allocated.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total = sum(line_totals)
    num_lines = len(line_totals)

    if total == 0:
        return [0] * num_lines

    # Calculate base allocations using float division to retain precision
    # and then truncate to integer cents.
    # Store (fractional_part, original_index) to distribute remainder.
    allocations_with_fractions = []
    for i, cents in enumerate(line_totals):
        exact_allocation = (cents * total_discount_cents) / total
        base_allocation = int(exact_allocation)
        fractional_part = exact_allocation - base_allocation
        allocations_with_fractions.append((fractional_part, i, base_allocation))

    # Sort by fractional part in descending order.
    # If fractional parts are equal, sort by original index in ascending order
    # to ensure deterministic behavior.
    allocations_with_fractions.sort(key=lambda x: (x[0], -x[1]), reverse=True)

    # Initialize final allocations with base allocations
    final_allocations = [0] * num_lines
    for _, original_index, base_allocation in allocations_with_fractions:
        final_allocations[original_index] = base_allocation

    # Calculate the remainder that needs to be distributed
    current_allocated_sum = sum(final_allocations)
    remainder = total_discount_cents - current_allocated_sum

    # Distribute the remainder, one cent at a time, to the lines
    # with the largest fractional parts (or earliest by index if fractions are equal).
    for i in range(remainder):
        # We use the sorted list to pick which line gets an extra cent.
        # The element at index `i` in the sorted list corresponds to the (i+1)th
        # line to receive an extra cent.
        # The `original_index` is used to update the `final_allocations` list correctly.
        original_index_to_update = allocations_with_fractions[i][1]
        final_allocations[original_index_to_update] += 1

    return final_allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))

=== FILE: tests/test_discount.py ===
from discount import prorate_discount, allocate_discounts

def test_even_split():
    assert prorate_discount([100, 100], 20) == [10, 10]
    res = allocate_discounts([200, 100], 30)
    assert res.discounts == [20, 10]
    assert res.total_allocated == 30

def test_zero_total():
    assert prorate_discount([0, 0], 50) == [0, 0]

def test_proportional_distribution():
    res = prorate_discount([100, 300], 100)
    assert res == [25, 75]

def test_fractional_split():
    # Example where 1 cent needs to be distributed
    assert prorate_discount([35, 35], 35) == [18, 17]
    assert prorate_discount([35, 35], 36) == [18, 18] # 18+18 = 36. 36/70 * 35 = 18.
    assert prorate_discount([33, 33, 34], 100) == [33, 33, 34] # 100/100 * 33 = 33, 33, 34

    # Test with more complex fractions
    assert prorate_discount([10, 20, 30], 10) == [2, 3, 5] # 10/60 * 10 = 1.66, 20/60 * 10 = 3.33, 30/60 * 10 = 5
                                                           # Base: [1, 3, 5]. Remainder: 10 - 9 = 1.
                                                           # Fractions: (0.66, 0), (0.33, 1), (0.0, 2)
                                                           # Sorted: (0.66,0), (0.33,1), (0.0,2)
                                                           # Add 1 to index 0: [2, 3, 5]

    assert prorate_discount([10, 10, 10], 1) == [1, 0, 0]
    assert prorate_discount([10, 10, 10], 2) == [1, 1, 0]
    assert prorate_discount([10, 10, 10], 3) == [1, 1, 1]

def test_allocation_result_fully_allocated():
    res = allocate_discounts([35, 35], 35)
    assert res.discounts == [18, 17]
    assert res.total_allocated == 35
    assert res.is_fully_allocated is True

    res = allocate_discounts([10, 20, 30], 10)
    assert res.discounts == [2, 3, 5]
    assert res.total_allocated == 10
    assert res.is_fully_allocated is True

def test_no_lines():
    # If there are no lines, discount cannot be allocated.
    assert prorate_discount([], 50) == []
    res = allocate_discounts([], 50)
    assert res.discounts == []
    assert res.total_allocated == 0
    assert res.is_fully_allocated is False # No lines means 0 allocated, but 50 was total_discount_cents

def test_negative_discount():
    # A negative discount should still be prorated, though it implies a charge.
    # The current implementation handles it mathematically; business logic might require checks.
    assert prorate_discount([100, 100], -20) == [-10, -10]
    res = allocate_discounts([200, 100], -30)
    assert res.discounts == [-20, -10]
    assert res.total_allocated == -30
    assert res.is_fully_allocated is True
