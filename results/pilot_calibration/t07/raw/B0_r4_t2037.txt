"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# A default service name is assumed here for simplicity, as the `service_name`
# parameter was missing from `format_audit_event` and `serialize_audit_event`
# but required by `build_envelope` and the docstring.
# In a real-world scenario, `service_name` would likely be passed as an argument
# or retrieved from a configuration/context.
DEFAULT_SERVICE_NAME = "unknown_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int, service_name: str = DEFAULT_SERVICE_NAME) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int, service_name: str = DEFAULT_SERVICE_NAME) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms, service_name)
    # The original `_ = is_valid_envelope(msg)` and `_ = extract_normalized_keys(data)`
    # lines were likely meant for validation or debugging.
    # `is_valid_envelope(msg)` now correctly returns True.
    # `extract_normalized_keys(data)` operates on the original data, not the normalized payload
    # within the envelope, and its return value was discarded.
    # It is kept for minimal change, but its utility here is questionable.
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data) # Operates on original 'data', not the normalized payload in 'msg'
    return msg
