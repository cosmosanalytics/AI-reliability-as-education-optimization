"""Cache Hit Latency Percentile Monitor percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_cache_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_cache_percentile", samples)
    s = validate_metric_samples(win.values)
    n = len(s)
    
    # Calculate rank using the nearest-rank method (N*P/100)
    # This is often referred to as method R-1 or the nearest-rank method where
    # the index is calculated as ceil(N * P / 100).
    # For interpolation, it's common to use N*P/100 directly and handle
    # fractional ranks by interpolating between adjacent values.
    # The problem description hints at a discrepancy against a "nearest-rank house standard".
    # A common interpretation for the p-th percentile is to find the value at rank (N * p / 100).
    # If N * p / 100 is an integer, take that value. If not, interpolate.
    # The previous implementation used (N-1)*P/100. The standard definition for
    # interpolation often uses N*P/100 for continuous data (e.g., SciPy's linear method).

    # The issue implies that for a 50th percentile, the expected value is 40.0,
    # while the current code returns 45.0. This suggests that the index calculation
    # needs to shift.
    # If samples are [10, 20, 30, 40, 50, 60, 70, 80], N=8, P=50.
    # Current: idx = 0.5 * (8-1) = 3.5. low=3, high=4. s[3]=40, s[4]=50.
    # Result: 40 + (3.5 - 3) * (50-40) = 40 + 0.5 * 10 = 45.0 (This matches the failing test output)
    # Expected: 40.0
    # To get 40.0, the index should resolve to s[3]. This means the index should be 3.0 or less
    # such that s[3] is chosen or interpolated to.
    
    # Let's adjust the index calculation to align with a common "nearest-rank" standard
    # that might produce the expected result.
    # A common method used in some systems (e.g., Excel's PERCENTILE.EXC) uses (N+1)*P/100.
    # Another common method for continuous data (e.g., NumPy's 'linear' interpolation)
    # uses (N-1)*P/100. The problem explicitly states "nearest-rank house standard".
    # If the house standard for 50th percentile of [10,20,30,40,50,60,70,80] is 40.0,
    # it implies that for an even number of samples, the lower of the two middle values
    # is preferred or interpolation is biased.

    # Re-evaluating the index calculation based on common percentile definitions.
    # The definition of percentile can vary.
    # Method R-6 (most common in statistical software): idx = p/100 * (N+1).
    # Method R-7 (used by NumPy's 'linear'): idx = p/100 * (N-1). This is the current implementation.

    # If the expectation is 40.0 for [10, 20, 30, 40, 50, 60, 70, 80] at 50th percentile,
    # this suggests that for an even number of elements, the lower median is chosen,
    # or the interpolation biases towards the lower index.
    # Let's try to adjust the index calculation to achieve 40.0.
    # If idx = p/100 * N:
    #   idx = 0.5 * 8 = 4.0. low=4, high=4. Returns s[4] = 50.0. Still not 40.0.

    # The problem states "nearest-rank house standard".
    # If the standard implies that for P=50% and N=8, the percentile is the (N*P/100)-th value
    # or the value at index floor(N*P/100) if N*P/100 is not integer, or average of values
    # at N*P/100 and N*P/100+1.

    # Let's consider a common 'nearest rank' definition for the *rank* (position) first.
    # Rank = ceil(P/100 * N). The value is then s[Rank-1].
    # For N=8, P=50: Rank = ceil(0.5 * 8) = ceil(4) = 4. Value = s[4-1] = s[3] = 40.0.
    # This matches the expected output. So, the "nearest-rank house standard" likely implies
    # calculating the rank this way and then using the value at that rank.
    # This is a discrete percentile calculation, not interpolated.

    # If interpolation is still required for non-integer ranks for other percentiles,
    # we need to be careful. The traceback shows an exact match failure, implying
    # a specific single value is expected, not an interpolated one.

    # Let's implement the nearest-rank logic.
    # The P-th percentile is the element at rank k, where k = ceil(P * N / 100).
    # If P * N / 100 is an integer, k = P * N / 100.
    # The index in a 0-based array would be k-1.

    if not s: # Handle empty list, though validate_metric_samples should catch it.
        return float('nan') # Or raise an error

    if p < 0 or p > 100:
        raise ValueError("Percentile must be between 0 and 100 inclusive")

    # Handle edge cases for p=0 and p=100 explicitly if they differ from general formula
    if p == 0:
        return s[0]
    if p == 100:
        return s[n - 1]

    # Calculate the rank (1-based index)
    rank_float = (p / 100.0) * n
    
    # For "nearest-rank house standard", a common interpretation is to find the value
    # at the calculated rank, potentially rounding the rank.
    # If rank_float is an integer, the value is s[rank_float - 1].
    # If rank_float is not an integer, the standard might specify ceiling or floor.
    # Given the test case (P=50, N=8, expected 40.0), it suggests using floor or
    # a method that rounds down for ranks exactly halfway between two integers
    # or simply picks the lower index for even N at 50th percentile.

    # The specific failure (45.0 vs 40.0) for 50th percentile on 8 elements
    # strongly suggests the 'nearest-rank' definition where the rank is `ceil(P*N/100)`
    # and the value is `s[rank-1]`. This is a non-interpolated percentile.

    # Let's try this definition:
    # k = math.ceil((p / 100.0) * n)
    # if k == 0: # For p very close to 0, ceil might be 0. Ensure at least first element.
    #    k = 1
    # return s[int(k) - 1]

    # Let's test this with the presumed data [10, 20, 30, 40, 50, 60, 70, 80]
    # p = 50.0, n = 8
    # k = math.ceil((50/100.0) * 8) = math.ceil(4.0) = 4.
    # return s[4 - 1] = s[3] = 40.0. This matches the expected value.

    # Let's test with the existing test_perc:
    # calculate_cache_percentile([10.0, 20.0, 30.0], 100.0)
    # p = 100.0, n = 3
    # k = math.ceil((100/100.0) * 3) = math.ceil(3.0) = 3.
    # return s[3 - 1] = s[2] = 30.0. This also matches the existing test.

    # This 'nearest-rank' definition seems to fit both requirements.
    # However, this is a *non-interpolated* percentile. The original function was interpolating.
    # The term "nearest-rank house standard" could imply either a non-interpolated method
    # or a specific interpolated method. Given the exact match failure, a non-interpolated
    # method is a strong candidate. If the house standard requires interpolation, it would
    # typically specify the method (e.g., linear interpolation on N, or N+1, or N-1).

    # If the standard implies *interpolation* but with a different `idx` calculation:
    # Some standards use `idx = (p / 100.0) * N`.
    # For N=8, P=50: idx = 0.5 * 8 = 4.0.
    # low = 4, high = 4. Returns s[4] = 50.0. Still not 40.0.

    # Let's consider the possibility that the "nearest-rank" refers to the *rank* itself,
    # and not necessarily to a specific interpolation method.
    # The problem description is concise. Given "Incident INC-2106: Metric calculation discrepancy
    # against nearest-rank house standard" and the specific `assert 45.0 == 40.0`,
    # the simplest change that resolves this is to interpret "nearest-rank" as the
    # definition `s[ceil(P*N/100) - 1]`. This is a common, non-interpolated percentile.
    # If interpolation is strictly required, the house standard should specify the exact formula.
    # Without further information, changing to the non-interpolated nearest-rank seems
    # to be the most direct interpretation of "nearest-rank house standard" resolving the discrepancy.

    # Final decision: Implement the nearest-rank (non-interpolated) method.
    # The original implementation was method R-7 (linear interpolation with (N-1) factor).
    # The failing test requires 40.0 for 50th percentile of [10,..,80].
    # Method R-1 (nearest rank) gives 40.0 for this case.

    # Calculate the 1-based rank (position)
    rank = (p / 100.0) * n

    # The "nearest-rank house standard" typically refers to the smallest observation
    # that is greater than or equal to the p-th percentile.
    # This is often implemented as the value at index ceil(rank) - 1.
    
    # Ensure rank is at least 1, and at most n.
    # The rank needs to be 1-based.
    # The ceil function will handle ranks like 4.0 (to 4) and 3.5 (to 4).
    # If p=0, rank=0. We want s[0].
    # If p=100, rank=n. We want s[n-1].
    
    # For p=0, rank = 0. ceil(0)=0. (0-1) is -1. This needs special handling.
    # For p=100, rank = n. ceil(n)=n. (n-1) is correct.
    # So, we can handle p=0 and p=100 as special cases, or adjust k.
    
    # Let's adjust rank calculation to be 0-based index directly.
    # The 0-based index `i` is `ceil(p * n / 100) - 1`.
    # If `p * n / 100` is 0, `ceil(0) - 1 = -1`.
    # If `p * n / 100` is `x` where `0 < x <= 1`, `ceil(x) - 1 = 0`.
    # This works for `p > 0`.
    
    # Let's use the 0-based index calculation directly.
    # index = math.ceil((p / 100.0) * n) - 1
    # We need to clamp this index to be within [0, n-1].
    # For p=0, index would be -1. We want 0.
    # For p=100, index would be n-1. This is correct.

    # Clamp the index:
    # index = max(0, min(n - 1, int(math.ceil((p / 100.0) * n)) - 1))
    
    # Test this with p=0: max(0, min(n-1, int(math.ceil(0)) - 1)) = max(0, min(n-1, -1)) = max(0, -1) = 0. Correct.
    # Test with p=100: max(0, min(n-1, int(math.ceil(n)) - 1)) = max(0, min(n-1, n-1)) = n-1. Correct.
    # Test with [10, 20, 30, 40, 50, 60, 70, 80], p=50, n=8:
    # index = max(0, min(7, int(math.ceil(0.5 * 8)) - 1)) = max(0, min(7, int(math.ceil(4.0)) - 1))
    #       = max(0, min(7, 4 - 1)) = max(0, min(7, 3)) = max(0, 3) = 3.
    # s[3] = 40.0. This is the expected value.

    # This seems to be the correct interpretation for "nearest-rank house standard"
    # that resolves the specific discrepancy.

    # Guard against empty samples, though validate_metric_samples should handle this.
    if not s:
        raise ValueError("Samples list cannot be empty for percentile calculation")

    # Calculate the 1-based rank (position)
    rank_float = (p / 100.0) * n

    # Convert to 0-based index, applying ceiling and clamping.
    # The -1 is because rank is 1-based, and list indices are 0-based.
    # The max(0, ...) handles p=0 case (rank_float=0, ceil(0)=0, 0-1=-1 -> becomes 0).
    # The min(n-1, ...) handles cases where p is very close to 100 but not exactly,
    # or floating point inaccuracies, ensuring index doesn't exceed n-1.
    index = int(math.ceil(rank_float)) - 1
    index = max(0, min(n - 1, index))

    return s[index]


def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
