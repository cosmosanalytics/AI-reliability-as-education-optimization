"""Worker Job Execution Duration Analyzer percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def compute_duration_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("compute_duration_percentile", samples)
    s = validate_metric_samples(win.values)
    n = len(s)

    # The problem description and failing test (`assert 45.0 == 40.0` for 50th percentile)
    # indicate that the percentile calculation needs to adhere to a standard
    # where the 50th percentile of an even-sized list like [10, 20, 30, 40, 50, 60, 70, 80]
    # (which currently yields 45.0) should be 40.0.
    # This implies using a method that picks the lower of the two middle values
    # for even-sized lists, or a different interpolation method.
    #
    # A common "nearest-rank" method that aligns with the expectation of 40.0 for
    # the 50th percentile in the example [10, 20, 30, 40, 50, 60, 70, 80] is
    # to use the formula where the index is calculated as p/100 * (n - 1) and
    # then rounding *up* to the nearest integer index (1-based) and selecting
    # that value, or for interpolation, using a specific formula.
    #
    # The current interpolation method `idx = (p / 100.0) * (len(s) - 1)`
    # is often referred to as "method R-7" or linear interpolation between nearest ranks.
    # If the standard implies that for an even set like [10,20,30,40,50,60,70,80],
    # the 50th percentile is 40.0, it suggests a non-interpolated approach for
    # integer ranks, or a different interpolation that picks the lower bound.
    #
    # Let's consider the general formula for a percentile, where the rank `r` is:
    # r = p/100 * (n - 1) (0-indexed)
    # If r is an integer, the percentile is s[r].
    # If r is not an integer, let `i = floor(r)` and `f = r - i`.
    # The percentile is s[i] + f * (s[i+1] - s[i]). This is the current implementation.
    #
    # To get 40.0 instead of 45.0 for the 50th percentile of
    # [10, 20, 30, 40, 50, 60, 70, 80] (n=8):
    # p = 50.0
    # Current: idx = 0.5 * (8 - 1) = 3.5. low=3, high=4. Result: s[3] + 0.5*(s[4]-s[3]) = 40 + 0.5*(50-40) = 45.0
    #
    # If the expectation is 40.0, it means for the 50th percentile (median),
    # in an even-sized list, the lower of the two middle values is chosen,
    # or a different percentile definition is used.
    #
    # A common alternative for discrete nearest-rank is:
    # index = ceil(p/100 * n) - 1 (for 0-indexed list)
    #
    # Let's test this alternative:
    # For [10, 20, 30, 40, 50, 60, 70, 80] (n=8), p=50:
    # index = ceil(0.5 * 8) - 1 = ceil(4.0) - 1 = 4 - 1 = 3.
    # s[3] = 40.0. This matches the expectation.
    #
    # For [10, 20, 30] (n=3), p=100:
    # Current: idx = 1.0 * (3-1) = 2.0. low=2, high=2. Result: s[2] = 30.0. (Correct)
    # Alternative: index = ceil(1.0 * 3) - 1 = ceil(3.0) - 1 = 3 - 1 = 2.
    # s[2] = 30.0. (Correct)
    #
    # This "nearest-rank" method (often called R-1 or closest-rank in some contexts,
    # or simply finding the element at a specific ceiling-based rank)
    # seems to fit the "nearest-rank house standard" and the expected `40.0` value.
    # This method does not involve interpolation between values.

    if not s: # Handle empty list after validation (though validate_metric_samples already checks)
        raise ValueError("Cannot compute percentile for an empty list of samples.")

    # Calculate the 1-based rank (position) for the percentile
    # This formula (N*P/100) is common for nearest-rank methods that don't interpolate
    # and directly pick a value from the sorted list.
    # Using math.ceil to find the rank, then adjusting for 0-based indexing.
    # Example: 50th percentile of 8 elements. (8 * 50 / 100) = 4. ceil(4) = 4.
    # 0-indexed: 4-1 = 3. s[3] is the 4th element.
    # For [10, 20, 30, 40, 50, 60, 70, 80], s[3] = 40.0.
    # This aligns with the expected value.

    # Ensure p is within valid range [0, 100]
    if not (0.0 <= p <= 100.0):
        raise ValueError("Percentile 'p' must be between 0 and 100.")

    if p == 0.0:
        return s[0] # 0th percentile is the minimum value
    if p == 100.0:
        return s[n - 1] # 100th percentile is the maximum value

    # Calculate the 1-based rank
    rank = (p / 100.0) * n

    # For "nearest-rank" standard, we typically pick an element directly.
    # The problem description and test suggest a specific interpretation.
    # If rank is an integer, we might pick that element. If not, we round.
    # The `ceil(rank) - 1` approach for 0-indexed list directly selects an element.
    #
    # Example: n=8, p=50 -> rank = 4.0. ceil(4.0)-1 = 3. s[3] = 40.0.
    # Example: n=3, p=33 -> rank = 0.99. ceil(0.99)-1 = 0. s[0] = 10.0.
    # Example: n=3, p=50 -> rank = 1.5. ceil(1.5)-1 = 1. s[1] = 20.0.
    # Example: n=3, p=66 -> rank = 1.98. ceil(1.98)-1 = 1. s[1] = 20.0.
    # Example: n=3, p=67 -> rank = 2.01. ceil(2.01)-1 = 2. s[2] = 30.0.

    # This index calculation matches the behavior implied by the failing test.
    # It corresponds to the "nearest rank" method (e.g., NumPy's 'lower' or 'higher'
    # depending on rounding, or R type 1).
    # Specifically, `math.ceil(p * n / 100.0) - 1` gives the 0-indexed position
    # of the element at or above the percentile rank.
    index = int(math.ceil(rank)) - 1

    # Ensure index is within bounds (should be handled by p=0 and p=100 cases,
    # but good to be explicit for intermediate values).
    # The smallest possible index is 0 (for p > 0 and rank > 0).
    # The largest possible index is n-1 (for p=100, rank=n).
    return s[max(0, min(index, n - 1))]


def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
