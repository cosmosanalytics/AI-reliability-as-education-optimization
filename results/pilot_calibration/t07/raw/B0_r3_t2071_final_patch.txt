"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses.

    This implementation uses the "nearest rank" method, which is robust for
    small datasets and common in simple percentile calculations.
    The index is calculated as p/100 * (n - 1) for 0-indexed arrays,
    then rounded to the nearest integer.
    """
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)
    # The percentile index calculation should use math.ceil for consistency
    # with common definitions that ensure the p-th percentile is at least
    # the value at the calculated rank, or to align with specific requirements
    # for upward rounding of indices for SLA purposes.
    # Using math.ceil for the index calculation handles cases where the
    # percentile falls between two values by selecting the higher rank,
    # which is often desired for SLAs (e.g., 99th percentile means 99%
    # of requests are *faster* than this value).
    # For example, with [10, 20, 30] and 50th percentile:
    # (50/100) * (3-1) = 1.0. ceil(1.0) = 1. s[1] = 20.0
    # With [10, 20, 30] and 90th percentile:
    # (90/100) * (3-1) = 1.8. ceil(1.8) = 2. s[2] = 30.0
    # This aligns with common interpretations for performance metrics where
    # we want to be conservative.
    idx = int(math.ceil((p / 100.0) * (n - 1)))
    # Ensure index is within bounds [0, n-1]
    idx = max(0, min(idx, n - 1))
    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
