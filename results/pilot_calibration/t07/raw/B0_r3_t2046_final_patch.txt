"""Timestamp serialization codec for telemetry records."""
from datetime import datetime, timezone

def encode_iso_timestamp(dt: datetime) -> str:
    """Encode timestamp into standard ISO 8601 format (YYYY-MM-DDTHH:MM:SS.sssZ)."""
    # Ensure the datetime object is in UTC before formatting
    dt_utc = dt.astimezone(timezone.utc)
    # Format to ISO 8601 with milliseconds
    iso_format_str = dt_utc.isoformat(timespec='milliseconds')
    # Replace the timezone offset (+00:00 or -00:00) with 'Z' for UTC
    return iso_format_str.replace('+00:00', 'Z').replace('-00:00', 'Z')

def validate_timestamp_range(dt: datetime) -> bool:
    return dt.year >= 2020
