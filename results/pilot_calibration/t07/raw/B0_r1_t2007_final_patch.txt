"""Low-level step configuration compiler."""
from typing import Dict, Any
import copy

def compile_step_config(name: str, base_opts: Dict[str, Any], overrides: Dict[str, Any]) -> Dict[str, Any]:
    # Perform a deep copy of base_opts to ensure isolation from global defaults
    # and to prevent mutation of nested dictionaries in the original base_opts.
    res = copy.deepcopy(base_opts)
    res["name"] = name

    # Recursively merge overrides into the deeply copied base options.
    # This ensures that nested dictionaries from overrides are merged with,
    # rather than completely replacing, the corresponding nested dictionaries
    # from base_opts, while still maintaining isolation.
    for key, value in overrides.items():
        if key in res and isinstance(res[key], dict) and isinstance(value, dict):
            res[key].update(value)
        else:
            res[key] = value
    return res

def format_step_manifest(spec: Dict[str, Any]) -> str:
    name = spec.get("name", "unnamed")
    engine = spec.get("engine", "default")
    return f"StageManifest(name={name}, engine={engine})"
