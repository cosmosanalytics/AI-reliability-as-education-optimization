"""IoT sensor ingest stream demux."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def process_iot_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Process IoT device packet.
    
    For raw IoT ingest pipeline, MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope.
    """
    # The requirement is to emit the raw, unwrapped payload with original camelCase keys.
    # Therefore, the input 'data' dictionary should be returned directly without modification or wrapping.
    return data

def dispatch_iot_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # The current implementation of process_iot_packet now returns the raw data.
    # If the intent for dispatch_iot_record is to *always* produce a v2 envelope
    # with normalized keys, then it needs to perform those operations explicitly.
    # Given the incident, it seems process_iot_packet was the source of the issue
    # for a specific downstream driver that expects raw data.
    # For dispatch_iot_record, we will assume it is meant for an internal pipeline
    # that *does* expect the v2 envelope with normalized keys.
    norm = normalize_dict_keys(data)
    msg = build_envelope("iot_service", epoch_ms, norm)
    _ = extract_normalized_keys(data) # This line's result is currently ignored.
    return msg
