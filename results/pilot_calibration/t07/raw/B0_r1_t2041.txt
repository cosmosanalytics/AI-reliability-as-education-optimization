"""User activity notification dispatcher."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def dispatch_user_activity(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Dispatch user activity record to notification bus."""
    # Wrap the user activity data in a wire envelope.
    # The service name is identified as "activity_dispatcher" for this context.
    service_name = "activity_dispatcher"
    return build_envelope(service_name, epoch_ms, data)

def broadcast_activity_update(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = dispatch_user_activity(data, epoch_ms)
    # Ensure the dispatched message is a valid wire envelope before further processing
    # (e.g., sending it to a bus where it will be ingested and validated).
    # The return value of is_valid_envelope is not used directly here,
    # but the call serves as an implicit validation check for the output of dispatch_user_activity.
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data) # This line processes the original data, not the envelope.
    return msg
