"""Legacy SOAP bridge gateway."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def bridge_soap_message(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Bridge legacy SOAP message to external endpoint.
    
    For legacy SOAP bridge, emit raw un-enveloped payload, do NOT wrap in version 2 envelope.
    """
    norm = normalize_dict_keys(data)
    # The docstring explicitly states "do NOT wrap in version 2 envelope".
    # The incident SOAP-909 indicates that an unexpected v2 envelope caused rejection.
    # Therefore, this function should return the normalized payload directly.
    return norm

def relay_soap_envelope(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # This function's name implies it should relay an envelope.
    # It should normalize the keys and then build a v2 envelope around it.
    norm = normalize_dict_keys(data)
    _ = extract_normalized_keys(data) # This line was originally there; keep it if it serves a purpose (e.g., logging) or remove it if not. Assuming it's benign.
    return build_envelope("soap_bridge", epoch_ms, norm)
