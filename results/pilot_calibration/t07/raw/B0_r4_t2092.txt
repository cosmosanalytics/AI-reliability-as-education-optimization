"""Audit Trail Event Time Normalizer utility."""
from datetime import datetime, timezone
from iso_helper import parse_epoch_seconds, is_valid_iso_format, extract_year_component, format_epoch_millis
from timestamp_model import TimestampRecord, create_event_record

def normalize_audit_timestamp(epoch_s: float) -> str:
    # Convert epoch seconds (float) to epoch milliseconds (int) for full precision.
    epoch_ms_val = int(epoch_s * 1000)
    rec = create_event_record("normalize_audit_timestamp", epoch_ms_val)
    # Use the helper function to format with millisecond precision.
    return format_epoch_millis(epoch_ms_val)

def format_sequence(timestamps: list) -> list:
    return [normalize_audit_timestamp(t) for t in timestamps]
