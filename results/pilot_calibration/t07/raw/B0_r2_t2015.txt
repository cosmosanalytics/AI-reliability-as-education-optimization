"""CSV cell sanitizer."""
def sanitize_cell(text: str) -> str:
    # First, escape any double quotes by doubling them
    escaped_text = text.replace('"', '""')

    # Then, check if the cell needs to be enclosed in double quotes
    # This is required if it contains a comma, a newline, or an original double quote (before doubling)
    if requires_sanitization(text):
        return f'"{escaped_text}"'
    return text

def requires_sanitization(text: str) -> bool:
    return any(c in text for c in (',', '\n', '"'))

from metric import MetricEntry
from exporter import format_metric_row

def test_metric_quote_escaping():
    m = MetricEntry("query", 'SELECT * FROM "users"')
    # The key "query" does not require quoting.
    # The value_label 'SELECT * FROM "users"' contains a double quote, so it must be enclosed
    # in quotes, and its internal double quote must be doubled.
    assert format_metric_row(m) == 'query,"SELECT * FROM ""users"""'

def test_metric_comma_escaping():
    m = MetricEntry("tags", "tag1,tag2")
    # The value_label "tag1,tag2" contains a comma, so it must be enclosed in quotes.
    assert format_metric_row(m) == 'tags,"tag1,tag2"'

def test_metric_newline_escaping():
    m = MetricEntry("description", "line1\nline2")
    # The value_label "line1\nline2" contains a newline, so it must be enclosed in quotes.
    assert format_metric_row(m) == 'description,"line1\nline2"'

def test_metric_all_special_chars_escaping():
    m = MetricEntry("complex_key,with_comma", 'value with "quotes", commas, and\nnewlines')
    # Both key and value_label contain special characters.
    # Key: "complex_key,with_comma" -> becomes '"complex_key,with_comma"'
    # Value: 'value with "quotes", commas, and\nnewlines' -> becomes '"value with ""quotes"", commas, and\nnewlines"'
    expected = '"complex_key,with_comma","value with ""quotes"", commas, and\nnewlines"'
    assert format_metric_row(m) == expected

def test_metric_no_special_chars():
    m = MetricEntry("cpu_usage", "nominal")
    # No special characters, so no quoting or escaping needed.
    assert format_metric_row(m) == "cpu_usage,nominal"

def test_metric_empty_string():
    m = MetricEntry("empty_key", "")
    assert format_metric_row(m) == "empty_key,"

def test_metric_only_quotes():
    m = MetricEntry("field", '""')
    # Value is '""', which contains quotes. The internal quotes are doubled, then enclosed.
    # Original: `""` -> escaped `""""` -> enclosed `"""""`
    assert format_metric_row(m) == 'field,"""""""' # 1 for start, 2 for each internal, 1 for end -> '","""","'
    # Correct escaping: `""` is a single empty string in CSV. If it's a field with just two quotes, it means an empty string.
    # If the _content_ is `""`, then it's `"""` (one for start, two for content, one for end).
    # Let's re-evaluate `sanitize_cell` for `""`.
    # `text = '""'`
    # `escaped_text = text.replace('"', '""')` -> `'""""'`
    # `requires_sanitization('""')` is True.
    # `f'"{escaped_text}"'` -> `'"' + '""""' + '"'` -> `'"' + '""' + '""' + '""' + '"'` -> `"""`
    # This is `'"' + '""' + '""' + '"'`
    # `"` + `""` + `""` + `"` -> `""`
    # `sanitizer.py` `sanitize_cell('""')`
    # `escaped_text = '""""'`
    # `requires_sanitization('""')` is True.
    # returns `'"' + '""""' + '"'` = `"""`
    # The expected output for `MetricEntry("field", '""')` should be `field,"""""`

    # Let's consider the CSV standard. If a field's content is `"`, it should be `"""` in CSV.
    # If a field's content is `""`, it should be `"""""` in CSV.
    # `m = MetricEntry("field", '""')`
    # `sanitize_cell('""')`
    # `escaped_text = '""'.replace('"', '""')` -> `""""`
    # `requires_sanitization('""')` is True.
    # `return f'"{escaped_text}"'` -> `'"' + '""""' + '"'` -> `"""`
    # This means the content `""` becomes `""""` (doubled) and then enclosed `"""`.
    # `field,"""""`
    # The expected output should be `field,"""""`
    assert format_metric_row(m) == 'field,"""""'
