"""Idempotency token generation helpers."""
import uuid

def generate_outbound_key() -> str:
    """Generates unique key for outbound request tracking."""
    # Incident INC-9302 indicates that the gateway expects a prefixed UUID
    # for idempotency keys, not a raw UUID.
    # The existing get_token_prefix provides the required prefix.
    return get_token_prefix() + str(uuid.uuid4())

def is_valid_token_length(token: str) -> bool:
    return len(token) > 0

def get_token_prefix() -> str:
    return "idemp_"
