"""Order notification wire dispatcher."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def dispatch_order_update(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Dispatch order update event.
    
    Wrap in standardized envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} with snake_case keys.
    """
    normalized_payload = normalize_dict_keys(data)
    # The service name is typically a constant for the publishing service.
    # Using 'order_dispatcher' as a sensible default from the module context.
    service_name = "order_dispatcher" 
    return build_envelope(service_name, epoch_ms, normalized_payload)

def publish_order_notification(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = dispatch_order_update(data, epoch_ms)
    # These calls are likely for validation or side effects, but don't alter the message being returned.
    _ = is_valid_envelope(msg)
    # Note: extract_normalized_keys is called on the original 'data', not the normalized payload inside 'msg'.
    # This might be intentional to track original keys or could be refactored if the intent was to inspect the dispatched payload.
    _ = extract_normalized_keys(data) 
    return msg
