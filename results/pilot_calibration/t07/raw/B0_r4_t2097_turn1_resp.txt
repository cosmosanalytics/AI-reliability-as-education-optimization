"""API Endpoint Response Time Evaluator percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_endpoint_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_endpoint_percentile", samples)
    s = validate_metric_samples(win.values)
    n = len(s)
    if n == 0:
        raise ValueError("Cannot calculate percentile for empty samples list")

    # The issue description refers to a "nearest-rank house standard".
    # This implies using a nearest-rank method rather than linear interpolation.
    # A common nearest-rank method calculates the rank k as ceil((p/100)*n).
    # The percentile value is then the element at index k-1 (0-indexed).
    rank = math.ceil((p / 100.0) * n)
    # Ensure rank is within bounds [1, n]
    rank = max(1, min(n, int(rank)))
    return s[rank - 1]

def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
