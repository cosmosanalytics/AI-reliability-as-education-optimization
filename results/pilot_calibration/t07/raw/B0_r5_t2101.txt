"""System Log TSV Serializer TSV line serializer."""
from typing import List
from tsv_helper import escape_tsv_cell
from tsv_model import create_record

def serialize_log_line(level: str, msg: str, target: str) -> str:
    rec = create_record(level, msg, target)
    # The ingestion system expects backslash-escaped quotes and tabs,
    # and does not expect RFC 4180 double-quote wrapping.
    # The `escape_tsv_cell` function now handles this custom escaping.
    escaped_msg = escape_tsv_cell(rec.message)
    return f"{rec.level}\t{escaped_msg}\t{rec.target}"

def validate_line(line: str) -> bool:
    # This function is not used in the serialization path and does not need modification
    # for the current incident. It's related to parsing, not serialization.
    from tsv_helper import count_tsv_fields
    return count_tsv_fields(line) == 3

"""TSV serialization validation utility."""
from typing import List, Dict

def parse_tsv_line(line: str) -> List[str]:
    """Split raw tab-delimited record into field tokens."""
    return line.split("\t")

def count_tsv_fields(line: str) -> int:
    """Count number of tab-delimited columns in row."""
    return len(parse_tsv_line(line))

def join_tsv_fields(fields: List[str]) -> str:
    """Join list of string fields using single tab delimiter."""
    return "\t".join(fields)

def validate_tsv_columns(line: str, expected_cols: int) -> bool:
    """Verify line conforms to expected column count."""
    return count_tsv_fields(line) == expected_cols

def escape_tsv_cell(val: str) -> str:
    """
    Escape control characters and special TSV delimiters for safe single-cell embedding.
    This implementation uses backslash escaping consistent with the ingestion system's
    expectations for quotes and tabs, and also escapes backslashes themselves to
    prevent ambiguity.
    """
    # Escape backslashes first to avoid escaping newly introduced backslashes
    val = val.replace("\\", "\\\\")
    # Escape quotes
    val = val.replace('"', '\\"')
    # Escape tabs
    val = val.replace("\t", "\\t")
    # Escape carriage returns (already existing logic)
    val = val.replace("\r", "\\r")
    # Escape newlines (often good practice in TSV cells)
    val = val.replace("\n", "\\n")
    return val

def sanitize_tsv_record(line: str) -> str:
    """Trim trailing newline or carriage return from line."""
    return line.rstrip("\r\n")

def is_comment_line(line: str) -> bool:
    """Check if row begins with comment prefix hash."""
    return line.startswith("#")
