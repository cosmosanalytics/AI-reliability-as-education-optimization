"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    This function implements the "nearest rank" method (R-1),
    which is a simpler and often used method for small datasets
    or when a specific data point is desired rather than interpolation.
    It's also known as the inclusive method.
    """
    if not values:
        raise ValueError("Empty values")
    if not (0 <= p <= 100):
        raise ValueError("Percentile must be between 0 and 100 inclusive")

    s = sorted(values)
    n = len(s)

    # Handle edge cases for p=0 and p=100
    if p == 0:
        return float(s[0])
    if p == 100:
        return float(s[n - 1])

    # Calculate rank using the nearest-rank method (R-1)
    # The rank is 1-indexed, so we subtract 1 for 0-indexed list access.
    # rank = ceil(p/100 * n)
    rank = math.ceil((p / 100.0) * n)
    # Ensure rank is at least 1 and at most n
    rank = max(1, min(int(rank), n))

    return float(s[rank - 1])
