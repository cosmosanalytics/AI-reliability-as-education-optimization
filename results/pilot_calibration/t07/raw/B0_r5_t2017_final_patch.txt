"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    allowed_pattern = policy.allowed_pattern

    if clean_req == allowed_pattern:
        return True

    # Handle double asterisk wildcard (matches arbitrary multi-segment prefixes)
    if allowed_pattern.endswith('::**'):
        prefix = allowed_pattern[:-4] # Remove '::**'
        return clean_req.startswith(prefix) and (len(prefix) == 0 or clean_req[len(prefix):].startswith('::') or clean_req == prefix)
    elif allowed_pattern == '**':
        return True # Matches any scope

    # Split patterns into segments for more granular matching
    allowed_segments = allowed_pattern.split('::')
    request_segments = clean_req.split('::')

    if len(request_segments) < len(allowed_segments):
        return False

    for i, allowed_segment in enumerate(allowed_segments):
        if allowed_segment == '**':
            # '**' at any position means it matches the rest of the request_segments
            # if the part before '**' matches.
            return True
        elif allowed_segment == '*':
            # '*' matches exactly one segment, so current request segment must exist
            # and not be empty.
            if i >= len(request_segments) or not request_segments[i]:
                return False
        elif allowed_segment != request_segments[i]:
            return False

    # If we reached here, all non-wildcard allowed segments matched.
    # Now check if the request_scope is an exact match or a child path.
    # If policy pattern does not end with '*', it must be an exact match or a deeper path.
    # Example: "a::b" matches "a::b" and "a::b::c"
    # Example: "a::b::*" matches "a::b::c" but NOT "a::b::c::d"
    # Example: "a::b::**" matches "a::b", "a::b::c", "a::b::c::d"

    # If the allowed pattern does not end with a wildcard,
    # the request scope must either be an exact match (already handled)
    # or be a deeper path (e.g., policy "a::b", request "a::b::c")
    if allowed_pattern.endswith('*'):
        # If the pattern ends with '*', it must match exactly the number of segments
        # or it is a prefix match if the pattern ends with '**' (already handled).
        # For a single '*', the segment count must be equal.
        return len(request_segments) == len(allowed_segments)
    else:
        # If no wildcard at the end, the request can be an exact match (handled)
        # or a child path (e.g. "a::b" allows "a::b::c")
        # This implies request_segments must be equal or longer than allowed_segments
        # and the prefix must match exactly.
        # This is already covered by the loop and the len check at the beginning.
        # The crucial part here is to disallow "a::b_suffix" if policy is "a::b".
        # The loop already ensures "a::b" matches "a::b" and "a::b::c".
        # If allowed_pattern is "corp::reports" and request is "corp::reports_secret",
        # the loop will fail at the last segment comparison "reports" != "reports_secret".
        return True

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
