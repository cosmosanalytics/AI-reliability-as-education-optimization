"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import remove_query_string

def _normalize_path_for_matching(path: str) -> str:
    """Removes query string and normalizes trailing slashes for path matching."""
    cleaned_path = remove_query_string(path)
    # Remove trailing slash unless it's the root path "/"
    if cleaned_path.endswith('/') and len(cleaned_path) > 1:
        return cleaned_path[:-1]
    return cleaned_path

def is_route_permitted(policy: RoutePolicy, request_path: str) -> bool:
    """Check if request_path is authorized under policy.allowed_prefix.
    Path matching contract:
    - Normalizes trailing slashes before matching.
    - Matches only full path segments (e.g. '/api' covers '/api' and '/api/v1', but NOT '/api-v2').
    """
    normalized_prefix = _normalize_path_for_matching(policy.allowed_prefix)
    normalized_request_path = _normalize_path_for_matching(request_path)

    # A wildcard prefix matches all paths
    if normalized_prefix == "/":
        return True

    # Check if the request path starts with the policy prefix
    if not normalized_request_path.startswith(normalized_prefix):
        return False

    # Ensure full path segment matching:
    # The character immediately after the prefix in the request path must be
    # either the end of the string or a '/'
    prefix_len = len(normalized_prefix)
    if len(normalized_request_path) == prefix_len:
        # Exact match (e.g., policy="/api", request="/api")
        return True
    elif normalized_request_path[prefix_len] == '/':
        # Matches a sub-path (e.g., policy="/api", request="/api/v1")
        return True
    
    # If it starts with the prefix but the next character is not '/'
    # (e.g., policy="/reports", request="/reports-confidential"), it's not a match.
    return False

def evaluate_request(policies: list, role: str, path: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_route_permitted(p, path):
            return True
    return False

def count_permitted_routes(policy: RoutePolicy, paths: list) -> int:
    return sum(1 for p in paths if is_route_permitted(policy, p))

=== FILE: tests/test_guard.py ===
from policy import RoutePolicy
from guard import is_route_permitted, _normalize_path_for_matching

def test_normalize_path_for_matching():
    assert _normalize_path_for_matching("/") == "/"
    assert _normalize_path_for_matching("/api") == "/api"
    assert _normalize_path_for_matching("/api/") == "/api"
    assert _normalize_path_for_matching("/api/v1") == "/api/v1"
    assert _normalize_path_for_matching("/api/v1/") == "/api/v1"
    assert _normalize_path_for_matching("/api?q=1") == "/api"
    assert _normalize_path_for_matching("/api/?q=1") == "/api"

def test_nominal_match():
    p = RoutePolicy("admin", "/api/v1")
    assert is_route_permitted(p, "/api/v1/users") is True
    assert is_route_permitted(p, "/api/v1") is True
    assert is_route_permitted(p, "/api/v1/") is True
    assert is_route_permitted(p, "/other") is False
    assert is_route_permitted(p, "/api") is False # Should not match parent path

def test_sibling_path_boundary():
    p = RoutePolicy("analyst", "/reports")
    assert is_route_permitted(p, "/reports-secret") is False
    assert is_route_permitted(p, "/reports/daily") is True
    assert is_route_permitted(p, "/reports") is True
    assert is_route_permitted(p, "/reports/") is True

def test_wildcard_policy():
    p = RoutePolicy("admin", "/")
    assert is_route_permitted(p, "/any/path") is True
    assert is_route_permitted(p, "/") is True
    assert is_route_permitted(p, "/admin/reports") is True

def test_trailing_slashes_in_policy_prefix():
    p_slash = RoutePolicy("admin", "/api/")
    p_no_slash = RoutePolicy("admin", "/api")

    assert is_route_permitted(p_slash, "/api") is True
    assert is_route_permitted(p_slash, "/api/") is True
    assert is_route_permitted(p_slash, "/api/v1") is True
    assert is_route_permitted(p_slash, "/api-v2") is False

    assert is_route_permitted(p_no_slash, "/api") is True
    assert is_route_permitted(p_no_slash, "/api/") is True
    assert is_route_permitted(p_no_slash, "/api/v1") is True
    assert is_route_permitted(p_no_slash, "/api-v2") is False

def test_query_strings():
    p = RoutePolicy("user", "/profile")
    assert is_route_permitted(p, "/profile?id=123") is True
    assert is_route_permitted(p, "/profile/edit?id=123") is True
    assert is_route_permitted(p, "/profile/?id=123") is True
    assert is_route_permitted(p, "/profiles?id=123") is False # Sibling path
