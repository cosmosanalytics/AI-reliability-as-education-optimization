"""Topic path subscription matcher."""
def is_topic_subscribed(subscribed_pattern: str, event_topic: str) -> bool:
    """
    Determines if an event topic matches a subscription pattern.

    A subscription pattern matches an event topic if:
    1. The event topic is an exact match for the subscribed pattern.
    2. The event topic is a sub-topic of the subscribed pattern,
       indicated by the pattern followed by '::'.
       (e.g., 'orders' subscribes to 'orders::created')
    """
    if event_topic == subscribed_pattern:
        return True
    if event_topic.startswith(f"{subscribed_pattern}::"):
        return True
    return False

def clean_topic_segment(topic: str) -> str:
    return topic.strip().lower()

def is_wildcard_subscription(pattern: str) -> bool:
    return pattern.strip() in ("*", "**")
