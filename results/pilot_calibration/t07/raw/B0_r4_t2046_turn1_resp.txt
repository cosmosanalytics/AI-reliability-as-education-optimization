"""Timestamp serialization codec for telemetry records."""
from datetime import datetime, timezone

def encode_iso_timestamp(dt: datetime) -> str:
    """Encode timestamp into standard ISO 8601 format with 'Z' for UTC."""
    # Ensure the datetime object is UTC.
    # If dt is naive, we assume it's UTC as per common telemetry practices.
    # If dt is timezone-aware, convert it to UTC.
    if dt.tzinfo is None:
        dt_utc = dt.replace(tzinfo=timezone.utc)
    else:
        dt_utc = dt.astimezone(timezone.utc)

    # Format as YYYY-MM-DDTHH:MM:SS.sssZ
    # isoformat() with timespec='milliseconds' will produce YYYY-MM-DDTHH:MM:SS.sss+00:00
    # for UTC datetimes. We replace '+00:00' with 'Z' to match the desired format.
    return dt_utc.isoformat(timespec='milliseconds').replace('+00:00', 'Z')

def validate_timestamp_range(dt: datetime) -> bool:
    return dt.year >= 2020
