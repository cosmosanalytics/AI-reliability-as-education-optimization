"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def format_audit_event(data: Dict[str, Any], epoch_ms: int, service_name: str) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int, service_name: str) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms, service_name)
    _ = is_valid_envelope(msg) # This line is likely for internal validation/logging, not directly related to the fix
    # The original line `_ = extract_normalized_keys(data)` was incorrect as it operated on raw data.
    # The keys are normalized *inside* format_audit_event, and we don't need to re-extract them here.
    # If the intention was to validate the normalized keys in the payload, it would need to extract the payload first.
    # For now, let's assume this line was an artifact or for a different purpose and can remain as-is or be removed if unused.
    # Given the ticket is about the envelope, focusing on the envelope first.
    return msg

from audit_formatter import format_audit_event, serialize_audit_event

def test_nominal_audit_execution():
    raw = {"action": "login"}
    service = "test-service"
    msg = format_audit_event(raw, 1000, service)
    assert isinstance(msg, dict)
    assert serialize_audit_event(raw, 1000, service) == msg
