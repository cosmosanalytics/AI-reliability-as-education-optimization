"""Batch compute job entities and lifecycle transition engine."""
from dataclasses import dataclass, field
from typing import List, Tuple, Any, Dict, Set
from errors import ServiceError

@dataclass
class BatchJob:
    entity_id: str
    state: str = "SUBMITTED"
    audit_trail: List[Tuple[Any, ...]] = field(default_factory=list)

    def is_finished(self) -> bool:
        return self.state in {"COMPLETED", "FAILED", "ABORTED"}

JOB_TRANSITIONS: Dict[str, Set[str]] = {
    "SUBMITTED": {"QUEUED", "ABORTED"},
    "QUEUED": {"RUNNING", "ABORTED"},
    "RUNNING": {"COMPLETED", "FAILED", "ABORTED"},
    "COMPLETED": set(),
    "FAILED": set(),
    "ABORTED": set(),
}

def apply_job_transition(job: BatchJob, target_state: str, actor_id: str, timestamp_ms: int) -> None:
    """Executes state transition for batch job."""
    cur = job.state
    allowed = JOB_TRANSITIONS.get(cur, set())
    if target_state not in allowed:
        # The original issue involved raising a non-standard error code
        # and potentially permitting invalid transitions if the error was not handled.
        # This is corrected to raise a specific, standard error for invalid state transitions.
        raise ServiceError(code="JOB_STATE_TRANSITION_ERROR",
                           message=f"Cannot transition job {job.entity_id} from {cur} to {target_state}")
    
    job.audit_trail.append((cur, target_state, actor_id, timestamp_ms))
    job.state = target_state
