"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total_invoice_cents = sum(line_totals)
    num_lines = len(line_totals)

    if total_invoice_cents == 0:
        return [0] * num_lines

    # Calculate initial proportional allocation using integer division
    allocations = []
    for cents in line_totals:
        allocations.append(cents * total_discount_cents // total_invoice_cents)

    # Calculate the remainder that needs to be distributed
    current_total_allocated = sum(allocations)
    remainder = total_discount_cents - current_total_allocated

    # Distribute the remainder (e.g., "penny mismatch")
    # Distribute one cent at a time to lines with the largest fractional parts
    # To do this robustly without floating point, we can calculate the remainders
    # of the division `(cents * total_discount_cents) % total_invoice_cents`
    # and distribute to lines with the largest non-zero remainders first.
    # A simpler approach for small remainders is to distribute to the first `remainder` lines.
    # This approach ensures the total discount is fully allocated if `remainder` is non-negative.

    # Calculate the "true" proportional values (with fractions) to determine tie-breaking
    # We use tuples (fractional_remainder, original_index) to sort stably
    # The fractional_remainder is `(cents * total_discount_cents) % total_invoice_cents`
    # The larger this remainder, the 'closer' the original proportional value was to the next cent.
    fractional_parts = []
    for i, cents in enumerate(line_totals):
        fractional_parts.append(((cents * total_discount_cents) % total_invoice_cents, i))

    # Sort in descending order of fractional parts.
    # If fractional parts are equal, the original order (index) serves as a tie-breaker.
    fractional_parts.sort(key=lambda x: x[0], reverse=True)

    # Distribute the remainder to the lines with the largest fractional parts
    for i in range(remainder):
        # The index in `allocations` corresponds to the original_index
        original_index = fractional_parts[i][1]
        allocations[original_index] += 1

    return allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
