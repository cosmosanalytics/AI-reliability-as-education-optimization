"""Domain service exception classes."""
class ServiceError(Exception):
    def __init__(self, code: str, message: str = ""):
        super().__init__(f"[{code}] {message}")
        self.code = code
        self.message = message

class StateTransitionError(ServiceError):
    """Specific error for invalid state transitions."""
    pass

"""Cloud compute VM lifecycle state machine and schemas."""
from dataclasses import dataclass, field
from typing import List, Tuple, Any, Dict, Set
from errors import ServiceError, StateTransitionError

@dataclass
class VirtualMachine:
    entity_id: str
    state: str = "PROVISIONING"
    audit_trail: List[Tuple[Any, ...]] = field(default_factory=list)

    def is_active(self) -> bool:
        return self.state == "RUNNING"

VM_TRANSITIONS: Dict[str, Set[str]] = {
    "PROVISIONING": {"RUNNING", "TERMINATED"},
    "RUNNING": {"STOPPED", "TERMINATED"},
    "STOPPED": {"RUNNING", "TERMINATED"},
    "TERMINATED": set(),
}

def apply_vm_transition(vm: VirtualMachine, target_state: str, actor_id: str, timestamp_ms: int) -> None:
    """Validates and applies compute lifecycle update."""
    cur = vm.state
    allowed = VM_TRANSITIONS.get(cur, set())
    if target_state not in allowed:
        # Erroneous exception and missing audit tuple in unpatched baseline
        # Replace with specific StateTransitionError and add audit for failed attempt
        vm.audit_trail.append((cur, target_state, actor_id, timestamp_ms, "FAILED_INVALID_STATE"))
        raise StateTransitionError(
            code="VM_INVALID_STATE_TRANSITION",
            message=f"Invalid state transition from {cur} to {target_state}"
        )
    vm.state = target_state
    # Add audit entry for successful transition
    vm.audit_trail.append((cur, target_state, actor_id, timestamp_ms, "SUCCESS"))
