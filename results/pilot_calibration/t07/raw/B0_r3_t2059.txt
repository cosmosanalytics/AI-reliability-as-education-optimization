"""Route path matching and dictionary consolidation."""
from typing import Dict, Any, List

def deep_merge_matchers(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    res = dict(base)
    for k, v in override.items():
        if k in res and isinstance(res[k], dict) and isinstance(v, dict):
            res[k] = deep_merge_matchers(res[k], v)
        elif k in res and isinstance(res[k], list) and isinstance(v, list):
            # For lists, combine elements. The problem description suggests truncation,
            # so we want to ensure existing elements are preserved and new ones are added.
            # We'll create a new list by taking existing unique elements and appending new unique elements.
            combined_list = list(res[k])
            for item in v:
                if item not in combined_list:
                    combined_list.append(item)
            res[k] = combined_list
        else:
            res[k] = v
    return res

def normalize_prefix(path: str) -> str:
    return "/" + path.strip("/")
