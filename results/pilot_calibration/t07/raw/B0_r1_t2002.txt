"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value, handling fractional cents.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")

    total_line_value = sum(line_totals)
    num_lines = len(line_totals)

    if total_line_value == 0:
        return [0] * num_lines

    if total_discount_cents == 0:
        return [0] * num_lines

    # Calculate base allocations and remainders
    allocations = [0] * num_lines
    remainders = []
    
    for i, cents in enumerate(line_totals):
        # Calculate the proportional share using float division to preserve fractional parts
        raw_allocation = (cents * total_discount_cents) / total_line_value
        
        # Take the integer part as the base allocation
        base_allocation = int(raw_allocation)
        allocations[i] = base_allocation
        
        # Store the remainder for later distribution
        remainders.append((raw_allocation - base_allocation, i))

    # Calculate the sum of base allocations
    sum_allocated = sum(allocations)
    
    # Determine the unallocated amount (the "pennies" to distribute)
    unallocated_cents = total_discount_cents - sum_allocated

    # Distribute the unallocated cents to lines with the largest remainders
    # Sort remainders in descending order. If remainders are equal, original index can break ties.
    remainders.sort(key=lambda x: x[0], reverse=True)

    for i in range(unallocated_cents):
        # Add 1 cent to the line with the current largest remainder
        # The index is taken from the sorted remainders list (x[1])
        allocations[remainders[i][1]] += 1
            
    return allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
