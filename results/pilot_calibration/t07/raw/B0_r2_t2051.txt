"""Token issuance time encoding."""
from datetime import datetime, timezone

def serialize_issuance_time(dt: datetime) -> str:
    # Ensure the datetime object is timezone-aware and in UTC for consistent serialization.
    if dt.tzinfo is None:
        # If naive, assume UTC as per common practice for token issuance times
        # in absence of explicit timezone, and convert to UTC.
        dt = dt.replace(tzinfo=timezone.utc)
    elif dt.tzinfo != timezone.utc:
        # Convert to UTC if it's in a different timezone
        dt = dt.astimezone(timezone.utc)

    # Format the datetime object to ISO 8601 string with milliseconds.
    # Example: 2026-10-25T18:00:00.000+00:00
    iso_str = dt.isoformat(timespec='milliseconds')

    # Replace the '+00:00' offset with 'Z' for UTC times to comply with
    # the "non-standard timestamp serialization" requirement.
    if iso_str.endswith('+00:00'):
        return iso_str[:-6] + 'Z'
    return iso_str

def compute_token_expiry_epoch(dt: datetime, lifetime_seconds: int = 3600) -> int:
    return int(dt.timestamp()) + lifetime_seconds
