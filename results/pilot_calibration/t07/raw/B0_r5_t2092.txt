"""Audit Trail Event Time Normalizer utility."""
from datetime import datetime, timezone
from iso_helper import parse_epoch_seconds, is_valid_iso_format, extract_year_component
from timestamp_model import TimestampRecord, create_event_record

def normalize_audit_timestamp(epoch_s: float) -> str:
    # The TimestampRecord's epoch_ms field should store milliseconds for precision.
    # The name epoch_ms implies milliseconds, so convert epoch_s (seconds) to epoch_ms (milliseconds).
    rec = create_event_record("normalize_audit_timestamp", int(epoch_s * 1000))
    dt = parse_epoch_seconds(epoch_s)
    # Format the datetime object to include milliseconds.
    # The microsecond attribute holds microseconds, so divide by 1000 to get milliseconds.
    # Use f-string formatting to ensure three digits with leading zeros.
    return dt.strftime(f"%Y-%m-%dT%H:%M:%S.{dt.microsecond // 1000:03d}Z")

def format_sequence(timestamps: list) -> list:
    return [normalize_audit_timestamp(t) for t in timestamps]
