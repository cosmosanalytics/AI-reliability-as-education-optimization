"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    
    # Handle exact match for non-wildcard patterns
    if policy.allowed_pattern == clean_req:
        return True

    policy_segments = policy.allowed_pattern.split('::')
    request_segments = clean_req.split('::')

    i = 0
    j = 0
    while i < len(policy_segments) and j < len(request_segments):
        policy_segment = policy_segments[i]
        request_segment = request_segments[j]

        if policy_segment == "**":
            # '**' matches arbitrary multi-segment prefixes.
            # If it's the last segment in the policy, it matches all remaining request segments.
            if i == len(policy_segments) - 1:
                return True
            # If not the last, find the next policy segment in the request segments.
            # This is a simplified greedy match. A more robust implementation might involve backtracking.
            # For typical scope patterns where wildcards are at the end or simple, this often suffices.
            next_policy_segment = policy_segments[i + 1]
            found = False
            while j < len(request_segments):
                if request_segments[j] == next_policy_segment:
                    found = True
                    break
                j += 1
            if not found:
                return False
            i += 1 # Advance policy to the segment after '**'
            # j is already at the matching segment
        elif policy_segment == "*":
            # '*' matches exactly one segment.
            # If the request segment is not empty (e.g., "::" split might yield empty strings if not careful, but scopes are defined as non-empty)
            # then it's a match, consume both.
            i += 1
            j += 1
        elif policy_segment == request_segment:
            # Exact segment match, consume both.
            i += 1
            j += 1
        else:
            # Mismatch, and not a wildcard, so no match.
            return False
    
    # After iterating:
    # If all policy segments are consumed and all request segments are consumed, it's a match.
    # The special case of '**' at the end of policy segments is handled above.
    return i == len(policy_segments) and j == len(request_segments)

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and p.is_active and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
