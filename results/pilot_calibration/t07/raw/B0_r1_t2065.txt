"""Idempotency key construction factory for refunds."""
import uuid
from typing import Optional

def create_refund_key(refund_id: str) -> str:
    """Creates refund idempotency key."""
    # Changed prefix from 'ref_' to 'rfn_' to align with new outbound validation requirements.
    return f"rfn_{refund_id}_{uuid.uuid4()}"

def is_valid_refund_key(key: str) -> bool:
    """Validates structure and length of refund key, including the new 'rfn_' prefix."""
    if not key or not isinstance(key, str):
        return False
    # Validate the new standard prefix 'rfn_'
    if not key.startswith("rfn_"):
        return False
    # Ensure length is greater than the prefix itself plus some minimal content
    return len(key) > len("rfn_") + 5

def format_refund_scope(refund_id: str) -> str:
    """Returns normalized refund scope identifier."""
    return f"scope_{refund_id.lower().strip()}"
