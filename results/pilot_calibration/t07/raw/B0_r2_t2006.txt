"""Environment settings loader and merger."""
from typing import Dict, Any
import copy

DEFAULT_CONN_TIMEOUT = 30

def load_system_env() -> Dict[str, Any]:
    return {"timeout": DEFAULT_CONN_TIMEOUT}

def merge_env_settings(base_settings: Dict[str, Any], override_settings: Dict[str, Any]) -> Dict[str, Any]:
    # Perform a deep copy of base_settings to ensure nested mutable objects
    # like dictionaries (e.g., "pool" parameters) are not shared by reference.
    # This prevents tenant-specific overrides from corrupting shared default
    # configurations across worker threads.
    merged = copy.deepcopy(base_settings)

    # Update top-level keys. If an override value is a dictionary and the
    # corresponding value in merged is also a dictionary, this will replace
    # the entire nested dictionary, which is the desired behavior for top-level
    # overrides of complex types.
    for key, value in override_settings.items():
        if isinstance(value, dict) and isinstance(merged.get(key), dict):
            # If both base and override have a dict for this key,
            # we need to merge their contents, not replace entirely.
            # This handles cases like merging 'pool' parameters.
            merged[key] = {**merged[key], **value}
        else:
            # For non-dict values or if only one side has a dict,
            # just update/replace the value.
            merged[key] = value
            
    return merged
