"""User activity notification dispatcher."""
import os
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a service name for the envelope.
# This could be made configurable via environment variables or a config file.
SERVICE_NAME = os.getenv("ACTIVITY_SERVICE_NAME", "user_activity_service")

def dispatch_user_activity(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Dispatch user activity record to notification bus.
    The record is wrapped in a wire envelope with normalized payload keys.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(SERVICE_NAME, epoch_ms, normalized_payload)

def broadcast_activity_update(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = dispatch_user_activity(data, epoch_ms)
    # The following lines are currently for side effects (e.g., logging or validation checks)
    # but the primary goal is to return the enveloped message.
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data) # Note: This extracts keys from the original 'data', not the normalized payload in 'msg'.
    return msg
