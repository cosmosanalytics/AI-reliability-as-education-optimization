"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a service name for the telemetry stream.
# In a real application, this would likely be configured or passed in.
SERVICE_NAME = "telemetry_stream_service"

def prepare_metric_packet(data: Dict[str, Any]) -> Dict[str, Any]:
    """Prepare telemetry metric packet by normalizing its keys."""
    return normalize_dict_keys(data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """
    Prepare and publish a telemetry stream packet, ensuring it has the required wire envelope.
    The input `data` will be normalized and then wrapped as the payload in the wire envelope.
    """
    # First, normalize the keys of the raw telemetry data
    normalized_payload = prepare_metric_packet(data)

    # Then, build the wire envelope around the normalized payload
    enveloped_packet = build_envelope(SERVICE_NAME, epoch_ms, normalized_payload)

    # The original checks below were not modifying the packet, they are now redundant
    # as build_envelope guarantees a valid envelope structure.
    # _ = is_valid_envelope(pkt)
    # _ = extract_normalized_keys(data) # This was checking the raw input data, not the packet

    return enveloped_packet
