"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    The remaining cents due to rounding are distributed one by one to the lines
    that received the largest fractional discount.

    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")

    num_lines = len(line_totals)
    if num_lines == 0:
        return []

    total_line_value = sum(line_totals)
    if total_line_value == 0:
        return [0] * num_lines

    allocations = [0] * num_lines
    remainder_cents = total_discount_cents

    # Calculate base allocations and track fractional remainders
    # We use a list of (remainder, index) tuples to sort and distribute
    fractions = []
    for i, cents in enumerate(line_totals):
        # Calculate full proportional value (as float)
        proportional_value = (cents * total_discount_cents) / total_line_value
        
        # Take the integer part as the initial allocation
        base_allocation = int(proportional_value)
        allocations[i] = base_allocation
        remainder_cents -= base_allocation
        
        # Store the fractional part for later distribution of remaining pennies
        # We multiply by -1 to sort in descending order of fractional part
        fractions.append((proportional_value - base_allocation, i))

    # Distribute remaining cents (pennies) to lines with the largest fractional parts
    # Sort by fractional part in descending order (largest fraction first)
    # If fractions are equal, distribute to lines with smaller original index (stable tie-breaking)
    fractions.sort(key=lambda x: x[0], reverse=True)

    for i in range(min(remainder_cents, num_lines)):
        # Give one cent to the line with the largest remaining fractional part
        allocations[fractions[i][1]] += 1
    
    return allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
