def deep_merge(base, override):
    """Return a NEW dict: base merged with override. Nested dicts merge recursively; any other value in override replaces
    the base value (lists are replaced, not concatenated). A value of None in override REMOVES that key from the result.
    Neither input is modified and the result shares no mutable containers with either input."""
    result = {}
    # Start with a deep copy of base to ensure no shared mutable containers
    # and to handle keys present only in base.
    for k, v in base.items():
        if isinstance(v, dict):
            result[k] = deep_merge(v, {})  # Deep copy nested dicts
        elif isinstance(v, list):
            result[k] = list(v)  # Copy lists
        else:
            result[k] = v  # Copy other types

    for k, v_override in override.items():
        if v_override is None:
            # A value of None in override REMOVES that key from the result.
            if k in result:
                del result[k]
        elif k in result and isinstance(result[k], dict) and isinstance(v_override, dict):
            # Nested dicts merge recursively
            result[k] = deep_merge(result[k], v_override)
        elif isinstance(v_override, list):
            # Lists are replaced, not concatenated.
            result[k] = list(v_override)  # Copy the override list
        elif isinstance(v_override, dict):
            # If override value is a dict but base value is not, replace with a deep copy of override dict
            result[k] = deep_merge({}, v_override)
        else:
            # Any other value in override replaces the base value
            result[k] = v_override
    return result

def parse_value(text):
    """Convert an environment string: "true"/"yes"/"on"/"1" (any case, surrounding spaces ignored) -> True,
    "false"/"no"/"off"/"0" -> False, a decimal integer like "42" or "-7" -> int, a decimal number like "2.5" -> float,
    "" or "null"/"none" (any case) -> None, anything else -> the stripped string unchanged. Note "1" and "0" are booleans."""
    s = text.strip()
    lower_s = s.lower()

    # Booleans
    if lower_s in ("true", "yes", "on", "1"):
        return True
    if lower_s in ("false", "no", "off", "0"):
        return False

    # None
    if s == "" or lower_s in ("null", "none"):
        return None

    # Numbers
    try:
        # Try integer first
        return int(s)
    except ValueError:
        try:
            # Then float
            return float(s)
        except ValueError:
            # If not a number, return the stripped string
            return s
