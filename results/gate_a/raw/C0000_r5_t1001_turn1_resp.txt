from decimal import Decimal, ROUND_HALF_UP, ROUND_HALF_DOWN


def _round_decimal_to_int_half_away_from_zero(d_value):
    """
    Rounds a Decimal value to the nearest integer, rounding halves away from zero.
    e.g., 28.5 -> 29, -100.5 -> -101.
    """
    target_precision = Decimal('1')
    if d_value >= 0:
        return d_value.quantize(target_precision, rounding=ROUND_HALF_UP)
    else:
        # For negative numbers, ROUND_HALF_DOWN rounds -X.5 to -X-1 (e.g., -100.5 -> -101).
        return d_value.quantize(target_precision, rounding=ROUND_HALF_DOWN)


def to_cents(x):
    """Convert a dollar amount (int, float or str such as "1.10") to integer cents. Round half away from zero at the
    cent: 0.285 -> 29, 2.675 -> 268, -1.005 -> -101. Never use binary float arithmetic for the conversion."""
    # Convert input to Decimal first, using str() for floats to avoid binary float precision issues.
    d = Decimal(str(x))
    # Multiply by 100 to get the value in cents as a Decimal.
    cents_decimal = d * 100
    # Apply the specified rounding rule and convert to integer.
    return int(_round_decimal_to_int_half_away_from_zero(cents_decimal))


def fmt(cents):
    """Format integer cents as dollars: 1999 -> "$19.99", 5 -> "$0.05", -150 -> "-$1.50", -5 -> "-$0.05"."""
    # Handle negative cents by taking the absolute value for division/modulo, then prepending the sign.
    sign = "-" if cents < 0 else ""
    abs_cents = abs(cents)
    dollars = abs_cents // 100
    remaining_cents = abs_cents % 100
    return "%s$%d.%02d" % (sign, dollars, remaining_cents)

from decimal import Decimal
from money import to_cents, _round_decimal_to_int_half_away_from_zero


def line_total(price, qty):
    """Cents for qty items at a dollar price."""
    return to_cents(price) * qty


def invoice_total(lines, discount_pct=0):
    """lines is a list of (price_in_dollars, qty). discount_pct (0-100, may be fractional) is taken off the subtotal;
    the discount amount is rounded half away from zero to a whole cent. Returns total cents."""
    sub = sum(line_total(p, q) for p, q in lines)

    # Calculate discount amount using Decimal for precision.
    # Convert sub to Decimal for calculation.
    sub_decimal = Decimal(sub)
    # Convert discount_pct to Decimal, using str() to avoid float precision issues if it's a float.
    discount_pct_decimal = Decimal(str(discount_pct))

    # Calculate raw discount amount.
    raw_discount_decimal = sub_decimal * discount_pct_decimal / Decimal('100')

    # Round the discount amount using the specified rule and convert to integer.
    rounded_discount = _round_decimal_to_int_half_away_from_zero(raw_discount_decimal)

    return sub - int(rounded_discount)

from decimal import Decimal
from money import _round_decimal_to_int_half_away_from_zero


def vat(cents, rate):
    """VAT in cents on `cents` at fractional `rate` (0.2 = 20%), rounded half away from zero to a whole cent."""
    # Convert cents to Decimal for calculation.
    cents_decimal = Decimal(cents)
    # Convert rate to Decimal, using str() to avoid float precision issues if it's a float.
    rate_decimal = Decimal(str(rate))

    # Calculate raw VAT amount.
    raw_vat_decimal = cents_decimal * rate_decimal

    # Round the VAT amount using the specified rule and convert to integer.
    rounded_vat = _round_decimal_to_int_half_away_from_zero(raw_vat_decimal)

    return int(rounded_vat)


def gross(cents, rate):
    """cents plus its VAT."""
    return cents + vat(cents, rate)
