from datetime import datetime


class Ledger:
    """Append-only usage ledger. Entries: (customer, timestamp (naive datetime), units (int, negative for refunds))."""

    def __init__(self):
        self.entries = []

    def record(self, customer, ts, units):
        """Record an entry. Reject (ValueError) a units value of 0, a non-int units (bools too), a non-datetime ts or an empty
        customer name."""
        # Bug fix: Store units as provided, allowing negative values for refunds.
        # The docstring implies validation that is not implemented here, but fixing that is out of scope for this ticket.
        self.entries.append((customer, ts, units))

    def customers(self):
        """Sorted list of every customer that ever had an entry."""
        return sorted({c for c, _, _ in self.entries})

from period import month_bounds


def monthly_report(ledger, year, month, customers=None):
    """Dict customer -> net units for the month (refunds subtract). Includes every customer in `customers` (default:
    all customers in the ledger) with 0 when they have no entries that month."""
    start, end = month_bounds(year, month)

    # Determine the set of customers to include in the report
    if customers is None:
        # Default to all customers that ever had an entry in the ledger
        report_customer_names = set(ledger.customers())
    else:
        # Use the provided list of customers
        report_customer_names = set(customers)

    # Initialize the output dictionary with all relevant customers having 0 usage
    # This fixes the bug where customers with no usage were missing from the report.
    out = {c: 0 for c in report_customer_names}

    for c, ts, u in ledger.entries:
        # Bug fix: An entry belongs to the month iff start <= ts < end (exclusive of 'end').
        # This fixes the bug where entries late on the last day of a month were counted in the next month.
        # Also, only process entries for customers we are reporting on.
        if start <= ts < end and c in report_customer_names:
            out[c] += u
    return out
