import csv
import io

def is_plain_number(s):
    """Checks if a string represents a plain number (int or float)."""
    try:
        float(s)
        return True
    except ValueError:
        return False

def apply_injection_guard(field_str):
    """Applies spreadsheet injection guard to a string field."""
    if field_str and field_str[0] in ('=', '+', '-', '@'):
        if not is_plain_number(field_str):
            return "'" + field_str
    return field_str

def write_row(fields):
    """Encode one record as a CSV line WITHOUT trailing newline. Fields are str/int/float/None. None -> empty field.
    A field is quoted (with double quotes, inner quotes doubled) iff it contains a comma, a double quote, CR or LF, or has
    leading/trailing spaces. Numbers use str(). Spreadsheet-injection guard: a str field starting with =, +, - or @
    (but a str that parses as a plain number such as "-5" is left alone) gets a leading apostrophe prepended BEFORE quoting."""
    processed_fields = []
    for f in fields:
        # 1. Handle None -> empty string
        s = "" if f is None else str(f)

        # 2. Apply Spreadsheet Injection Guard
        # The guard applies only if the original field was a string AND it starts with special chars
        # AND it's not a plain number.
        if isinstance(f, str):
            s = apply_injection_guard(s)

        # 3. Determine if quoting is needed
        needs_quoting = False
        if "," in s or '"' in s or "\r" in s or "\n" in s:
            needs_quoting = True
        elif s.strip() != s: # Check for leading/trailing spaces
            needs_quoting = True

        # 4. Apply quoting
        if needs_quoting:
            # Double inner quotes
            s = s.replace('"', '""')
            # Enclose in double quotes
            s = f'"{s}"'
        
        processed_fields.append(s)
    
    return ",".join(processed_fields)


def read_rows(text):
    """Parse CSV text (rows separated by \n or \r\n, quoted fields may contain commas, quotes (doubled) and newlines) into a
    list of lists of str. A trailing newline does not create an empty row. An empty field is "". Raise ValueError on an
    unterminated quote."""
    f = io.StringIO(text)
    reader = csv.reader(f)
    rows = []
    try:
        for row in reader:
            rows.append(row)
    except csv.Error as e:
        # Wrap csv.Error in ValueError as per docstring
        raise ValueError(f"CSV parsing error: {e}") from e
    
    return rows
