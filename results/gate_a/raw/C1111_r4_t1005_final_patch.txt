class Groups:
    """Group membership. Groups may contain users and other groups (nesting). Cycles must not hang."""

    def __init__(self):
        self.members = {}

    def add(self, group, member):
        self.members.setdefault(group, set()).add(member)

    def remove(self, group, member):
        """Remove member from group; removing a non-member is a no-op. An emptied group still exists."""
        if group in self.members:
            self.members[group].discard(member)

    def groups_of(self, user):
        """Set of all groups the user belongs to, directly or through nested groups."""
        all_groups = set()
        q = [user]  # Queue for BFS, starting with the user itself
        processed_members = set()  # To prevent cycles and redundant work

        while q:
            current_member = q.pop(0)  # BFS
            if current_member in processed_members:
                continue
            processed_members.add(current_member)

            # Check if current_member is a member of any group
            for group_name, members_set in self.members.items():
                if current_member in members_set:
                    if group_name not in all_groups:
                        all_groups.add(group_name)
                        # If this group is a member of other groups, we need to explore those too.
                        # Add the group itself to the queue to find groups it might be a member of.
                        q.append(group_name)
        return all_groups

def allowed(groups, acl, user, path, action):
    """acl: list of (path_prefix, subject, action, effect) with effect "allow" or "deny"; subject is a user name or group
    name. A rule applies if its prefix is a path-segment prefix of path ("/a" covers "/a" and "/a/b" but NOT "/ab"; "/" covers
    everything), its action equals `action` or is "*", and its subject is the user or one of the user's groups (nested
    included). The MOST SPECIFIC (longest) matching prefix decides; at equal length deny beats allow. No match -> False."""
    best_match_len = -1
    best_effect = None

    user_and_groups = {user} | groups.groups_of(user)

    for prefix, subject, act, effect in acl:
        # Path prefix matching: "/a" covers "/a" and "/a/b" but NOT "/ab"
        if not path.startswith(prefix):
            continue
        # Additional check for path-segment prefix: if path is longer than prefix, the next char must be '/'
        if len(path) > len(prefix) and path[len(prefix)] != '/':
            continue
        # Special case: if prefix is "/" and path is also "/", it's a match.
        # The above check handles this correctly: len(path) == len(prefix) so the condition is false.

        # Action matching: action equals `action` or is "*"
        if act != action and act != "*":
            continue
        
        # Subject matching: subject is the user or one of the user's groups (nested included)
        if subject not in user_and_groups:
            continue
        
        # Determine the best rule based on prefix length and deny beats allow
        if len(prefix) > best_match_len:
            best_match_len = len(prefix)
            best_effect = effect
        elif len(prefix) == best_match_len:
            # At equal length, deny beats allow.
            # If current effect is "deny", it overrides an "allow" rule.
            # If current effect is "allow", it does not override a "deny" rule.
            if effect == "deny":
                best_effect = effect
            
    return best_effect == "allow"
