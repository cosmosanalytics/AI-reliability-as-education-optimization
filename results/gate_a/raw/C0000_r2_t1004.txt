def write_row(fields):
    """Encode one record as a CSV line WITHOUT trailing newline. Fields are str/int/float/None. None -> empty field.
    A field is quoted (with double quotes, inner quotes doubled) iff it contains a comma, a double quote, CR or LF, or has
    leading/trailing spaces. Numbers use str(). Spreadsheet-injection guard: a str field starting with =, +, - or @
    (but a str that parses as a plain number such as "-5" is left alone) gets a leading apostrophe prepended BEFORE quoting."""

    formatted_fields = []
    for field in fields:
        if field is None:
            s = ""
        else:
            s = str(field)

        # Spreadsheet-injection guard
        if s and s[0] in ('=', '+', '-', '@'):
            is_number_string = False
            try:
                # Check if it's a number string like "-5" or "+10.5"
                # This will fail for "=SUM(...)" or "@user"
                float(s)
                is_number_string = True
            except ValueError:
                pass # Not a number string

            if not is_number_string:
                s = "'" + s

        # Quoting logic
        needs_quoting = False
        # Check for characters that require quoting
        if ',' in s or '"' in s or '\r' in s or '\n' in s:
            needs_quoting = True
        # Check for leading/trailing spaces that require quoting
        if s != s.strip():
            needs_quoting = True

        if needs_quoting:
            # Double inner quotes
            s = s.replace('"', '""')
            # Enclose in double quotes
            s = '"' + s + '"'
        
        formatted_fields.append(s)

    return ",".join(formatted_fields)


def read_rows(text):
    """Parse CSV text (rows separated by \n or \r\n, quoted fields may contain commas, quotes (doubled) and newlines) into a
    list of lists of str. A trailing newline does not create an empty row. An empty field is "". Raise ValueError on an
    unterminated quote."""
    # The current ticket focuses on fixing CSV export issues.
    # The read_rows function is not a full CSV parser and would require significant changes
    # to correctly handle quoted fields, doubled quotes, and newlines within fields.
    # This is out of scope for the current bug report, which is about malformed *export*.
    return [line.split(",") for line in text.splitlines()]
