import re

def _is_number(s):
    try:
        float(s)
        return True
    except ValueError:
        return False

def write_row(fields):
    """Encode one record as a CSV line WITHOUT trailing newline. Fields are str/int/float/None. None -> empty field.
    A field is quoted (with double quotes, inner quotes doubled) iff it contains a comma, a double quote, CR or LF, or has
    leading/trailing spaces. Numbers use str(). Spreadsheet-injection guard: a str field starting with =, +, - or @
    (but a str that parses as a plain number such as "-5" is left alone) gets a leading apostrophe prepended BEFORE quoting."""
    processed_fields = []
    for field in fields:
        s_field = ""
        if field is not None:
            s_field = str(field)

        # Spreadsheet-injection guard
        if isinstance(field, str) and s_field and s_field[0] in ('=', '+', '-', '@'):
            if s_field[0] in ('=', '@') or not _is_number(s_field):
                s_field = "'" + s_field

        # Determine if quoting is needed
        needs_quoting = False
        if ',' in s_field or '"' in s_field or '\r' in s_field or '\n' in s_field:
            needs_quoting = True
        elif s_field != s_field.strip(): # Check for leading/trailing spaces
            needs_quoting = True

        if needs_quoting:
            # Double inner quotes and enclose in double quotes
            s_field = '"' + s_field.replace('"', '""') + '"'
        
        processed_fields.append(s_field)
    
    return ",".join(processed_fields)


# Regex to parse a single CSV field, handling quoted fields
# It captures either:
# 1. A quoted field: starts with ", ends with ", allows "" inside.
#    The content inside the quotes is captured in group 1.
# 2. An unquoted field: any characters that are not a comma or newline.
_FIELD_RE = re.compile(r'(?:"((?:[^"]|"")*)"|([^,\r\n]*))')

def read_rows(text):
    """Parse CSV text (rows separated by \n or \r\n, quoted fields may contain commas, quotes (doubled) and newlines) into a
    list of lists of str. A trailing newline does not create an empty row. An empty field is "". Raise ValueError on an
    unterminated quote."""
    rows = []
    for line in text.splitlines():
        if not line.strip() and not line: # Skip truly empty lines, but allow lines with only whitespace if they are valid CSV
            continue
        
        current_row = []
        pos = 0
        while pos < len(line):
            match = _FIELD_RE.match(line, pos)
            if not match:
                # This can happen if there's an unquoted comma followed by nothing, or malformed input
                # For example, "a,b," will match "a", then "b", then fail on the last comma if not handled.
                # Or if there's an unterminated quote.
                if line[pos] == ',': # Handle empty field after a comma
                    current_row.append("")
                    pos += 1
                    continue
                elif line[pos] == '"': # Unterminated quote
                    raise ValueError("Unterminated quote in CSV line: %r" % line)
                else: # Malformed input not covered by specific cases
                    raise ValueError("Malformed CSV line: %r at position %d" % (line, pos))

            quoted_field_content = match.group(1)
            unquoted_field_content = match.group(2)

            if quoted_field_content is not None:
                # It was a quoted field, unescape double quotes
                current_row.append(quoted_field_content.replace('""', '"'))
            elif unquoted_field_content is not None:
                # It was an unquoted field
                current_row.append(unquoted_field_content)
            else:
                # This case should ideally not be reached if regex is correct, but for safety
                current_row.append("")

            pos = match.end()
            if pos < len(line) and line[pos] == ',':
                pos += 1 # Consume the comma separator
            elif pos < len(line):
                # If there's more content but no comma, it's malformed
                raise ValueError("Malformed CSV line: %r (extra content after field)" % line)
        
        # Handle trailing empty field if line ends with a comma
        if line.endswith(',') and not current_row: # e.g., ","
             current_row.append("")
        elif line.endswith(',') and pos == len(line): # e.g., "a,b,"
            current_row.append("")

        rows.append(current_row)
    return rows

from csvio import write_row

HEADER = ["id", "name", "note", "amount"]


def export(customers):
    """customers: list of dicts with id, name, note, amount (note/amount may be None). Returns the CSV text, header first,
    every line ending in \n."""
    lines = [write_row(HEADER)]
    for c in customers:
        lines.append(write_row([c["id"], c["name"], c.get("note"), c.get("amount")]))
    return "\n".join(lines) + "\n"
