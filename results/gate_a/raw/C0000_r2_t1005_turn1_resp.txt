class Groups:
    """Group membership. Groups may contain users and other groups (nesting). Cycles must not hang."""

    def __init__(self):
        self.members = {}

    def add(self, group, member):
        self.members.setdefault(group, set()).add(member)

    def remove(self, group, member):
        """Remove member from group; removing a non-member is a no-op. An emptied group still exists."""
        if group in self.members:
            self.members[group].discard(member)

    def groups_of(self, user):
        """Set of all groups the user belongs to, directly or through nested groups."""
        user_groups = set()
        q = [user] # Queue for BFS/DFS, starting with the user itself

        # Keep track of visited entities to prevent infinite loops in case of cycles
        visited = set()

        while q:
            current_member = q.pop(0) # Use pop(0) for BFS, pop() for DFS

            if current_member in visited:
                continue
            visited.add(current_member)

            for group, members_set in self.members.items():
                if current_member in members_set:
                    if group not in user_groups: # Avoid re-adding to queue if already processed as a group
                        user_groups.add(group)
                        q.append(group) # Add the group itself to the queue to find its parent groups
        return user_groups

def allowed(groups, acl, user, path, action):
    """acl: list of (path_prefix, subject, action, effect) with effect "allow" or "deny"; subject is a user name or group
    name. A rule applies if its prefix is a path-segment prefix of path ("/a" covers "/a" and "/a/b" but NOT "/ab"; "/" covers
    everything), its action equals `action` or is "*", and its subject is the user or one of the user's groups (nested
    included). The MOST SPECIFIC (longest) matching prefix decides; at equal length deny beats allow. No match -> False."""
    best_match_len = -1
    best_effect = None # "allow" or "deny"

    user_and_groups = groups.groups_of(user)
    user_and_groups.add(user) # Add the user itself to the set of subjects to check

    for prefix, subject, act, effect in acl:
        # Check path prefix match
        if not path.startswith(prefix):
            continue
        # Ensure path segment prefix match: "/a" matches "/a" and "/a/b", but not "/ab"
        # This means if prefix is not "/", the character immediately after the prefix in path must be "/" or path must end there.
        if len(prefix) > 0 and prefix != "/" and len(path) > len(prefix) and path[len(prefix)] != '/':
            continue

        # Check action match (action or wildcard "*")
        if act != action and act != "*":
            continue

        # Check subject match (user or any of user's groups)
        if subject not in user_and_groups:
            continue

        # Determine if this rule is "better" than the current best
        if len(prefix) > best_match_len:
            # More specific rule, this one wins
            best_match_len = len(prefix)
            best_effect = effect
        elif len(prefix) == best_match_len:
            # Equal specificity, "deny" beats "allow"
            if effect == "deny" and best_effect == "allow":
                best_effect = effect # Deny overrides allow
            # If current is "allow" and best is "deny", best_effect remains "deny"
            # If both are "allow" or both are "deny", best_effect remains unchanged (first one found wins, which is fine)

    return best_effect == "allow"
