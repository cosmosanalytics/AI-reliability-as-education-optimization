from decimal import Decimal, ROUND_HALF_UP

def to_cents(x):
    """Convert a dollar amount (int, float or str such as "1.10") to integer cents. Round half away from zero at the
    cent: 0.285 -> 29, 2.675 -> 268, -1.005 -> -101. Never use binary float arithmetic for the conversion."""
    # Convert input to Decimal, ensuring floats are converted via string to avoid precision issues.
    # Then multiply by 100 to shift to cents, and quantize to an integer using ROUND_HALF_UP.
    # ROUND_HALF_UP rounds half away from zero, which matches the requirement.
    if isinstance(x, float):
        d = Decimal(str(x))
    else:
        d = Decimal(x)
    
    return int(d.quantize(Decimal('0.01'), rounding=ROUND_HALF_UP) * 100)


def fmt(cents):
    """Format integer cents as dollars: 1999 -> "$19.99", 5 -> "$0.05", -150 -> "-$1.50", -5 -> "-$0.05"."""
    return "$%d.%02d" % (cents // 100, abs(cents % 100))

from money import to_cents
from decimal import Decimal, ROUND_HALF_UP


def line_total(price, qty):
    """Cents for qty items at a dollar price."""
    return to_cents(price) * qty


def invoice_total(lines, discount_pct=0):
    """lines is a list of (price_in_dollars, qty). discount_pct (0-100, may be fractional) is taken off the subtotal;
    the discount amount is rounded half away from zero to a whole cent. Returns total cents."""
    sub = sum(line_total(p, q) for p, q in lines)
    
    # Convert subtotal to Decimal for precise discount calculation.
    # Calculate discount amount, then quantize to a whole cent using ROUND_HALF_UP.
    # Finally, convert back to int.
    sub_d = Decimal(sub)
    discount_pct_d = Decimal(str(discount_pct))
    
    discount_amount_d = (sub_d * discount_pct_d / 100).quantize(Decimal('1'), rounding=ROUND_HALF_UP)
    
    return int(sub_d - discount_amount_d)

from decimal import Decimal, ROUND_HALF_UP

def vat(cents, rate):
    """VAT in cents on `cents` at fractional `rate` (0.2 = 20%), rounded half away from zero to a whole cent."""
    # Convert cents and rate to Decimal for precise VAT calculation.
    # Calculate VAT amount, then quantize to a whole cent using ROUND_HALF_UP.
    # Finally, convert back to int.
    cents_d = Decimal(cents)
    rate_d = Decimal(str(rate)) # Convert rate to string first if it's a float
    
    return int((cents_d * rate_d).quantize(Decimal('1'), rounding=ROUND_HALF_UP))


def gross(cents, rate):
    """cents plus its VAT."""
    return cents + vat(cents, rate)
