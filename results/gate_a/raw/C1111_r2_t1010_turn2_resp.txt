class Event:
    """Event object passed to subscribers."""

    def __init__(self, name, data=None):
        self.name = name
        self.data = data or {}
        self.cancelled = False

    def stop_propagation(self):
        """Halt dispatching to all subsequent subscribers immediately."""
        self.cancelled = True

from subscriber import Subscriber


class EventDispatcher:
    def __init__(self):
        self._subscribers = {}
        self._dispatching = False # Flag to indicate if dispatch is in progress
        self._pending_unsubscribes = [] # List of (name, handler) for pending unsubs

    def subscribe(self, name, handler, priority=0, once=False):
        sub = Subscriber(handler, priority, once)
        self._subscribers.setdefault(name, []).append(sub)
        # Maintain insertion order for stable sort
        self._subscribers[name].sort(key=lambda s: (s.priority, len(self._subscribers[name])))
        return sub

    def unsubscribe(self, name, handler):
        if self._dispatching:
            # If dispatching, defer unsubscribe to avoid modifying list during iteration
            self._pending_unsubscribes.append((name, handler))
        else:
            self._perform_unsubscribe(name, handler)

    def _perform_unsubscribe(self, name, handler):
        if name in self._subscribers:
            # Filter out the subscriber with the matching handler
            self._subscribers[name] = [
                s for s in self._subscribers[name] if s.handler != handler
            ]
            # If the list becomes empty, remove the event name entry
            if not self._subscribers[name]:
                del self._subscribers[name]

    def dispatch(self, event):
        """Dispatch event to subscribers sorted by priority (descending).
        Equal priority subscribers must execute in FIFO order.
        If event.cancelled is set, halt execution immediately."""
        
        # Get a snapshot of subscribers for the event, ensuring stable sort
        # The original list is already sorted by priority and insertion order
        # so a simple copy is sufficient for iteration.
        subs_for_event = list(self._subscribers.get(event.name, []))

        self._dispatching = True
        try:
            for s in subs_for_event:
                # Check if event propagation has been stopped
                if event.cancelled:
                    break

                # Check if the subscriber was unsubscribed during this dispatch cycle
                # This handles cases where a handler unsubscribes another handler
                # or itself (if not 'once') before it's reached in the iteration.
                if event.name not in self._subscribers or s not in self._subscribers[event.name]:
                    continue

                s.handler(event)

                # Handle 'once' subscribers
                if s.once:
                    # Unsubscribe immediately if not already pending
                    if (event.name, s.handler) not in self._pending_unsubscribes:
                        self.unsubscribe(event.name, s.handler)
        finally:
            self._dispatching = False
            # Process any pending unsubscriptions after dispatch is complete
            for name, handler in self._pending_unsubscribes:
                self._perform_unsubscribe(name, handler)
            self._pending_unsubscribes.clear()
