from datetime import datetime


class Ledger:
    """Append-only usage ledger. Entries: (customer, timestamp (naive datetime), units (int, negative for refunds))."""

    def __init__(self):
        self.entries = []

    def record(self, customer, ts, units):
        """Record an entry. Reject (ValueError) a units value of 0, a non-int units (bools too), a non-datetime ts or an empty
        customer name."""
        if not customer:
            raise ValueError("customer name cannot be empty")
        if not isinstance(ts, datetime):
            raise ValueError("timestamp must be a datetime object")
        # Check for int, but not bool (since bools are subclasses of int)
        if not isinstance(units, int) or isinstance(units, bool):
            raise ValueError("units must be an integer (not a boolean)")
        if units == 0:
            raise ValueError("units value cannot be 0")

        # Original code had abs(units), which was incorrect for refunds.
        # Store units as-is to allow negative values for refunds.
        self.entries.append((customer, ts, units))

    def customers(self):
        """Sorted list of every customer that ever had an entry."""
        return sorted({c for c, _, _ in self.entries})

from datetime import datetime


def month_bounds(year, month):
    """Return (start, end) with start = first instant of the month and end = first instant of the NEXT month; an entry
    belongs to the month iff start <= ts < end. month 1..12, otherwise ValueError."""
    if not (1 <= month <= 12):
        raise ValueError("month must be between 1 and 12")

    start = datetime(year, month, 1)
    # Handle December to January transition for end date
    if month == 12:
        end = datetime(year + 1, 1, 1)
    else:
        end = datetime(year, month + 1, 1)
    return start, end

from period import month_bounds


def monthly_report(ledger, year, month, customers=None):
    """Dict customer -> net units for the month (refunds subtract). Includes every customer in `customers` (default:
    all customers in the ledger) with 0 when they have no entries that month."""
    start, end = month_bounds(year, month)

    if customers is None:
        # Get all customers that ever had an entry in the ledger
        all_customers = ledger.customers()
    else:
        # Use the provided list of customers. Ensure it's iterable and sorted for consistent behavior.
        all_customers = sorted(list(customers))

    # Initialize output dictionary with all relevant customers and zero usage
    out = {c: 0 for c in all_customers}

    for c, ts, u in ledger.entries:
        # An entry belongs to the month iff start <= ts < end.
        # The original code used <= end, which incorrectly included the first instant of the next month.
        if start <= ts < end:
            # Only update if the customer is in the target set (either all_customers or the provided 'customers')
            if c in out:
                out[c] += u # Units can be negative for refunds, correctly subtracting
    return out
