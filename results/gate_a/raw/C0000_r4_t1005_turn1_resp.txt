class Groups:
    """Group membership. Groups may contain users and other groups (nesting). Cycles must not hang."""

    def __init__(self):
        self.members = {}

    def add(self, group, member):
        self.members.setdefault(group, set()).add(member)

    def remove(self, group, member):
        """Remove member from group; removing a non-member is a no-op. An emptied group still exists."""
        if group in self.members:
            self.members[group].discard(member)

    def groups_of(self, user):
        """Set of all groups the user belongs to, directly or through nested groups."""
        # Use a worklist algorithm to find all groups the user belongs to, directly or indirectly.
        # This handles cycles by only processing each group once.
        user_groups = set()
        worklist = {user} # Start with the user itself

        while worklist:
            current_member = worklist.pop()
            for group_name, members_set in self.members.items():
                if current_member in members_set and group_name not in user_groups:
                    user_groups.add(group_name)
                    worklist.add(group_name) # Add the group itself to the worklist to check for nesting

        return user_groups

def allowed(groups, acl, user, path, action):
    """acl: list of (path_prefix, subject, action, effect) with effect "allow" or "deny"; subject is a user name or group
    name. A rule applies if its prefix is a path-segment prefix of path ("/a" covers "/a" and "/a/b" but NOT "/ab"; "/" covers
    everything), its action equals `action` or is "*", and its subject is the user or one of the user's groups (nested
    included). The MOST SPECIFIC (longest) matching prefix decides; at equal length deny beats allow. No match -> False."""
    best_match_len = -1
    best_effect = None

    # Pre-calculate all groups the user belongs to, including nested ones, for efficiency.
    user_and_groups = {user} | groups.groups_of(user)

    for prefix, subject, act, effect in acl:
        # Check path prefix match
        if not path.startswith(prefix):
            continue
        # Ensure prefix matches whole path segments (e.g., "/a" matches "/a/b" but not "/ab")
        if len(path) > len(prefix) and path[len(prefix)] != '/':
            continue
        if len(path) == len(prefix) and path != prefix: # Exact match required if lengths are equal
            continue

        # Check action match
        if act != action and act != "*":
            continue

        # Check subject match
        if subject not in user_and_groups:
            continue

        # This rule is a candidate. Determine if it's the "best" so far.
        current_len = len(prefix)

        if current_len > best_match_len:
            # This rule is more specific (longer prefix)
            best_match_len = current_len
            best_effect = effect
        elif current_len == best_match_len:
            # Same specificity, apply "deny beats allow" tie-breaker
            if effect == "deny" and best_effect == "allow":
                best_effect = effect # Deny overrides allow
            # If both are deny, or both are allow, or current is allow and best is deny, keep best_effect.

    return best_effect == "allow"
