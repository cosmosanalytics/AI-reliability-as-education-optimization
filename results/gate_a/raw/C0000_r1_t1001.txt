import decimal

def to_cents(x):
    """Convert a dollar amount (int, float or str such as "1.10") to integer cents. Round half away from zero at the
    cent: 0.285 -> 29, 2.675 -> 268, -1.005 -> -101. Never use binary float arithmetic for the conversion."""
    # Convert float inputs to string representation to avoid binary float precision issues
    # when creating a Decimal object.
    if isinstance(x, float):
        x = str(x)
    
    # Convert to Decimal, multiply by 100 to get cents as a Decimal,
    # then quantize to a whole number using ROUND_HALF_UP.
    # ROUND_HALF_UP rounds .5 away from zero (e.g., 28.5 -> 29, -100.5 -> -101).
    cents_decimal = decimal.Decimal(x) * 100
    return int(cents_decimal.quantize(decimal.Decimal('0'), rounding=decimal.ROUND_HALF_UP))


def fmt(cents):
    """Format integer cents as dollars: 1999 -> "$19.99", 5 -> "$0.05", -150 -> "-$1.50", -5 -> "-$0.05"."""
    # Use divmod for correct handling of negative cents for both quotient and remainder
    dollars, remainder_cents = divmod(abs(cents), 100)
    sign = "-" if cents < 0 else ""
    return f"{sign}${dollars}.{remainder_cents:02d}"

from money import to_cents
import decimal

def line_total(price, qty):
    """Cents for qty items at a dollar price."""
    # Ensure price is converted to cents precisely before multiplication
    return to_cents(price) * qty


def invoice_total(lines, discount_pct=0):
    """lines is a list of (price_in_dollars, qty). discount_pct (0-100, may be fractional) is taken off the subtotal;
    the discount amount is rounded half away from zero to a whole cent. Returns total cents."""
    sub = sum(line_total(p, q) for p, q in lines)
    
    # Calculate discount using Decimal for precision and apply specified rounding.
    # Convert subtotal to Decimal for calculation.
    sub_decimal = decimal.Decimal(sub)
    discount_pct_decimal = decimal.Decimal(str(discount_pct)) # Convert to string to avoid float issues
    
    discount_amount_decimal = sub_decimal * discount_pct_decimal / 100
    
    # Round discount amount half away from zero to a whole cent.
    # This is equivalent to quantizing to 0 decimal places with ROUND_HALF_UP.
    rounded_discount_amount = int(discount_amount_decimal.quantize(decimal.Decimal('0'), rounding=decimal.ROUND_HALF_UP))
    
    return sub - rounded_discount_amount

import decimal

def vat(cents, rate):
    """VAT in cents on `cents` at fractional `rate` (0.2 = 20%), rounded half away from zero to a whole cent."""
    # Calculate VAT using Decimal for precision and apply specified rounding.
    # Convert cents and rate to Decimal for calculation.
    cents_decimal = decimal.Decimal(cents)
    rate_decimal = decimal.Decimal(str(rate)) # Convert to string to avoid float issues
    
    vat_amount_decimal = cents_decimal * rate_decimal
    
    # Round VAT amount half away from zero to a whole cent.
    # This is equivalent to quantizing to 0 decimal places with ROUND_HALF_UP.
    return int(vat_amount_decimal.quantize(decimal.Decimal('0'), rounding=decimal.ROUND_HALF_UP))


def gross(cents, rate):
    """cents plus its VAT."""
    return cents + vat(cents, rate)
