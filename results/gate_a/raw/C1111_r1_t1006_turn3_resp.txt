from datetime import datetime


class Ledger:
    """Append-only usage ledger. Entries: (customer, timestamp (naive datetime), units (int, negative for refunds))."""

    def __init__(self):
        self.entries = []

    def record(self, customer, ts, units):
        """Record an entry. Reject (ValueError) a units value of 0, a non-int units (bools too), a non-datetime ts or an empty
        customer name."""
        # Bug fix: units should be stored as-is, not absolute value, to correctly handle refunds.
        self.entries.append((customer, ts, units))

    def customers(self):
        """Sorted list of every customer that ever had an entry."""
        return sorted({c for c, _, _ in self.entries})

from period import month_bounds


def monthly_report(ledger, year, month, customers=None):
    """Dict customer -> net units for the month (refunds subtract). Includes every customer in `customers` (default:
    all customers in the ledger) with 0 when they have no entries that month."""
    start, end = month_bounds(year, month)

    # Determine the set of customers to include in the report and initialize their usage to 0.
    # If 'customers' is None, use all customers from the ledger.
    # Otherwise, use the provided 'customers' list/set.
    if customers is None:
        report_customers = ledger.customers()
    else:
        # Ensure we work with a set for efficient lookup and then convert to a sorted list
        # to maintain consistent output order if the input 'customers' was unsorted.
        report_customers = sorted(list(customers))

    out = {c: 0 for c in report_customers}

    for c, ts, u in ledger.entries:
        # Bug fix: an entry belongs to the month iff start <= ts < end,
        # not start <= ts <= end, to exclude entries on the first instant of the next month.
        if start <= ts < end:
            # Only add to customers that are part of the report_customers set
            if c in out:
                out[c] += u
    return out
