import decimal


def to_cents(x):
    """Convert a dollar amount (int, float or str such as "1.10") to integer cents. Round half away from zero at the
    cent: 0.285 -> 29, 2.675 -> 268, -1.005 -> -101. Never use binary float arithmetic for the conversion."""
    # Convert input to Decimal, handling float inputs safely via string conversion
    if isinstance(x, float):
        d = decimal.Decimal(str(x))
    else:
        d = decimal.Decimal(x)

    # Multiply by 100 to get cents as a Decimal
    cents_decimal = d * 100

    # Round half away from zero
    # For positive numbers, ROUND_HALF_UP rounds .5 up.
    # For negative numbers, ROUND_HALF_DOWN rounds .5 away from zero.
    if cents_decimal >= 0:
        rounded_cents = cents_decimal.to_integral_value(rounding=decimal.ROUND_HALF_UP)
    else:
        rounded_cents = cents_decimal.to_integral_value(rounding=decimal.ROUND_HALF_DOWN)

    return int(rounded_cents)


def fmt(cents):
    """Format integer cents as dollars: 1999 -> "$19.99", 5 -> "$0.05", -150 -> "-$1.50", -5 -> "-$0.05"."""
    return "$%d.%02d" % (cents // 100, abs(cents) % 100)

from money import to_cents
import decimal


def line_total(price, qty):
    """Cents for qty items at a dollar price."""
    return to_cents(price) * qty


def invoice_total(lines, discount_pct=0):
    """lines is a list of (price_in_dollars, qty). discount_pct (0-100, may be fractional) is taken off the subtotal;
    the discount amount is rounded half away from zero to a whole cent. Returns total cents."""
    sub = sum(line_total(p, q) for p, q in lines)

    if discount_pct == 0:
        return sub

    # Convert subtotal to Decimal for precise calculation
    sub_decimal = decimal.Decimal(sub)
    discount_pct_decimal = decimal.Decimal(str(discount_pct))

    # Calculate discount amount
    discount_value = sub_decimal * discount_pct_decimal / decimal.Decimal('100')

    # Round discount amount half away from zero
    if discount_value >= 0:
        rounded_discount = discount_value.to_integral_value(rounding=decimal.ROUND_HALF_UP)
    else:
        rounded_discount = discount_value.to_integral_value(rounding=decimal.ROUND_HALF_DOWN)

    return int(sub_decimal - rounded_discount)

import decimal


def vat(cents, rate):
    """VAT in cents on `cents` at fractional `rate` (0.2 = 20%), rounded half away from zero to a whole cent."""
    # Convert cents to Decimal for precise calculation
    cents_decimal = decimal.Decimal(cents)
    rate_decimal = decimal.Decimal(str(rate))

    # Calculate VAT amount
    vat_amount = cents_decimal * rate_decimal

    # Round VAT amount half away from zero
    if vat_amount >= 0:
        rounded_vat = vat_amount.to_integral_value(rounding=decimal.ROUND_HALF_UP)
    else:
        rounded_vat = vat_amount.to_integral_value(rounding=decimal.ROUND_HALF_DOWN)

    return int(rounded_vat)


def gross(cents, rate):
    """cents plus its VAT."""
    return cents + vat(cents, rate)
