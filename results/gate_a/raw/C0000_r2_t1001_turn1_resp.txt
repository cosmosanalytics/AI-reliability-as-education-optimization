from decimal import Decimal, ROUND_HALF_UP, ROUND_HALF_DOWN


def to_cents(x):
    """Convert a dollar amount (int, float or str such as "1.10") to integer cents. Round half away from zero at the
    cent: 0.285 -> 29, 2.675 -> 268, -1.005 -> -101. Never use binary float arithmetic for the conversion."""
    # Convert input to Decimal, ensuring floats are converted via string to preserve precision
    if isinstance(x, float):
        d_val = Decimal(str(x))
    else:
        d_val = Decimal(x)

    # Multiply by 100 to get cents as a Decimal
    cents_decimal = d_val * 100

    # Apply "round half away from zero"
    # For positive numbers, round half up (e.g., 28.5 -> 29)
    # For negative numbers, round half down (e.g., -100.5 -> -101)
    if cents_decimal >= 0:
        rounded_cents = cents_decimal.quantize(Decimal('1'), rounding=ROUND_HALF_UP)
    else:
        rounded_cents = cents_decimal.quantize(Decimal('1'), rounding=ROUND_HALF_DOWN)

    return int(rounded_cents)


def fmt(cents):
    """Format integer cents as dollars: 1999 -> "$19.99", 5 -> "$0.05", -150 -> "-$1.50", -5 -> "-$0.05"."""
    sign = "-" if cents < 0 else ""
    abs_cents = abs(cents)
    dollars = abs_cents // 100
    remaining_cents = abs_cents % 100
    return "%s$%d.%02d" % (sign, dollars, remaining_cents)

from money import to_cents
from decimal import Decimal, ROUND_HALF_UP, ROUND_HALF_DOWN


def line_total(price, qty):
    """Cents for qty items at a dollar price."""
    return to_cents(price) * qty


def invoice_total(lines, discount_pct=0):
    """lines is a list of (price_in_dollars, qty). discount_pct (0-100, may be fractional) is taken off the subtotal;
    the discount amount is rounded half away from zero to a whole cent. Returns total cents."""
    sub = sum(line_total(p, q) for p, q in lines) # sub is in cents (integer)

    if discount_pct == 0:
        return sub

    # Convert sub and discount_pct to Decimal for precise calculation
    sub_d = Decimal(sub)
    # Ensure discount_pct is converted via string if it's a float to preserve precision
    if isinstance(discount_pct, float):
        discount_pct_d = Decimal(str(discount_pct))
    else:
        discount_pct_d = Decimal(discount_pct)

    # Calculate discount amount
    discount_amount_d = sub_d * discount_pct_d / Decimal('100')

    # Round discount amount half away from zero
    if discount_amount_d >= 0:
        rounded_discount = discount_amount_d.quantize(Decimal('1'), rounding=ROUND_HALF_UP)
    else:
        rounded_discount = discount_amount_d.quantize(Decimal('1'), rounding=ROUND_HALF_DOWN)

    return sub - int(rounded_discount)

from decimal import Decimal, ROUND_HALF_UP, ROUND_HALF_DOWN

def vat(cents, rate):
    """VAT in cents on `cents` at fractional `rate` (0.2 = 20%), rounded half away from zero to a whole cent."""
    # cents is an integer. rate may be fractional (e.g., 0.2)
    # Convert to Decimal for precise calculation
    cents_d = Decimal(cents)
    # Ensure rate is converted via string if it's a float to preserve precision
    if isinstance(rate, float):
        rate_d = Decimal(str(rate))
    else:
        rate_d = Decimal(rate)

    vat_amount_d = cents_d * rate_d

    # Round VAT amount half away from zero
    if vat_amount_d >= 0:
        rounded_vat = vat_amount_d.quantize(Decimal('1'), rounding=ROUND_HALF_UP)
    else:
        rounded_vat = vat_amount_d.quantize(Decimal('1'), rounding=ROUND_HALF_DOWN)

    return int(rounded_vat)


def gross(cents, rate):
    """cents plus its VAT."""
    return cents + vat(cents, rate)
