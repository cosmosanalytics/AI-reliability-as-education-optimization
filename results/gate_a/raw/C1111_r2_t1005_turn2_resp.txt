class Groups:
    """Group membership. Groups may contain users and other groups (nesting). Cycles must not hang."""

    def __init__(self):
        self.members = {}

    def add(self, group, member):
        self.members.setdefault(group, set()).add(member)

    def remove(self, group, member):
        """Remove member from group; removing a non-member is a no-op. An emptied group still exists."""
        if group in self.members:
            self.members[group].discard(member)

    def groups_of(self, user):
        """Set of all groups the user belongs to, directly or through nested groups."""
        # This method needs to handle nested groups and cycles.
        # We'll use a worklist algorithm to find all groups the user is a member of,
        # directly or indirectly.
        
        # The set of groups the user is directly a member of.
        direct_groups = {g for g, m in self.members.items() if user in m}
        
        # The set of all groups the user belongs to, directly or indirectly.
        # Initialize with direct groups.
        all_groups = set(direct_groups)
        
        # Worklist for groups whose members we still need to check.
        # Initialize with direct groups, as these might be members of other groups.
        worklist = list(direct_groups)
        
        # Keep track of visited groups to prevent infinite loops in case of cycles.
        visited = set(direct_groups)

        while worklist:
            current_group = worklist.pop(0) # Use pop(0) for BFS-like traversal

            # Find all groups that contain current_group as a member.
            for parent_group, members_set in self.members.items():
                if current_group in members_set:
                    if parent_group not in visited:
                        all_groups.add(parent_group)
                        visited.add(parent_group)
                        worklist.append(parent_group)
        
        return all_groups

def allowed(groups, acl, user, path, action):
    """acl: list of (path_prefix, subject, action, effect) with effect "allow" or "deny"; subject is a user name or group
    name. A rule applies if its prefix is a path-segment prefix of path ("/a" covers "/a" and "/a/b" but NOT "/ab"; "/" covers
    everything), its action equals `action` or is "*", and its subject is the user or one of the user's groups (nested
    included). The MOST SPECIFIC (longest) matching prefix decides; at equal length deny beats allow. No match -> False."""
    
    # Get all groups the user belongs to, including nested ones.
    user_and_groups = {user} | groups.groups_of(user)

    best_match_len = -1
    best_effect = None # Can be "allow", "deny", or None if no match

    for prefix, subject, act, effect in acl:
        # Check path prefix match
        # "/a" covers "/a" and "/a/b" but NOT "/ab"
        # This means the path must start with the prefix, AND
        # if the prefix is not the entire path, the character immediately after the prefix must be '/'
        if not path.startswith(prefix):
            continue
        if len(path) > len(prefix) and path[len(prefix)] != '/':
            # This handles the "/ab" case for prefix "/a"
            # However, the original description says "/a" covers "/a" and "/a/b".
            # The current `startswith` is correct for this.
            # The "NOT /ab" implies that if prefix is "/a", path "/ab" should not match.
            # The original `path.startswith(prefix)` is actually correct for the description.
            # Let's re-read: "prefix is a path-segment prefix of path".
            # This means if prefix is "/a", then path "/a" matches, "/a/b" matches, but "/ab" does not.
            # The current `path.startswith(prefix)` would match "/ab" for prefix "/a".
            # We need to ensure it's a path segment prefix.
            if prefix != path and not path.startswith(prefix + '/'):
                continue
            # Special case: if prefix is "/", it matches everything.
            # If prefix is "/a" and path is "/a", it matches.
            # If prefix is "/a" and path is "/a/b", it matches.
            # If prefix is "/a" and path is "/ab", it should NOT match.
            # The condition `path.startswith(prefix)` is not sufficient.
            # It should be `path == prefix` OR `path.startswith(prefix + '/')`.
            # The only exception is if prefix is "/", which matches everything.
            if prefix != "/" and not (path == prefix or path.startswith(prefix + '/')):
                continue


        # Check action match
        if act != action and act != "*":
            continue

        # Check subject match
        if subject not in user_and_groups:
            continue

        # This rule applies. Now compare with the best match found so far.
        current_prefix_len = len(prefix)

        if current_prefix_len > best_match_len:
            # More specific prefix, this rule wins.
            best_match_len = current_prefix_len
            best_effect = effect
        elif current_prefix_len == best_match_len:
            # Equal length prefix, "deny" beats "allow".
            if best_effect == "allow" and effect == "deny":
                best_effect = "deny"
            # If best_effect is already "deny", or current effect is "allow" and best is "allow",
            # the existing best_effect remains.
            # The problem description says "deny beats allow", so if we have a deny, it takes precedence.
            # If we find a deny, and the current best is allow, we update.
            # If we find an allow, and the current best is deny, we keep deny.
            # If both are allow, we keep allow.
            # If both are deny, we keep deny.
            # So, if effect is "deny", it always becomes the best_effect for this length.
            if effect == "deny":
                best_effect = "deny"

    return best_effect == "allow"
