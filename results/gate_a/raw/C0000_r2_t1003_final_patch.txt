from datetime import datetime, timedelta
from dateutil.relativedelta import relativedelta


def next_occurrence(start, rule, after):
    """start: naive datetime of the first occurrence. rule: "daily", "weekly" or "monthly". Return the first occurrence
    STRICTLY after `after` (a naive datetime), or start itself if start > after. Occurrences are start + n periods, n>=0.
    Weekly = same weekday and wall-clock time every 7 days. Monthly = same day-of-month and time each month; if the month
    is too short use that month's LAST day (31 Jan -> 28/29 Feb -> 31 Mar, always counted from the ORIGINAL start day, so
    the day returns to 31 in March). Raise ValueError for an unknown rule."""

    # Store the original start day for monthly recurrence logic
    original_start_day = start.day

    # Initialize cur to start. If start is already after `after`, it's the first occurrence.
    # Otherwise, we need to find the next one.
    cur = start

    # If start is already strictly after `after`, it's the first occurrence.
    # The loop below will handle the case where start <= after.
    if cur > after:
        return cur

    # Find the first occurrence that is strictly after `after`.
    # The loop must ensure `cur` becomes strictly greater than `after`.
    # If `start` itself is `after`, we need to advance `cur` at least once.
    # If `start` is before `after`, we need to advance `cur` until it's strictly after `after`.
    while True:
        if rule == "daily":
            cur += timedelta(days=1)
        elif rule == "weekly":
            # Use relativedelta to ensure wall-clock time and weekday are preserved,
            # handling potential DST shifts more robustly than fixed timedelta(days=7)
            # for naive datetimes when the system's interpretation of "wall-clock time"
            # might implicitly involve timezone rules.
            cur += relativedelta(weeks=1)
        elif rule == "monthly":
            # Calculate the next month based on the current date
            next_month_date = cur + relativedelta(months=1)
            
            # Attempt to set the day to the original_start_day
            try:
                cur = next_month_date.replace(day=original_start_day)
            except ValueError:
                # If the original_start_day is too large for the next month,
                # use the last day of that month.
                # This is achieved by replacing day=1 and then adding one month and subtracting one day.
                # Or more simply, by replacing day=1 and then adding relativedelta(months=1, days=-1)
                # to get the last day of the *current* month, but we need the last day of the *next* month.
                # A robust way is to go to the first day of the month after next, then subtract one day.
                cur = (next_month_date.replace(day=1) + relativedelta(months=1)) - timedelta(days=1)
        else:
            raise ValueError(rule)
        
        # Once cur is strictly after `after`, we've found our next occurrence.
        if cur > after:
            return cur

from recur import next_occurrence


def upcoming(start, rule, after, n):
    """The next n occurrences strictly after `after`, as a list."""
    out = []
    cur = after # Start searching from `after`
    for _ in range(n):
        cur = next_occurrence(start, rule, cur)
        out.append(cur)
    return out
