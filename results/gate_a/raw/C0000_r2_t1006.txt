from datetime import datetime


class Ledger:
    """Append-only usage ledger. Entries: (customer, timestamp (naive datetime), units (int, negative for refunds))."""

    def __init__(self):
        self.entries = []

    def record(self, customer, ts, units):
        """Record an entry. Reject (ValueError) a units value of 0, a non-int units (bools too), a non-datetime ts or an empty
        customer name."""
        # Bug fix: units should be stored as provided, not absolute, to correctly handle refunds.
        self.entries.append((customer, ts, units))

    def customers(self):
        """Sorted list of every customer that ever had an entry."""
        return sorted({c for c, _, _ in self.entries})

from period import month_bounds


def monthly_report(ledger, year, month, customers=None):
    """Dict customer -> net units for the month (refunds subtract). Includes every customer in `customers` (default:
    all customers in the ledger) with 0 when they have no entries that month."""
    start, end = month_bounds(year, month)

    # Bug fix: Initialize all relevant customers to 0, even if they have no usage.
    # If 'customers' is provided, use that list; otherwise, use all customers from the ledger.
    all_customers_for_report = customers if customers is not None else ledger.customers()
    out = {c: 0 for c in all_customers_for_report}

    for c, ts, u in ledger.entries:
        # Bug fix: The upper bound for the month should be exclusive (`< end`), not inclusive (`<= end`).
        # An entry at the first instant of the next month belongs to the next month, not the current one.
        if start <= ts < end:
            # Only add to customers that are part of the report scope
            if c in out:
                out[c] = out.get(c, 0) + u
    return out
