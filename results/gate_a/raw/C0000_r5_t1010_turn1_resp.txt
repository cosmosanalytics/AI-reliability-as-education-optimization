class Event:
    """Event object passed to subscribers."""

    def __init__(self, name, data=None):
        self.name = name
        self.data = data or {}
        self.cancelled = False
        self._propagation_stopped = False

    def stop_propagation(self):
        """Halt dispatching to all subsequent subscribers immediately."""
        self._propagation_stopped = True

    @property
    def propagation_stopped(self):
        """Check if propagation has been stopped."""
        return self._propagation_stopped

from subscriber import Subscriber


class EventDispatcher:
    def __init__(self):
        # _subscribers stores {event_name: {priority: [Subscriber, ...]}}
        self._subscribers = {}

    def subscribe(self, name, handler, priority=0, once=False):
        sub = Subscriber(handler, priority, once)
        self._subscribers.setdefault(name, {}).setdefault(priority, []).append(sub)
        return sub

    def unsubscribe(self, name, handler):
        if name in self._subscribers:
            for priority in list(self._subscribers[name].keys()): # Iterate over a copy of keys
                self._subscribers[name][priority] = [
                    s for s in self._subscribers[name][priority] if s.handler != handler
                ]
                if not self._subscribers[name][priority]:
                    del self._subscribers[name][priority]
            if not self._subscribers[name]:
                del self._subscribers[name]

    def dispatch(self, event):
        """Dispatch event to subscribers sorted by priority (descending).
        Equal priority subscribers must execute in FIFO order.
        If event.propagation_stopped is set, halt execution immediately."""
        if event.name not in self._subscribers:
            return

        # Create a flattened, sorted list of active subscribers for this dispatch.
        # This snapshot ensures that modifications to _subscribers during dispatch
        # (e.g., via unsubscribe or s.once) do not cause RuntimeError or skip/re-execute
        # handlers incorrectly within the current dispatch loop.
        # Subscribers are sorted by priority (descending), then by their original
        # registration order (FIFO for equal priority).
        active_subs = []
        for priority in sorted(self._subscribers[event.name].keys(), reverse=True):
            # Append a copy of the list to maintain FIFO order for equal priorities
            active_subs.extend(list(self._subscribers[event.name][priority]))

        # Keep track of handlers to unsubscribe after the dispatch loop
        # to avoid modifying the _subscribers structure during iteration.
        handlers_to_unsubscribe = []

        for s in active_subs:
            if event.propagation_stopped:
                break

            # Check if the subscriber is still active in the main _subscribers structure.
            # This handles cases where a handler unsubscribes another handler
            # that is later in the active_subs snapshot.
            # This check is efficient because _subscribers is now structured by priority.
            if s not in self._subscribers.get(event.name, {}).get(s.priority, []):
                continue

            s.handler(event)

            if s.once:
                handlers_to_unsubscribe.append((event.name, s.handler))

        # Perform unsubscribes for 'once' handlers after the dispatch loop
        for name, handler in handlers_to_unsubscribe:
            self.unsubscribe(name, handler)
