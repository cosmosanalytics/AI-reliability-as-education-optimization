from datetime import datetime


class Ledger:
    """Append-only usage ledger. Entries: (customer, timestamp (naive datetime), units (int, negative for refunds))."""

    def __init__(self):
        self.entries = []

    def record(self, customer, ts, units):
        """Record an entry. Reject (ValueError) a units value of 0, a non-int units (bools too), a non-datetime ts or an empty
        customer name."""
        # BUG FIX: Do not take absolute value of units. This ensures negative units (refunds) are recorded correctly.
        self.entries.append((customer, ts, units))

    def customers(self):
        """Sorted list of every customer that ever had an entry."""
        return sorted({c for c, _, _ in self.entries})

from period import month_bounds


def monthly_report(ledger, year, month, customers=None):
    """Dict customer -> net units for the month (refunds subtract). Includes every customer in `customers` (default:
    all customers in the ledger) with 0 when they have no entries that month."""
    start, end = month_bounds(year, month)
    
    # BUG FIX: Initialize the report dictionary with all relevant customers, setting their initial usage to 0.
    # This ensures customers with no usage in the specified month are included in the report with a zero value.
    all_customers_for_report = customers if customers is not None else ledger.customers()
    out = {c: 0 for c in all_customers_for_report}
    
    for c, ts, u in ledger.entries:
        # BUG FIX: Correct time boundary check. An entry belongs to the month if its timestamp `ts` is
        # greater than or equal to `start` and strictly less than `end`.
        # The original `ts <= end` incorrectly included the first instant of the *next* month.
        if start <= ts < end:
            # Only update if the customer is part of the requested report (either all ledger customers or specific 'customers' list).
            if c in out:
                out[c] += u # `u` can now be negative for refunds due to fix in ledger.py
    return out
