def write_row(fields):
    """Encode one record as a CSV line WITHOUT trailing newline. Fields are str/int/float/None. None -> empty field.
    A field is quoted (with double quotes, inner quotes doubled) iff it contains a comma, a double quote, CR or LF, or has
    leading/trailing spaces. Numbers use str(). Spreadsheet-injection guard: a str field starting with =, +, - or @
    (but a str that parses as a plain number such as "-5" is left alone) gets a leading apostrophe prepended BEFORE quoting."""
    processed_fields = []
    for f in fields:
        if f is None:
            field_str = ""
        elif isinstance(f, (int, float)):
            field_str = str(f)
        else: # Assume it's a string or convertible to string
            field_str = str(f)

            # Apply spreadsheet injection guard
            is_numeric = False
            try:
                # Check if it parses as a plain number
                float(field_str)
                is_numeric = True
            except ValueError:
                pass

            if field_str.startswith('='):
                field_str = "'" + field_str
            elif field_str.startswith('@'):
                field_str = "'" + field_str
            elif (field_str.startswith('+') or field_str.startswith('-')) and not is_numeric:
                field_str = "'" + field_str

        # Determine if quoting is needed
        needs_quoting = (
            ',' in field_str or
            '"' in field_str or
            '\r' in field_str or
            '\n' in field_str or
            field_str.strip() != field_str # leading/trailing spaces
        )

        if needs_quoting:
            field_str = '"' + field_str.replace('"', '""') + '"'
        processed_fields.append(field_str)

    return ",".join(processed_fields)


def read_rows(text):
    """Parse CSV text (rows separated by \n or \r\n, quoted fields may contain commas, quotes (doubled) and newlines) into a
    list of lists of str. A trailing newline does not create an empty row. An empty field is "". Raise ValueError on an
    unterminated quote."""
    if not text:
        return []

    rows = []
    current_row_fields = []
    current_field_chars = []
    in_quoted_field = False
    i = 0
    while i < len(text):
        char = text[i]

        if in_quoted_field:
            if char == '"':
                # Check for doubled quote (escaped quote)
                if i + 1 < len(text) and text[i+1] == '"':
                    current_field_chars.append('"')
                    i += 1 # Consume the second quote
                else:
                    # End of quoted field
                    in_quoted_field = False
            else:
                current_field_chars.append(char)
        else: # Not in a quoted field
            if char == '"':
                # Start of a quoted field
                if current_field_chars:
                    # Malformed CSV: quote found in unquoted field
                    raise ValueError(f"Malformed CSV: quote found in unquoted field at position {i}")
                in_quoted_field = True
            elif char == ',':
                current_row_fields.append("".join(current_field_chars))
                current_field_chars = []
            elif char == '\n' or char == '\r':
                current_row_fields.append("".join(current_field_chars)) # Add last field of the row
                current_field_chars = []
                rows.append(current_row_fields)
                current_row_fields = [] # Reset for next row

                if char == '\r' and i + 1 < len(text) and text[i+1] == '\n':
                    i += 1 # Consume the \n
            else:
                current_field_chars.append(char)
        i += 1

    if in_quoted_field:
        raise ValueError("Unterminated quote at end of text")

    # Handle the very last field/row if not terminated by a newline
    if current_field_chars or current_row_fields:
        current_row_fields.append("".join(current_field_chars))
        rows.append(current_row_fields)

    # "A trailing newline does not create an empty row."
    # If the input ends with a newline, and the last row in `rows` is `[""]`, then remove that last row.
    if rows and len(rows[-1]) == 1 and rows[-1][0] == "" and text.endswith(('\n', '\r')):
        rows.pop()

    return rows
