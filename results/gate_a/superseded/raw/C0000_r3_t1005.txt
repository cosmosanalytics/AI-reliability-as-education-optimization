class Groups:
    """Group membership. Groups may contain users and other groups (nesting). Cycles must not hang."""

    def __init__(self):
        self.members = {}

    def add(self, group, member):
        self.members.setdefault(group, set()).add(member)

    def remove(self, group, member):
        """Remove member from group; removing a non-member is a no-op. An emptied group still exists."""
        # Fix for issue #1: Actually remove the member from the group's set.
        if group in self.members:
            self.members[group].discard(member) # Use discard to avoid KeyError if member is not present

    def groups_of(self, user):
        """Set of all groups the user belongs to, directly or through nested groups."""
        # Fix for issue #3: Traverse nested groups using a BFS-like approach.
        all_groups = set()
        q = [user] # Queue for members to process (can be users or groups)
        visited_members = set() # To prevent cycles and redundant processing

        while q:
            current_member = q.pop(0) # Dequeue the current member

            if current_member in visited_members:
                continue
            visited_members.add(current_member)

            # Find all groups where 'current_member' is a direct member
            for group_name, direct_members in self.members.items():
                if current_member in direct_members:
                    if group_name not in all_groups: # Only add to queue if this is a new group found
                        all_groups.add(group_name)
                        q.append(group_name) # Add the group itself to the queue to find its parent groups

        return all_groups

def allowed(groups, acl, user, path, action):
    """acl: list of (path_prefix, subject, action, effect) with effect "allow" or "deny"; subject is a user name or group
    name. A rule applies if its prefix is a path-segment prefix of path ("/a" covers "/a" and "/a/b" but NOT "/ab"; "/" covers
    everything), its action equals `action` or is "*", and its subject is the user or one of the user's groups (nested
    included). The MOST SPECIFIC (longest) matching prefix decides; at equal length deny beats allow. No match -> False."""
    
    candidate_rules = []
    # Optimize by getting all user's groups once, and include the user itself as a subject.
    user_and_groups = groups.groups_of(user)
    user_and_groups.add(user)

    for prefix, subject, act, effect in acl:
        # 1. Path prefix matching: "/a" covers "/a" and "/a/b" but NOT "/ab"
        if not path.startswith(prefix):
            continue
        # Ensure it's a path-segment prefix (e.g., "/a" matches "/a/b" but not "/ab")
        if prefix != "/" and len(path) > len(prefix) and path[len(prefix)] != '/':
            continue

        # 2. Action matching: "its action equals `action` or is `*`"
        if act != action and act != "*":
            continue

        # 3. Subject matching: "its subject is the user or one of the user's groups (nested included)"
        if subject not in user_and_groups:
            continue
        
        # If all checks pass, this rule is a candidate
        candidate_rules.append((len(prefix), effect))

    if not candidate_rules:
        return False

    # Find the maximum prefix length among all candidate rules
    max_len = max(rule_len for rule_len, _ in candidate_rules)

    # Filter rules to only those with the maximum prefix length
    longest_rules_effects = [effect for rule_len, effect in candidate_rules if rule_len == max_len]

    # Fix for issue #2: Apply "deny beats allow" tie-breaker for equal length prefixes
    if "deny" in longest_rules_effects:
        return False # If any "deny" rule exists among the longest, it takes precedence
    elif "allow" in longest_rules_effects:
        return True # Otherwise, if any "allow" rule exists, it allows access
    else:
        # This case should ideally not be reached if candidate_rules was not empty
        # and max_len was found, as effects are either "allow" or "deny".
        return False
