import decimal

def _round_half_away_from_zero_to_int(d_val):
    """Helper to round a Decimal value half away from zero to the nearest integer."""
    if d_val >= 0:
        # For positive numbers, ROUND_HALF_UP rounds ties away from zero.
        return int(d_val.quantize(decimal.Decimal('1'), rounding=decimal.ROUND_HALF_UP))
    else:
        # For negative numbers, take absolute value, round, then reapply negative sign.
        # Example: -100.5 -> abs(100.5) -> rounds to 101 -> reapply sign -> -101.
        abs_d_val = d_val.copy_abs()
        rounded_abs = abs_d_val.quantize(decimal.Decimal('1'), rounding=decimal.ROUND_HALF_UP)
        return int(rounded_abs.copy_sign(d_val))


def to_cents(x):
    """Convert a dollar amount (int, float or str such as "1.10") to integer cents. Round half away from zero at the
    cent: 0.285 -> 29, 2.675 -> 268, -1.005 -> -101. Never use binary float arithmetic for the conversion."""
    # Convert input to Decimal. Safest for floats is str(x) to avoid binary float precision issues.
    if isinstance(x, float):
        d_val = decimal.Decimal(str(x))
    else:
        d_val = decimal.Decimal(x)

    # Multiply by 100 to get cents as a Decimal
    cents_decimal = d_val * 100

    # Round to the nearest integer cent using half away from zero rule
    return _round_half_away_from_zero_to_int(cents_decimal)


def fmt(cents):
    """Format integer cents as dollars: 1999 -> "$19.99", 5 -> "$0.05", -150 -> "-$1.50", -5 -> "-$0.05"."""
    sign = "-" if cents < 0 else ""
    abs_cents = abs(cents)
    return "%s$%d.%02d" % (sign, abs_cents // 100, abs_cents % 100)

import decimal
from money import to_cents, _round_half_away_from_zero_to_int


def line_total(price, qty):
    """Cents for qty items at a dollar price."""
    return to_cents(price) * qty


def invoice_total(lines, discount_pct=0):
    """lines is a list of (price_in_dollars, qty). discount_pct (0-100, may be fractional) is taken off the subtotal;
    the discount amount is rounded half away from zero to a whole cent. Returns total cents."""
    sub = sum(line_total(p, q) for p, q in lines)

    # Calculate discount amount using Decimal for precision
    sub_d = decimal.Decimal(sub)
    # Convert discount_pct via string to preserve precision if it's a float
    discount_pct_d = decimal.Decimal(str(discount_pct))

    discount_amount_d = sub_d * discount_pct_d / 100

    # Round the discount amount half away from zero to a whole cent
    rounded_discount = _round_half_away_from_zero_to_int(discount_amount_d)

    return sub - rounded_discount

import decimal
from money import _round_half_away_from_zero_to_int


def vat(cents, rate):
    """VAT in cents on `cents` at fractional `rate` (0.2 = 20%), rounded half away from zero to a whole cent."""
    # Calculate VAT amount using Decimal for precision
    cents_d = decimal.Decimal(cents)
    # Convert rate via string to preserve precision if it's a float
    rate_d = decimal.Decimal(str(rate))

    vat_amount_d = cents_d * rate_d

    # Round the VAT amount half away from zero to a whole cent
    return _round_half_away_from_zero_to_int(vat_amount_d)


def gross(cents, rate):
    """cents plus its VAT."""
    return cents + vat(cents, rate)
