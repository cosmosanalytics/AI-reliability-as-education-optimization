import re
import csv
import io

# Helper for write_row
_SPECIAL_CHARS_RE = re.compile(r'[,"\r\n]')

def _needs_quoting(field_str):
    # A field is quoted iff it contains a comma, a double quote, CR or LF, or has
    # leading/trailing spaces.
    return (
        _SPECIAL_CHARS_RE.search(field_str) is not None
        or field_str.strip() != field_str # leading/trailing spaces
    )

def _apply_spreadsheet_guard(field_str):
    # A str field starting with =, +, - or @
    # (but a str that parses as a plain number such as "-5" is left alone)
    # gets a leading apostrophe prepended BEFORE quoting.
    if field_str and field_str[0] in ('=', '+', '-', '@'):
        try:
            # Check if it's a plain number (int or float).
            # If it can be converted to float, it's considered a number.
            float(field_str)
            return field_str # It's a number, leave it alone
        except ValueError:
            pass # Not a number, apply guard
        return "'" + field_str
    return field_str


def write_row(fields):
    """Encode one record as a CSV line WITHOUT trailing newline. Fields are str/int/float/None. None -> empty field.
    A field is quoted (with double quotes, inner quotes doubled) iff it contains a comma, a double quote, CR or LF, or has
    leading/trailing spaces. Numbers use str(). Spreadsheet-injection guard: a str field starting with =, +, - or @
    (but a str that parses as a plain number such as "-5" is left alone) gets a leading apostrophe prepended BEFORE quoting."""
    processed_fields = []
    for f in fields:
        if f is None:
            field_str = ""
        else:
            field_str = str(f)

        # Apply spreadsheet injection guard
        field_str = _apply_spreadsheet_guard(field_str)

        # Apply quoting rules
        if _needs_quoting(field_str):
            # Double inner quotes
            field_str = field_str.replace('"', '""')
            # Enclose in double quotes
            field_str = '"' + field_str + '"'
        processed_fields.append(field_str)
    return ",".join(processed_fields)


def read_rows(text):
    """Parse CSV text (rows separated by \n or \r\n, quoted fields may contain commas, quotes (doubled) and newlines) into a
    list of lists of str. A trailing newline does not create an empty row. An empty field is "". Raise ValueError on an
    unterminated quote."""
    try:
        # The standard csv.reader handles all specified parsing requirements:
        # - Different line endings (\n or \r\n)
        # - Quoted fields containing commas, quotes (doubled), and newlines
        # - Trailing newlines not creating empty rows
        # - Empty fields as ""
        reader = csv.reader(io.StringIO(text))
        return list(reader)
    except csv.Error as e:
        # Map csv.Error to ValueError as per docstring requirement.
        raise ValueError(f"CSV parsing error: {e}") from e
