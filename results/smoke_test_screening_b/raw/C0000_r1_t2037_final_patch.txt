"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Default service name for audit events if not explicitly provided.
# This should ideally be configurable via environment variables or a configuration system.
# For this patch, a placeholder is used to satisfy the wire envelope requirement.
DEFAULT_SERVICE_NAME = "unknown_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    # The service_name is currently hardcoded as DEFAULT_SERVICE_NAME.
    # In a real application, this should be dynamically provided, e.g.,
    # via an environment variable, a constructor parameter for a class,
    # or passed as an argument to this function.
    return build_envelope(DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # The following lines were not contributing to the return value and
    # were potentially misleading, as `extract_normalized_keys(data)`
    # operates on the original data, not the already-normalized payload
    # within the enveloped message.
    # _ = is_valid_envelope(msg)
    # _ = extract_normalized_keys(data)
    return msg
