"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys
from wire_envelope import build_envelope
from typing import Dict, Any

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    # Rule H08: All dictionary keys inside the payload must be strictly normalized to snake_case.
    normalized_payload = normalize_dict_keys(data)
    
    # Rule H08: The _src identifier is the producing service's domain name
    # (the first word of the module's file name before the underscore) followed by _service.
    # For 'audit_formatter.py', this should be 'audit_service'.
    service_name = "audit_service"
    
    # Rule H08: All internal wire messages must be wrapped in a standardized envelope.
    enveloped_message = build_envelope(service_name, epoch_ms, normalized_payload)
    
    return enveloped_message

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # This function should ensure the audit event is properly formatted and enveloped.
    # The previous intermediate calls to is_valid_envelope and extract_normalized_keys
    # were not modifying the message and their return values were discarded.
    # By calling format_audit_event, we ensure the message adheres to Rule H08.
    return format_audit_event(data, epoch_ms)
