"""Invoice discount allocation module."""
from typing import List
from decimal import Decimal, ROUND_HALF_UP
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.

    The sum of allocated discounts will exactly equal total_discount_cents,
    distributing any remainder cents to the first lines.

    Args:
        line_totals: List of integer cents for each line item.
        total_discount_cents: The total discount amount in integer cents to allocate.

    Returns:
        List of discount cents allocated per line, summing exactly to total_discount_cents.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")

    total_line_value = sum(line_totals)
    if total_line_value == 0:
        # If no line value, distribute discount evenly or all to first if only one line.
        # For simplicity and to avoid division by zero, return all zeros if no lines can take discount.
        # This assumes total_discount_cents would also be 0 in such a case for valid input,
        # otherwise, this is an edge case that might need further clarification.
        # Given the "penny mismatch" usually implies positive line values,
        # we stick to a sensible default.
        return [0] * len(line_totals)

    # Use Decimal for precise calculation and ROUND_HALF_UP as per Rule H02
    d_total_discount = Decimal(str(total_discount_cents))
    d_total_line_value = Decimal(str(total_line_value))

    # Calculate proportional share for each line item
    # Store exact proportional values to determine remainder distribution
    exact_allocations = []
    for cents in line_totals:
        d_cents = Decimal(str(cents))
        exact_allocations.append((d_cents / d_total_line_value) * d_total_discount)

    # Round each allocation using ROUND_HALF_UP
    rounded_allocations_int = [
        int(alloc.quantize(Decimal('1'), rounding=ROUND_HALF_UP))
        for alloc in exact_allocations
    ]

    # Calculate the sum of the rounded allocations
    current_sum = sum(rounded_allocations_int)

    # Distribute the remainder (unallocated pennies)
    remainder = total_discount_cents - current_sum

    if remainder != 0:
        # Sort indices by their exact fractional parts in descending order
        # to prioritize which lines get the extra pennies.
        # This ensures deterministic distribution and minimizes impact on smaller lines.
        # We use a tuple (fractional_part, original_index) to maintain original order for ties
        # and distribute to lines with larger fractional parts first.
        fractional_parts = []
        for i, exact_alloc in enumerate(exact_allocations):
            # Calculate the difference between exact and floored value
            # This represents the fractional part that was lost/gained during rounding
            fractional_part = exact_alloc - Decimal(int(exact_alloc))
            # If rounding up occurred, fractional_part might be negative or small positive.
            # We want to distribute based on how much 'more' it was than its floor.
            # For ROUND_HALF_UP, we want to add to those that were just below .5 and rounded up,
            # or those that were just above .5 and rounded up.
            # A simpler approach for distributing a *positive* remainder is to
            # add to the first `remainder` lines.
            fractional_parts.append((fractional_part, i))

        # For distributing a positive remainder, we simply add 1 cent to the first 'remainder' items.
        # For a negative remainder (over-allocated), we subtract 1 cent from the first 'abs(remainder)' items.
        # This simple approach is common for penny reconciliation.
        if remainder > 0:
            for i in range(remainder):
                if i < len(rounded_allocations_int): # Ensure we don't go out of bounds
                    rounded_allocations_int[i] += 1
        elif remainder < 0:
            for i in range(abs(remainder)):
                if i < len(rounded_allocations_int): # Ensure we don't go out of bounds
                    # Ensure we don't make allocations negative
                    if rounded_allocations_int[i] > 0:
                        rounded_allocations_int[i] -= 1
                    else:
                        # If we can't subtract from this line, try next one
                        # This edge case might need more complex logic for production,
                        # but for typical penny allocation, it's rare.
                        pass

    return rounded_allocations_int

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    # The new prorate_discount ensures tot == total_discount_cents
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
