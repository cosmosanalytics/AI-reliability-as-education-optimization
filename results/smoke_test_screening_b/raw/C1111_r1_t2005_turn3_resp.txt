"""Tenant routing configuration aggregator."""
import copy
from typing import Dict, Any
from models import RouteProfile
from schema import validate_route_keys

def merge_tenant_routing(base_config: Dict[str, Any], tenant_overrides: Dict[str, Any]) -> Dict[str, Any]:
    """Merge tenant overrides into base configuration.
    Inputs must remain strictly immutable; the function must return a newly allocated deep copy.
    
    This function performs a deep merge for nested dictionaries:
    - It starts with a deep copy of the `base_config` to ensure immutability of the original.
    - For each key in `tenant_overrides`:
      - If the key exists in both `merged` (the deep copy of base) and `tenant_overrides`,
        and both values are dictionaries, they are recursively deep-merged. This ensures
        that nested dictionary values from the `tenant_overrides` extend or override
        corresponding values in the `base_config` without losing other base settings.
      - Otherwise (for non-dictionary values or new keys), the value from `tenant_overrides`
        is deep copied and assigned to the `merged` dictionary.
    """
    merged = copy.deepcopy(base_config)
    for key, value in tenant_overrides.items():
        if key in merged and isinstance(merged[key], dict) and isinstance(value, dict):
            # Recursively deep-merge nested dictionaries
            # This ensures that inner dictionaries are also merged, not just replaced.
            merged[key] = merge_tenant_routing(merged[key], value)
        else:
            # For non-dict values or new keys, deep copy the override value
            merged[key] = copy.deepcopy(value)
    return merged

def build_route_profile(tenant_id: str, base_config: Dict[str, Any], tenant_overrides: Dict[str, Any]) -> RouteProfile:
    cfg = merge_tenant_routing(base_config, tenant_overrides)
    # The validation function itself should not mutate the config.
    # It's assumed validate_route_keys performs read-only checks.
    validate_route_keys(cfg)
    return RouteProfile(tenant_id=tenant_id, settings=cfg)
