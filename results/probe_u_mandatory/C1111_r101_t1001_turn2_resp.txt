from decimal import Decimal, ROUND_HALF_UP

def to_cents(x):
    """Convert a dollar amount (int, float or str such as "1.10") to integer cents. Round half away from zero at the
    cent: 0.285 -> 29, 2.675 -> 268, -1.005 -> -101. Never use binary float arithmetic for the conversion."""
    # Convert input to Decimal. For floats, convert to string first to avoid binary floating-point inaccuracies.
    if isinstance(x, float):
        d = Decimal(str(x))
    else:
        d = Decimal(x)

    # Multiply by 100 to get the value in cents as a Decimal.
    cents_decimal = d * 100

    # Round half away from zero.
    # This means for positive numbers, X.5 rounds up (e.g., 2.5 -> 3).
    # For negative numbers, X.5 rounds down (e.g., -2.5 -> -3).
    # The standard library's `decimal` module's `quantize` method with `ROUND_HALF_UP`
    # rounds half towards positive infinity. This works for positive numbers (2.5 -> 3)
    # but for negative numbers, it rounds towards zero (-2.5 -> -2), which is not "away from zero".
    # To achieve "round half away from zero" for both positive and negative numbers,
    # we can check the sign and apply `ROUND_HALF_UP` or `ROUND_HALF_DOWN` accordingly,
    # or use a simpler approach: add 0.5 for positive numbers and subtract 0.5 for negative numbers
    # before truncating (int conversion).

    # A robust way to implement "round half away from zero" for Decimal:
    # Add 0.5 if positive, subtract 0.5 if negative, then truncate.
    # This is equivalent to `int(x + 0.5)` for positive x and `int(x - 0.5)` for negative x.
    # Example:
    # 2.675 * 100 = 267.5 -> 267.5 + 0.5 = 268.0 -> int(268.0) = 268
    # 0.285 * 100 = 28.5 -> 28.5 + 0.5 = 29.0 -> int(29.0) = 29
    # -1.005 * 100 = -100.5 -> -100.5 - 0.5 = -101.0 -> int(-101.0) = -101
    if cents_decimal >= 0:
        return int(cents_decimal + Decimal('0.5'))
    else:
        return int(cents_decimal - Decimal('0.5'))


def fmt(cents):
    """Format integer cents as dollars: 1999 -> "$19.99", 5 -> "$0.05", -150 -> "-$1.50", -5 -> "-$0.05"."""
    # Ensure correct handling of negative cents for modulo and division
    sign = "-" if cents < 0 else ""
    abs_cents = abs(cents)
    dollars = abs_cents // 100
    remaining_cents = abs_cents % 100
    return f"{sign}${dollars}.{remaining_cents:02d}"

from money import to_cents
from decimal import Decimal, ROUND_HALF_UP


def line_total(price, qty):
    """Cents for qty items at a dollar price."""
    return to_cents(price) * qty


def invoice_total(lines, discount_pct=0):
    """lines is a list of (price_in_dollars, qty). discount_pct (0-100, may be fractional) is taken off the subtotal;
    the discount amount is rounded half away from zero to a whole cent. Returns total cents."""
    sub = sum(line_total(p, q) for p, q in lines)

    # Convert subtotal to Decimal for precise discount calculation
    sub_decimal = Decimal(sub)
    discount_pct_decimal = Decimal(str(discount_pct)) # Convert float to string first if it could be float

    # Calculate discount amount
    discount_amount_decimal = sub_decimal * discount_pct_decimal / 100

    # Round discount amount half away from zero.
    # Similar logic as to_cents for rounding.
    if discount_amount_decimal >= 0:
        rounded_discount = int(discount_amount_decimal + Decimal('0.5'))
    else:
        rounded_discount = int(discount_amount_decimal - Decimal('0.5'))

    return sub - rounded_discount

from decimal import Decimal, ROUND_HALF_UP

def vat(cents, rate):
    """VAT in cents on `cents` at fractional `rate` (0.2 = 20%), rounded half away from zero to a whole cent."""
    # Convert cents and rate to Decimal for precise calculation
    cents_decimal = Decimal(cents)
    rate_decimal = Decimal(str(rate)) # Convert float to string first if it could be float

    vat_amount_decimal = cents_decimal * rate_decimal

    # Round VAT amount half away from zero.
    # Similar logic as to_cents for rounding.
    if vat_amount_decimal >= 0:
        return int(vat_amount_decimal + Decimal('0.5'))
    else:
        return int(vat_amount_decimal - Decimal('0.5'))


def gross(cents, rate):
    """cents plus its VAT."""
    return cents + vat(cents, rate)
