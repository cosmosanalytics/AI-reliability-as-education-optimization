from datetime import datetime


class Ledger:
    """Append-only usage ledger. Entries: (customer, timestamp (naive datetime), units (int, negative for refunds))."""

    def __init__(self):
        self.entries = []

    def record(self, customer, ts, units):
        """Record an entry. Reject (ValueError) a units value of 0, a non-int units (bools too), a non-datetime ts or an empty
        customer name."""
        # Bug fix: Do not take absolute value of units, to allow negative units for refunds.
        self.entries.append((customer, ts, units))

    def customers(self):
        """Sorted list of every customer that ever had an entry."""
        return sorted({c for c, _, _ in self.entries})

from period import month_bounds


def monthly_report(ledger, year, month, customers=None):
    """Dict customer -> net units for the month (refunds subtract). Includes every customer in `customers` (default:
    all customers in the ledger) with 0 when they have no entries that month."""
    start, end = month_bounds(year, month)

    # Determine the set of customers to report on.
    # Bug fix: If customers is None, use all customers from the ledger.
    # Bug fix: Ensure customers with no usage in the month are included with 0.
    if customers is None:
        customers_to_report = ledger.customers()
    else:
        # Ensure the provided customers are included, and sort for consistent output if needed.
        # Convert to set for efficient lookup, then back to sorted list for consistent initialization order.
        customers_to_report = sorted(list(set(customers)))

    # Initialize output dictionary with all relevant customers having 0 usage.
    out = {c: 0 for c in customers_to_report}

    for c, ts, u in ledger.entries:
        # Bug fix: An entry belongs to the month iff start <= ts < end.
        # The original code used <= end, which incorrectly included entries from the next month.
        if start <= ts < end:
            # Only update if the customer is in the set of customers to report on.
            # This handles cases where `customers` was provided and an entry exists for a customer not in that list.
            if c in out:
                out[c] = out.get(c, 0) + u
    return out
