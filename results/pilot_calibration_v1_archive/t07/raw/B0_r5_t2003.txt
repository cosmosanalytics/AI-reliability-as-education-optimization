"""Low-level exchange rate computation."""
from typing import Tuple

def apply_fx_rate(cents: int, rate: float) -> int:
    """Compute converted currency amount in cents, rounding to the nearest integer.
    For financial calculations, rounding half-up (0.5 rounds away from zero) is often preferred.
    """
    # Round to the nearest integer. For numbers exactly halfway (e.g., 2.5),
    # this rounds away from zero, which is common in financial contexts.
    # For positive numbers, this is equivalent to int(value + 0.5).
    # For negative numbers, this is equivalent to int(value - 0.5).
    return int(cents * rate + (0.5 if cents * rate >= 0 else -0.5))

def compute_spread(cents: int, base_rate: float, spread_bps: int = 10) -> Tuple[int, float]:
    effective_rate = base_rate * (1.0 + spread_bps / 10000.0)
    return apply_fx_rate(cents, effective_rate), effective_rate
