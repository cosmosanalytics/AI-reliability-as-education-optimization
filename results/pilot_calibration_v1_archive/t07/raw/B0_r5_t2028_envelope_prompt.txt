"""DNS cache resolver client."""
import time
from typing import Dict, Optional, Callable
from dns_record import DnsEntry
from dns_utils import normalize_domain_name

class DnsCache:
    def __init__(self, clock: Optional[Callable[[], float]] = None):
        self._clock = clock or time.time
        self._entries: Dict[str, DnsEntry] = {}

    def put_record(self, domain: str, ip: str, ttl: float) -> None:
        clean_domain = normalize_domain_name(domain)
        now = self._clock()
        self._entries[clean_domain] = DnsEntry(clean_domain, ip, now + ttl)

    def lookup(self, domain: str) -> Optional[str]:
        clean_domain = normalize_domain_name(domain)
        entry = self._entries.get(clean_domain)
        if not entry:
            return None
        now = self._clock()
        if now > entry.expires_at:
            del self._entries[clean_domain]
            return None
        return entry.ip_addr

    def count(self) -> int:
        return len(self._entries)

from dns_resolver import DnsCache

def test_dns_lookup():
    cache = DnsCache()
    cache.put_record("example.com", "93.184.216.34", 10.0)
    assert cache.lookup("example.com") == "93.184.216.34"

def test_clock_advance():
    clock_calls = [0]
    def fake_clock():
        clock_calls[0] += 1
        return 1000.0
    cache = DnsCache(clock=fake_clock)
    cache.put_record("test.org", "1.1.1.1", 5.0)
