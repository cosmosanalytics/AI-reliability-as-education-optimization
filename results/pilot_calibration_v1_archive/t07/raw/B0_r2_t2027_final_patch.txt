"""Lease validity checker."""
def is_lease_valid(valid_until: float, current_time: float) -> bool:
    # A lease is valid up to, but not including, the valid_until timestamp.
    # This prevents split-brain issues where a lease might be considered valid
    # at the exact expiration second, allowing two nodes to simultaneously
    # believe they hold the lock.
    return current_time < valid_until

def remaining_lease_time(valid_until: float, current_time: float) -> float:
    return max(0.0, valid_until - current_time)
