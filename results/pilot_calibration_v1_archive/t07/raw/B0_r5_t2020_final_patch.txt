"""Topic path subscription matcher."""
def is_topic_subscribed(subscribed_pattern: str, event_topic: str) -> bool:
    if subscribed_pattern == event_topic:
        return True
    # Ensure that if the event topic starts with the subscribed pattern, it must be
    # at a segment boundary. This prevents "orders" from matching "orders_admin".
    return event_topic.startswith(subscribed_pattern + '/')

def clean_topic_segment(topic: str) -> str:
    return topic.strip().lower()

def is_wildcard_subscription(pattern: str) -> bool:
    return pattern.strip() in ("*", "#", "/")

from webhook import WebhookEvent
from scope_checker import should_dispatch_webhook

def test_nominal_topic():
    evt = WebhookEvent("orders/created", "{'id': 1}")
    assert should_dispatch_webhook("orders", evt) is True
    assert should_dispatch_webhook("billing", evt) is False

def test_admin_topic_leak_prevention():
    # Test case for EVT-918: "orders" subscription should not receive "orders_admin" events
    admin_evt = WebhookEvent("orders_admin/internal_event", "{'id': 1337}")
    assert should_dispatch_webhook("orders", admin_evt) is False

    # Ensure regular sub-topics still work
    regular_evt = WebhookEvent("orders/refunded", "{'id': 101}")
    assert should_dispatch_webhook("orders", regular_evt) is True

    # Test exact match
    exact_evt = WebhookEvent("orders", "{'id': 102}")
    assert should_dispatch_webhook("orders", exact_evt) is True

    # Test similar prefix but not a segment boundary
    similar_prefix_evt = WebhookEvent("orders_v2/created", "{'id': 200}")
    assert should_dispatch_webhook("orders", similar_prefix_evt) is False
