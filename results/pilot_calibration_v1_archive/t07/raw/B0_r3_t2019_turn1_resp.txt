"""Storage path normalization engine."""
def normalize_bucket_path(path: str) -> str:
    # Ensure path starts with a single '/' and does not end with a '/' unless it's just '/'
    normalized = "/" + path.strip("/")
    return normalized

def validate_prefix_match(scope_prefix: str, asset_path: str) -> bool:
    """
    Validates if an asset's path falls within a given scope prefix, respecting directory boundaries.
    For example, '/assets' should match '/assets/image.png' but not '/assets-private/doc.pdf'.
    Trailing slashes are normalized before comparison.
    """
    normalized_scope = normalize_bucket_path(scope_prefix)
    normalized_asset_path = normalize_bucket_path(asset_path)

    # If the asset path is exactly the scope path, it's a match
    if normalized_asset_path == normalized_scope:
        return True

    # If the asset path starts with the scope path, check if the character
    # immediately following the scope path in the asset path is a directory separator.
    # This enforces directory boundary matching.
    if normalized_asset_path.startswith(normalized_scope):
        # The character after the scope must be a '/' to qualify as being 'inside' the directory
        # e.g., for scope='/assets', asset='/assets/image.png', then normalized_scope='/assets'
        # normalized_asset_path[len(normalized_scope)] would be '/'
        # For scope='/assets', asset='/assets-private/doc.pdf', then normalized_scope='/assets'
        # normalized_asset_path[len(normalized_scope)] would be '-'
        if len(normalized_asset_path) > len(normalized_scope) and \
           normalized_asset_path[len(normalized_scope)] == '/':
            return True
    return False

def get_scope_depth(path: str) -> int:
    return len([seg for seg in path.split("/") if seg])
