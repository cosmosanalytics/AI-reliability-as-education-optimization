"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total = sum(line_totals)
    if total == 0:
        return [0] * len(line_totals)

    # Calculate base allocations using float division to preserve precision
    # and store the fractional parts.
    allocations_with_fractions = []
    for cents in line_totals:
        allocations_with_fractions.append((cents * total_discount_cents) / total)

    # Get the integer part of each allocation
    allocs = [int(alloc) for alloc in allocations_with_fractions]

    # Calculate the remaining cents to distribute
    remaining_cents = total_discount_cents - sum(allocs)

    # Distribute the remaining cents one by one to the lines with the largest fractional parts
    # To do this, we sort the lines by their fractional part in descending order.
    # We use a list of tuples (fractional_part, original_index) to maintain order.
    fractional_parts = [(alloc - int(alloc), i) for i, alloc in enumerate(allocations_with_fractions)]
    fractional_parts.sort(key=lambda x: x[0], reverse=True)

    # Add one cent to the lines with the largest fractional parts until remaining_cents is 0
    for i in range(remaining_cents):
        # The original index is stored in the second element of the tuple
        original_index = fractional_parts[i][1]
        allocs[original_index] += 1

    return allocs

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
