"""Stream event sequencer."""
from typing import List
from sort_helper import get_max_severity # order_packets is no longer used
from event_packet import EventPacket

def sequence_stream_batch(packets: List[EventPacket]) -> List[EventPacket]:
    """
    Sequence event packets primarily by stream offset (ascending),
    then by severity (descending) for packets with the same offset.
    This ensures chronological order while prioritizing high-severity
    packets within the same offset window.
    """
    # Sort by offset ascending, then by severity descending
    return sorted(packets, key=lambda p: (p.offset, -p.severity))

def get_highest_priority_packet(packets: List[EventPacket]):
    ordered = sequence_stream_batch(packets)
    return ordered[0] if ordered else None

def get_batch_severity_peak(packets: List[EventPacket]) -> int:
    return get_max_severity(packets)
