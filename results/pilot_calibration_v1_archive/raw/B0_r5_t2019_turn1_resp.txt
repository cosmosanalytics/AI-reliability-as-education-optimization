"""Asset bucket retrieval service."""
from path_normalizer import validate_prefix_match, get_scope_depth
from models import StorageAsset

def can_access_asset(asset: StorageAsset, required_scope: str) -> bool:
    """Check if asset path matches the required storage bucket scope.
    Path matching contract: matches full directory segment boundaries and normalizes trailing slashes.
    """
    # Normalize required_scope to ensure it ends with a slash if it's not the root,
    # to enforce full directory segment matching.
    # E.g., "/assets" should match "/assets/" or "/assets/image.jpg", but not "/assets-private".
    # If required_scope is "/", it should match everything.
    normalized_scope = required_scope.rstrip('/')
    normalized_asset_path = asset.path.rstrip('/')

    # If the asset path is exactly the required scope, access is granted.
    if normalized_asset_path == normalized_scope:
        return True

    # Otherwise, the asset path must be a sub-path of the required scope.
    # This means the asset path must start with the required scope followed by a '/'.
    # For example, if required_scope is "/assets", it should match "/assets/image.jpg"
    # but not "/assets-private/doc.pdf".
    # We append a '/' to the normalized_scope to ensure segment boundary matching.
    prefix_to_match = normalized_scope + '/'
    return validate_prefix_match(prefix_to_match, asset.path)

def filter_accessible_assets(assets: list, required_scope: str) -> list:
    return [a for a in assets if can_access_asset(a, required_scope)]

def get_asset_hierarchy_level(asset: StorageAsset) -> int:
    return get_scope_depth(asset.path)
