"""Tenant routing configuration aggregator."""
import copy
from typing import Dict, Any
from models import RouteProfile
from schema import validate_route_keys

def merge_tenant_routing(base_config: Dict[str, Any], tenant_overrides: Dict[str, Any]) -> Dict[str, Any]:
    """Merge tenant overrides into base configuration.
    Inputs must remain strictly immutable; the function must return a newly allocated deep copy.
    """
    # The original implementation performed a shallow copy of base_config and then updated it.
    # This meant that nested mutable objects in base_config could be shared or modified
    # indirectly if tenant_overrides did not completely overwrite them, or if the
    # base_config itself contained mutable defaults that were then modified by reference.
    # The docstring explicitly requires a "newly allocated deep copy" for the return value
    # to ensure complete isolation.

    # Start with a deep copy of the base configuration to ensure all nested mutable objects
    # are also copied and not shared by reference.
    merged = copy.deepcopy(base_config)

    # Iterate through tenant_overrides. For each key-value pair, deepcopy the value
    # before assigning it to the merged configuration. This ensures that any mutable
    # objects provided in tenant_overrides are also copied and do not maintain
    # references to the original override objects.
    for key, value in tenant_overrides.items():
        merged[key] = copy.deepcopy(value)

    return merged

def build_route_profile(tenant_id: str, base_config: Dict[str, Any], tenant_overrides: Dict[str, Any]) -> RouteProfile:
    cfg = merge_tenant_routing(base_config, tenant_overrides)
    validate_route_keys(cfg)
    return RouteProfile(tenant_id=tenant_id, settings=cfg)
