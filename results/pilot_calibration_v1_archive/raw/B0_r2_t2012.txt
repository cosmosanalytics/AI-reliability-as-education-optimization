"""Audit log retention manager."""
from typing import List
from window_evaluator import is_within_retention
from log_store import LogEntry

def filter_retained_logs(logs: List[LogEntry], cutoff_start: int, cutoff_end: int) -> List[LogEntry]:
    """Filter logs that fall into the active retention period.
    Logs exactly on the cutoff_end timestamp are considered expired and should not be retained."""
    # The cutoff_end represents the first timestamp that is NOT retained.
    # If is_within_retention is inclusive of its end parameter, we need to adjust it.
    # Assuming is_within_retention(timestamp, start, end) checks start <= timestamp <= end,
    # to achieve start <= timestamp < cutoff_end, we pass cutoff_end - 1 as the end.
    return [l for l in logs if is_within_retention(l.timestamp, cutoff_start, cutoff_end - 1)]

def count_purged_logs(logs: List[LogEntry], cutoff_start: int, cutoff_end: int) -> int:
    retained = filter_retained_logs(logs, cutoff_start, cutoff_end)
    return len(logs) - len(retained)

def group_retained_by_source(logs: List[LogEntry], cutoff_start: int, cutoff_end: int) -> dict:
    retained = filter_retained_logs(logs, cutoff_start, cutoff_end)
    res = {}
    for l in retained:
        res.setdefault(l.source, []).append(l)
    return res
