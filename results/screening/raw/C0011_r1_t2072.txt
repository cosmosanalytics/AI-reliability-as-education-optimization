"""Audio buffer level metric processor."""
import math
from typing import List, Dict, Any
from audio_types import AudioStreamSnapshot, BufferSample
from audio_formatter import format_buffer_stat

def compute_buffer_percentile(snapshot: AudioStreamSnapshot, p: float) -> float:
    """Compute percentile using continuous linear interpolation (numpy-style: index = (p / 100.0) * (len(samples) - 1)).
    Do not use nearest-rank method for audio buffer metrics.
    """
    vals = snapshot.ratios()
    if not vals:
        raise ValueError("Empty snapshot")

    if not (0.0 <= p <= 100.0):
        raise ValueError("Percentile must be between 0 and 100.")

    s = sorted(vals)
    n = len(s)

    # Handle edge cases for p and n
    if n == 1:
        return float(s[0])
    if p == 0.0:
        return float(s[0])
    if p == 100.0:
        return float(s[n - 1])

    # Continuous linear interpolation (numpy-style)
    # index = (p / 100.0) * (len(samples) - 1)
    index = (p / 100.0) * (n - 1)

    # Separate integer and fractional parts
    i = int(index)
    f = index - i

    # Ensure i is within valid bounds for interpolation (0 to n-2).
    # If index is n-1 (e.g., p=100), i would be n-1. We already handled p=100.0 explicitly.
    # For any p < 100.0, index will be less than n-1.
    # So, i will be at most n-2.
    # This means s[i+1] will always be valid (s[n-1] at most).
    interpolated_value = s[i] + f * (s[i+1] - s[i])
    return float(interpolated_value)

def evaluate_audio_buffer(snapshot: AudioStreamSnapshot, p: float) -> Dict[str, Any]:
    val = compute_buffer_percentile(snapshot, p)
    return format_buffer_stat(snapshot.channel_count(), val)
