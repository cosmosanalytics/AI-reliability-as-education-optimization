"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total_line_value = sum(line_totals)
    num_lines = len(line_totals)

    if total_line_value == 0:
        return [0] * num_lines

    # Calculate base allocations using floor division
    allocations = [cents * total_discount_cents // total_line_value for cents in line_totals]

    # Distribute remaining cents
    current_total_allocated = sum(allocations)
    remainder = total_discount_cents - current_total_allocated

    # Calculate initial loss for each line due to floor division
    # This helps in distributing the remainder to the lines that lost the most
    losses = [(cents * total_discount_cents % total_line_value, i) for i, cents in enumerate(line_totals)]
    # Sort by loss (descending) to prioritize lines with higher fractional loss,
    # then by original index (ascending) for stable tie-breaking.
    # Note: `(value % divisor)` gives the remainder. A larger remainder means more was "lost"
    # by flooring, relative to the ideal fractional share.
    losses.sort(key=lambda x: (-x[0], x[1]))

    for _ in range(remainder):
        if not losses: # Should not happen if remainder is positive and num_lines > 0
            break
        # Take the line with the largest "loss" (or highest remainder from division)
        # and distribute one cent to it.
        # We don't need to re-sort or remove; we just iterate 'remainder' times
        # over the sorted list, effectively distributing to the top 'remainder' items.
        # If remainder > num_lines, it will wrap around correctly.
        idx_to_add = losses[_ % num_lines][1]
        allocations[idx_to_add] += 1

    return allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
