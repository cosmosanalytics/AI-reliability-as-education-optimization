"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    allowed_pattern = policy.allowed_pattern

    # Handle the global wildcard case
    if allowed_pattern == "**":
        return True

    allowed_segments = allowed_pattern.split('::')
    request_segments = clean_req.split('::')

    len_allowed = len(allowed_segments)
    len_request = len(request_segments)

    # Check for the double asterisk wildcard at the end
    if len_allowed > 0 and allowed_segments[-1] == "**":
        # If the pattern ends with '**', it matches if the prefix before '**' matches
        # and the request scope has at least as many segments as the prefix.
        prefix_segments = allowed_segments[:-1]
        len_prefix = len(prefix_segments)

        # The prefix must match exactly
        if len_request < len_prefix:
            return False
        for i in range(len_prefix):
            if prefix_segments[i] != request_segments[i]:
                return False
        return True # Prefix matches, and '**' matches the rest (including empty rest)

    # Handle single asterisk or exact segment matching
    # Both patterns must have the same number of segments if no trailing '**'
    if len_allowed != len_request:
        return False

    for i in range(len_allowed):
        allowed_seg = allowed_segments[i]
        request_seg = request_segments[i]

        if allowed_seg == "*":
            # '*' matches exactly one segment, so any non-empty segment is fine
            # (already covered by len_allowed == len_request, and segment is not empty)
            continue
        elif allowed_seg != request_seg:
            # Mismatch in exact segments
            return False

    return True

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
