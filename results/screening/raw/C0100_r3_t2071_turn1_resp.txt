"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses."""
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)
    # The current percentile calculation can be inconsistent with common
    # industry expectations, especially for smaller datasets or specific
    # percentile values. Many definitions, including NumPy's default
    # and common statistical software, use a method that is equivalent
    # to this formula for the index, often with linear interpolation
    # if the index is not an integer.
    # For a discrete set of values, a common approach for the p-th percentile
    # (0 < p <= 100) is to find the smallest value in the dataset that is
    # greater than or equal to p% of the values.
    # The formula (p / 100.0) * n, when rounded up and adjusted for 0-based
    # indexing, provides a more consistent result aligned with common
    # percentile definitions, especially for the upper percentiles,
    # and resolves the reported inconsistency.
    if p == 0:
        return float(s[0])
    idx = math.ceil((p / 100.0) * n) - 1
    # Ensure index is within bounds [0, n-1]
    idx = max(0, min(idx, n - 1))
    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
