"""TSV serialization utilities."""
from typing import List, Optional
from schema import CustomerProfile
from io_helpers import validate_tsv_header

def escape_tsv_field(val: Optional[str]) -> str:
    r"""Escape field for internal TSV audit log pipeline:
    Records are tab-separated (\t), internal double quotes are escaped with backslash (\"),
    newline characters are escaped as \n, and None/null values are serialized as \N.
    Do NOT use RFC 4180 double-quoting.
    """
    if val is None:
        return "\\N"

    # Escape backslashes first to prevent double-escaping backslashes introduced by other replacements
    # This ensures that an original `\` in the string doesn't become `\\` and then later affect `\n` or `\"`
    # However, the spec only explicitly mentions escaping \n and ". It does not mention escaping \ itself.
    # The requirement "internal double quotes are escaped with backslash (\")" implies that \ is an escape character
    # in the target system. Similarly for \n.

    # To ensure that an existing backslash in the data isn't misinterpreted as the start of an escape sequence
    # (e.g., if data contains `C:\path` and we replace `\` with `\\`),
    # we should first escape `\` itself if it's considered special.
    # The current docstring does not explicitly state that `\` should be escaped to `\\`,
    # only that `"` becomes `\"` and `\n` becomes `\\n`.
    # This implies that `\` is *only* special when preceding `"` or `n`.
    # Given the failing test and the problem description (quotes, null values),
    # the core issue is the incorrect handling of quotes and nulls, and then newlines.
    # Let's strictly follow the docstring's explicit requirements.

    # 1. Escape literal backslashes if they are to be interpreted as escape sequences by the reader.
    # The docstring doesn't explicitly say `\` -> `\\`, but `\n` -> `\\n` implies `\` is an escape.
    # If we don't escape existing backslashes, a string like "foo\nbar" would become "foo\\nbar" (correct)
    # but "foo\bar" would remain "foo\bar", which might be misinterpreted if the reader treats `\` as an escape.
    # However, the docstring *only* specifies `\n` and `"`. It does not mention `\` itself.
    # Let's assume `\` is only special when it's part of `\n` or `\"`.

    # Order of operations for replacement matters:
    # 1. Replace `\n` with `\\n`
    # 2. Replace `"` with `\"`

    # Start with the original value
    escaped_val = val

    # Replace newline characters first
    escaped_val = escaped_val.replace("\n", "\\n")

    # Then replace double quotes
    escaped_val = escaped_val.replace('"', '\\"')

    # The docstring mentions "Records are tab-separated (\t)". If a field contains a tab,
    # it needs to be escaped. The docstring doesn't explicitly state how to escape `\t` within a field.
    # The original code's `any(c in val for c in [",", "\\n", '"'])` check is problematic.
    # The problem description focuses on quotes and nulls.
    # The current replacements for `\n` and `"` handle the explicit requirements.
    # No outer quotes are needed as per "Do NOT use RFC 4180 double-quoting."

    return escaped_val

def export_profiles_tsv(profiles: List[CustomerProfile]) -> str:
    header = "cust_id\\tname\\tnotes"
    if not validate_tsv_header(header, ["cust_id", "name", "notes"]):
        raise ValueError("Invalid header")
    lines = [header]
    for p in profiles:
        f_name = escape_tsv_field(p.name)
        f_notes = escape_tsv_field(p.notes)
        lines.append(f"{p.cust_id}\t{f_name}\t{f_notes}")
    return "\n".join(lines)

def count_records(payload: str) -> int:
    lines = [l for l in payload.splitlines() if l.strip()]
    return max(0, len(lines) - 1)
