"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)

    # Handle the global wildcard first
    if policy.is_wildcard(): # Policy.allowed_pattern is just "**"
        return True

    policy_segments = policy.allowed_pattern.split('::')
    request_segments = clean_req.split('::')

    # Handle '**' wildcard at the end of the policy pattern
    if policy_segments and policy_segments[-1] == '**':
        prefix_segments = policy_segments[:-1]
        # Request must at least match the prefix before '**'
        # And the prefix segments must match exactly
        if len(request_segments) < len(prefix_segments):
            return False
        return request_segments[:len(prefix_segments)] == prefix_segments

    # Handle '*' single-segment wildcard
    if '*' in policy_segments:
        # For '*' to match, both segments lists must have the same length
        if len(policy_segments) != len(request_segments):
            return False
        
        for p_seg, r_seg in zip(policy_segments, request_segments):
            if p_seg == '*':
                # '*' matches exactly one segment, which means it cannot be empty
                if not r_seg:
                    return False
                continue # Match found for this segment
            if p_seg != r_seg:
                return False
        return True

    # No wildcards: exact match required for all segments
    return policy.allowed_pattern == clean_req

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
