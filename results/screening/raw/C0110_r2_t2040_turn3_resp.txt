"""IoT sensor ingest stream demux."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def process_iot_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Process IoT device packet.
    
    For raw IoT ingest pipeline, MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope.
    """
    # The original implementation violated its docstring and the ticket's requirement
    # by normalizing keys and wrapping the payload in a v2 envelope.
    # To fix IOT-703, this function must return the raw, original 'data' dictionary.
    return data

def dispatch_iot_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Dispatch an IoT record, wrapping it in a v2 envelope with normalized keys.
    
    This function processes the data by normalizing its keys and then wraps the
    result in a version 2 wire envelope.
    """
    # This function appears to be for a different pipeline that *does* require
    # key normalization and wrapping in a v2 envelope, contrasting with the
    # raw ingest pipeline handled by process_iot_packet.
    norm = normalize_dict_keys(data)
    # The call to extract_normalized_keys(data) was previously unused.
    # Assuming it was vestigial or for a feature not yet implemented,
    # it is removed to simplify the function, as it does not affect the return value.
    return build_envelope("iot_service", epoch_ms, norm)
