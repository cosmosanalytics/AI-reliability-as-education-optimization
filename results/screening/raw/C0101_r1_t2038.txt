"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

_SERVICE_NAME = "billing_service"

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers."""
    # The record's keys should be normalized for consistency.
    return normalize_dict_keys(data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # The record should have its keys normalized before being included in the envelope.
    record = normalize_dict_keys(data)
    # The billing payload must be enveloped before broadcasting to meet message bus requirements.
    enveloped_message = build_envelope(_SERVICE_NAME, epoch_ms, record)
    # Ensure the enveloped message is valid.
    _ = is_valid_envelope(enveloped_message)
    # The original test `test_nominal_billing_emit` expected `broadcast_billing_event`
    # to return the same non-enveloped record as `emit_billing_record`.
    # To satisfy this existing test without modifying it, and still address the incident,
    # we must ensure `broadcast_billing_event` returns the payload only when the test expects it,
    # while still performing the enveloping internally for the incident's requirement.
    # However, the primary ticket "BILL-402" states the message bus rejected *un-enveloped* payload.
    # The `test_billing_cases.py` failure confirms `broadcast_billing_event` must return an enveloped message.
    # The `test_nominal_billing_emit` test's assertion `assert broadcast_billing_event(raw, 500) == res`
    # where `res` is the non-enveloped record, is fundamentally incompatible with the ticket's requirement.
    # Since test files cannot be changed, the only way to pass `test_nominal_billing_emit`
    # is to revert `broadcast_billing_event` to return an un-enveloped record,
    # which would re-introduce the BILL-402 incident.
    # The ticket explicitly states "Ledger message bus rejected un-enveloped billing payload"
    # and the failing test `tests/test_billing_cases.py::test_service_execution - AssertionError: assert None == 2`
    # indicates `broadcast_billing_event` must produce a valid enveloped message.
    # Therefore, the previous fix that returns `enveloped_message` is correct for the ticket.
    # The `test_nominal_billing_emit` test is asserting an incorrect behavior for `broadcast_billing_event`
    # given the ticket's requirements and cannot be satisfied without breaking the fix for BILL-402.
    # Given the constraint "Do not modify files in tests/", the only path forward is to ensure
    # `broadcast_billing_event` correctly returns an enveloped message, as per the ticket.
    # The `_ = extract_normalized_keys(data)` line was redundant and has been removed.
    return enveloped_message
