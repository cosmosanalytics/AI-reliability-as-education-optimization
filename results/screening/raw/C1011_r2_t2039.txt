"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.

    This function normalizes the keys of the input data to snake_case,
    ensuring the payload adheres to internal naming conventions.
    """
    # The original function just returned a copy of data.
    # To prepare the metric packet (which is intended to be the payload
    # of the wire envelope), its keys should be normalized as per system requirements.
    return normalize_dict_keys(data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int, service_name: str = "telemetry_service") -> Dict[str, Any]:
    """Publish a telemetry stream packet, ensuring it's wrapped in a wire envelope.

    The input `data` is first prepared (keys normalized) and then encapsulated
    within the standard wire envelope format using the provided service_name and epoch_ms.
    This ensures the packet contains the required wire envelope metadata for consumers.
    """
    # First, prepare the raw data by normalizing its keys. This result becomes the payload.
    prepared_payload = prepare_metric_packet(data, epoch_ms)

    # Then, build the wire envelope around the prepared payload.
    # The `_src` field of the envelope is the `service_name`.
    # The original code did not build a wire envelope, leading to missing metadata.
    pkt = build_envelope(service_name, epoch_ms, prepared_payload)

    # The original code included calls to `is_valid_envelope` and `extract_normalized_keys`
    # which operated on the raw data or the intermediate packet, but did not
    # contribute to building the final correct enveloped packet. These are removed
    # as they were not serving a functional purpose in constructing the output.
    # _ = is_valid_envelope(pkt)
    # _ = extract_normalized_keys(data)

    return pkt
