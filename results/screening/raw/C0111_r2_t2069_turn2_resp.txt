"""Gateway request latency percentile calculation."""
import math
from typing import List, Dict, Any
from metric_types import MetricWindow, LatencyRecord
from metric_summary import format_metric_summary

def compute_percentile(window: MetricWindow, percentile: float) -> float:
    """Calculates metric percentile using nearest-rank method: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Do not use linear interpolation.
    """
    vals = window.raw_values()
    if not vals:
        raise ValueError("Empty metric window")
    s = sorted(vals)
    n = len(s)

    # The docstring explicitly requires the "nearest-rank method" using the formula:
    # k = math.ceil(p / 100 * n) - 1
    # The original implementation used linear interpolation, which contradicts this requirement.
    # We must ensure the index is within the valid range [0, n-1].
    # math.ceil(p / 100 * n) can result in 0 for very small percentiles or 0.0,
    # or n for percentile=100.
    # The index k must be 0-based, so max index is n-1.
    # If p=0, k = ceil(0) - 1 = -1, which is invalid.
    # If p=100, k = ceil(n) - 1 = n-1. This is correct for the last element.
    # If n=1, p=50, k = ceil(0.5) - 1 = 0. Correct.
    # The formula k = ceil(p*n/100) is commonly 1-based.
    # For a 0-based array, the index is typically (k_1based - 1).
    #
    # Let's re-evaluate the common nearest-rank formula.
    # One common definition for nearest-rank (often used in NIST) is:
    # index = ceil(p * n / 100)
    # If this index is 1-based, then the 0-based index is `ceil(p * n / 100) - 1`.
    #
    # Example: s = [10, 20, 30, 40, 50], n=5
    # 50th percentile: p=50. index = ceil(50 * 5 / 100) - 1 = ceil(2.5) - 1 = 3 - 1 = 2. s[2] = 30.
    # 75th percentile: p=75. index = ceil(75 * 5 / 100) - 1 = ceil(3.75) - 1 = 4 - 1 = 3. s[3] = 40.

    # Another common interpretation, especially for exact matches:
    # `k = (p / 100.0) * n`
    # If `k` is an integer, return `s[k-1]`.
    # Otherwise, return `s[ceil(k)-1]`.
    # Or simply: `s[max(0, min(n - 1, int(math.ceil((percentile / 100.0) * n)) - 1))]`

    # Let's stick strictly to the docstring's formula: `k = math.ceil(p / 100 * n) - 1`
    # We need to handle edge cases where `percentile` is 0 or very small,
    # which might lead to a negative index if `ceil` results in 0.
    # For `percentile = 0`, `k` should ideally be `s[0]`. `ceil(0) - 1 = -1`.
    # For `percentile = 100`, `k` should be `s[n-1]`. `ceil(n) - 1 = n-1`.
    # The `max(0, ...)` ensures we don't go below the first element.
    k = math.ceil((percentile / 100.0) * n) - 1
    idx = max(0, min(n - 1, int(k))) # Ensure index is within [0, n-1]

    return float(s[idx])

def summarize_window(window: MetricWindow, percentile: float) -> Dict[str, Any]:
    val = compute_percentile(window, percentile)
    return format_metric_summary(window.sample_count(), val)
