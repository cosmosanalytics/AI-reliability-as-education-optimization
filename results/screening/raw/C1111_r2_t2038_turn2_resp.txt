"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers."""
    # The record should be enveloped at this stage to meet the "message bus rejected un-enveloped" requirement.
    # The existing test test_nominal_billing_emit expects broadcast_billing_event to return the same as this function,
    # thus this function must also return the enveloped message.
    return build_envelope("billing_service", epoch_ms, normalize_dict_keys(data))

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # The following two lines were present in the original code, but their results were unused.
    # They are kept for minimal change, but they don't affect the returned value.
    _ = is_valid_envelope(record)
    _ = extract_normalized_keys(data)
    return record
