"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# A placeholder service name is used here.
# In a real application, this would likely be configured or passed in.
# The ticket states "Downstream aggregator dropped raw un-enveloped audit records missing version 2 wrapper."
# and the docstring for format_audit_event mentions `_src: service_name`.
# Since `service_name` is not provided to `format_audit_event` or `serialize_audit_event`
# in the existing signature or tests, a default is necessary to fulfill the envelope requirement.
# If a specific service name were required per-call, the function signatures would need to be updated,
# which would break existing callers and tests.
DEFAULT_SERVICE_NAME = "unknown_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    # Use the default service name as per the decision to maintain backward compatibility
    # with the existing function signature and tests.
    return build_envelope(DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # The following lines were not modifying msg and are not necessary for the
    # core functionality of returning the formatted message.
    # _ = is_valid_envelope(msg)
    # _ = extract_normalized_keys(data)
    return msg
