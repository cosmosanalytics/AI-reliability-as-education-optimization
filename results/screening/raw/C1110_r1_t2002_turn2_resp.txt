"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total = sum(line_totals)
    if total == 0:
        return [0] * len(line_totals)

    # Calculate initial allocations using floor division
    allocations = []
    for cents in line_totals:
        allocations.append(cents * total_discount_cents // total)

    # Distribute the remainder
    remainder = total_discount_cents - sum(allocations)
    if remainder > 0:
        # Calculate fractional parts to determine distribution priority
        # Use a list of (fractional_part, original_index) tuples
        fractions = []
        for i, cents in enumerate(line_totals):
            # Calculate the true proportion with float, then subtract the integer part
            # to get the fractional remainder.
            fractional_part = (cents * total_discount_cents / total) - allocations[i]
            fractions.append((fractional_part, i))

        # Sort by fractional part in descending order to distribute to those "most in need"
        fractions.sort(key=lambda x: x[0], reverse=True)

        # Add 1 cent to the lines with the largest fractional parts until the remainder is gone
        for _ in range(remainder):
            if not fractions: # Should not happen if remainder > 0 and line_totals is not empty
                break
            # Take the line with the largest current fractional part
            _, original_index = fractions.pop(0) # Pop the largest one
            allocations[original_index] += 1
            # Re-sort or re-evaluate if needed, but for simple "+1" distribution, this works.
            # For simplicity, we assume one pass is sufficient for distributing cents.
            # If a line gets +1, its fractional part effectively becomes negative for the purpose of this logic.
            # However, since we are popping, it won't be considered again for this remainder loop.

    return allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
