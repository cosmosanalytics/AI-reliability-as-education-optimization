"""TSV serialization utilities."""
from typing import List, Optional
from schema import CustomerProfile
from io_helpers import validate_tsv_header

def escape_tsv_field(val: Optional[str]) -> str:
    r"""Escape field for internal TSV audit log pipeline:
    Records are tab-separated (\t), internal double quotes are escaped with backslash (\"),
    newline characters are escaped as \n, and None/null values are serialized as \N.
    Do NOT use RFC 4180 double-quoting.
    """
    if val is None:
        return r"\N" # Serialize None as \N
    # Escape internal double quotes with backslash (\")
    # Escape newline characters as \n (literal backslash-n)
    # The docstring explicitly states "Do NOT use RFC 4180 double-quoting."
    # The original implementation was doing RFC 4180, which is incorrect.
    # The current implementation was also not handling literal \n correctly.
    # Fix: replace " with \" and actual newline char \n with literal \n
    return val.replace('"', r'\"').replace('\n', r'\n')


def export_profiles_tsv(profiles: List[CustomerProfile]) -> str:
    # The header and line delimiters must be actual tabs and newlines for validate_tsv_header
    # and for the final output string to be correctly interpreted as TSV.
    # The escape_tsv_field function handles the escaping of characters *within* the fields
    # to literal sequences like \", \n, \N.
    header = "cust_id\tname\tnotes"
    if not validate_tsv_header(header, ["cust_id", "name", "notes"]):
        raise ValueError("Invalid header")
    lines = [header]
    for p in profiles:
        f_id = str(p.cust_id) # cust_id is already an int, no special escaping needed, just convert to string
        f_name = escape_tsv_field(p.name)
        f_notes = escape_tsv_field(p.notes) # notes can be None
        lines.append(f"{f_id}\t{f_name}\t{f_notes}")
    return "\n".join(lines)

def count_records(payload: str) -> int:
    lines = [l for l in payload.splitlines() if l.strip()]
    return max(0, len(lines) - 1)
