"""Timestamp serialization codec for telemetry records."""
from datetime import datetime, timezone

def encode_iso_timestamp(dt: datetime) -> str:
    """Encode timestamp into standard format (ISO 8601 with milliseconds and 'Z' for UTC)."""
    # Ensure the datetime object is in UTC before formatting with 'Z'.
    if dt.tzinfo is None:
        # Assume naive datetimes are UTC for this context, or convert if explicit local handling needed.
        # For this specific issue, the input is UTC, so this path might not be hit or might need refinement
        # if naive datetimes can represent other timezones.
        dt_utc = dt.replace(tzinfo=timezone.utc)
    else:
        dt_utc = dt.astimezone(timezone.utc)

    # Format to ISO 8601 with milliseconds and 'Z' suffix.
    # isoformat() with timespec='milliseconds' gives '+00:00' for UTC.
    # We replace '+00:00' with 'Z'.
    iso_str = dt_utc.isoformat(timespec='milliseconds')
    if iso_str.endswith('+00:00'):
        return iso_str[:-6] + 'Z'
    return iso_str

def validate_timestamp_range(dt: datetime) -> bool:
    return dt.year >= 2020
