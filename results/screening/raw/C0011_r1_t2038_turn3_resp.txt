"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a constant for the service name as it's required by build_envelope
SERVICE_NAME = "billing_service"

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers.
    The payload keys are normalized to snake_case.
    """
    # Original implementation returned dict(data).
    # The incident implies that the payload itself might also need normalization
    # before being enveloped, or that this function should prepare the 'record' part.
    # Given the existence of key_normalizer, applying it here makes sense
    # for a "billing record" preparation step.
    return normalize_dict_keys(data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Broadcast a billing event by enveloping the billing record.
    The payload keys are normalized to snake_case before enveloping.
    """
    # The incident states "Ledger message bus rejected un-enveloped billing payload."
    # This function is responsible for broadcasting, implying it should produce
    # an enveloped message.
    # First, prepare the billing record, including key normalization.
    record_payload = emit_billing_record(data, epoch_ms)

    # Then, build the wire envelope around the prepared record.
    enveloped_message = build_envelope(SERVICE_NAME, epoch_ms, record_payload)

    # The original code had these lines. While they don't modify the return value,
    # they might be intended as internal assertions or logging.
    # We keep them for now, ensuring they operate on the correct (enveloped) message.
    _ = is_valid_envelope(enveloped_message) # Checks the newly created envelope
    _ = extract_normalized_keys(data) # Still extracts keys from the original data, not used for modification

    return enveloped_message
