"""Cache Hit Latency Percentile Monitor percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_cache_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_cache_percentile", samples)
    s = validate_metric_samples(win.values) # This already validates for empty list and sorts.
    n = len(s)

    if not (0 <= p <= 100):
        raise ValueError("Percentile 'p' must be between 0 and 100, inclusive.")

    # Nearest-rank method (Type 1): L = ceil(p * N / 100)
    # The value is S[L-1] for 0-indexed S.
    # Handle p=0 and p=100 explicitly for clarity and to match common interpretations
    # for nearest-rank, which maps 0 to min and 100 to max.
    if p == 0:
        return s[0]
    if p == 100:
        return s[n - 1]

    # Calculate rank L (1-indexed)
    rank = math.ceil(p / 100.0 * n)
    # Ensure rank is within valid bounds [1, n] for 1-based indexing,
    # though ceil(p*N/100) should naturally handle this for 0 < p < 100.
    rank = max(1, min(int(rank), n))
    return s[rank - 1] # Convert to 0-indexed

def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
