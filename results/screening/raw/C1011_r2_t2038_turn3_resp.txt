"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers."""
    # Incident BILL-402: Ledger message bus rejected un-enveloped billing payload.
    # The payload must be wrapped in a wire envelope.
    # Assuming 'billing_emitter' as the service name for this module.
    return build_envelope(service_name="billing_emitter", epoch_ms=epoch_ms, payload=data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # These calls are likely for logging/validation and should operate on the correct data.
    # is_valid_envelope checks the structure of the returned record.
    _ = is_valid_envelope(record)
    # extract_normalized_keys operates on the original payload data.
    _ = extract_normalized_keys(data)
    return record
