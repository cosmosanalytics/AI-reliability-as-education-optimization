"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys
from wire_envelope import build_envelope
from typing import Dict, Any

_SERVICE_NAME = "telemetry_stream_service" # Default service name for the envelope

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This includes normalizing keys and encapsulating the payload in a wire envelope."""
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(_SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The original function had calls to is_valid_envelope and extract_normalized_keys
    # but discarded their results. These operations are now implicitly handled or
    # can be performed on the returned 'pkt' if needed elsewhere.
    return pkt
