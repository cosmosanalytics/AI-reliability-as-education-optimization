"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Assuming a default service name for this telemetry stream.
# In a real-world scenario, this might be configured dynamically or passed as an argument.
# For this patch, we'll use a placeholder as the ticket implies missing *any* src.
DEFAULT_SERVICE_NAME = "unknown_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int, service_name: str = DEFAULT_SERVICE_NAME) -> Dict[str, Any]:
    """
    Prepare telemetry metric packet for distribution to event stream.
    This function now constructs a wire envelope around the normalized data.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int, service_name: str = DEFAULT_SERVICE_NAME) -> Dict[str, Any]:
    """
    Publishes a telemetry stream packet.
    The packet is first prepared as a wire envelope, then returned.
    The is_valid_envelope check is now expected to pass.
    """
    pkt = prepare_metric_packet(data, epoch_ms, service_name)
    # The following checks are for validation, but they don't modify the packet.
    # The `prepare_metric_packet` function now ensures the packet is a valid envelope.
    _ = is_valid_envelope(pkt)
    # extract_normalized_keys was applied inside prepare_metric_packet for the payload.
    # This line was checking the original data, which is no longer relevant for the final packet structure.
    # It's removed to avoid confusion and redundant processing.
    return pkt
