"""IoT sensor ingest stream demux."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def process_iot_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Process IoT device packet.
    
    For raw IoT ingest pipeline, MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope.
    """
    # The docstring explicitly states that for the raw IoT ingest pipeline,
    # the payload dictionary should be emitted directly with original camelCase keys,
    # and not wrapped in a version 2 envelope.
    # The current implementation normalizes keys and builds an envelope,
    # which contradicts this requirement.
    # To satisfy the requirement, we return the original data directly.
    return data

def dispatch_iot_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # This function now uses a slightly different logic to produce the enveloped and normalized output.
    # We create a normalized payload, then build the envelope with it.
    norm_payload = normalize_dict_keys(data)
    msg = build_envelope("iot_service", epoch_ms, norm_payload)
    _ = extract_normalized_keys(data) # This line remains unchanged as it might be used for side effects or logging elsewhere.
    return msg
