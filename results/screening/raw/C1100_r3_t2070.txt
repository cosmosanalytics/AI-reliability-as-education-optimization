"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    Uses linear interpolation between the two nearest ranks, corresponding
    to the "R-7" method often used in statistical software.
    """
    if not values:
        raise ValueError("Empty values")
    if not (0 <= p <= 100):
        raise ValueError("Percentile p must be between 0 and 100 inclusive")

    s = sorted(values)
    n = len(s)

    # Use the formula (p / 100.0) * (n - 1) for the index, which is a common
    # method for calculating percentiles (e.g., NumPy's default linear interpolation).
    # This formula maps p=0 to index 0 and p=100 to index n-1.
    idx = (p / 100.0) * (n - 1)

    low_idx = int(math.floor(idx))
    high_idx = int(math.ceil(idx))

    # Handle edge cases where low_idx or high_idx might be out of bounds
    # due to floating point inaccuracies when idx is exactly n-1.
    # For p=100, idx should be n-1.
    if low_idx < 0:
        low_idx = 0
    if high_idx >= n:
        high_idx = n - 1

    # If the indices are the same, or if n=1 (so low_idx=high_idx=0),
    # then there's no interpolation needed.
    if low_idx == high_idx:
        return float(s[low_idx])

    # Perform linear interpolation
    frac = idx - low_idx
    return float(s[low_idx] + frac * (s[high_idx] - s[low_idx]))
