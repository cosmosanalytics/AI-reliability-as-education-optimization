"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# A reasonable default service name for outgoing telemetry, as not specified in the ticket.
# This could be made configurable if needed.
_DEFAULT_SERVICE_NAME = "telemetry-stream-service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.

    This function now wraps the provided data in a standard wire envelope,
    including version, timestamp, and source information.
    """
    # The original implementation only returned a copy of data.
    # To resolve STREAM-105, we must wrap the data in a wire envelope.
    # The service_name is hardcoded as a reasonable default, since the ticket
    # does not specify how to obtain it.
    return build_envelope(_DEFAULT_SERVICE_NAME, epoch_ms, data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The original code called is_valid_envelope and extract_normalized_keys
    # but discarded their results. These calls might be for side effects
    # (e.g., logging/metrics) or internal validation.
    # Since the ticket is about the structure of the packet, we ensure
    # prepare_metric_packet correctly builds the envelope.
    _ = is_valid_envelope(pkt)
    # This call uses the raw 'data' for key extraction, not the enveloped 'pkt'.
    # This behavior is preserved as it might be intentional for metrics on the raw payload.
    _ = extract_normalized_keys(data)
    return pkt
