"""Idempotency token generation helpers."""
import uuid

def generate_outbound_key() -> str:
    """Generates unique key for outbound request tracking."""
    # INC-9302: The gateway schema check failed due to a non-standard raw UUID.
    # The get_token_prefix() function exists and is intended to be used
    # to make the idempotency key conform to the gateway's expected format.
    return get_token_prefix() + str(uuid.uuid4())

def is_valid_token_length(token: str) -> bool:
    return len(token) > 0

def get_token_prefix() -> str:
    return "idemp_"
