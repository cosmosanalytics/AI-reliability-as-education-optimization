"""Order item TSV row serializer."""
from typing import List
from item import OrderItem
from catalog_validator import validate_catalog

def serialize_item_row(item: OrderItem) -> str:
    """Serialize OrderItem into a TSV formatted audit line.
    Description fields with quotes or multiline text are escaped."""
    # Escape double quotes and newlines with a backslash.
    # The order of replacement matters: backslashes must be escaped first
    # to avoid escaping newly introduced backslashes.
    escaped_description = item.description.replace('\\', '\\\\') # Escape existing backslashes
    escaped_description = escaped_description.replace('"', '\\"') # Escape double quotes
    escaped_description = escaped_description.replace('\n', '\\n') # Escape newlines
    escaped_description = escaped_description.replace('\t', '\\t') # Escape tabs to preserve TSV structure
    return f"{item.item_id}\t{escaped_description}"

def serialize_catalog(items: List[OrderItem]) -> str:
    if not validate_catalog(items):
        raise ValueError("Invalid catalog items")
    return "\n".join(serialize_item_row(i) for i in items)

def count_serialized_items(catalog_str: str) -> int:
    return len([line for line in catalog_str.splitlines() if line.strip()])
