"""Document scope RBAC filter."""
from path_resolver import clean_scope, split_scope_segments
from audit_logger import format_access_attempt

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern.
    Access is granted if the requested scope is a direct descendant or the same as
    the allowed scope, respecting segment boundaries. A wildcard '::*' at the end
    of the allowed_scope grants access to all descendants."""
    cleaned_allowed = clean_scope(allowed_scope)
    cleaned_requested = clean_scope(requested_scope)

    if not cleaned_allowed:
        # An empty allowed_scope does not grant access to anything.
        allowed = False
    elif cleaned_allowed.endswith("::*"):
        # Handle wildcard pattern: match prefix up to the wildcard
        prefix_allowed = cleaned_allowed[:-3] # Remove "::*"
        allowed = cleaned_requested.startswith(prefix_allowed) and \
                  (cleaned_requested == prefix_allowed or cleaned_requested.startswith(prefix_allowed + "::"))
    else:
        # Exact segment matching
        allowed_segs = split_scope_segments(cleaned_allowed)
        requested_segs = split_scope_segments(cleaned_requested)

        if not allowed_segs: # This covers cases like allowed_scope="  ::  " which results in empty segments.
            allowed = False
        elif len(requested_segs) < len(allowed_segs):
            allowed = False
        else:
            # Check if allowed segments are an exact prefix of requested segments
            # and that the segment boundary is respected at the end of allowed_segs.
            allowed = requested_segs[:len(allowed_segs)] == allowed_segs

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
