"""Topic path subscription matcher."""
def is_topic_subscribed(subscribed_pattern: str, event_topic: str) -> bool:
    # A topic is subscribed if the event topic either exactly matches the subscribed pattern,
    # or the event topic starts with the subscribed pattern followed by "::".
    # This prevents partial matches like "orders" matching "orders_admin".
    return event_topic == subscribed_pattern or event_topic.startswith(f"{subscribed_pattern}::")

def clean_topic_segment(topic: str) -> str:
    return topic.strip().lower()

def is_wildcard_subscription(pattern: str) -> bool:
    return pattern.strip() in ("*", "**")
