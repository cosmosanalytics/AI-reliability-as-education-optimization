"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    allowed_pattern = policy.allowed_pattern

    if policy.is_wildcard():  # Handles allowed_pattern == "**"
        return True

    # Split patterns into segments
    policy_segments = allowed_pattern.split('::')
    request_segments = clean_req.split('::')

    p_idx = 0
    r_idx = 0

    while p_idx < len(policy_segments) and r_idx < len(request_segments):
        policy_segment = policy_segments[p_idx]
        request_segment = request_segments[r_idx]

        if policy_segment == "**":
            # If '**' is the last segment in the policy pattern, it matches any remaining request segments
            return True
        elif policy_segment == "*":
            # '*' matches exactly one segment. Move to the next segments in both.
            p_idx += 1
            r_idx += 1
        elif policy_segment == request_segment:
            # Literal match. Move to the next segments in both.
            p_idx += 1
            r_idx += 1
        else:
            # Mismatch for non-wildcard segments
            return False

    # Check if all policy segments have been consumed.
    # If there are remaining policy segments, they must all be '**'.
    # However, given "arbitrary multi-segment prefixes", '**' is expected at the end.
    # The loop condition `p_idx < len(policy_segments)` covers this:
    # if we reach here and `p_idx == len(policy_segments)` it means all policy segments matched.
    # If `p_idx < len(policy_segments)` and we are out of `request_segments`, it's not a match,
    # unless the remaining policy segments are all `**` (which is covered by the `**` check inside the loop).

    # For a match, all policy segments must be consumed (or end with '**' which returns True earlier),
    # and all request segments up to the point of match must be consumed.
    # If policy pattern ends exactly at the request pattern, it's a match.
    # E.g., policy="a::b", request="a::b"
    return p_idx == len(policy_segments) and r_idx == len(request_segments)


def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
