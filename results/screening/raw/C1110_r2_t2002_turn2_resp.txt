"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total_invoice_value = sum(line_totals)
    num_lines = len(line_totals)

    if total_invoice_value == 0:
        return [0] * num_lines

    # Calculate initial allocations using float for precision
    # and store the integer part and the remainder.
    allocations_with_remainders = []
    for cents in line_totals:
        proportion = cents * total_discount_cents / total_invoice_value
        int_part = int(proportion)
        remainder = proportion - int_part
        allocations_with_remainders.append((int_part, remainder))

    # Convert to a mutable list of integers for final adjustments
    final_allocations = [item[0] for item in allocations_with_remainders]

    # Calculate the sum of currently allocated cents
    current_total_allocated = sum(final_allocations)

    # Determine how many cents are still unallocated
    unallocated_cents = total_discount_cents - current_total_allocated

    # Distribute the remaining unallocated cents one by one
    # to the lines with the largest remainders, breaking ties by original index.
    # We sort by remainder in descending order, and then by original index in ascending order.
    # The original index is implicitly preserved by Python's stable sort if remainders are equal.
    # To be explicit and robust against non-stable sorts (not an issue for Python's Timsort),
    # we can add the original index to the tuple for sorting.
    indexed_remainders = sorted(
        [(allocations_with_remainders[i][1], i) for i in range(num_lines)],
        key=lambda x: (-x[0], x[1]) # Sort by remainder (desc), then by index (asc)
    )

    for i in range(unallocated_cents):
        # Pick the line with the largest remainder (and smallest original index for ties)
        # and add 1 cent to its allocation.
        line_index_to_adjust = indexed_remainders[i][1]
        final_allocations[line_index_to_adjust] += 1

    return final_allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
