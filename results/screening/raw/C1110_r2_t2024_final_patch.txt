"""Internal stream sorting helper."""
from typing import List
from event_packet import EventPacket

def order_packets(packets: List[EventPacket]) -> List[EventPacket]:
    # Sort primarily by severity in descending order (higher severity first).
    # For packets with the same severity, use the stream_id as a secondary tie-breaker
    # in ascending lexicographical order to ensure deterministic sequencing.
    # The original issue mentioned "Out-of-order stream tiebreaker", suggesting the
    # previous tie-breaking was insufficient or not consistently applied across systems.
    # Sorting by stream_id provides a stable, deterministic order for same-severity packets.
    return sorted(packets, key=lambda p: (-p.severity, p.stream_id))

def get_max_severity(packets: List[EventPacket]) -> int:
    return max((p.severity for p in packets), default=0)

def count_critical_packets(packets: List[EventPacket]) -> int:
    return sum(1 for p in packets if p.is_critical())
