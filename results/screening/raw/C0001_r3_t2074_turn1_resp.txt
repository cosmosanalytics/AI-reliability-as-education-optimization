"""Network jitter SLA evaluation service."""
import math
from typing import List, Dict, Any
from jitter_types import JitterStream, JitterSample
import jitter_core

def check_jitter_sla(stream: JitterStream, p: float, max_jitter: float) -> bool:
    """SLA compliance requires nearest-rank percentile estimation: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Returns True if compliance limit is respected.
    """
    vals = stream.raw_jitters()
    if not vals:
        # If there are no samples, it cannot be compliant.
        return False
    
    s = sorted(vals)
    n = len(s)
    
    # Calculate nearest-rank percentile index as per SLA requirement.
    # index k = math.ceil(p / 100 * n) - 1
    # Note: p is 1-100, n is 1-based count.
    # The index k is 0-based.
    # Ensure p is within valid range (0 < p <= 100)
    if not (0 < p <= 100):
        raise ValueError("Percentile 'p' must be between 0 and 100 (exclusive of 0, inclusive of 100).")

    # The formula math.ceil(p / 100 * n) gives a 1-based rank.
    # Subtract 1 to get a 0-based index.
    rank_1_based = math.ceil((p / 100.0) * n)
    # Ensure the calculated rank is at least 1, as ceil(x) for x close to 0 could be 0.
    # For p > 0, (p/100.0 * n) > 0, so ceil will be >= 1.
    
    # Adjust for 0-based indexing:
    k_0_based = int(rank_1_based) - 1

    # Handle edge case where k_0_based could be negative if n=0, though we check for vals.
    # For p=100, k_0_based will be n-1, which is the last element.
    # For p close to 0, k_0_based will be 0, which is the first element.
    
    percentile_val = s[k_0_based]
    return percentile_val <= max_jitter

def get_stream_jitter_metrics(stream: JitterStream, p: float) -> Dict[str, Any]:
    vals = stream.raw_jitters()
    # This function uses the general-purpose percentile calculation from jitter_core.
    q = jitter_core.evaluate_jitter_percentile(vals, p)
    return {"count": stream.sample_count(), "jitter_p": q}
