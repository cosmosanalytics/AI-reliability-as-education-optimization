"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# TODO: The service_name should ideally be configurable or passed as an argument.
# For now, a placeholder is used to ensure the envelope is built correctly.
_DEFAULT_SERVICE_NAME = "unknown_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # The telemetry packet must be wrapped in a wire envelope.
    # The 'data' argument serves as the payload of the envelope.
    return build_envelope(service_name=_DEFAULT_SERVICE_NAME, epoch_ms=epoch_ms, payload=data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following lines were likely intended for validation or further processing
    # of the *prepared packet*, not the raw input data.
    # However, for this patch, we ensure 'pkt' is a valid envelope.
    # The return value of is_valid_envelope is not used, but it's a good sanity check.
    _ = is_valid_envelope(pkt)
    # This line extracts normalized keys from the original 'data', not the envelope.
    # Its purpose here is unclear without further context, but it's not directly
    # related to the envelope issue, so it remains unchanged.
    _ = extract_normalized_keys(data)
    return pkt
