"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    clean_pattern = clean_scope(policy.allowed_pattern)

    req_segments = clean_req.split('::')
    pattern_segments = clean_pattern.split('::')

    # Helper function for recursive matching
    def _match_segments(pat_idx: int, req_idx: int) -> bool:
        # Base case: if all pattern segments are consumed
        if pat_idx == len(pattern_segments):
            # If all request segments are also consumed, it's a match
            return req_idx == len(req_segments)

        # Base case: if all request segments are consumed but pattern segments remain
        # This is only a match if remaining pattern segments are all '**'
        if req_idx == len(req_segments):
            return all(p == '**' for p in pattern_segments[pat_idx:])

        current_pattern_segment = pattern_segments[pat_idx]

        if current_pattern_segment == '**':
            # '**' can match 0 or more segments in the request
            # Option 1: '**' matches 0 segments. Try to match the rest of the pattern with current request segment.
            if _match_segments(pat_idx + 1, req_idx):
                return True
            # Option 2: '**' matches 1 or more segments. Consume current request segment and try again with '**'.
            if _match_segments(pat_idx, req_idx + 1):
                return True
            return False
        elif current_pattern_segment == '*':
            # '*' matches exactly one segment in the request
            return _match_segments(pat_idx + 1, req_idx + 1)
        else:
            # Literal segment match
            if current_pattern_segment == req_segments[req_idx]:
                return _match_segments(pat_idx + 1, req_idx + 1)
            else:
                return False

    return _match_segments(0, 0)


def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
