"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# A placeholder service name. In a real application, this would likely be
# configured or passed into the function. For this patch, we assume a default.
# The ticket implies a problem with missing metadata, not incorrect service name.
DEFAULT_SERVICE_NAME = "unknown_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This function normalizes the payload keys and builds the wire envelope."""
    # Normalize keys of the actual telemetry payload
    normalized_payload = normalize_dict_keys(data)
    
    # Build the wire envelope around the normalized payload
    # Using a placeholder service name as it's not provided in the signature.
    # A more robust solution might pass service_name as an argument.
    return build_envelope(DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following two lines were likely meant for validation or side effects,
    # but the primary issue is the packet construction.
    # Keeping them for now, but they don't affect the returned pkt.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # Note: this operates on original data, not the packet.
    return pkt
