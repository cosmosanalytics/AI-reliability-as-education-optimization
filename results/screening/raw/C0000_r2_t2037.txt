"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# A default service name is introduced to maintain compatibility with existing
# call sites that do not provide a service_name, as per the constraint
# not to modify test files or other call sites.
# In a real-world scenario, 'service_name' should ideally be passed explicitly
# or configured at a higher level.
DEFAULT_SERVICE_NAME = "unknown_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int, service_name: str = DEFAULT_SERVICE_NAME) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # The original implementation had these lines, they are kept for consistency
    # but their return values are not currently used.
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data) # Note: This still operates on the original 'data', not the normalized payload within 'msg'.
    return msg
