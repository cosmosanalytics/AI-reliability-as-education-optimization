"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    This implementation uses the 'nearest rank' method (R-1), which is
    common for SLA calculations where a specific sample value is desired
    rather than an interpolated value. The index is calculated as ceil(p/100 * n),
    and the value at this rank (0-indexed) is returned.
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)

    # Use the nearest rank method (R-1) for percentile calculation.
    # The index is 1-based, so subtract 1 for 0-based list access.
    # This method is often preferred for SLA calculations where we want
    # to identify a specific data point rather than an interpolated value.
    idx = math.ceil((p / 100.0) * n)

    # Ensure the index is within valid bounds.
    # For p=0, idx would be 0, which would map to s[-1] if not careful.
    # For p=100, idx would be n, which maps to s[n-1].
    # The smallest index is 1 (for p > 0), so max(1, idx).
    # The largest index is n (for p=100).
    # Since idx is 1-based, we convert to 0-based by subtracting 1.
    final_idx = max(0, int(idx) - 1)

    return float(s[final_idx])
