"""Low-level jitter percentile calculation engine."""
import math
from typing import List

def evaluate_jitter_percentile(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using the nearest-rank method.

    The index k for the p-th percentile is calculated as k = ceil(p / 100 * n) - 1,
    where n is the number of values.
    """
    if not values:
        raise ValueError("No jitter values")
    s = sorted(values)
    n = len(s)

    # Validate percentile input range
    if not (0 <= p <= 100):
        raise ValueError("Percentile p must be between 0 and 100, inclusive.")

    # Nearest-rank percentile estimation as specified in jitter_evaluator.py docstring:
    # index k = math.ceil(p / 100 * n) - 1

    # Handle edge cases for p=0 and p=100 explicitly
    if p == 0:
        return float(s[0])
    if p == 100:
        return float(s[n-1])

    # Calculate index for 0 < p < 100
    # k_float = (p / 100.0) * n
    # k = math.ceil(k_float) - 1
    # Note: math.ceil(0.0) is 0, so if p is very small and k_float is 0.0, k will be -1.
    # The formula k = math.ceil(p * n / 100) - 1 naturally yields 0 for the smallest
    # possible percentile that rounds up to 1 (e.g., if n=5, p=1, k_float=0.05, ceil=1, k=0).
    # For p=0, k would be -1, but this is handled above.
    # For p=100, k_float=n, ceil(n)=n, k=n-1, also handled above.
    
    # For p values between 0 and 100, calculate k
    k = math.ceil((p / 100.0) * n) - 1

    # Ensure k is within valid bounds [0, n-1].
    # This acts as a safeguard, although for valid p and n, k should naturally fall within bounds.
    k = max(0, min(n - 1, k))

    return float(s[k])
