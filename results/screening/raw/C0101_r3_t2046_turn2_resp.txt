"""Timestamp serialization codec for telemetry records."""
from datetime import datetime, timezone

def encode_iso_timestamp(dt: datetime) -> str:
    """Encode timestamp into standard format (ISO 8601 with milliseconds and UTC 'Z' suffix)."""
    # Ensure the datetime object is timezone-aware and in UTC for the 'Z' suffix.
    # Convert to UTC if it's not already. If naive, assume UTC for telemetry exports.
    if dt.tzinfo is None:
        dt_utc = dt.replace(tzinfo=timezone.utc)
    else:
        dt_utc = dt.astimezone(timezone.utc)

    # Use isoformat with milliseconds and replace the UTC offset with 'Z'
    return dt_utc.isoformat(timespec='milliseconds').replace('+00:00', 'Z')

def validate_timestamp_range(dt: datetime) -> bool:
    return dt.year >= 2020
