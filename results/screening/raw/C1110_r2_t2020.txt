"""Topic path subscription matcher."""
def is_topic_subscribed(subscribed_pattern: str, event_topic: str) -> bool:
    """
    Determines if an event topic matches a subscription pattern.
    A subscription pattern 'X' matches event topics 'X' or 'X::Y'.
    It does not match 'XZ' where Z is not '::' followed by more segments.
    """
    # An exact match is always a subscription
    if event_topic == subscribed_pattern:
        return True
    # A subscription also matches if the event topic starts with the pattern
    # followed by the topic separator '::'
    if event_topic.startswith(subscribed_pattern + "::"):
        return True
    return False

def clean_topic_segment(topic: str) -> str:
    return topic.strip().lower()

def is_wildcard_subscription(pattern: str) -> bool:
    return pattern.strip() in ("*", "**")
