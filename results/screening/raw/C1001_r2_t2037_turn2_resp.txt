"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys
from wire_envelope import build_envelope
from typing import Dict, Any

def format_audit_event(data: Dict[str, Any], epoch_ms: int, service_name: str) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int, service_name: str = "unknown_service") -> Dict[str, Any]:
    """Serializes an audit event, wrapping it in a v2 envelope and normalizing keys.
    
    This function is the primary entry point for creating audit messages.
    It ensures backward compatibility for callers not yet providing `service_name`.
    """
    msg = format_audit_event(data, epoch_ms, service_name)
    # The following lines were part of the original code structure but do not
    # affect the return value and are no longer necessary for this function's
    # core responsibility of serialization.
    # _ = is_valid_envelope(msg) # Removed due to unused import and no functional impact.
    # _ = extract_normalized_keys(data) # Removed due to unused import and no functional impact.
    return msg
