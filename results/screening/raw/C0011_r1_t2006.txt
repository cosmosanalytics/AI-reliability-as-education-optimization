"""Database pool manager."""
from typing import Dict, Any
from env_loader import merge_env_settings
from options import ConnectionOptions

def configure_pool(base_template: Dict[str, Any], overrides: Dict[str, Any]) -> ConnectionOptions:
    """Configure connection pool options from template and overrides.
    Returns ConnectionOptions instance.
    """
    settings = merge_env_settings(base_template, overrides)
    dsn = settings.get("dsn", "sqlite:///:memory:")

    # Extract pool-specific parameters, prioritizing direct ConnectionOptions attributes
    # from the merged settings.
    pool_params = settings.get("pool", {})
    
    # Initialize ConnectionOptions, explicitly passing any overrides for
    # max_overflow and timeout_seconds found in the merged settings.
    # If these keys are not present in settings, fall back to the dataclass's
    # own defined default values for max_overflow and timeout_seconds.
    opts = ConnectionOptions(
        dsn=dsn,
        pool_params=pool_params,
        max_overflow=settings.get("max_overflow", ConnectionOptions.max_overflow),
        timeout_seconds=settings.get("timeout_seconds", ConnectionOptions.timeout_seconds),
    )
    _ = opts.summary()
    return opts
