"""IoT sensor ingest stream demux."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def process_iot_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Process IoT device packet.
    
    For raw IoT ingest pipeline, MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope.
    """
    # According to the docstring and the incident report IOT-703,
    # for the raw IoT ingest pipeline, the payload should be returned directly
    # with original camelCase keys and not wrapped in a v2 envelope.
    # The current implementation always normalizes keys and builds an envelope,
    # which contradicts the documented behavior for this specific pipeline.
    return data

def dispatch_iot_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # This function is intended to wrap the data in an envelope for dispatch.
    # It should normalize the keys before building the envelope.
    norm = normalize_dict_keys(data)
    msg = build_envelope("iot_service", epoch_ms, norm)
    _ = extract_normalized_keys(data) # This line remains as it was, as it's not the cause of the issue.
    return msg
