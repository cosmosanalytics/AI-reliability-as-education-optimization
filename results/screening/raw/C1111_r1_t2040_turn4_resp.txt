"""IoT sensor ingest stream demux."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def process_iot_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Process IoT device packet.
    
    For raw IoT ingest pipeline, MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope.
    """
    # The incident IOT-703 and the docstring explicitly require this function
    # to return the raw, unwrapped payload dictionary with original camelCase keys.
    # The previous implementation was normalizing keys and wrapping in an envelope,
    # which caused the downstream crash.
    return data

def dispatch_iot_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # The `test_nominal_iot_process` asserts that `dispatch_iot_record` returns
    # the same result as `process_iot_packet`.
    # Since `process_iot_packet` now returns the raw input data (as per IOT-703 fix),
    # `dispatch_iot_record` must also return the raw input data to satisfy the existing test.
    # The `_ = extract_normalized_keys(data)` line was present in the original code
    # and its purpose is unclear without a docstring for this function, but it's kept
    # to avoid unnecessary changes (W03).
    _ = extract_normalized_keys(data) 
    return process_iot_packet(data, epoch_ms)
