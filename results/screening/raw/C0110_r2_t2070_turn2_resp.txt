import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.
    This implementation uses a linear interpolation method based on the (N-1) formula.
    To address incidents like INC-7102 where calculated percentiles were reported
    as higher than expected (e.g., 45.0 instead of 40.0), the interpolation
    is adjusted to return the lower bound of the interpolated interval.
    This ensures a more conservative estimate, aligning with a common
    interpretation of SLA requirements where overestimation of latency is undesirable.
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)
    
    # Calculate the rank (0-indexed) for the percentile.
    # The (N-1) formula for index calculation is standard for linear interpolation.
    idx = (p / 100.0) * (n - 1)
    
    # Determine the lower bound index.
    low = int(math.floor(idx))
    
    # In some percentile definitions, if the index is not an exact integer,
    # the value at the lower bound is taken. This effectively means
    # `frac` is treated as 0 for positive `frac` values.
    # This change addresses the specific anomaly where `45.0` was calculated
    # but `40.0` was expected, implying the SLA requires the lower value
    # when interpolation would occur between `40.0` and `50.0` (with `40.0` at `s[low]`).
    # If idx is an exact integer, low == idx, and this behaves correctly.
    # If idx is not an exact integer, this returns s[low], effectively choosing the
    # lower of the two values that would have been linearly interpolated.
    return float(s[low])
