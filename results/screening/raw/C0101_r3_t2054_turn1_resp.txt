"""Low-level dictionary and list hierarchy merger."""
from typing import Dict, Any, List

def merge_spec_values(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """Recursively merges dictionary values, concatenating lists if both base and override values are lists."""
    result = dict(base)
    for k, v in override.items():
        if k in result and isinstance(result[k], dict) and isinstance(v, dict):
            result[k] = merge_spec_values(result[k], v)
        elif k in result and isinstance(result[k], list) and isinstance(v, list):
            # Concatenate lists, preserving order and allowing duplicates.
            # Convert to set for unique elements then back to list to ensure original base elements are preserved
            # and new override elements are added, while avoiding duplicates from the combined set.
            # The order of elements from the base is preserved first, then new unique elements from the override.
            merged_list = list(result[k])
            for item in v:
                if item not in merged_list:
                    merged_list.append(item)
            result[k] = merged_list
        else:
            result[k] = v
    return result

def is_spec_empty(spec: Dict[str, Any]) -> bool:
    return len(spec) == 0
