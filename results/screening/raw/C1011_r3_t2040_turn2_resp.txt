"""IoT sensor ingest stream demux."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def process_iot_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Process IoT device packet.
    
    For raw IoT ingest pipeline, MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope.
    """
    # The docstring explicitly states:
    # "MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope."
    # The previous implementation normalized keys and then wrapped the payload,
    # which caused the incident IOT-703.
    # This behavior is changed to simply return the original data as specified.
    return data

def dispatch_iot_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # The incident report and docstring for process_iot_packet indicate that
    # process_iot_packet should return the raw data.
    # However, dispatch_iot_record is expected to produce an enveloped message
    # with normalized keys for internal processing, as it uses `_ = extract_normalized_keys(data)`
    # which implies normalization is intended somewhere in this path, and the system
    # typically works with v2 envelopes.
    # To fix IOT-703, process_iot_packet must return raw data.
    # To maintain the expected behavior for dispatch_iot_record (which likely
    # feeds into other internal systems that expect the envelope and normalized keys),
    # we apply the normalization and envelope building here.
    norm = normalize_dict_keys(data)
    msg = build_envelope("iot_service", epoch_ms, norm)
    # The line below is kept as it was in the original implementation,
    # indicating an intent to process normalized keys, even if the result was ignored.
    _ = extract_normalized_keys(data) 
    return msg
