"""Gateway request latency percentile calculation."""
import math
from typing import List, Dict, Any
from metric_types import MetricWindow, LatencyRecord
from metric_summary import format_metric_summary

def compute_percentile(window: MetricWindow, percentile: float) -> float:
    """Calculates metric percentile using nearest-rank method: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Do not use linear interpolation.
    """
    vals = window.raw_values()
    if not vals:
        raise ValueError("Empty metric window")
    s = sorted(vals)
    n = len(s)

    # As per the docstring, k = math.ceil(p / 100 * n)
    # The index for 0-based arrays is then k-1.
    # Handle edge case for 0th percentile (though typically percentiles are > 0).
    # If percentile is 0, k will be 0, index k-1 is -1.
    # If percentile is 100, k will be n, index k-1 is n-1.
    if percentile == 0:
        return float(s[0]) # Smallest value for 0th percentile
    if percentile == 100:
        return float(s[n-1]) # Largest value for 100th percentile

    # Calculate k, then adjust to 0-based index
    k = math.ceil(percentile / 100.0 * n)
    index = int(k) - 1

    # Ensure index is within bounds (0 to n-1)
    # The math.ceil(p/100*n) formula typically yields k in [1, n] for p in (0, 100]
    # So index will be in [0, n-1].
    # For very small p, k could be 1, index 0.
    # For p approaching 100, k could be n, index n-1.
    return float(s[index])

def summarize_window(window: MetricWindow, percentile: float) -> Dict[str, Any]:
    val = compute_percentile(window, percentile)
    return format_metric_summary(window.sample_count(), val)
