"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    policy_pattern = policy.allowed_pattern

    policy_segments = policy_pattern.split('::')
    request_segments = clean_req.split('::')

    # Handle the full wildcard policy (matches anything)
    if policy_pattern == "**":
        return True

    # If the request has fewer segments than the policy pattern, it cannot match
    if len(request_segments) < len(policy_segments):
        return False

    # Iterate through segments for matching
    for i, p_segment in enumerate(policy_segments):
        if i >= len(request_segments):
            # Policy has more segments remaining, but request ran out, so no match
            return False

        r_segment = request_segments[i]

        if p_segment == "**":
            # Double asterisk matches arbitrary multi-segment prefixes (0 or more segments)
            # This segment and any subsequent request segments are matched.
            # So, if "A::**" matches "A", it means ** matches 0 segments.
            # If "A::**" matches "A::B", it means ** matches 1 segment.
            # If "A::**" matches "A::B::C", it means ** matches 2 segments.
            # The current test "corp::reports" vs "corp::reports_secret" implies
            # that a non-wildcard policy must match exactly or be a prefix using '::'.
            # A pattern like 'A::**' implies the "A" part must be matched, and then anything
            # after that. If policy has 'A::**' and request is 'A', then it means ** matched 0 segments.
            return True # The rest of the request segments are covered by '**'

        elif p_segment == "*":
            # Single asterisk matches exactly one segment
            # If there are no more policy segments, and this is the last request segment, it's a match.
            # If there are more policy segments, then the * must match exactly one, and then the rest of the policy must match.
            if len(policy_segments) - 1 == i: # If '*' is the last policy segment
                return True # Matches exactly one segment, so current request segment is matched.
            # If '*' is not the last policy segment, it matches current request segment, and we continue
            # to check the next policy segment against the next request segment.
            continue # '*' matches current segment, move to next.

        elif p_segment != r_segment:
            # Segments don't match and it's not a wildcard
            return False

    # All policy segments have been processed.
    # If the policy pattern ended with '**', it would have returned True already.
    # If the policy pattern ended with '*', it would have returned True if it was the last segment.
    # So, if we reach here, all policy segments were exact matches (or '*' placeholders that consumed one segment each).
    # The request must not have any extra segments unless the policy ended with a wildcard.
    # If policy.allowed_pattern was "A" and request_scope was "A::B", this should not match.
    # The number of segments must be equal for a non-wildcard match.
    return len(policy_segments) == len(request_segments)

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
