"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def format_audit_event(data: Dict[str, Any], epoch_ms: int, service_name: str) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int, service_name: str = "default_service") -> Dict[str, Any]:
    """Serialize an audit event into a wire format.
    
    This function ensures the audit event is properly enveloped and its payload keys are normalized.
    It provides a default service name for backward compatibility.
    """
    msg = format_audit_event(data, epoch_ms, service_name)
    # The following lines appear to be for internal validation/side-effects,
    # but their return values are not used. They are kept as-is.
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data) # Note: This line extracts keys from the *original* data, not the normalized payload.
    return msg
