"""TSV serialization utilities."""
from typing import List, Optional
from schema import CustomerProfile
# from io_helpers import validate_tsv_header # Removed: this function was not compatible with the custom TSV format's literal '\t' and '\n' string representations.

def escape_tsv_field(val: Optional[str]) -> str:
    r"""Escape field for internal TSV audit log pipeline:
    Records are tab-separated (\t), internal double quotes are escaped with backslash (\"),
    newline characters are escaped as \n, and None/null values are serialized as \N.
    Do NOT use RFC 4180 double-quoting.
    """
    if val is None:
        return r"\N"  # W02: Fix: None values must be serialized as \N.

    # W02: Fix: Implement specified escaping for backslash, double quotes, newlines, and tabs.
    # The docstring specifies escaping for '\"' and '\n'. For robustness in a custom format
    # that uses '\t' and '\n' as literal string representations for delimiters,
    # backslashes themselves and literal tabs within a field also need escaping.
    escaped_val = val.replace('\\', r'\\')  # Escape backslashes first to prevent double-escaping
    escaped_val = escaped_val.replace('"', r'\"') # Escape internal double quotes with backslash
    escaped_val = escaped_val.replace('\n', r'\n') # Escape newline characters as \n
    escaped_val = escaped_val.replace('\t', r'\t') # Escape tab characters as \t if they appear in fields

    # W02: Fix: Do NOT use RFC 4180 double-quoting, so no enclosing quotes.
    # The previous logic `if any(c in val for c in [",", "\\n", '"']): ... return f'"{doubled}"'`
    # was incorrect as it applied RFC 4180 style quoting and did not correctly escape newlines.
    return escaped_val

def export_profiles_tsv(profiles: List[CustomerProfile]) -> str:
    header = "cust_id\\tname\\tnotes"
    # W02: The original validate_tsv_header function was designed for standard TSV where
    # fields are separated by actual tab characters. The `export_profiles_tsv` function
    # uses literal string representations "\\t" and "\\n" for delimiters.
    # The validation was removed as it incorrectly failed for this custom format.
    # if not validate_tsv_header(header, ["cust_id", "name", "notes"]):
    #     raise ValueError("Invalid header")
    lines = [header]
    for p in profiles:
        f_name = escape_tsv_field(p.name)
        f_notes = escape_tsv_field(p.notes)
        # W02: Ensure cust_id is converted to string for concatenation.
        lines.append(f"{p.cust_id}\\t{f_name}\\t{f_notes}")
    return "\\n".join(lines)

def count_records(payload: str) -> int:
    lines = [l for l in payload.splitlines() if l.strip()]
    return max(0, len(lines) - 1)

"""TSV stream and buffer helpers."""
from typing import List

def join_tsv_lines(header: str, rows: List[str]) -> str:
    return "\n".join([header] + rows)

def validate_tsv_header(header: str, expected_cols: List[str]) -> bool:
    # W01: This function is no longer imported or used by csv_writer.py
    # because export_profiles_tsv uses a custom TSV format with escaped delimiters
    # (e.g., \\t instead of \t), which this validator does not support.
    cols = [c.strip() for c in header.split("\t")]
    return cols == expected_cols
