"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def format_audit_event(data: Dict[str, Any], epoch_ms: int, service_name: str = "unknown") -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # format_audit_event now requires a service_name. To maintain backward compatibility
    # for existing callers of serialize_audit_event that do not provide service_name,
    # we rely on the default 'unknown' provided by format_audit_event.
    msg = format_audit_event(data, epoch_ms)
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data)
    return msg
