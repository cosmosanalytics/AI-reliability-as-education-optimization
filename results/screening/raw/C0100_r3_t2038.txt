"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

_SERVICE_NAME = "billing"

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers."""
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name=_SERVICE_NAME, epoch_ms=epoch_ms, payload=normalized_payload)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # The original checks here were for demonstration purposes and did not affect the return value.
    # They can be kept or removed based on whether they serve a future purpose (e.g., logging).
    # For this ticket, we just need to ensure the correct enveloped record is returned.
    _ = is_valid_envelope(record) # This check is now meaningful as `record` *should* be valid.
    _ = extract_normalized_keys(data) # This line operates on the original data, not the enveloped record.
    return record
