"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

_SERVICE_NAME = "telemetry_stream"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # The incident report indicates that the wire envelope metadata was missing.
    # The 'prepare_metric_packet' function should now return a fully enveloped packet.
    # This aligns with its responsibility to "Prepare telemetry metric packet for distribution to event stream."
    # The `publish_stream_packet` function will then simply return this prepared packet.
    normalized_payload = normalize_dict_keys(data)
    enveloped_pkt = build_envelope(service_name=_SERVICE_NAME, epoch_ms=epoch_ms, payload=normalized_payload)
    # Validate the envelope immediately after building it to ensure correctness
    if not is_valid_envelope(enveloped_pkt):
        # This case should ideally not be reached if build_envelope is correct,
        # but it serves as a defensive check.
        raise ValueError("Failed to build a valid wire envelope for the telemetry packet.")
    return enveloped_pkt

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # The packet should already be fully prepared and enveloped by prepare_metric_packet.
    # This function now acts as a pass-through or a final publishing step.
    pkt = prepare_metric_packet(data, epoch_ms)
    # The original line `_ = extract_normalized_keys(data)` was likely for side effects or validation.
    # It should operate on the original data, not the already enveloped packet.
    _ = extract_normalized_keys(data)
    return pkt
