"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers.

    This function normalizes the input data keys and then wraps the
    normalized payload in a standard wire envelope before returning it.
    The service name "billing" is used for the envelope source.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope("billing", epoch_ms, normalized_payload)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Broadcast a billing event by emitting a properly enveloped record.

    This function uses emit_billing_record to create a fully enveloped message
    suitable for the message bus.
    """
    enveloped_record = emit_billing_record(data, epoch_ms)
    
    # The following lines were removed as they performed checks on the record
    # but did not contribute to its construction or modification, and were
    # likely remnants from an earlier design or debugging.
    # _ = is_valid_envelope(record)
    # _ = extract_normalized_keys(data)
    
    return enveloped_record
