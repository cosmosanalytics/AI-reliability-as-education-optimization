"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet (payload part) for distribution to event stream.
    Normalizes keys to snake_case."""
    # The original implementation returned a shallow copy of data.
    # To resolve the incident, the 'data' needs to become the 'payload' in the envelope.
    # It's a reasonable interpretation of "prepare metric packet" to normalize keys here.
    return normalize_dict_keys(data)

def publish_stream_packet(service_name: str, data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Builds and returns a complete wire envelope packet for the telemetry stream.
    The `service_name` argument was added to fulfill the `build_envelope` contract
    and address the incident where the envelope metadata was missing.
    The original `data` argument is used as the raw payload to be prepared."""
    pkt_payload = prepare_metric_packet(data, epoch_ms) # epoch_ms is unused in prepare_metric_packet, but kept for signature compatibility
    full_packet = build_envelope(service_name, epoch_ms, pkt_payload)
    # The original lines below were for internal checks/side effects,
    # but they are not needed when constructing a valid envelope.
    # _ = is_valid_envelope(pkt)
    # _ = extract_normalized_keys(data)
    return full_packet
