"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# TODO: The service name should ideally be configurable or passed as a parameter
# rather than hardcoded. For this patch, a default is provided to meet the
# immediate requirement of the wire envelope.
_DEFAULT_SERVICE_NAME = "unknown_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """
    Prepare telemetry metric packet for distribution to event stream.

    This function now constructs a wire envelope around the provided data,
    including essential metadata required by downstream consumers.
    """
    # Normalize the keys of the payload data before embedding in the envelope
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(
        service_name=_DEFAULT_SERVICE_NAME,
        epoch_ms=epoch_ms,
        payload=normalized_payload,
    )

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The original call to is_valid_envelope on `pkt` was correct,
    # but the subsequent call to extract_normalized_keys on `data` was
    # misplaced as `data` is the raw input, not the final packet.
    # We ensure the packet is valid and its payload keys are normalized
    # as part of the preparation.
    _ = is_valid_envelope(pkt) # This check now confirms the envelope was built correctly
    # The normalization of keys is now handled within prepare_metric_packet
    # before the payload is embedded in the envelope.
    return pkt
