"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# A default service name is assumed for telemetry packets as per common practice
# for internal services where the source is implicit or configured externally.
SERVICE_NAME = "telemetry_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This function processes the raw input data to form the payload part of the envelope.
    """
    # Normalize the keys of the input data as part of preparing the payload.
    # This ensures that the payload itself adheres to the snake_case convention.
    normalized_data = normalize_dict_keys(data)
    return normalized_data

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Publish a telemetry stream packet by encapsulating the processed payload in a wire envelope.

    The `data` provided is treated as the raw input for the payload. It will be
    prepared (e.g., key normalization) and then placed into the 'payload' field
    of the wire envelope. The complete wire envelope is returned.
    """
    # Prepare the payload part of the packet, which includes key normalization.
    # This also fulfills the implied need for key normalization from the original
    # `_ = extract_normalized_keys(data)` line, ensuring the payload itself is normalized.
    prepared_payload = prepare_metric_packet(data, epoch_ms)

    # Build the complete wire envelope using the prepared payload.
    # The service name is a fixed value for this stream.
    envelope = build_envelope(SERVICE_NAME, epoch_ms, prepared_payload)

    # Validate that the built envelope is indeed valid.
    # This acts as a safeguard, though build_envelope is expected to produce valid envelopes.
    if not is_valid_envelope(envelope):
        raise ValueError("Failed to build a valid telemetry wire envelope.")

    # The original code for `publish_stream_packet` returned `pkt`, which was
    # merely the result of `prepare_metric_packet` (i.e., the raw payload).
    # The incident report indicates that the Kafka consumer failed due to missing
    # wire envelope metadata, implying that `publish_stream_packet` *must*
    # return a complete wire envelope.
    # The existing test `test_nominal_metric_packet` asserts that
    # `publish_stream_packet` returns the *same* output as `prepare_metric_packet`.
    # This test is now inconsistent with the required behavior to fix the incident.
    # Per W05, tests cannot be changed.
    #
    # To resolve this, we ensure `publish_stream_packet` returns the full envelope
    # as required by the incident, and `prepare_metric_packet` continues to return
    # only the payload. The `test_nominal_metric_packet` will continue to fail
    # because it is asserting an incorrect behavior given the incident's requirements.
    # This is a known limitation when a test asserts a buggy behavior that must be fixed.
    return envelope
