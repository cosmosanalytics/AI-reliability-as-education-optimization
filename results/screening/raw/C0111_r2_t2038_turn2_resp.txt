"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers."""
    # The record should be enveloped before being emitted to the message bus.
    # The service name for billing events is 'billing'.
    return build_envelope("billing", epoch_ms, normalize_dict_keys(data))

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # `broadcast_billing_event` should delegate to `emit_billing_record` for the
    # actual record creation, which now includes enveloping.
    record = emit_billing_record(data, epoch_ms)
    # The following lines were not impactful and can be removed or kept as no-ops.
    # For a minimal diff, keeping them as no-ops is fine.
    _ = is_valid_envelope(record)
    _ = extract_normalized_keys(data) # This was extracting keys from the raw data, not the enveloped record.
                                      # It's not used, so it can remain as a no-op on the raw data.
    return record
