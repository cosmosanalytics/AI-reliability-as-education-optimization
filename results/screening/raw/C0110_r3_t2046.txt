"""Timestamp serialization codec for telemetry records."""
from datetime import datetime, timezone

def encode_iso_timestamp(dt: datetime) -> str:
    """Encode timestamp into standard ISO 8601 format (YYYY-MM-DDTHH:MM:SS.sssZ)."""
    # Ensure the datetime object is in UTC. The 'Z' suffix implies UTC.
    # We assume `dt` is already a UTC datetime object.
    
    # Format YYYY-MM-DDTHH:MM:SS
    base_format = dt.strftime("%Y-%m-%dT%H:%M:%S")
    # Add milliseconds (3 digits) and 'Z' for UTC
    milliseconds = dt.microsecond // 1000
    return f"{base_format}.{milliseconds:03d}Z"

def validate_timestamp_range(dt: datetime) -> bool:
    return dt.year >= 2020
