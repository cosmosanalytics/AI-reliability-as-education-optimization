"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define the service name for this telemetry stream as it's required by build_envelope.
# This assumes a fixed service name for this specific stream context.
SERVICE_NAME = "telemetry-service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This function now constructs the full wire envelope around the payload.
    """
    # The payload keys should also be normalized before being enveloped.
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The original call to is_valid_envelope was checking the raw data,
    # and extract_normalized_keys was also applied to raw data.
    # Now, prepare_metric_packet already produces a valid envelope with normalized keys inside.
    # The checks below are for validation or side effects, but not directly used for return.
    _ = is_valid_envelope(pkt) # This now correctly checks the built envelope
    # The original intent of extract_normalized_keys on 'data' was likely to process the payload.
    # Given that `prepare_metric_packet` now normalizes the keys *inside* the payload,
    # this line can be removed or adapted if 'extract_normalized_keys' was meant for the outer envelope.
    # For this fix, we assume the normalization happens within prepare_metric_packet.
    # If a specific check on the raw keys was needed, it would be done before normalization.
    return pkt
