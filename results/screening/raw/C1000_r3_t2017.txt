"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    pattern = policy.allowed_pattern

    pattern_segments = pattern.split('::')
    scope_segments = clean_req.split('::')

    p_idx = 0
    s_idx = 0

    while p_idx < len(pattern_segments) and s_idx < len(scope_segments):
        p_segment = pattern_segments[p_idx]
        s_segment = scope_segments[s_idx]

        if p_segment == "**":
            # If '**' is the last segment in the pattern, it matches any remaining scope segments.
            if p_idx == len(pattern_segments) - 1:
                return True
            
            # If '**' is not the last segment, it matches zero or more segments
            # Look for the next pattern segment in the remaining scope segments.
            # If found, try to match the rest of the pattern from that point in the scope.
            next_p_segment = pattern_segments[p_idx + 1]
            temp_s_idx = s_idx
            while temp_s_idx < len(scope_segments):
                # Recursive call or equivalent logic is needed here for a robust match.
                # For this specific problem, we can simplify: if the next pattern segment
                # matches any subsequent scope segment, and the rest of the pattern matches
                # the rest of the scope, then it's a match.
                # A simpler approach for '**' is to consume segments in the scope
                # until the *rest* of the pattern matches the *rest* of the scope.
                # This suggests a helper or a more complex loop.
                # Let's try to match greedily.
                
                # The current segment p_segment is "**".
                # We need to check if the rest of the pattern matches the rest of the scope
                # starting from any point from s_idx onwards.
                # This is a classic wildcard matching problem, often solved with recursion or dynamic programming.
                # For simplicity and given the constraints, let's implement a direct check.

                # If the next segment after '**' in the pattern matches the current scope segment
                # or any subsequent scope segment, we can potentially "jump" to that point.
                # The problem statement implies '**' matches prefixes, which is simpler than full globbing.
                # Let's re-evaluate the docstring: "double asterisk '**' matches arbitrary multi-segment prefixes."
                # This means "corp::**" should match "corp::billing::invoices".
                # If "corp::**::details" is the pattern, it implies "corp::(anything)::details".
                # This is more complex than a simple prefix match.

                # Let's consider the simpler interpretation:
                # `A::B::**` matches `A::B::C`, `A::B::C::D`
                # `A::**` matches `A::B`, `A::B::C`
                # `**` matches anything.

                # If `**` is encountered:
                # If it's the last pattern segment, it matches all remaining scope segments.
                if p_idx == len(pattern_segments) - 1:
                    return True
                
                # If not the last pattern segment, it matches zero or more scope segments.
                # We need to find if the *rest* of the pattern (`pattern_segments[p_idx+1:]`)
                # matches the *rest* of the scope (`scope_segments[s_idx:]`).
                # This requires trying all possible matches for '**'.
                # The most straightforward way is to iterate through possible matches for '**'
                # and recursively check the remaining parts.
                
                # Let's implement this iteratively by checking suffixes.
                # The pattern after '**' must match a suffix of the scope after s_idx.
                remaining_pattern = pattern_segments[p_idx + 1:]
                
                # If there's no remaining pattern, then '**' must match exactly up to the end of the scope.
                if not remaining_pattern:
                    return True # '**' matches remaining scope segments
                
                # Search for the first segment of the remaining pattern in the scope.
                found_match_for_suffix = False
                for k in range(s_idx, len(scope_segments)):
                    if _match_segments(remaining_pattern, scope_segments[k:]):
                        return True # Found a match for the suffix
                return False # No match found for the suffix after '**'
                
        elif p_segment == "*":
            # '*' matches exactly one segment.
            # If the scope has no more segments, it's a mismatch.
            if s_idx >= len(scope_segments):
                return False
            # Otherwise, consume one segment from both pattern and scope.
            p_idx += 1
            s_idx += 1
        elif p_segment == s_segment:
            # Exact segment match. Consume both.
            p_idx += 1
            s_idx += 1
        else:
            # Mismatch.
            return False

    # After loop, both must have exhausted their segments for a full match.
    return p_idx == len(pattern_segments) and s_idx == len(scope_segments)

# Helper function for matching suffixes after a '**' wildcard
def _match_segments(pattern_suffix: list, scope_suffix: list) -> bool:
    """Helper to check if a pattern suffix matches a scope suffix."""
    ps_idx = 0
    ss_idx = 0

    while ps_idx < len(pattern_suffix) and ss_idx < len(scope_suffix):
        ps_segment = pattern_suffix[ps_idx]
        ss_segment = scope_suffix[ss_idx]

        if ps_segment == "**":
            # This helper is called for suffixes *after* a '**'.
            # A '**' in a suffix implies the original pattern had multiple '**'s or it's part of a recursive call.
            # For simplicity, if we encounter another '**', it can match any remaining scope.
            # This is a simplification based on common globbing behavior where multiple '**' are often redundant
            # or treated as a single '**'. Assuming a single '**' is the primary intent.
            # If multiple '**' are truly intended to be distinct, the logic would be more complex.
            if ps_idx == len(pattern_suffix) - 1:
                return True
            
            # If not the last segment, search for the next pattern segment
            next_ps_segment = pattern_suffix[ps_idx + 1]
            found_next_segment = False
            for k in range(ss_idx, len(scope_suffix)):
                if _match_segments(pattern_suffix[ps_idx+1:], scope_suffix[k:]):
                    return True
            return False

        elif ps_segment == "*":
            if ss_idx >= len(scope_suffix):
                return False # '*' needs one segment, but scope exhausted
            ps_idx += 1
            ss_idx += 1
        elif ps_segment == ss_segment:
            ps_idx += 1
            ss_idx += 1
        else:
            return False # Mismatch

    return ps_idx == len(pattern_suffix) and ss_idx == len(scope_suffix)

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
