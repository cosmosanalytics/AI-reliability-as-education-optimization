"""Database pool manager."""
from typing import Dict, Any
from env_loader import merge_env_settings
from options import ConnectionOptions

def configure_pool(base_template: Dict[str, Any], overrides: Dict[str, Any]) -> ConnectionOptions:
    """Configure connection pool options from template and overrides.
    Returns ConnectionOptions instance.
    """
    settings = merge_env_settings(base_template, overrides)
    dsn = settings.get("dsn", "sqlite:///:memory:")
    # Ensure pool_params is a new dictionary instance to prevent shared mutable state
    # when base_template contains a 'pool' dictionary that is not fully overridden
    # and subsequent modifications to pool_params would affect other ConnectionOptions
    # instances sharing the same dictionary reference.
    pool_params = settings.get("pool", {}).copy()
    opts = ConnectionOptions(dsn=dsn, pool_params=pool_params)
    _ = opts.summary()
    return opts
