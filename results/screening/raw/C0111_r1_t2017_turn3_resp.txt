"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def _split_scope_segments(scope: str) -> list[str]:
    """Splits a scope string into segments by '::'."""
    return scope.split("::")

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    pattern_segments = _split_scope_segments(policy.allowed_pattern)
    request_segments = _split_scope_segments(clean_req)

    p_idx = 0  # Pointer for pattern_segments
    r_idx = 0  # Pointer for request_segments

    while p_idx < len(pattern_segments) and r_idx < len(request_segments):
        pattern_segment = pattern_segments[p_idx]
        request_segment = request_segments[r_idx]

        if pattern_segment == "**":
            # Double asterisk matches zero or more segments.
            # If '**' is the last pattern segment, it matches everything remaining in the request.
            if p_idx == len(pattern_segments) - 1:
                return True
            # Otherwise, we need to find the next non-wildcard pattern segment in the request.
            # Advance request_segments pointer until the next pattern segment is found or end of request.
            next_pattern_segment = pattern_segments[p_idx + 1]
            found_match_for_next_segment = False
            while r_idx < len(request_segments):
                if request_segments[r_idx] == next_pattern_segment:
                    # Potential match found for the segment after '**'.
                    # Try to match the rest of the pattern from this point.
                    # This is effectively a recursive call or a backtracking step.
                    # To avoid explicit recursion, we'll try to find the earliest match.
                    # If the remaining pattern matches the remaining request from this r_idx, it's a match.
                    # This is a bit tricky to implement iteratively without full backtracking.
                    # A simpler approach for the iterative solution is to check if the suffix matches.
                    # If we find a match for next_pattern_segment, we try to match the rest.
                    # If it fails, we continue looking for the next_pattern_segment further in request_segments.
                    # This implies a nested loop or a helper function.

                    # Let's simplify: if we see "**", we try to match the rest of the pattern
                    # from the current request_segment pointer, moving it forward.
                    # This requires a look-ahead.
                    # A common way to handle this is to iterate backwards from the end of the request_segments
                    # when '**' is encountered, to find the longest possible match for the preceding parts.

                    # For a forward pass, when '**' is hit:
                    # We need to find `pattern_segments[p_idx+1:]` in `request_segments[r_idx:]`.
                    # Let's use a helper for suffix matching.
                    # If a suffix match is found, then the whole thing matches.
                    remaining_pattern = pattern_segments[p_idx + 1:]
                    remaining_request = request_segments[r_idx:]

                    # Check if the remaining pattern (after **) matches any suffix of the remaining request
                    # starting from current r_idx.
                    # This becomes a sub-problem.
                    # To avoid complex iteration, for '**', we can make a simplifying assumption:
                    # if '**' is found, we effectively skip segments in request until the next pattern segment matches.

                    # Let's re-evaluate the greedy approach for '**':
                    # If `**` is not the last segment, it means we must match `pattern_segments[p_idx+1]`
                    # somewhere in `request_segments[r_idx:]`.
                    # Once we find it, we can continue matching.
                    temp_r_idx = r_idx
                    found_next_segment_in_request = False
                    while temp_r_idx < len(request_segments):
                        if request_segments[temp_r_idx] == next_pattern_segment:
                            # Found the segment. Now, try to match the rest of the pattern
                            # from this point.
                            # This is where a recursive helper would be clean.
                            # Without recursion, we essentially need to jump p_idx and temp_r_idx,
                            # and hope the rest matches. This is not robust for multiple '**'.
                            # Let's use a standard implementation for glob-like matching.

                            # The simplest iterative way for `**` is to consider it matching zero segments.
                            # If `pattern_segments[p_idx+1:]` matches `request_segments[r_idx:]`, then `**` matched zero.
                            # Otherwise, try matching `**` to one segment, then two, etc.

                            # This problem is a variation of string matching with wildcards,
                            # which is often solved with dynamic programming or a two-pointer approach
                            # with backtracking/recursion.
                            # For a small, correct patch, let's implement the two-pointer with backtracking
                            # for the `**` case.

                            # To handle `**` matching multiple segments:
                            # We can try to match the rest of the pattern starting from `p_idx + 1`
                            # against the request starting from various `r_idx_candidate` positions.
                            # This is best done with a helper function that can be called recursively or
                            # simulate recursion.

                            # Let's try an alternative: split into functions.
                            # `_match_segments(pattern_idx, request_idx)`
                            pass # Re-thinking this part.
                    # If this loop completes without a return, it means we need to find a match.
                    # The current simple `while` loop structure makes `**` hard.

                    # Let's try matching from the end when '**' is encountered.
                    # This is also complex.

                    # A simpler iterative approach for `**` is to advance `p_idx` and `r_idx` as normal
                    # for non-`**` segments. When `**` is hit, store its position.
                    # If a mismatch occurs later, backtrack `p_idx` to `**`'s stored position,
                    # advance `r_idx` past where `**` previously matched, and try again.

                    # This is the standard "glob matching" algorithm.
                    # p_idx: current index in pattern_segments
                    # r_idx: current index in request_segments
                    # asterisk_p_idx: last seen '*' or '**' pattern index
                    # asterisk_r_idx: request index where the last '*' or '**' match started

                    return _match_segments_recursive(pattern_segments, request_segments, 0, 0)

        elif pattern_segment == "*":
            # Single asterisk matches exactly one segment.
            p_idx += 1
            r_idx += 1
        elif pattern_segment == request_segment:
            # Exact match.
            p_idx += 1
            r_idx += 1
        else:
            # Mismatch.
            return False

    # unless a '**' consumed the rest of the request.
    # The current `while` loop structure is not suitable for the `**` case.
    # I need to rewrite the entire `is_scope_permitted` logic to handle this.
    # Let's use a helper recursive function.
    return _match_segments_recursive(pattern_segments, request_segments, 0, 0)


def _match_segments_recursive(pattern_segments: list[str], request_segments: list[str], p_idx: int, r_idx: int) -> bool:
    """Helper for recursive glob-style matching."""
    # Base case: All pattern segments matched.
    if p_idx == len(pattern_segments):
        return r_idx == len(request_segments) # Request must also be fully consumed.

    # Base case: All request segments consumed, but pattern still remains.
    if r_idx == len(request_segments):
        # Only remaining pattern segments can be '**' that matches zero.
        # Check if all remaining pattern segments are '**'.
        for i in range(p_idx, len(pattern_segments)):
            if pattern_segments[i] != "**":
                return False
        return True

    pattern_segment = pattern_segments[p_idx]

    if pattern_segment == "**":
        # '**' can match zero segments or one or more segments.
        # Option 1: '**' matches zero segments. Try matching the rest of the pattern against current request_segment.
        if _match_segments_recursive(pattern_segments, request_segments, p_idx + 1, r_idx):
            return True
        # Option 2: '**' matches one or more segments. Consume current request_segment and try again with '**'.
        if _match_segments_recursive(pattern_segments, request_segments, p_idx, r_idx + 1):
            return True
        return False
    elif pattern_segment == "*":
        # '*' matches exactly one segment.
        return _match_segments_recursive(pattern_segments, request_segments, p_idx + 1, r_idx + 1)
    else:
        # Literal segment match.
        if pattern_segment == request_segments[r_idx]:
            return _match_segments_recursive(pattern_segments, request_segments, p_idx + 1, r_idx + 1)
        else:
            return False


def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
