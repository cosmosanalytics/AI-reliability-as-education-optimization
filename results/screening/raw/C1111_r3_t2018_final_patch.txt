"""Document scope RBAC filter."""
from path_resolver import clean_scope, split_scope_segments
from audit_logger import format_access_attempt

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern.

    A scope is permitted if:
    1. The `allowed_scope` is an exact match for the `requested_scope` (segment by segment).
    2. The `allowed_scope` ends with `::*` and the `requested_scope` starts with the
       `allowed_scope` prefix (before `::*`), segment by segment.
    3. The `allowed_scope` is a segment-by-segment prefix of `requested_scope` (e.g.,
       `corp::public` allows `corp::public::readme`). This implies that
       `corp::public` does *not* allow `corp::public_internal`.
    """
    cleaned_allowed = clean_scope(allowed_scope)
    cleaned_requested = clean_scope(requested_scope)

    requested_segments = split_scope_segments(cleaned_requested)

    allowed = False
    if cleaned_allowed.endswith("::*"):
        # Handle wildcard access: "corp::public::*" allows "corp::public::readme"
        # The prefix segments are everything before "::*"
        prefix_str = cleaned_allowed.removesuffix("::*")
        prefix_segments = split_scope_segments(prefix_str)

        if len(requested_segments) >= len(prefix_segments):
            # Check if the requested scope starts with the prefix segments
            allowed = requested_segments[:len(prefix_segments)] == prefix_segments
    else:
        # Handle exact segment prefix access: "corp::public" allows "corp::public::readme"
        # but not "corp::public_internal::secret".
        # This requires the allowed_scope segments to be a prefix of the requested_scope segments.
        allowed_segments = split_scope_segments(cleaned_allowed)
        if len(requested_segments) >= len(allowed_segments):
            # Check if the requested scope starts with the allowed scope segments
            allowed = requested_segments[:len(allowed_segments)] == allowed_segments

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
