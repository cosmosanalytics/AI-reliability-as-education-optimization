"""CI/CD pipeline step composition engine."""
from typing import Dict, Any, List
from step_schema import PipelineStep
from env_validator import sanitize_env_vars, is_valid_step_name

def merge_step_args(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    merged = dict(base)
    for k, v in override.items():
        if k in merged:
            if isinstance(merged[k], dict) and isinstance(v, dict):
                # Recursively merge dictionaries
                merged[k] = merge_step_args(merged[k], v)
            elif isinstance(merged[k], list) and isinstance(v, list):
                # Concatenate lists, preserving base elements and adding override elements
                # Ensure uniqueness if elements are hashable, or just extend if order/duplicates matter.
                # Given 'compilation flags', uniqueness is usually desired.
                # Convert to set for unique elements, then back to list.
                # Preserve order of base elements, then add new override elements.
                unique_elements = list(merged[k]) # Start with base elements
                for item in v:
                    if item not in unique_elements: # Add only if not already present
                        unique_elements.append(item)
                merged[k] = unique_elements
            else:
                # Overwrite for other types
                merged[k] = v
        else:
            # Add new key from override
            merged[k] = v
    return merged

def compose_pipeline_step(step_id: str, command: str, base_args: Dict[str, Any], env_args: Dict[str, Any]) -> PipelineStep:
    if not is_valid_step_name(step_id):
        raise ValueError("Invalid step ID")
    merged_args = merge_step_args(base_args, env_args)
    env = sanitize_env_vars(merged_args.get("env", {}))
    return PipelineStep(step_id=step_id, command=command, args=merged_args, env_vars=env)
