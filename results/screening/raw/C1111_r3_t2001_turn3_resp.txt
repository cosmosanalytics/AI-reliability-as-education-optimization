"""Low-level tax rate computation engine."""
from typing import Dict, Any

DEFAULT_TAX_JURISDICTIONS = {
    "standard": 0.20,
    "reduced": 0.10,
    "exempt": 0.0,
}

def get_rate_for_jurisdiction(code: str) -> float:
    return DEFAULT_TAX_JURISDICTIONS.get(code, 0.20)

def calculate_vat(net_cents: int, vat_rate: float) -> int:
    """Compute VAT in integer cents for given net amount.
    Uses half-up rounding (ties broken towards positive infinity) on exact half-cents.
    """
    # Implement half-up rounding with ties broken towards positive infinity.
    # Adding 0.5 and then truncating (int()) achieves this for both positive and negative numbers.
    # E.g., 13.5 becomes 14.0 -> 14.
    # E.g., -13.5 becomes -13.0 -> -13.
    # E.g., 13.4 becomes 13.9 -> 13.
    # E.g., -13.6 becomes -13.1 -> -13.
    return int(net_cents * vat_rate + 0.5)
