"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a default service name for the telemetry stream, as required by the wire envelope.
# This can be configured or passed as an argument in a more complex setup.
_DEFAULT_SERVICE_NAME = "telemetry_stream_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This includes normalizing keys and wrapping the data in a wire envelope.
    """
    # Normalize the keys of the payload data
    normalized_payload = normalize_dict_keys(data)
    # Build the wire envelope around the normalized payload, including required metadata.
    return build_envelope(_DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following lines are for validation or side effects, and their return values
    # are currently unused. They remain as per existing code structure and do not
    # affect the primary fix.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # Note: this operates on original 'data', not the 'normalized_payload' within 'pkt'
    return pkt
