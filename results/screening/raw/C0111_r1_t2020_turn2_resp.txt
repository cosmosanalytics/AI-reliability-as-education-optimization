"""Topic path subscription matcher."""
def is_topic_subscribed(subscribed_pattern: str, event_topic: str) -> bool:
    # A subscription matches if the event topic is exactly the subscribed pattern,
    # or if the event topic starts with the subscribed pattern followed by the '::' delimiter.
    # This prevents 'orders' from matching 'orders_admin'.
    return event_topic == subscribed_pattern or event_topic.startswith(subscribed_pattern + "::")

def clean_topic_segment(topic: str) -> str:
    return topic.strip().lower()

def is_wildcard_subscription(pattern: str) -> bool:
    return pattern.strip() in ("*", "**")
