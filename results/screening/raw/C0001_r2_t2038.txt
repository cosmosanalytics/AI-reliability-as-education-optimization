"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers.

    This function normalizes the keys of the input data to snake_case,
    preparing it as the payload for an enveloped message. The `epoch_ms`
    argument is passed along but not directly used in this step.
    """
    # Normalize keys as implied by the import of normalize_dict_keys
    # and the module's responsibility for preparing billing records.
    return normalize_dict_keys(data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Broadcast a billing event by enveloping the billing record.

    The input `data` is first processed by `emit_billing_record` to
    normalize its keys, and then this prepared payload is wrapped in a
    wire envelope suitable for the message bus. The service name for
    the envelope is hardcoded as "billing".
    """
    # Prepare the billing record payload by normalizing its keys.
    record_payload = emit_billing_record(data, epoch_ms)

    # Build the wire envelope around the prepared payload.
    # The service name "billing" is derived from the module context.
    enveloped_message = build_envelope(service_name="billing", epoch_ms=epoch_ms, payload=record_payload)

    # The original calls to is_valid_envelope and extract_normalized_keys were
    # side-effect-free and their return values were ignored.
    # We now return the properly enveloped message.
    return enveloped_message
