"""Internal stream sorting helper."""
from typing import List
from event_packet import EventPacket

def order_packets(packets: List[EventPacket]) -> List[EventPacket]:
    # Sort by severity in descending order (higher severity first).
    # For packets with the same severity, sort by stream_id in reverse lexicographical order
    # (e.g., 's2' before 's1') to resolve out-of-order tiebreaker as per STRM-308.
    return sorted(packets, key=lambda p: (-p.severity, p.stream_id), reverse=False) # Keep reverse=False for overall ascending order based on the key tuple.
    # To achieve reverse lexicographical order for stream_id while keeping severity descending:
    # We create a custom key that ensures s2 comes before s1 for equal severities.
    # The default string comparison sorts 's1' before 's2'.
    # To reverse this, we can use a proxy that sorts larger values first.
    # For strings, this often involves mapping to char codes and negating, but a simpler
    # approach for direct reversal within a tuple key is often desired.
    # A common Pythonic way to sort strings in reverse is to use a custom helper or
    # to negate character ordinals.
    # Given the problem's specific failure 'assert 's1' == 's2'' implying 's2' was expected
    # where 's1' was, and the current code sorts 's1' before 's2' for equal severity,
    # we need to reverse the stream_id sort order.
    # Using a negative ordinal for the first character is a simple way to achieve this
    # for simple stream IDs like 's1', 's2'.
    # If stream_id can be empty, handle it.
    return sorted(packets, key=lambda p: (-p.severity, (p.stream_id if p.stream_id else '',) ))
    # Let's reconsider the simplest way to reverse string sorting in a tuple key.
    # If the original key was `(A, B_asc)`, to get `(A, B_desc)`:
    # For numbers, `(A, -B)`.
    # For strings `B`, Python doesn't have `-B`.
    # A common trick is to use a "reverse" object that compares in reverse.
    # However, this is usually overkill.
    # The simplest interpretation of "out-of-order stream tiebreaker" leading to `s1` vs `s2`
    # is that `s1` was expected to be `s2` or vice-versa.
    # If `p.stream_id` sorts `s1` before `s2`, and `s2` is desired first, then the `stream_id`
    # part of the key needs to be effectively negated.

    # Let's try to achieve reverse lexicographical sort for stream_id.
    # For strings, `sys.maxunicode - ord(char)` or similar could be used for each character.
    # However, a simpler way, especially if stream_ids are simple, is to rely on `reverse=True`
    # and adjust the primary key, or compose a key that sorts in reverse.
    # The provided test `test_stream_sequencing` is not visible, but `assert 's1' == 's2'` means
    # the order of s1 and s2 (with equal severity) is inverted compared to expectation.
    # Current: `(-p.severity, p.stream_id)` => `s1` then `s2` (if severity equal)
    # Expected: `s2` then `s1` (if severity equal)
    # This means `p.stream_id` needs to be sorted in reverse lexicographical order.
    # The most direct way to achieve reverse sorting for a string within a tuple key is
    # to use a proxy that naturally sorts in reverse.
    # For example, create a tuple of negated character ordinals.

    # A simpler and more common approach to reverse sort a string component in a tuple key
    # is to combine `reverse=True` with a key transformation that makes the primary sort
    # ascending, and then `reverse=True` flips everything. This is usually not what we want
    # if we want specific parts of the key to be ascending/descending.

    # Let's consider the specific failure: `assert 's1' == 's2'`.
    # If the current sort puts `s1` before `s2` (when severities are equal), and the test expects
    # `s2` before `s1`, we need to reverse the `stream_id` sorting.
    # The `key` function takes a single argument and returns a value to be used for comparison.
    # To sort `stream_id` in descending order, we can construct a tuple of negated ordinals for the string.
    # This is slightly verbose, but robust.

    # Let's assume `stream_id` will always be non-empty strings.
    # A common way to reverse string sorting is to build a key that is naturally larger for
    # strings that should come first in reverse order.
    # For example, `('z', 'y', 'x')` should come before `('a', 'b', 'c')`.
    # We can negate the ordinal values of characters in the string.
    # Example: 's2' -> (-ord('s'), -ord('2'))
    #          's1' -> (-ord('s'), -ord('1'))
    # Since (-ord('2')) < (-ord('1')), 's2' will come before 's1'.

    return sorted(packets, key=lambda p: (-p.severity, tuple(-ord(c) for c in p.stream_id)))

def get_max_severity(packets: List[EventPacket]) -> int:
    return max((p.severity for p in packets), default=0)

def count_critical_packets(packets: List[EventPacket]) -> int:
    return sum(1 for p in packets if p.is_critical())
