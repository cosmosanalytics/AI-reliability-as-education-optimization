"""Database pool manager."""
from typing import Dict, Any
from env_loader import merge_env_settings
from options import ConnectionOptions

def configure_pool(base_template: Dict[str, Any], overrides: Dict[str, Any]) -> ConnectionOptions:
    """Configure connection pool options from template and overrides.
    Returns ConnectionOptions instance.
    """
    settings = merge_env_settings(base_template, overrides)
    dsn = settings.get("dsn", "sqlite:///:memory:")

    # The default empty dictionary for 'pool' in settings.get("pool", {})
    # is a mutable object. If the 'pool' key is not present in settings,
    # settings.get("pool", {}) would return the same default empty dictionary
    # object every time. If this shared default dictionary is then modified
    # by one ConnectionOptions instance, it would affect all other instances
    # that received the same default, leading to corrupted state.
    # To prevent this, we ensure that ConnectionOptions always receives a
    # fresh, independent copy of the pool parameters.
    pool_settings_from_config = settings.get("pool")
    if pool_settings_from_config is None:
        # If 'pool' key is not present, provide a new empty dictionary.
        pool_settings = {}
    else:
        # If 'pool' key is present, create a copy of the dictionary
        # to ensure isolation from the original settings object and
        # other ConnectionOptions instances.
        pool_settings = dict(pool_settings_from_config)

    opts = ConnectionOptions(dsn=dsn, pool_params=pool_settings)
    _ = opts.summary()
    return opts
