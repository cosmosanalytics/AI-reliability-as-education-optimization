"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Incident WIRE-801: Downstream aggregator dropped raw un-enveloped audit records missing version 2 wrapper.
# The `service_name` for the envelope is hardcoded as per the incident context
# and the absence of a `service_name` parameter in `format_audit_event`.
SERVICE_NAME = "audit_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(SERVICE_NAME, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # The original _ = is_valid_envelope(msg) and _ = extract_normalized_keys(data)
    # calls were likely for debugging or partial implementation.
    # The format_audit_event now correctly wraps and normalizes,
    # so these explicit checks are not strictly necessary for the return value,
    # but keeping them if they serve an unstated purpose.
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data) # This extracts from original `data`, not `msg`'s payload
    return msg
