"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# TODO: The service name should ideally be configured or passed dynamically.
# For now, a placeholder is used to ensure the envelope is built correctly.
_DEFAULT_SERVICE_NAME = "unknown_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This includes normalizing keys and wrapping the payload in a wire envelope.
    """
    normalized_payload = normalize_dict_keys(data)
    # The build_envelope function requires a service_name.
    # For now, a default service name is used.
    # Future work may involve passing this dynamically.
    return build_envelope(_DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Publish a telemetry stream packet after preparing it."""
    pkt = prepare_metric_packet(data, epoch_ms)
    # The original lines checking is_valid_envelope and extracting normalized keys
    # were effectively no-ops and did not alter the packet structure being returned.
    # The prepare_metric_packet function now correctly builds the full enveloped packet.
    return pkt
