"""Low-level jitter percentile calculation engine."""
import math
from typing import List

def evaluate_jitter_percentile(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using the nearest-rank method.
    Index k = math.ceil(p / 100 * n) - 1 on sorted values.
    """
    if not values:
        raise ValueError("No jitter values")
    
    s = sorted(values)
    n = len(s)

    # The ticket specifies nearest-rank percentile estimation: index k = math.ceil(p / 100 * n) - 1
    # This method is often called 'R-1' or 'Nearest Rank' method.
    # It directly picks an observation from the sorted list.
    
    # Ensure p is within [0, 100]
    if not (0 <= p <= 100):
        raise ValueError("Percentile 'p' must be between 0 and 100.")

    if p == 0: # 0th percentile is the minimum value
        return s[0]
    if p == 100: # 100th percentile is the maximum value
        return s[n - 1]

    # Calculate rank based on the nearest-rank method
    rank = (p / 100.0) * n
    
    # Use math.ceil to get the 1-based index, then subtract 1 for 0-based index
    k = int(math.ceil(rank)) - 1
    
    # Ensure k is within bounds [0, n-1]
    # For p > 0 and p <= 100, k will naturally be between 0 and n-1.
    # E.g., if rank is 0.1, ceil(0.1) = 1, k=0. If rank is n, ceil(n)=n, k=n-1.
    
    return float(s[k])
