"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# WIRE-801: Downstream aggregator requires a service_name in the envelope.
# The docstring for format_audit_event implies a 'service_name' but it's not
# in the signature. Assuming a default service name for existing callers
# based on the incident report context, as changing the signature is not
# explicitly requested and would break existing callers without modification.
DEFAULT_SERVICE_NAME = "audit-service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    # Normalize payload keys as per docstring
    normalized_payload = normalize_dict_keys(data)
    
    # Build the version 2 envelope using the default service name
    return build_envelope(DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # The following lines are likely for internal checks or debugging and do not
    # affect the return value or external behavior, but they are retained as
    # per W03 (smallest change).
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data) # Note: this extracts keys from raw data, not the normalized payload
    return msg
