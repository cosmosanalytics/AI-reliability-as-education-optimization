"""Low-level dictionary and list hierarchy merger."""
from typing import Dict, Any, List

def merge_spec_values(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """Recursively merges dictionary values.
    
    When a key exists in both base and override, and both values are lists,
    the lists are combined (concatenated) to ensure all elements are present.
    For other types, override values replace base values.
    """
    result = dict(base)
    for k, v in override.items():
        if k in result and isinstance(result[k], dict) and isinstance(v, dict):
            result[k] = merge_spec_values(result[k], v)
        elif k in result and isinstance(result[k], list) and isinstance(v, list):
            # Combine lists to ensure all elements from both base and override are present.
            # Order is preserved, and duplicates are allowed as per the incident's implication
            # of ensuring availability rather than strict uniqueness.
            result[k] = result[k] + v
        else:
            result[k] = v
    return result

def is_spec_empty(spec: Dict[str, Any]) -> bool:
    return len(spec) == 0
