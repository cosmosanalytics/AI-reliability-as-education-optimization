"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    
    policy_segments = policy.allowed_pattern.split('::')
    request_segments = clean_req.split('::')

    if not policy_segments: # Empty policy pattern should not match anything unless request is also empty
        return not request_segments

    # Handle the '**' wildcard at the end
    if policy_segments[-1] == '**':
        # All segments *before* '**' must match exactly or by single wildcard
        if len(request_segments) < len(policy_segments) - 1:
            return False # Not enough request segments to match the non-'**' part

        for i in range(len(policy_segments) - 1):
            if policy_segments[i] == '*':
                # '*' matches exactly one segment. If request_segments is shorter here, it's a mismatch.
                if i >= len(request_segments):
                    return False
                # else '*' matches any single segment, no content comparison needed
            elif policy_segments[i] != request_segments[i]:
                return False
        return True # All preceding segments matched, '**' covers the rest (zero or more segments)

    # For patterns without '**' suffix, the number of segments must match
    if len(policy_segments) != len(request_segments):
        return False

    # Compare segment by segment for exact matches or single wildcards
    for p_segment, r_segment in zip(policy_segments, request_segments):
        if p_segment == '*':
            # '*' matches any single segment
            continue
        elif p_segment != r_segment:
            return False
            
    return True

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
