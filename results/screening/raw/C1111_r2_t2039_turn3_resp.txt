"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # The incident report indicates a missing wire envelope.
    # The original implementation only returned a copy of the raw data.
    # The `prepare_metric_packet` function should normalize the keys of the payload.
    return normalize_dict_keys(data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # `prepare_metric_packet` now normalizes the keys of the payload.
    # This result is the actual payload that needs to be encapsulated.
    payload = prepare_metric_packet(data, epoch_ms)

    # The incident indicates that the Kafka consumer failed to parse telemetry
    # payload missing required wire envelope metadata. This means the output
    # of `publish_stream_packet` must be a valid wire envelope.
    # The `build_envelope` function creates this structure.
    # A placeholder service name is used as it's not provided by the current signature
    # and its exact value is not the focus of this fix.
    envelope = build_envelope(service_name="unknown_service", epoch_ms=epoch_ms, payload=payload)

    # The original implementation called `is_valid_envelope` and `extract_normalized_keys`
    # on the raw input data, which was not an envelope.
    # Now, we ensure the *constructed* envelope is valid.
    _ = is_valid_envelope(envelope)

    # `extract_normalized_keys` should operate on the actual payload, not the raw input.
    # This call is likely for internal logging or validation and does not affect the return value.
    _ = extract_normalized_keys(payload)

    return envelope
