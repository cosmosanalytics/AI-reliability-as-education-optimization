"""Low-level dictionary and list hierarchy merger."""
from typing import Dict, Any, List

def merge_spec_values(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """Recursively merges dictionary values.

    For dictionary values, merges them recursively.
    For list values, extends the base list with unique items from the override list,
    preserving order.
    For all other types, the override value replaces the base value.
    """
    result = dict(base)
    for k, v in override.items():
        if k in result and isinstance(result[k], dict) and isinstance(v, dict):
            result[k] = merge_spec_values(result[k], v)
        elif k in result and isinstance(result[k], list) and isinstance(v, list):
            # Merge lists: append unique items from override to base, preserving order
            merged_list = list(result[k]) # Start with base list elements
            for item in v:
                if item not in merged_list:
                    merged_list.append(item)
            result[k] = merged_list
        else:
            result[k] = v
    return result

def is_spec_empty(spec: Dict[str, Any]) -> bool:
    return len(spec) == 0
