"""Cache Hit Latency Percentile Monitor percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_cache_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_cache_percentile", samples)
    s = validate_metric_samples(win.values)

    # Adjust percentile calculation to align with "nearest-rank house standard".
    # This typically means that the index is calculated based on the total number of
    # samples (n), and for non-integer ranks, a specific interpolation method
    # (often linear interpolation between the two nearest ranks) is used.
    # The common formula for index 'i' for the p-th percentile in a 0-indexed
    # sorted array of n elements is i = (p/100) * (n-1) for Method 7 (NIST),
    # or sometimes i = (p/100) * n or i = (p/100) * (n+1) for other definitions.
    # The traceback indicates that for p=50.0 and even number of samples,
    # the expected result is the lower of the two middle values, not their average.
    # This aligns with a definition where the index is 1-based, k = ceil((p/100)*n),
    # and the result is s[k-1]. For linear interpolation, this implies using
    # a 0-based index `i = (p / 100.0) * len(s) - 1` and then clamping.
    # A simpler approach for "nearest-rank" that matches the observed failure
    # is to compute the index based on `p * n / 100` and then use `math.floor`
    # for the index.
    
    if not s: # validate_metric_samples already checks this, but good for robustness
        raise ValueError("Samples list cannot be empty")

    n = len(s)
    # The "nearest-rank house standard" for 50th percentile on an even list
    # like [10, 20, 30, 40, 50, 60, 70, 80] expecting 40.0 (s[3]) instead of 45.0
    # implies that for a fractional index, it should round down to the nearest
    # existing sample for the percentile calculation, rather than interpolating.
    # A common formula for this is `k = ceil(p * n / 100)`.
    # Using 0-based indexing, this means `index = math.ceil(p * n / 100) - 1`.
    # However, this method for p=50, n=8 gives ceil(4)-1 = 3 -> s[3]=40.
    # For p=100, n=3, gives ceil(3)-1 = 2 -> s[2]=30.
    # This seems to match the requirements.

    # Ensure p is within valid range [0, 100]
    if not (0.0 <= p <= 100.0):
        raise ValueError("Percentile 'p' must be between 0.0 and 100.0")

    if p == 0.0:
        return s[0]
    if p == 100.0:
        return s[n - 1]

    # Calculate the 1-based rank k
    k = math.ceil((p / 100.0) * n)
    
    # Convert to 0-based index
    idx = int(k) - 1
    
    # Ensure index is within bounds (should be handled by k calculation for 0 < p < 100)
    # The result should directly be the value at this index.
    return s[idx]


def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
