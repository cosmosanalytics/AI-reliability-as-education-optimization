"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    This implementation uses the "linear interpolation between the closest two ranks"
    method (often referred to as Method 7 in NIST, or the default in NumPy's
    `percentile` function).
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)

    # Handle edge cases for p=0 and p=100
    if p == 0:
        return float(s[0])
    if p == 100:
        return float(s[n - 1])

    # The index calculation for percentile.
    # The original formula `(p / 100.0) * (n - 1)` is correct for Method 7.
    # The issue description indicates an off-by-one or off-by-half error for a specific case.
    # The formula for `idx` should be `(p / 100.0) * (n - 1)`.
    # However, for consistency with common statistical software like Pandas/Excel,
    # which often use an index calculation equivalent to `(p / 100.0) * n - 0.5`
    # or similar variations, the issue might be expecting a different percentile definition.
    #
    # Given the specific error (45.0 vs 40.0), it implies an off-by-half error.
    # If the desired value was `s[k]` but `s[k] + 0.5 * (s[k+1] - s[k])` was calculated,

    #
    # Let's consider the standard definition of percentile where `q` is the desired quantile
    # (p/100), and `i = q * n`.
    # If `i` is an integer, the value is `(s[i-1] + s[i]) / 2`.
    # If `i` is not an integer, the value is `s[floor(i)]`.
    #
    # The existing implementation calculates the index `idx` such that `s[floor(idx)]`
    # is the value for `p = 100 * floor(idx) / (n-1)`.
    #
    # A common alternative index calculation, often used by systems expecting a
    # slightly different interpolation behavior, is `(p / 100.0) * n`.
    # However, this usually leads to larger discrepancies.
    #
    # The most probable cause for 45.0 vs 40.0 (where 40.0 is `s[k]` and 45.0 is `s[k] + 0.5 * (s[k+1]-s[k])`)
    # implies that `idx` was `k + 0.5` instead of `k`.
    # This scenario points to a mismatch in how `idx` is calculated or interpreted.
    #
    # Let's consider the standard NumPy/Scipy `linear` interpolation method,
    # which calculates `idx = (n - 1) * p / 100.0`. This is the current formula.
    #
    # Another common method (e.g., in R, SAS, Excel) is the "nearest rank" method or
    # variations that use `(p / 100.0) * n`.
    #
    # Given the specific error `assert 45.0 == 40.0`, and assuming the `s` array
    # contains `... 40, 50 ...` at indices `k, k+1`, the current formula
    # produced an index `k.5` (leading to `40 + 0.5 * (50-40) = 45`),
    # when it should have produced `k` (leading to `40`).
    #
    # This suggests that `idx` might be consistently off by 0.5.
    # A common cause for this is using `(p / 100.0) * n` instead of `(p / 100.0) * (n - 1)`

    # Let's adjust the index calculation to align with a definition that provides
    # the exact value when the index is an integer, and interpolates otherwise,
    # often slightly shifted compared to the `(n-1)` method, particularly for
    # lower percentiles in small datasets.
    #
    # The formula `(p / 100.0) * n` is used in many statistical contexts.
    # If `idx = (p / 100.0) * n` is used:
    # `idx` is the rank.
    # If `idx` is an integer, the value is typically `s[idx-1]` (0-indexed).
    # If `idx` is not an integer, it's `s[floor(idx)]`.
    # This is slightly different from the interpolation logic here.
    #
    # The most robust fix, given the ambiguity of the precise percentile definition
    # expected by the failing test, is to use a definition that is less prone
    # to this specific type of half-step error.
    #
    # Let's consider the definition used by `scipy.stats.percentileofscore` or

    # If `idx` is an integer, `low = high = idx`, `frac = 0`, result is `s[idx]`.
