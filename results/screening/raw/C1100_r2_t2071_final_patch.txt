"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses.

    Uses the linear interpolation method, which is common for continuous data
    and consistent with widely used statistical packages.
    """
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)

    # The percentile calculation uses the linear interpolation method (R-1),
    # where the index L is calculated as p/100 * (n - 1).
    # If L is an integer, the value is s[L].
    # If L is not an integer, the value is interpolated between s[floor(L)] and s[ceil(L)].
    # This method is robust for various dataset sizes and percentile values.
    # See NIST: https://www.itl.nist.gov/div898/handbook/prc/section2/prc252.htm
    
    if n == 1:
        return float(s[0])

    # Ensure p is within [0, 100] range
    p = max(0.0, min(100.0, p))

    l_index = (p / 100.0) * (n - 1)

    if l_index.is_integer():
        return float(s[int(l_index)])
    else:
        lower_idx = math.floor(l_index)
        upper_idx = math.ceil(l_index)
        
        v1 = s[int(lower_idx)]
        v2 = s[int(upper_idx)]
        
        # Linear interpolation
        return float(v1 + (l_index - lower_idx) * (v2 - v1))

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
