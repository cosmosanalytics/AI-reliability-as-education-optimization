"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

_SERVICE_NAME = "billing_service"

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers."""
    # The billing payload needs to be normalized and then enveloped
    # before being emitted to the message bus.
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(_SERVICE_NAME, epoch_ms, normalized_payload)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # These lines are likely for verification or logging purposes.
    # The record is already correctly enveloped by emit_billing_record.
    _ = is_valid_envelope(record)
    # The keys of the original data are extracted here, but this is not
    # directly used in the final record, as emit_billing_record normalizes
    # the keys of the payload it envelops.
    _ = extract_normalized_keys(data)
    return record
