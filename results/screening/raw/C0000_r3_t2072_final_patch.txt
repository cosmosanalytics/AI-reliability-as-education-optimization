"""Audio buffer level metric processor."""
import math
from typing import List, Dict, Any
from audio_types import AudioStreamSnapshot, BufferSample
from audio_formatter import format_buffer_stat

def compute_buffer_percentile(snapshot: AudioStreamSnapshot, p: float) -> float:
    """Compute percentile using continuous linear interpolation (numpy-style: index = (p / 100.0) * (len(samples) - 1)).
    Do not use nearest-rank method for audio buffer metrics.
    """
    vals = snapshot.ratios()
    if not vals:
        raise ValueError("Empty snapshot")
    s = sorted(vals)
    n = len(s)

    # Handle edge cases for p outside [0, 100] or for very small/large n
    if p < 0.0:
        return float(s[0])
    if p >= 100.0:
        return float(s[n - 1])
    if n == 1:
        return float(s[0])

    # Calculate the index for continuous linear interpolation
    # Formula: index = (p / 100.0) * (n - 1)
    # This is consistent with numpy.percentile with default 'linear' interpolation.
    idx = (p / 100.0) * (n - 1)

    # Determine the two surrounding indices and the fractional part
    i = math.floor(idx)
    f = idx - i # fractional part

    # If the index is an integer (or very close due to float precision), return the exact value
    if f < 1e-9: # Check if f is effectively zero
        return float(s[i])
    
    # Linear interpolation between s[i] and s[i+1]
    # Ensure i+1 is within bounds, which it should be if idx < n-1
    return float(s[i] * (1.0 - f) + s[i+1] * f)

def evaluate_audio_buffer(snapshot: AudioStreamSnapshot, p: float) -> Dict[str, Any]:
    val = compute_buffer_percentile(snapshot, p)
    return format_buffer_stat(snapshot.channel_count(), val)
