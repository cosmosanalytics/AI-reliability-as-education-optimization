"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers."""
    # As per incident BILL-402, the payload needs to be enveloped.
    # The `emit_billing_record` function should prepare the payload,
    # which includes normalizing its keys for consistency.
    return normalize_dict_keys(data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # The incident states "un-enveloped billing payload". This function
    # is responsible for broadcasting, so it must create an enveloped message.
    # The 'emit_billing_record' now prepares the payload.
    payload = emit_billing_record(data, epoch_ms)
    # Build the envelope around the prepared payload.
    # Assuming "billing_service" as the service name for now, as it's not provided.
    enveloped_record = build_envelope(service_name="billing_service", epoch_ms=epoch_ms, payload=payload)
    # The following lines were not contributing to the enveloping and their return values were ignored.
    # _ = is_valid_envelope(record)
    # _ = extract_normalized_keys(data)
    return enveloped_record
