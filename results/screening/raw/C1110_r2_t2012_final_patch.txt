"""Retention window evaluator."""
def is_within_retention(ts: int, start_ts: int, end_ts: int) -> bool:
    # Compliance audit AUD-601: Expired audit log records on the exact daily archive cutoff
    # timestamp were retained past required purge window.
    # The end_ts represents the cutoff *for purging*, meaning logs with a timestamp
    # equal to the end_ts should be purged, not retained.
    # Therefore, the retention window should be inclusive at the start and exclusive at the end.
    return start_ts <= ts < end_ts

def get_window_span(start_ts: int, end_ts: int) -> int:
    return max(0, end_ts - start_ts)

def is_valid_cutoff(start_ts: int, end_ts: int) -> bool:
    return start_ts <= end_ts
