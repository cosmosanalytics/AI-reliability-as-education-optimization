"""Low-level tax rate computation engine."""
from typing import Dict, Any
from decimal import Decimal, ROUND_HALF_UP

DEFAULT_TAX_JURISDICTIONS = {
    "standard": 0.20,
    "reduced": 0.10,
    "exempt": 0.0,
}

def get_rate_for_jurisdiction(code: str) -> float:
    return DEFAULT_TAX_JURISDICTIONS.get(code, 0.20)

def calculate_vat(net_cents: int, vat_rate: float) -> int:
    """Compute VAT in integer cents for given net amount, using half-up rounding."""
    # Convert inputs to Decimal for precise calculation and rounding.
    # Quantize to an integer (0 decimal places) using ROUND_HALF_UP.
    net_decimal = Decimal(str(net_cents))
    rate_decimal = Decimal(str(vat_rate))
    vat_amount = net_decimal * rate_decimal
    rounded_vat = vat_amount.quantize(Decimal('1'), rounding=ROUND_HALF_UP)
    return int(rounded_vat)
