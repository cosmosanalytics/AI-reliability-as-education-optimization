"""Document scope RBAC filter."""
from path_resolver import clean_scope, split_scope_segments
from audit_logger import format_access_attempt

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern."""
    cleaned_allowed_scope = clean_scope(allowed_scope)
    cleaned_requested_scope = clean_scope(requested_scope)

    # Handle the wildcard case where allowed_scope ends with "::*"
    if cleaned_allowed_scope.endswith("::*"):
        prefix_scope = cleaned_allowed_scope[:-3]  # Remove "::*"
        allowed = cleaned_requested_scope.startswith(prefix_scope)
        # Ensure that if the prefix is "A", "A::B" is allowed, but "AB" is not.
        # This means the character immediately after the prefix must be '::' or the end of the string.
        if allowed and len(prefix_scope) < len(cleaned_requested_scope):
            if not cleaned_requested_scope[len(prefix_scope):].startswith("::"):
                allowed = False
        elif allowed and len(prefix_scope) == len(cleaned_requested_scope):
            # If prefix_scope is identical to requested_scope, it's allowed.
            pass
        elif not allowed: # if prefix doesn't match at all
            pass
        else: # if prefix_scope is empty, e.g. allowed_scope was just "::*"
            pass
    else:
        # For non-wildcard allowed scopes, perform strict segment-by-segment comparison
        # to prevent "corp::public" from matching "corp::public_internal".
        allowed_segments = split_scope_segments(cleaned_allowed_scope)
        requested_segments = split_scope_segments(cleaned_requested_scope)

        allowed = (
            len(requested_segments) >= len(allowed_segments) and
            all(req_seg == allow_seg for req_seg, allow_seg in zip(requested_segments, allowed_segments))
        )
        # Additionally, if allowed_scope does not end with '::*', it should only match itself
        # or direct children if the allowed_scope is a prefix.
        # The existing logic `len(requested_segments) >= len(allowed_segments)`
        # already handles `corp::public` allowing `corp::public` but not `corp::public::readme`
        # if the interpretation is strict (no implicit wildcard).
        # The nominal test `check_scope_access("corp::public", "corp::private::secret") is False`
        # passes with just the segment prefix check.
        # The original bug `check_scope_access("corp::public", "corp::public_internal::secret") is False`
        # is also handled by the segment prefix check.
        # The current logic for `else` block correctly implements strict prefix matching by segments.

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
