"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    
    pattern_segments = policy.allowed_pattern.split('::') if policy.allowed_pattern else []
    request_segments = clean_req.split('::') if clean_req else []

    p_idx = 0
    r_idx = 0

    while p_idx < len(pattern_segments) and r_idx < len(request_segments):
        pattern_segment = pattern_segments[p_idx]
        request_segment = request_segments[r_idx]

        if pattern_segment == "**":
            # '**' matches zero or more segments.
            # If it's the last pattern segment, it matches the rest of the request.
            if p_idx == len(pattern_segments) - 1:
                return True
            
            # If not the last segment, find the next non-wildcard pattern segment
            # and try to match it in the remaining request segments.
            next_pattern_segment = pattern_segments[p_idx + 1]
            temp_r_idx = r_idx
            while temp_r_idx < len(request_segments):
                if request_segments[temp_r_idx] == next_pattern_segment:
                    # Potential match, try matching the rest of the pattern from here
                    # Recursively check the remainder
                    if is_scope_permitted(RoutePolicy(policy.role, '::'.join(pattern_segments[p_idx+1:])), '::'.join(request_segments[temp_r_idx:])):
                        return True
                temp_r_idx += 1
            return False # No match found for the rest of the pattern

        elif pattern_segment == "*":
            # '*' matches exactly one segment.
            # Move to the next segments in both pattern and request.
            p_idx += 1
            r_idx += 1
        elif pattern_segment == request_segment:
            # Exact segment match.
            p_idx += 1
            r_idx += 1
        else:
            # Segments don't match and no wildcard applies.
            return False

    # After loop, check if all pattern segments have been consumed.
    # If there are remaining pattern segments, they must all be '**' to match nothing.
    while p_idx < len(pattern_segments):
        if pattern_segments[p_idx] != "**":
            return False
        p_idx += 1
    
    # All pattern segments matched. Now check if all request segments were consumed.
    # This must be true unless the last pattern segment was '**' (which is handled above by returning True)
    # or if the pattern was empty and request was also empty.
    return r_idx == len(request_segments)

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
