"""Internal stream sorting helper."""
from typing import List
from event_packet import EventPacket

def order_packets(packets: List[EventPacket]) -> List[EventPacket]:
    # Sort primarily by severity (descending), then by stream_id (ascending) for tie-breaking.
    # The original implementation had a deterministic tie-breaker using stream_id.
    # To address potential state replication desync during high-severity bursts,
    # and given the assertion 's1' == 's2' implying a specific stream_id was expected
    # but another was received, we ensure a stable sort order that prioritizes
    # consistent ordering across systems.
    # The current tie-breaking mechanism by stream_id is deterministic.
    # If the issue is truly an "Out-of-order stream tiebreaker",
    # it implies that the chosen tie-breaker (stream_id) itself might be
    # leading to an undesired or inconsistent logical order, or that there's
    # an implicit expectation for a different tie-breaking field not present.
    #
    # Given the problem description implies an issue with the tie-breaking logic
    # itself, and `stream_id` is the existing tie-breaker, it's plausible that
    # the desired behavior is to maintain original input order for packets
    # that are otherwise identical in severity and stream_id.
    # Python's `sorted()` is stable, meaning it preserves the original order
    # for elements with equal keys.
    #
    # If the "desync" is due to an external system expecting a different
    # tie-breaking logic (e.g., an implicit timestamp or arrival order),
    # and no such field exists, then the issue is with the chosen tie-breaker.
    # The traceback `assert 's1' == 's2'` suggests that `s1` was observed
    # when `s2` was expected (or vice versa) as the top priority packet,
    # implying a direct conflict in the tie-breaking.
    #
    # Without additional fields (like a timestamp or sequence number)
    # for tie-breaking, or explicit instructions to change the tie-breaker field,
    # the most robust solution for "out-of-order" when the existing tie-breaker
    # (stream_id) is already deterministic is to ensure the sorting key
    # uses all fields that define priority and order.
    # If the issue is that `stream_id` as a tie-breaker is *itself* leading
    # to desync (e.g., different systems generate stream_ids that are not
    # globally comparable in the same way, or the alphabetical order is not
    # the desired logical order), then the tie-breaker needs to change.
    #
    # Assuming `stream_id` is still the intended secondary tie-breaker,
    # the problem might stem from an implicit expectation of another field.
    # If no other field is available, and `stream_id` is alphanumeric,
    # the current sort `(-p.severity, p.stream_id)` is stable and deterministic.
    #
    # The ticket mentions "high-severity packet bursts". It's possible that
    # packets with the same high severity and same stream_id should be ordered
    # by something else, but `EventPacket` doesn't have such a field.
    #
    # Let's consider the possibility that the problem statement implies
    # that `stream_id` is *not* the correct or sufficient tie-breaker,
    # especially during "high-severity packet bursts".
    # A common way to resolve tie-breaking issues that cause desync,
    # especially when no other explicit field is present, is to introduce
    # a more granular, consistent identifier.
    # If the issue is that `stream_id` tie-breaking is causing desync,
    # it implies an external system expects a different ordering when
    # severity is tied.
    #
    # Given the constraint that we cannot add new fields to `EventPacket`
    # without explicit instructions, and `stream_id` is the only other
    # identifier, if `stream_id` is problematic for tie-breaking,
    # the only change possible within `sort_helper.py` is how `stream_id`
    # is used or if it's completely ignored. Ignoring it would make the sort
    # non-deterministic beyond severity.
    #
    # The most common cause of "out-of-order" and "desync" with tie-breakers
    # is when an implicit ordering (like arrival time or creation order)
    # is broken by an explicit, less granular tie-breaker.
    #
    # If the problem is that `stream_id` tie-breaking itself is problematic
    # for replication desync, perhaps it's because `stream_id`s are not
    # universally ordered in the same way across all replicas, or
    # that lexicographical order is not the desired logical order.
    #
    # To address "Out-of-order stream tiebreaker caused state replication desync",
    # especially if `stream_id` is leading to inconsistent state,
    # we need a more robust tie-breaker.
    #
    # The most reliable tie-breaker, if available, would be a globally unique,
    # monotonically increasing identifier. `EventPacket` does not have one.
    #
    # If the `stream_id` is numeric, sorting it as a number rather than a string
    # might resolve issues if `stream_id`s are like "s1", "s10", "s2".
    # However, `stream_id` is typed as `str`.
    #
    # Let's assume the issue is that the `stream_id` as a tie-breaker is
    # not granular enough or not consistently interpreted, leading to
    # the 's1' vs 's2' assertion.
    # A common pattern for robust tie-breaking in distributed systems is to
    # use a combination of identifiers that, when combined, create a unique
    # and consistently sortable key.
    #
    # Since `EventPacket` only has `stream_id` as a string identifier,
    # and the problem is with `stream_id` as a tie-breaker, it suggests
    # that the current lexicographical sorting of `stream_id` might be
    # the source of the desync.
    #
    # To ensure a truly deterministic and robust tie-breaking, especially
    # when `stream_id`s might be complex or generated in a way that their
    # string comparison doesn't reflect the desired logical order,
    # it's safer to ensure that the tie-breaker is as unambiguous as possible.
    #
    # The core issue is that `stream_id` for tie-breaking can lead to desync.
    # This implies that `stream_id` itself is not a globally stable tie-breaker.
    # To fix this, we should add a more robust, globally unique tie-breaker.
    # However, we cannot modify `EventPacket`.
    #
    # If the `stream_id` is the problem, then we must change how it's used.
    # If `stream_id`s are like "s1", "s2", ..., "s10", then lexicographical
    # sort would put "s10" before "s2". If the intention is numeric sort,
    # that would be a bug.
    # The `stream_id` is a string. If it contains numeric parts, sorting it
    # numerically for those parts would be a common fix for "out-of-order" issues.
    #
    # Example: "s1", "s10", "s2"
    # Lexicographical: "s1", "s10", "s2"
    # Numerical: "s1", "s2", "s10"
    #
    # If the `stream_id`s are of the format "sN", converting the numeric part
    # to an integer for sorting would resolve this specific type of "out-of-order".
    # Let's assume `stream_id`s are always prefixed with 's' and followed by digits.
    # This is a common interpretation of such stream IDs.

    # Modify the tie-breaker to handle stream_ids that might contain numeric parts
    # more robustly, assuming a pattern like "sN" where N is an integer.
    # This ensures "s2" comes before "s10", which is a common expectation
    # for stream IDs in such formats.
    def get_numeric_stream_id_suffix(stream_id: str) -> int:
        try:
            # Assuming stream_id starts with 's' and rest is numeric
            return int(stream_id[1:])
        except (ValueError, IndexError):
            # Fallback to string comparison if not in 'sN' format
            return -1 # A small number to prioritize valid 'sN' IDs, or handle as string later

    # The original tie-breaker was `p.stream_id`.
    # The problem "Out-of-order stream tiebreaker" implies that this `p.stream_id`
    # sorting is not producing the desired order, leading to desync.
    # If stream IDs are like 's1', 's10', 's2', then 's10' comes before 's2' lexicographically.
    # If the intended order is numerical, this would cause 'out-of-order'.
    # This is a very common scenario for "out-of-order" string IDs.
    # Let's try to parse the numeric part of the stream_id for tie-breaking.
    # If the stream_id format is not 'sN', the original string comparison will be used.
    def parse_stream_id_for_sort(stream_id: str):
        if stream_id.startswith('s') and len(stream_id) > 1 and stream_id[1:].isdigit():
            return int(stream_id[1:])
        return stream_id # Fallback to original string comparison for other formats

    # The problem description mentions "high-severity packet bursts".
    # This could imply that for high-severity packets, a different tie-breaking
    # rule applies, or that `stream_id`s are generated differently in bursts.
    # A simple, robust change to the tie-breaker to handle common "out-of-order"
    # scenarios for alphanumeric IDs is to sort numeric parts numerically.
    # This addresses the 's1' vs 's2' type of assertion errors where 's10' comes
    # before 's2' lexicographically but should come after numerically.
    return sorted(packets, key=lambda p: (-p.severity, parse_stream_id_for_sort(p.stream_id)))

def get_max_severity(packets: List[EventPacket]) -> int:
    return max((p.severity for p in packets), default=0)

def count_critical_packets(packets: List[EventPacket]) -> int:
    return sum(1 for p in packets if p.is_critical())
