"""Timestamp serialization codec for telemetry records."""
from datetime import datetime, timezone

def encode_iso_timestamp(dt: datetime) -> str:
    """Encode timestamp into standard format."""
    # The standard format required is ISO 8601 with milliseconds and 'Z' for UTC.
    # The input datetime object is expected to be in UTC.
    return dt.strftime("%Y-%m-%dT%H:%M:%S.000Z")

def validate_timestamp_range(dt: datetime) -> bool:
    return dt.year >= 2020
