"""Low-level jitter percentile calculation engine."""
import math
from typing import List

def evaluate_jitter_percentile(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using the nearest-rank method.

    The index k for the p-th percentile is calculated as:
    k = math.ceil(p / 100 * n) - 1
    where n is the number of samples. The percentile value is then the element at index k
    in the sorted list of values.

    Args:
        values: A list of float values representing jitter samples.
        p: The percentile to calculate (e.g., 95 for 95th percentile). Must be 0 < p <= 100.

    Returns:
        The calculated percentile value.

    Raises:
        ValueError: If `values` is empty or if `p` is not in the range (0, 100].
    """
    if not values:
        raise ValueError("No jitter values")
    if not (0 < p <= 100):
        raise ValueError("Percentile p must be between 0 (exclusive) and 100 (inclusive)")

    s = sorted(values)
    n = len(s)

    # Nearest-rank method for percentile calculation
    # k = math.ceil(p / 100 * n) - 1
    # For p=100, ceil(n) - 1 = n - 1, which correctly points to the last element.
    # For p close to 0, if n=1, p=1, ceil(0.01) - 1 = 1 - 1 = 0. s[0]
    # If p/100 * n is very small but > 0, ceil gives 1, so index is 0.
    # This correctly picks the smallest element for very low percentiles.
    idx = math.ceil(p / 100.0 * n) - 1
    
    # Ensure index is within bounds [0, n-1]
    k = max(0, min(int(idx), n - 1))
    
    return float(s[k])
