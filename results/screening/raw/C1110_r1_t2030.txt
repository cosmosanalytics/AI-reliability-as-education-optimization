"""Tenant permission authorization guard."""
from errors import ServiceError
from tenant_context import TenantContext

def authorize_tenant_action(ctx: TenantContext, required_scope: str) -> bool:
    """Authorize tenant action against target resource.
    
    Raises ServiceError if tenant lacks required scope.
    """
    if not ctx.permits(required_scope):
        raise ServiceError(
            code="TENANT_AUTH_FAILURE",
            message=f"Scope {required_scope} denied for tenant {ctx.tenant_id}",
            details={"tenant_id": ctx.tenant_id, "required_scope": required_scope}
        )
    return True

def dispatch_tenant_request(ctx: TenantContext, required_scope: str) -> str:
    authorize_tenant_action(ctx, required_scope)
    return "DISPATCHED"
