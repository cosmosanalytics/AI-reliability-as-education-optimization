"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total = sum(line_totals)
    if total == 0:
        return [0] * len(line_totals)

    allocations = []
    remainders = []
    for cents in line_totals:
        proportion = cents * total_discount_cents
        allocated = proportion // total
        remainder = proportion % total
        allocations.append(allocated)
        remainders.append(remainder)

    # Distribute remaining cents due to rounding
    current_total_allocated = sum(allocations)
    unallocated_cents = total_discount_cents - current_total_allocated

    # Sort by remainder in descending order to distribute the unallocated cents
    # to lines with the largest fractional parts.
    # If remainders are equal, distribute to earlier lines to ensure deterministic output.
    # Create a list of tuples (remainder, original_index) to maintain order for ties.
    indexed_remainders = sorted(
        [(remainders[i], i) for i in range(len(remainders))],
        key=lambda x: (x[0], -x[1]), # Sort by remainder DESC, then by original index ASC (negated for stable sort for ties)
        reverse=True
    )

    for i in range(unallocated_cents):
        # Add 1 cent to the line with the largest remainder (and smallest original index for ties)
        # The (remainder, original_index) tuple is already sorted by remainder descending,
        # and then by original index ascending for ties.
        # So, simply pick the next item in the sorted list.
        # We use a modulo operation to cycle through the sorted remainders if unallocated_cents > number of lines.
        original_idx = indexed_remainders[i % len(indexed_remainders)][1]
        allocations[original_idx] += 1

    return allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
