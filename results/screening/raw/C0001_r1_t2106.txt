"""Cache Hit Latency Percentile Monitor percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_cache_percentile(samples: List[float], p: float) -> float:
    """
    Calculate the p-th percentile of a list of samples using the nearest-rank method.

    The nearest-rank method calculates the rank `r = ceil(p/100 * n)`,
    where `n` is the number of samples. The percentile is then the value
    at that rank (using 1-based indexing for rank, or `r-1` for 0-based indexing).
    For an empty list of samples, a ValueError is raised by validate_metric_samples.
    If p is outside the range [0, 100], it is clamped to the nearest valid percentile:
    p < 0 effectively returns the minimum value (0th percentile), and
    p > 100 effectively returns the maximum value (100th percentile).
    """
    win = create_sample_window("calculate_cache_percentile", samples)
    s = validate_metric_samples(win.values) # s is sorted samples
    n = len(s)

    # Clamp p to the valid range [0, 100] for percentile calculation.
    # This aligns with common interpretations for 0th and 100th percentiles
    # as min and max values, respectively.
    p_clamped = max(0.0, min(100.0, p))

    # Calculate rank using the nearest-rank method (R-1 in NIST terminology)
    # r = ceil(p/100 * n)
    # For n=1, p=50 -> ceil(0.5) = 1. Index 0.
    # For n=5, p=50 -> ceil(2.5) = 3. Index 2.
    # For n=5, p=80 -> ceil(4.0) = 4. Index 3.
    rank = math.ceil((p_clamped / 100.0) * n)

    # Adjust rank to 0-based index.
    # The smallest rank is 1, corresponding to index 0.
    # The largest rank is n, corresponding to index n-1.
    # We subtract 1 from the 1-based rank to get the 0-based index.
    # If rank is 0 (e.g., p=0.0, n=any -> (0/100)*n = 0 -> ceil(0) = 0),
    # index will be -1. We need to clamp this to 0.
    index = int(rank) - 1

    # Ensure index is within bounds [0, n-1].
    # This handles cases where rank might be 0 (resulting in index -1)
    # or if there's any floating point peculiarity near n.
    index = max(0, min(index, n - 1))

    return s[index]

def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
