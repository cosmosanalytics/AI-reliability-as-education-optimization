"""Low-level tax rate computation engine."""
import math # Import math for floor function for specific rounding
from typing import Dict, Any

DEFAULT_TAX_JURISDICTIONS = {
    "standard": 0.20,
    "reduced": 0.10,
    "exempt": 0.0,
}

def get_rate_for_jurisdiction(code: str) -> float:
    return DEFAULT_TAX_JURISDICTIONS.get(code, 0.20)

def calculate_vat(net_cents: int, vat_rate: float) -> int:
    """
    Compute VAT in integer cents for given net amount, using half-up rounding
    (ties broken towards positive infinity).
    """
    # To achieve "half-up rounding" where ties are broken towards positive infinity,
    # we add 0.5 to the float value and then take the floor.
    # This works for both positive and negative numbers in the way specified.
    # E.g., 13.5 + 0.5 = 14.0 -> floor(14.0) = 14
    #       13.4 + 0.5 = 13.9 -> floor(13.9) = 13
    #      -13.5 + 0.5 = -13.0 -> floor(-13.0) = -13
    #      -13.6 + 0.5 = -13.1 -> floor(-13.1) = -14
    return math.floor(net_cents * vat_rate + 0.5)
