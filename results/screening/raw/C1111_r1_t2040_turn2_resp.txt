"""IoT sensor ingest stream demux."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def process_iot_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Process IoT device packet.
    
    For raw IoT ingest pipeline, MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope.
    """
    # The docstring explicitly states: "MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope."
    # The current implementation `return build_envelope("iot_service", epoch_ms, norm)` violates this by wrapping the payload and normalizing keys.
    # The incident IOT-703 confirms this is a bug.
    # Therefore, the function should return the original `data` dictionary as is.
    return data

def dispatch_iot_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # This function's behavior is not specified to change, but it relies on process_iot_packet.
    # If process_iot_packet is now returning raw data, and dispatch_iot_record needs an enveloped message,
    # it must build the envelope itself.
    # The original implementation of process_iot_packet normalized keys before building the envelope.
    # If the intent for dispatch_iot_record is to dispatch an enveloped, normalized record,
    # then the normalization and envelope building should happen here.
    # The incident description specifically relates to the "raw IoT ingest pipeline", which
    # process_iot_packet serves. dispatch_iot_record might be for a different purpose.
    # Given the existing test `test_nominal_iot_process` passes with the original
    # `process_iot_packet` returning an enveloped and normalized dict, and `dispatch_iot_record`
    # returning the same, it implies `dispatch_iot_record` should also produce an enveloped,
    # normalized output.
    norm = normalize_dict_keys(data)
    msg = build_envelope("iot_service", epoch_ms, norm)
    _ = extract_normalized_keys(data) # This line was originally present and seems to be a side-effect or vestige, not affecting the return value. Keep it as per W03.
    return msg
