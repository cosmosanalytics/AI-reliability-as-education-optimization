"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Default service name for audit events.
# This constant is added to fulfill the '_src' requirement of the envelope,
# as 'format_audit_event' does not receive a service_name argument.
DEFAULT_AUDIT_SERVICE_NAME = "unknown_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    # The original implementation returned raw data, violating the docstring.
    # This change ensures the payload is wrapped in the specified envelope
    # and its keys are normalized.
    return build_envelope(DEFAULT_AUDIT_SERVICE_NAME, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # This line checks envelope validity but does not use the result; it's kept for minimal change.
    _ = is_valid_envelope(msg)
    # This line extracts normalized keys from the original data, which is redundant
    # after format_audit_event normalizes the payload, but kept for minimal change.
    _ = extract_normalized_keys(data)
    return msg
