"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

SERVICE_NAME = "billing"

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers."""
    # Normalize the keys of the payload data as per common practice for billing records.
    normalized_data = normalize_dict_keys(data)
    # Envelope the normalized payload for downstream consumption.
    return build_envelope(SERVICE_NAME, epoch_ms, normalized_data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # The following lines were likely intended for validation or further processing,
    # but their results were ignored. The primary issue was the lack of enveloping.
    # With emit_billing_record now returning an enveloped message, is_valid_envelope
    # should return True. The extract_normalized_keys call is redundant here
    # as normalization now happens inside emit_billing_record.
    _ = is_valid_envelope(record)
    # _ = extract_normalized_keys(data) # This line can be removed or kept as is, as it has no side effect.
    return record
