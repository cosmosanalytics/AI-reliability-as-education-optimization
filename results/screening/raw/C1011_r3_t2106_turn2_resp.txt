"""Cache Hit Latency Percentile Monitor percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_cache_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_cache_percentile", samples)
    s = validate_metric_samples(win.values)
    # The incident report INC-2106 specifies adherence to a "nearest-rank house standard"
    # for percentile calculation. This standard typically corresponds to the
    # definition where the index is calculated as p/100 * N, and the value is
    # taken from the sample at the ceiling of this index (1-indexed), or
    # floor(index) (0-indexed) after adjusting for 0-indexing.
    # The previous implementation used linear interpolation (R-7 method),
    # which caused discrepancies.
    n = len(s)
    if n == 0:
        raise ValueError("Samples list cannot be empty for percentile calculation")
    # Calculate rank based on p/100 * N.
    # For 0-indexed list, the index is ceil(p/100 * N) - 1.
    # This aligns with the "nearest-rank" definition where the smallest value
    # in the dataset that is greater than or equal to the p-th percentile is chosen.
    # Example: For [10, 20, 30, 40, 50] and 50th percentile:
    # idx_float = 0.50 * 5 = 2.5
    # ceil(2.5) - 1 = 3 - 1 = 2. s[2] = 30.0
    # For [10, 20, 30, 40, 50, 60] and 50th percentile:
    # idx_float = 0.50 * 6 = 3.0
    # ceil(3.0) - 1 = 3 - 1 = 2. s[2] = 30.0
    # This matches common definitions where the 50th percentile of [10,20,30,40,50,60] is 30.
    #
    # However, the failing test expected 40.0 for 50th percentile.
    # This implies a different interpretation of "nearest-rank" or the specific dataset.
    # Let's consider the method often known as "nearest rank" where if the index is
    # not an integer, it is rounded up.
    # The issue's failing test implies that for [10, 20, 30, 40, 50, 60, 70, 80],
    # the 50th percentile should be 40.0, not 45.0 (current) or 50.0 (simple nearest-rank).
    #
    # Let's re-evaluate the "nearest-rank house standard" based on the expected outcome.
    # If calculate_cache_percentile(d1, 50.0) == 40.0 and it returned 45.0,
    # it implies d1 might be something like [10,20,30,40,50,60,70,80].
    # For this dataset, len(s) = 8.
    # p = 50.0.
    #
    # Original (R-7): idx = 0.5 * (8 - 1) = 3.5.
    # low = 3 (s[3]=40), high = 4 (s[4]=50).
    # result = 40 + (3.5 - 3) * (50 - 40) = 40 + 0.5 * 10 = 45.0. This matches the actual.
    # Expected: 40.0.
    #
    # If the expected value is s[3] (40.0), this means the percentile index should be 3.
    # A common way to get this is `floor(p/100 * n + 0.5) - 1` or similar, but the
    # simplest "nearest-rank" implies picking an existing value.
    #
    # Let's use the definition: `k = p/100 * (n + 1)`. If `k` is an integer, take `s[k-1]`.
    # If `k` is not an integer, round up to `ceil(k) - 1`.
    # For `n=8, p=50`: `k = 0.5 * (8 + 1) = 4.5`. `ceil(4.5) - 1 = 5 - 1 = 4`. `s[4]` is 50.0. (Still not 40.0)
    #
    # The ticket describes a "nearest-rank house standard". This usually means selecting an
    # actual data point rather than interpolating.
    # A common method, often called R-1, calculates `k = p/100 * n` and if `k` is not an integer,
    # it rounds up to `ceil(k)`. The percentile is then `s[ceil(k) - 1]` (0-indexed).
    # For `n=8, p=50`: `k = 0.5 * 8 = 4`. This is an integer.
    # If `k` is an integer, some definitions take `s[k-1]` (0-indexed). So `s[3]` which is 40.0.
    # This matches the expected output!
    #
    # So, the "nearest-rank house standard" is interpreted as:
    # 1. Calculate `k = p/100 * n`.
    # 2. If `k` is an integer, the percentile is `s[k-1]`.
    # 3. If `k` is not an integer, the percentile is `s[floor(k)]`. (Often R-6/R-8, but R-1/R-2 might use ceil).
    #
    # Let's try `idx = math.ceil(p / 100.0 * n) - 1`. This is common for 1-indexed to 0-indexed.
    # For `n=8, p=50`: `idx = ceil(0.5 * 8) - 1 = ceil(4) - 1 = 4 - 1 = 3`. Returns `s[3] = 40.0`. This works!
    #
    # For `n=5, p=50`, `s=[10,20,30,40,50]`: `idx = ceil(0.5 * 5) - 1 = ceil(2.5) - 1 = 3 - 1 = 2`. Returns `s[2]=30.0`.
    # This seems to be a consistent "nearest rank" method.

    # Ensure p is within valid range [0, 100]
    if not (0.0 <= p <= 100.0):
        raise ValueError("Percentile 'p' must be between 0.0 and 100.0")

    if p == 0.0:
        return s[0]
    if p == 100.0:
        return s[-1]

    # Calculate the rank (1-indexed)
    rank = (p / 100.0) * n
    # For nearest-rank, we often take the value at ceil(rank) (1-indexed).
    # Convert to 0-indexed: index = int(math.ceil(rank)) - 1.
    # This method is robust for both integer and non-integer ranks.
    # Example: n=8, p=50, rank=4.0. ceil(4.0)-1 = 3. s[3]=40.0.
    # Example: n=5, p=50, rank=2.5. ceil(2.5)-1 = 2. s[2]=30.0.
    # This matches the common "nearest rank" where for an integer rank, it takes that rank,
    # and for a non-integer rank, it rounds up.
    # This calculation works for the provided failing test.
    index = int(math.ceil(rank)) - 1

    # Ensure index is within bounds [0, n-1]
    # For p=0, index would be -1, handled by explicit check above.
    # For p=100, index could be n-1 if rank is n.
    # The ceil(rank)-1 formulation naturally handles index=n-1 for p=100.
    # e.g., n=3, p=100, rank=3.0, index=ceil(3.0)-1 = 2. s[2] is last element.
    return s[index]

def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
