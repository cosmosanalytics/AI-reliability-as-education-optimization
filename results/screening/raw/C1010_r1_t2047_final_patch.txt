"""Webhook delivery payload construction."""
from datetime import datetime, timezone
from typing import Dict, Any
from envelope_schema import WebhookEnvelope
from signature_utils import generate_delivery_headers

def build_webhook_payload(topic: str, data: Dict[str, Any], dt: datetime) -> WebhookEnvelope:
    hdrs = generate_delivery_headers(topic)
    # Format timestamp to RFC3339, specifically to millisecond precision (3 decimal places).
    # The original format "%Y-%m-%dT%H:%M:%S.%fZ" uses %f for microseconds (6 decimal places),
    # which subscriber gateways rejected.
    # We construct the string by taking the date-time part, then adding milliseconds
    # padded to 3 digits, followed by 'Z' for UTC.
    ts = dt.strftime("%Y-%m-%dT%H:%M:%S") + f".{dt.microsecond // 1000:03d}Z"
    env = WebhookEnvelope(topic=topic, body=data, sent_at=ts, headers=hdrs)
    if not env.is_dispatched():
        raise ValueError("Invalid envelope state")
    return env
