"""Low-level tax rate computation engine."""
from typing import Dict, Any

DEFAULT_TAX_JURISDICTIONS = {
    "standard": 0.20,
    "reduced": 0.10,
    "exempt": 0.0,
}

def get_rate_for_jurisdiction(code: str) -> float:
    return DEFAULT_TAX_JURISDICTIONS.get(code, 0.20)

def calculate_vat(net_cents: int, vat_rate: float) -> int:
    """Compute VAT in integer cents for given net amount, using half-up rounding.

    Tax calculation uses half-up rounding (ties broken towards positive infinity)
    on exact half-cents. For non-negative values, this is achieved by adding 0.5
    before converting to an integer.
    """
    # Apply half-up rounding for non-negative values.
    # E.g., 12.4 -> 12, 12.5 -> 13, 12.6 -> 13
    return int(net_cents * vat_rate + 0.5)
