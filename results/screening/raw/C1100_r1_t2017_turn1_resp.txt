"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    pattern = policy.allowed_pattern

    if pattern == "**":
        return True
    if pattern == "*": # Matches any single segment
        return "::" not in clean_req and bool(clean_req)

    pattern_parts = pattern.split("::")
    request_parts = clean_req.split("::")

    if not request_parts and not pattern_parts: # Both empty
        return True
    if not request_parts: # Request empty, pattern not empty (unless pattern was "**" or "*", handled above)
        return False
    if not pattern_parts: # Pattern empty, request not empty
        return False

    i = 0
    j = 0
    while i < len(pattern_parts) and j < len(request_parts):
        pattern_part = pattern_parts[i]
        request_part = request_parts[j]

        if pattern_part == "**":
            # If "**" is the last part, it matches the rest of the request_parts
            if i == len(pattern_parts) - 1:
                return True
            # If "**" is not the last part, find the next non-wildcard part
            # and try to match it in the remaining request_parts
            next_pattern_part = pattern_parts[i + 1]
            while j < len(request_parts):
                if is_scope_permitted(RoutePolicy("", next_pattern_part), request_parts[j]):
                    # If the next part matches, continue matching the rest of the pattern
                    # from the matched position in request_parts
                    return is_scope_permitted(
                        RoutePolicy("", "::".join(pattern_parts[i+1:])),
                        "::".join(request_parts[j:])
                    )
                j += 1
            return False # Could not find a match for the part after "**"

        elif pattern_part == "*":
            # '*' matches exactly one segment
            if "::" in request_part: # Should not happen with split, but for safety
                return False
            i += 1
            j += 1
        elif pattern_part == request_part:
            # Exact segment match
            i += 1
            j += 1
        else:
            # Mismatch
            return False

    # Check if both pattern and request parts have been fully consumed
    # If pattern_parts still has a remaining '**', it can match zero segments
    while i < len(pattern_parts) and pattern_parts[i] == "**":
        i += 1

    return i == len(pattern_parts) and j == len(request_parts)


def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
