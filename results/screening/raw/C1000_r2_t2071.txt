"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses.

    This function uses the "linear interpolation between the two nearest ranks" method,
    which is a common approach for calculating percentiles, especially for small datasets
    or when more precise interpolation is needed than simply picking the nearest rank.
    """
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)

    # The formula for the rank (index) using linear interpolation is:
    # L = (p / 100.0) * (n - 1)
    # The percentile value is then interpolated between the two nearest data points.
    # This is equivalent to numpy.percentile with method='linear' or 'fraction'.
    
    # Calculate the fractional rank
    rank = (p / 100.0) * (n - 1)

    # Get the integer and fractional parts of the rank
    irank = math.floor(rank)
    frank = rank - irank

    # Handle edge cases for rank
    if irank < 0:
        return float(s[0])
    if irank >= n - 1:
        return float(s[n - 1])

    # Perform linear interpolation
    lower_val = s[irank]
    upper_val = s[irank + 1]
    
    return lower_val + (upper_val - lower_val) * frank

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
