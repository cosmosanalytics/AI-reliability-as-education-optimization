"""Audio buffer level metric processor."""
import math
from typing import List, Dict, Any
from audio_types import AudioStreamSnapshot, BufferSample
from audio_formatter import format_buffer_stat

def compute_buffer_percentile(snapshot: AudioStreamSnapshot, p: float) -> float:
    """Compute percentile using continuous linear interpolation (numpy-style: index = (p / 100.0) * (len(samples) - 1)).
    Do not use nearest-rank method for audio buffer metrics.
    """
    vals = snapshot.ratios()
    if not vals:
        raise ValueError("Empty snapshot")
    
    s = sorted(vals)
    n = len(s)

    # Handle edge cases for p=0 and p=100 explicitly
    if p == 0.0:
        return float(s[0])
    if p == 100.0:
        return float(s[n - 1])

    # Calculate index using continuous linear interpolation formula
    # index = (p / 100.0) * (len(samples) - 1)
    # This formula ensures that for n=1, idx will be 0.0, correctly returning s[0]
    idx = (p / 100.0) * (n - 1)

    # If index is an integer, return the value at that index
    if idx == int(idx):
        return float(s[int(idx)])
    else:
        # Interpolate between the two nearest ranks
        i = math.floor(idx)
        j = math.ceil(idx)
        
        # Ensure indices are within bounds (though with idx calculation, they should be)
        i = max(0, min(n - 1, int(i)))
        j = max(0, min(n - 1, int(j)))

        # Linear interpolation: s[i] + (s[j] - s[i]) * (fractional_part_of_idx)
        return float(s[i] + (s[j] - s[i]) * (idx - i))

def evaluate_audio_buffer(snapshot: AudioStreamSnapshot, p: float) -> Dict[str, Any]:
    val = compute_buffer_percentile(snapshot, p)
    return format_buffer_stat(snapshot.channel_count(), val)
