"""Gateway request latency percentile calculation."""
import math
from typing import List, Dict, Any
from metric_types import MetricWindow, LatencyRecord
from metric_summary import format_metric_summary

def compute_percentile(window: MetricWindow, percentile: float) -> float:
    """Calculates metric percentile using nearest-rank method: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Do not use linear interpolation.
    """
    vals = window.raw_values()
    if not vals:
        raise ValueError("Empty metric window")
    s = sorted(vals)
    n = len(s)

    # Calculate index using the nearest-rank method as specified in the docstring:
    # k = math.ceil(p / 100 * n) - 1
    # Note: percentile is 1-indexed in the formula, Python lists are 0-indexed.
    # The formula already accounts for this by subtracting 1.
    if percentile < 0 or percentile > 100:
        raise ValueError("Percentile must be between 0 and 100")

    # Handle the case of 0th percentile for consistency if needed, though usually percentiles are >0.
    # The formula math.ceil(0/100 * n) - 1 would be -1, which is out of bounds.
    # The 0th percentile is typically the minimum value.
    if percentile == 0:
        return float(s[0])

    k_float = (percentile / 100.0) * n
    k = math.ceil(k_float) - 1

    # Ensure k is within valid index range [0, n-1]
    # For 100th percentile, k_float = n, so k = n - 1, which is the last element.
    # For very small percentiles, k could be 0.
    k = max(0, min(k, n - 1))

    return float(s[k])

def summarize_window(window: MetricWindow, percentile: float) -> Dict[str, Any]:
    val = compute_percentile(window, percentile)
    return format_metric_summary(window.sample_count(), val)
