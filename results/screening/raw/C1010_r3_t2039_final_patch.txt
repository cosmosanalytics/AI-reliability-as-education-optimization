"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # The issue indicates that the Kafka consumer failed due to missing wire envelope metadata.
    # The 'prepare_metric_packet' function is responsible for preparing the packet
    # for distribution. It should therefore construct a valid wire envelope.
    # A default service name is used as it's not provided in the function signature
    # and existing callers (like tests) do not pass it.
    service_name = "telemetry-service"
    return build_envelope(service_name, epoch_ms, data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The original code called is_valid_envelope and extract_normalized_keys
    # but didn't use their results. These lines are kept for consistency,
    # though they don't affect the return value.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # This operates on original 'data', not 'pkt'
    return pkt
