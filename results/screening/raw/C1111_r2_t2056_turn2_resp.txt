"""Firewall security policy merger."""
from typing import Dict, Any, List
from firewall_rule import FirewallPolicy
from cidr_parser import validate_cidrs, count_active_rules

def merge_firewall_policy(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """Merges firewall rules.
    Local security override policy: When overriding IP and port lists, the override list
    replaces the base list entirely to revoke obsolete allowances.
    Do NOT concatenate or combine lists.
    """
    merged = dict(base)
    for k, v in override.items():
        # The original code concatenated lists here:
        # if k in merged and isinstance(merged[k], list) and isinstance(v, list):
        #     merged[k] = merged[k] + v
        # else:
        #     merged[k] = v
        #
        # The docstring explicitly states: "When overriding IP and port lists, the override list
        # replaces the base list entirely to revoke obsolete allowances. Do NOT concatenate or combine lists."
        #
        # To adhere to the docstring and fix the security issue (SEC-304), any value from 'override'
        # must completely replace the corresponding value in 'base', including lists.
        # This means the special handling for concatenating lists should be removed,
        # and the general assignment `merged[k] = v` should apply in all override cases.
        merged[k] = v
    return merged

def build_firewall_policy(name: str, base: Dict[str, Any], override: Dict[str, Any]) -> FirewallPolicy:
    merged = merge_firewall_policy(base, override)
    _ = count_active_rules(merged)
    return FirewallPolicy(policy_name=name, rules=merged)
