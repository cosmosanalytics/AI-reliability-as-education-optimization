"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    policy_segments = policy.allowed_pattern.split('::')
    request_segments = clean_req.split('::')

    i = 0 # policy segment index
    j = 0 # request segment index

    while i < len(policy_segments) and j < len(request_segments):
        policy_segment = policy_segments[i]
        request_segment = request_segments[j]

        if policy_segment == "**":
            # '**' matches zero or more segments.
            # If it's the last policy segment, it matches everything remaining in the request.
            if i == len(policy_segments) - 1:
                return True
            # Otherwise, find the next policy segment in the request segments.
            # Advance request_segment pointer until it matches the next policy segment.
            next_policy_segment = policy_segments[i + 1]
            while j < len(request_segments) and request_segments[j] != next_policy_segment:
                j += 1
            # If we exhausted request segments without finding the next policy segment, it's not a match.
            if j == len(request_segments):
                return False
            # If we found it, advance policy segment pointer past '**' and the matched segment.
            i += 1 # Move past '**'
            # j is already at the matched segment, so we will continue the loop with i and j potentially matching
            # the next_policy_segment if it's not a wildcard.
            # The current j is the start of the match for the segment *after* the **.
            # The loop will then compare policy_segments[i] (which is next_policy_segment) with request_segments[j].
            # This is handled by the regular comparison logic.
            continue
        elif policy_segment == "*":
            # '*' matches exactly one segment.
            # If there's no request segment, or we're at the end of request segments but policy expects more, it's not a match.
            if j >= len(request_segments): # Should not happen if j is already checked by while condition
                return False
            i += 1
            j += 1
            continue
        elif policy_segment == request_segment:
            # Exact match, move to next segments.
            i += 1
            j += 1
            continue
        else:
            # Segments do not match and not a wildcard.
            return False

    # After loop, check if both policy and request segments are fully consumed.
    # If policy has remaining segments that are not '**', it's not a match.
    while i < len(policy_segments) and policy_segments[i] == "**":
        i += 1

    return i == len(policy_segments) and j == len(request_segments)

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
