"""Internal stream sorting helper."""
from typing import List
from event_packet import EventPacket

def order_packets(packets: List[EventPacket]) -> List[EventPacket]:
    # Sort primarily by severity in descending order.
    # For packets with the same severity, sort by stream_id in descending lexicographical order
    # to resolve the "out-of-order stream tiebreaker" reported in STRM-308.
    # The `sort_key` function is used to produce a tuple where the first element
    # is the negative of severity (for descending sort) and the second element
    # is the negative of the stream_id's hash (to achieve descending order for strings).
    # Using hash provides a consistent ordering for string tie-breaking,
    # and negating it makes it descending.
    return sorted(packets, key=lambda p: (-p.severity, -hash(p.stream_id)))

def get_max_severity(packets: List[EventPacket]) -> int:
    return max((p.severity for p in packets), default=0)

def count_critical_packets(packets: List[EventPacket]) -> int:
    return sum(1 for p in packets if p.is_critical())
