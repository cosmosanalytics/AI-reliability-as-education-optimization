"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.
    Uses the inclusive method, where the lowest value is at 0th percentile
    and the highest value is at 100th percentile, for consistency with
    common statistical software (e.g., NumPy's default linear interpolation).
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)

    # For p=0, return the minimum. For p=100, return the maximum.
    # For values in between, use linear interpolation.
    if p < 0 or p > 100:
        raise ValueError("Percentile p must be between 0 and 100 inclusive.")
    if p == 0:
        return float(s[0])
    if p == 100:
        return float(s[n-1])

    # Calculate the rank index. This formula (p/100.0) * (n-1) means 0th percentile
    # is s[0] and 100th percentile is s[n-1].
    idx = (p / 100.0) * (n - 1)
    
    # Floor and ceil to get the two surrounding indices
    low_idx = int(math.floor(idx))
    high_idx = int(math.ceil(idx))
    
    # If the index is an integer, it's an exact rank
    if low_idx == high_idx:
        return float(s[low_idx])
    
    # Otherwise, linearly interpolate between the two values
    frac = idx - low_idx
    return float(s[low_idx] + frac * (s[high_idx] - s[low_idx]))
