"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

_SERVICE_NAME = "billing_service"  # Define the service name for the envelope

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers."""
    # The record should be enveloped before emission.
    # Normalize keys of the payload data before building the envelope.
    normalized_data = normalize_dict_keys(data)
    return build_envelope(_SERVICE_NAME, epoch_ms, normalized_data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # The following lines are for demonstration/debugging and can be removed
    # or kept if they serve a purpose beyond the scope of this fix.
    # For now, ensure the record is valid and extract payload if needed.
    # The actual checks/extractions might happen downstream.
    _ = is_valid_envelope(record)
    _ = extract_normalized_keys(data) # This still operates on the original data keys
    return record
