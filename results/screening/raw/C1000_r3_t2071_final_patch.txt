"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses.

    Uses the "nearest rank" method, where the index is calculated as
    (p / 100) * n, rounded to the nearest whole number.
    """
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)

    # The most common definition for percentile rank uses (p/100) * n.
    # We round up to get the 1-based index, then convert to 0-based.
    # For a list of N items, the p-th percentile is the item at index
    # ceil((p / 100.0) * N) - 1 (0-indexed).
    # This ensures that for p=100, we get the last element (index N-1).
    # For p=0, it should conceptually be the first element, but this definition
    # usually applies for p > 0.
    # For p=0, we simply return the minimum value.
    if p == 0:
        return float(s[0])
    
    # Calculate the 1-based rank, then adjust to 0-based index.
    # Using math.ceil for rounding up to the nearest integer.
    rank = math.ceil((p / 100.0) * n)
    idx = int(rank) - 1

    # Ensure the index is within bounds [0, n-1]
    # In case of p=100 and a small N, ceil((100/100)*N) = N, so idx = N-1.
    # This correctly points to the last element.
    # In case of tiny p values where rank becomes 0, idx would be -1.
    # We ensure it's at least 0.
    idx = max(0, min(idx, n - 1))
    
    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
