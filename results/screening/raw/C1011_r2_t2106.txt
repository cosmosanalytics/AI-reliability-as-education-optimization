"""Cache Hit Latency Percentile Monitor percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_cache_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_cache_percentile", samples)
    s = validate_metric_samples(win.values) # Ensures samples is not empty and is sorted
    n = len(s)
    
    if p < 0.0 or p > 100.0:
        raise ValueError("Percentile must be between 0 and 100 inclusive.")

    # The issue specifies "nearest-rank house standard".
    # A common "nearest-rank" method (often Type 3 in NIST definitions) calculates
    # the 1-based rank R = P/100 * N, and then selects the element at index
    # math.ceil(R) - 1 (for 0-based indexing).
    # This method is robust for P=0 and P=100, and matches the expected behavior
    # for the failing test case (e.g., 50th percentile of [..., 40, 50, ...]
    # being 40.0, not 45.0).

    # Handle edge cases for p=0 and p=100 explicitly for clarity and precision.
    if p == 0.0:
        return s[0]
    if p == 100.0:
        return s[n - 1]

    # Calculate the 1-based rank.
    rank = (p / 100.0) * n
    
    # Convert to 0-based index by taking ceiling of rank and subtracting 1.
    # This effectively rounds up to the next integer rank if `rank` is not an integer.
    idx = int(math.ceil(rank)) - 1
    
    # The logic above guarantees idx will be within [0, n-1] for 0 < p < 100.
    # Example:
    # samples = [10, 20, 30, 40, 50, 60, 70, 80], n = 8
    # p = 50.0: rank = (50/100.0) * 8 = 4.0. idx = int(math.ceil(4.0)) - 1 = 4 - 1 = 3. Returns s[3] (40.0).
    # p = 25.0: rank = (25/100.0) * 8 = 2.0. idx = int(math.ceil(2.0)) - 1 = 2 - 1 = 1. Returns s[1] (20.0).
    # p = 12.5: rank = (12.5/100.0) * 8 = 1.0. idx = int(math.ceil(1.0)) - 1 = 1 - 1 = 0. Returns s[0] (10.0).
    
    return s[idx]

def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
