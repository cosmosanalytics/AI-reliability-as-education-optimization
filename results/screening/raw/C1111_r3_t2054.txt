"""Low-level dictionary and list hierarchy merger."""
from typing import Dict, Any, List

def merge_spec_values(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """Recursively merges dictionary values.
    If a key exists in both base and override, and both values are lists,
    the lists are merged by taking unique elements from base first, then
    adding unique elements from override that are not already present in base.
    For other types, values in `override` replace values in `base`.
    """
    result = dict(base)
    for k, v in override.items():
        if k in result and isinstance(result[k], dict) and isinstance(v, dict):
            result[k] = merge_spec_values(result[k], v)
        elif k in result and isinstance(result[k], list) and isinstance(v, list):
            # Merge lists by taking unique elements from base, then adding unique elements from override
            # that are not already in the merged list. This preserves existing elements and adds new ones.
            merged_list = list(result[k])
            seen = set(merged_list)
            for item in v:
                if item not in seen:
                    merged_list.append(item)
                    seen.add(item)
            result[k] = merged_list
        else:
            result[k] = v
    return result

def is_spec_empty(spec: Dict[str, Any]) -> bool:
    return len(spec) == 0
