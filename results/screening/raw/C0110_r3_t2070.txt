"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.
    Uses the "nearest rank" method: L = ceil(p * N / 100), where N is the number of values.
    The L-th value is returned, after sorting."""
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)
    # Calculate the rank (1-indexed)
    rank = math.ceil((p / 100.0) * n)
    # Adjust to 0-indexed list
    index = int(rank) - 1
    # Ensure index is within bounds (should be handled by ceil and sorted non-empty list)
    if index < 0:
        index = 0
    if index >= n:
        index = n - 1
    return s[index]
