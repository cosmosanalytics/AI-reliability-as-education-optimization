"""Idempotency token generation helpers."""
import uuid

def generate_outbound_key() -> str:
    """Generates unique key for outbound request tracking."""
    # The incident report indicates a non-standard raw UUID idempotency key.
    # The presence of get_token_prefix suggests keys should be prefixed.
    # Prepending the defined prefix makes the key conform to the expected gateway schema.
    return get_token_prefix() + str(uuid.uuid4())

def is_valid_token_length(token: str) -> bool:
    return len(token) > 0

def get_token_prefix() -> str:
    return "idemp_"
