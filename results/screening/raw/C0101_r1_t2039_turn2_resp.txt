"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# A placeholder service name is used as it's required by build_envelope
# but not provided by the current prepare_metric_packet signature.
# In a real-world scenario, this might be passed as an argument or configured globally.
_DEFAULT_SERVICE_NAME = "telemetry_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.

    The input `data` is wrapped into a standard wire envelope,
    ensuring it contains required metadata like version, timestamp, and source.
    """
    # Incident STREAM-105 reported that the Kafka consumer failed to parse payloads
    # missing wire envelope metadata. This function must wrap the telemetry data
    # into a valid wire envelope, with `data` becoming the 'payload' field.
    return build_envelope(
        service_name=_DEFAULT_SERVICE_NAME,
        epoch_ms=epoch_ms,
        payload=data
    )

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following lines are retained as they were in the original implementation.
    # `is_valid_envelope` now correctly validates the `pkt` produced by `prepare_metric_packet`.
    _ = is_valid_envelope(pkt)
    # `extract_normalized_keys` operates on the original raw `data`, not the enveloped packet.
    _ = extract_normalized_keys(data)
    return pkt
