"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers.
    The message will have its keys normalized and be wrapped in a wire envelope."""
    # Normalize keys for the billing record payload.
    normalized_payload = normalize_dict_keys(data)
    # The incident report indicates that payloads must be enveloped.
    # Use "billing" as the service name, consistent with the module's purpose.
    return build_envelope(service_name="billing", epoch_ms=epoch_ms, payload=normalized_payload)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # `emit_billing_record` now produces the fully enveloped message.
    # This function acts as a direct passthrough.
    return emit_billing_record(data, epoch_ms)
