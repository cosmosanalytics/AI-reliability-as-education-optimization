"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define the service name for billing events, as required by the wire envelope.
BILLING_SERVICE_NAME = "billing"

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers.
    The message is wrapped in a wire envelope with normalized keys in the payload."""
    # The payload itself should have its keys normalized before being enveloped.
    normalized_data = normalize_dict_keys(data)
    return build_envelope(BILLING_SERVICE_NAME, epoch_ms, normalized_data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # These checks ensure the emitted record is valid and that key normalization
    # is considered, but the results are not used directly by this function.
    _ = is_valid_envelope(record)
    _ = extract_normalized_keys(data) # Operates on original data as a check/hint
    return record
