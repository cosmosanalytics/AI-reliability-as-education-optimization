"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

_SERVICE_NAME = "telemetry-stream-processor" # Default service name for this module


def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.

    This function normalizes the keys of the input data and then wraps it
    in the standard wire envelope format, adding necessary metadata.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(_SERVICE_NAME, epoch_ms, normalized_payload)


def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following calls were previously for side effects or debugging,
    # but the primary responsibility of ensuring a valid envelope and
    # normalized keys is now handled by prepare_metric_packet.
    # We can keep them for explicit validation if desired, but they are
    # no longer strictly necessary for the packet's construction.
    _ = is_valid_envelope(pkt) # Validate the constructed packet
    _ = extract_normalized_keys(data) # This extracts keys from original data, not the prepared packet.
                                    # It might be intended for logging or other purposes.
    return pkt
