"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# A default service name is introduced as the ticket implies a missing envelope,
# and the service name is a required part of the envelope.
# In a real-world scenario, this might be configured or passed as an argument.
_DEFAULT_SERVICE_NAME = "telemetry_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This function now builds the standard wire envelope around the payload,
    normalizing the payload keys before embedding."""
    # Normalize keys of the payload data before embedding in the envelope.
    normalized_payload = normalize_dict_keys(data)
    # Build the complete wire envelope, including required metadata.
    return build_envelope(_DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Publish a telemetry stream packet.
    This function now correctly prepares an enveloped packet by calling
    prepare_metric_packet, which ensures the wire envelope metadata is present."""
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following lines were previously executing without their results being used
    # or affecting the return value. They are kept as-is to preserve existing logic,
    # although their direct impact on the return 'pkt' is minimal after the fix
    # in prepare_metric_packet.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # This still operates on the original 'data', not the enveloped 'pkt'.
    return pkt
