"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers.

    The input data's keys are normalized to snake_case and then enveloped
    with the service name "billing".
    """
    normalized_data = normalize_dict_keys(data)
    return build_envelope(service_name="billing", epoch_ms=epoch_ms, payload=normalized_data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # The following calls are for side effects or validation and do not alter the record.
    # The result of is_valid_envelope is discarded, but it might be used for logging in a real system.
    _ = is_valid_envelope(record)
    # extract_normalized_keys is called on the original data, not the normalized payload,
    # and its result is discarded. This might be an analytics hook.
    _ = extract_normalized_keys(data)
    return record
