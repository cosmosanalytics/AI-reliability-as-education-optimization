"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    This implementation uses the "nearest rank" method (also known as
    Method 1 or the R-1 method) for calculating the index,
    followed by linear interpolation if the index is not an integer.
    The index `idx` is calculated as `(p / 100.0) * (n - 1)`.
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)

    # Handle edge cases for p=0 and p=100 directly to avoid floating point issues
    # and ensure consistent behavior for min/max values.
    if p == 0.0:
        return float(s[0])
    if p == 100.0:
        return float(s[n - 1])

    # Calculate index using the R-1 method for linear interpolation.
    # The index is 0-based.
    idx = (p / 100.0) * (n - 1)

    low_idx = int(math.floor(idx))
    high_idx = int(math.ceil(idx))

    # If low_idx and high_idx are the same, the percentile is directly at that rank.
    if low_idx == high_idx:
        return float(s[low_idx])

    # Otherwise, linearly interpolate between the two nearest ranks.
    frac = idx - low_idx
    return float(s[low_idx] + frac * (s[high_idx] - s[low_idx]))
