"""Cache Hit Latency Percentile Monitor percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_cache_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_cache_percentile", samples)
    s = validate_metric_samples(win.values)
    # The "nearest-rank house standard" implies using the floor of the rank
    # index, effectively rounding down to the nearest available data point.
    # The standard formula for the rank index is (P/100) * N, where N is
    # the number of samples.
    #
    # For a percentile P and N samples, the rank is (P/100) * N.
    # The index into a 0-indexed sorted list is then floor((P/100) * N) - 1.
    #
    # Example: samples = [10, 20, 30, 40, 50, 60, 70, 80], N=8. P=50.
    # Rank = (50/100) * 8 = 4.0.
    # Index = floor(4.0) - 1 = 3.
    # Result = s[3] = 40.0.
    # This matches the expected value from the failing test: assert 45.0 == 40.0.
    
    # Calculate rank based on N (total number of samples)
    rank = (p / 100.0) * len(s)
    
    # Adjust rank to 0-based index and handle edge cases
    # If rank is an integer, the index is rank - 1.
    # If rank is not an integer, floor(rank) - 1 gives the index for the value
    # at or below the percentile, which matches the "nearest-rank" interpretation
    # for rounding down.
    # The smallest possible index is 0, so max(0, ...)
    # The largest possible index is len(s) - 1, so min(len(s) - 1, ...)
    
    # For a 0-indexed array, the rank `k` corresponds to index `k-1`.
    # If `rank` is 0, this means the 0th percentile, which should be the first element.
    # If `rank` is `len(s)`, this means the 100th percentile, which should be the last element.
    
    # The issue implies a direct index lookup rather than interpolation.
    # The rank is 1-indexed. To get a 0-indexed list index, we subtract 1.
    # We round up to get the "nearest rank" in the sense of a non-decreasing function.
    # However, the test case indicates a floor-like behavior for the index.
    # For P=50, N=8, rank=4. The expected value is s[3]=40.0.
    # This means the index should be floor(rank) - 1 if rank > 0.
    # Or more simply, if rank is the actual count, then the index is rank-1.
    # The index `k` in a 0-indexed array for the k-th smallest element.
    # For the `p`-th percentile, we need the element at `ceil(p/100 * N) - 1`.
    # Let's test this:
    # N=8, p=50. ceil(0.5 * 8) - 1 = ceil(4) - 1 = 4 - 1 = 3. s[3] = 40.0. This works.
    # N=3, p=100. ceil(1.0 * 3) - 1 = ceil(3) - 1 = 3 - 1 = 2. s[2] = 30.0. This works for test_perc.
    
    if p == 0:
        return s[0]
    if p == 100:
        return s[len(s) - 1]
    
    # Calculate the 1-based rank (position) and convert to 0-based index
    # using the nearest-rank method specified by the incident.
    # This uses the ceiling of (P/100 * N) and then adjusts for 0-indexing.
    index = math.ceil((p / 100.0) * len(s)) - 1
    
    # Ensure the index is within valid bounds for the list
    # The index can be -1 if p is very small and N=1.
    # The index can be len(s) if p=100.
    index = max(0, min(int(index), len(s) - 1))
    
    return s[index]

def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
