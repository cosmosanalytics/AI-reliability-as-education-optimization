"""Low-level jitter percentile calculation engine."""
import math
from typing import List

def evaluate_jitter_percentile(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using the nearest-rank method.

    The index k is calculated as math.ceil(p / 100 * n) - 1 on sorted values,
    where n is the number of samples. This corresponds to the definition
    specified for SLA compliance.
    """
    if not values:
        raise ValueError("No jitter values")
    if not (0 <= p <= 100):
        raise ValueError("Percentile p must be between 0 and 100")

    s = sorted(values)
    n = len(s)

    # Handle edge cases for p=0 and p=100 explicitly
    if p == 0:
        return float(s[0])
    if p == 100:
        return float(s[n - 1])

    # Calculate the 1-based rank, then convert to 0-based index
    # k = math.ceil(p / 100 * n) - 1 as specified for nearest-rank.
    k_1_based = math.ceil((p / 100.0) * n)
    k_0_based = int(k_1_based) - 1

    # The calculated k_0_based index will always be within [0, n-1]
    # for p in (0, 100).
    # For example:
    # n=1: k_1_based = ceil(p/100 * 1). If p=0.01, ceil(0.0001)=1. k_0_based=0. s[0].
    # If p=99.99, ceil(0.9999)=1. k_0_based=0. s[0].
    # n=5, p=1: k_1_based = ceil(0.01 * 5) = ceil(0.05) = 1. k_0_based = 0. s[0].
    # n=5, p=100: k_1_based = ceil(1.0 * 5) = 5. k_0_based = 4. s[4]. (handled by p=100 case)
    # n=5, p=99: k_1_based = ceil(0.99 * 5) = ceil(4.95) = 5. k_0_based = 4. s[4].
    return float(s[k_0_based])
