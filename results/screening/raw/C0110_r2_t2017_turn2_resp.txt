"""API gateway authorization guard."""
import re
from policy import RoutePolicy
from url_normalizer import clean_scope

# Pre-compile regex for splitting segments to avoid recompilation on each call
_SEGMENT_SPLITTER = re.compile(r"::")

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    allowed_segments = _SEGMENT_SPLITTER.split(policy.allowed_pattern)
    request_segments = _SEGMENT_SPLITTER.split(clean_req)

    i = 0  # index for allowed_segments
    j = 0  # index for request_segments

    while i < len(allowed_segments) and j < len(request_segments):
        allowed_segment = allowed_segments[i]
        request_segment = request_segments[j]

        if allowed_segment == "**":
            # If '**' is the last segment, it matches any remaining request segments (0 or more).
            # So, if we are at the end of the pattern, it matches.
            # Otherwise, it implies that the next segment in allowed_segments must be found later in request_segments.
            if i == len(allowed_segments) - 1:
                return True # Matches 0 or more remaining segments
            
            # Look for the next non-wildcard segment from the allowed pattern in the request
            next_allowed_segment = allowed_segments[i + 1]
            while j < len(request_segments):
                if request_segments[j] == next_allowed_segment:
                    break
                j += 1
            if j == len(request_segments): # Next allowed segment not found in request
                return False
            # Found the next allowed segment, now proceed to match it.
            i += 1 # Move past '**'
            # j is already at the matching segment, so it will be incremented in the next loop iteration.
            continue # Continue to the next iteration to match next_allowed_segment

        elif allowed_segment == "*":
            # '*' matches exactly one segment, so both indices must advance.
            i += 1
            j += 1
        elif allowed_segment == request_segment:
            # Exact match, advance both indices.
            i += 1
            j += 1
        else:
            # Mismatch, and not a wildcard that can handle it.
            return False

    # After the loop, check if both patterns are fully consumed.
    # If policy.allowed_pattern has remaining segments that are not '**', it's a mismatch.
    # If request_scope has remaining segments, and policy.allowed_pattern was fully consumed without a trailing '**', it's a mismatch.

    # Check for remaining wildcards in allowed_segments
    while i < len(allowed_segments):
        if allowed_segments[i] == "**":
            i += 1
        else:
            return False # Non-wildcard segment remaining in policy pattern
    
    # If all allowed_segments are consumed (or were '**'), and all request_segments are consumed, it's a match.
    return i == len(allowed_segments) and j == len(request_segments)


def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
