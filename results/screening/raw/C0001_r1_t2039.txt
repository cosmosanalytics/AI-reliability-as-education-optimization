"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys
from wire_envelope import build_envelope
from typing import Dict, Any

# TODO: service_name should ideally be configured or passed as an argument
# to the streaming context. For this patch, a placeholder is used.
_DEFAULT_SERVICE_NAME = "unknown_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """
    Prepare telemetry metric packet for distribution to event stream.
    Normalizes keys in the payload and wraps it in a wire envelope.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(_DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """
    Publishes a telemetry stream packet.
    Prepares the packet by normalizing keys and adding a wire envelope.
    """
    pkt = prepare_metric_packet(data, epoch_ms)
    # The original code had `_ = is_valid_envelope(pkt)` and
    # `_ = extract_normalized_keys(data)`.
    # `is_valid_envelope` check is no longer strictly necessary here as
    # `prepare_metric_packet` now guarantees a valid envelope structure.
    # `extract_normalized_keys` was a no-op on the raw input data.
    # Key normalization is now handled within `prepare_metric_packet`.
    return pkt
