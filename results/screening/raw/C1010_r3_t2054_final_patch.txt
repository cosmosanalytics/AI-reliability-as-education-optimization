"""Low-level dictionary and list hierarchy merger."""
from typing import Dict, Any, List

def merge_spec_values(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """Recursively merges dictionary values.

    For dictionary values, it recursively merges them.
    For list values, it combines them by extending the base list with unique elements
    from the override list, preserving the order of the base list first,
    then adding new elements from the override list in their original order.
    For all other types, it replaces the base value with the override value.
    """
    result = dict(base)
    for k, v in override.items():
        if k in result and isinstance(result[k], dict) and isinstance(v, dict):
            # Recursively merge dictionaries
            result[k] = merge_spec_values(result[k], v)
        elif k in result and isinstance(result[k], list) and isinstance(v, list):
            # Combine lists: extend base with unique elements from override
            # preserving order from base, then order from override for new elements.
            combined_list = list(result[k])
            for item in v:
                if item not in combined_list:
                    combined_list.append(item)
            result[k] = combined_list
        else:
            # For other types (or if types don't match for special merge), replace
            result[k] = v
    return result

def is_spec_empty(spec: Dict[str, Any]) -> bool:
    return len(spec) == 0
