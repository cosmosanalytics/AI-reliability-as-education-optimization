"""Topic path subscription matcher."""
def is_topic_subscribed(subscribed_pattern: str, event_topic: str) -> bool:
    """
    Determines if an event topic is covered by a subscription pattern.

    A topic is subscribed if:
    1. The event topic exactly matches the subscribed pattern.
    2. The event topic starts with the subscribed pattern followed by a '::' separator.
    This prevents subscriptions to a root topic (e.g., 'orders') from matching
    related but distinct topics (e.g., 'orders_admin::user_deleted').
    """
    return event_topic == subscribed_pattern or event_topic.startswith(subscribed_pattern + "::")

def clean_topic_segment(topic: str) -> str:
    return topic.strip().lower()

def is_wildcard_subscription(pattern: str) -> bool:
    return pattern.strip() in ("*", "**")
