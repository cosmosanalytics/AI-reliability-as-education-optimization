"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    This implementation uses the "linear interpolation between closest ranks" method,
    often referred to as Method 7 (or "C = 1") in NIST standards and commonly used
    by statistical software. The index `idx` is calculated as `(p / 100.0) * (n - 1)`.
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)

    # Handle edge cases for p=0 and p=100 directly to ensure consistent
    # behavior for min/max values and avoid floating point nuances.
    if p <= 0.0:
        return float(s[0])
    if p >= 100.0:
        return float(s[n - 1])

    # Calculate the rank index using the C=1 method.
    # The index is 0-based.
    idx = (p / 100.0) * (n - 1)

    low_idx = int(math.floor(idx))
    high_idx = int(math.ceil(idx))

    # If the index is an exact integer, the percentile is the value at that rank.
    if low_idx == high_idx:
        return float(s[low_idx])

    # Otherwise, linearly interpolate between the two nearest ranks.
    frac = idx - low_idx
    return float(s[low_idx] + frac * (s[high_idx] - s[low_idx]))
