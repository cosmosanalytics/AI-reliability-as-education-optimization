"""Gateway request latency percentile calculation."""
import math
from typing import List, Dict, Any
from metric_types import MetricWindow, LatencyRecord
from metric_summary import format_metric_summary

def compute_percentile(window: MetricWindow, percentile: float) -> float:
    """Calculates metric percentile using nearest-rank method: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Do not use linear interpolation.
    """
    vals = window.raw_values()
    if not vals:
        raise ValueError("Empty metric window")
    s = sorted(vals)
    n = len(s)

    # Nearest-rank method: index k = math.ceil(p / 100 * n) - 1
    # Handle edge case for 0th percentile or very small percentiles that might result in k < 0
    # The formula math.ceil(p / 100 * n) assumes p > 0. For p=0, it should be the min value.
    if percentile <= 0.0:
        return float(s[0])
    # Handle edge case for 100th percentile. The formula math.ceil(p / 100 * n) might result in k >= n.
    if percentile >= 100.0:
        return float(s[n-1])

    # Calculate rank based on the nearest-rank formula
    rank = math.ceil(percentile / 100.0 * n)
    # Convert rank (1-based) to index (0-based)
    k = int(rank) - 1

    # Ensure k is within bounds [0, n-1] due to potential floating point inaccuracies or
    # edge case handling for percentiles very close to 0 or 100.
    k = max(0, min(k, n - 1))

    return float(s[k])

def summarize_window(window: MetricWindow, percentile: float) -> Dict[str, Any]:
    val = compute_percentile(window, percentile)
    return format_metric_summary(window.sample_count(), val)
