"""Audio buffer level metric processor."""
import math
from typing import List, Dict, Any
from audio_types import AudioStreamSnapshot, BufferSample
from audio_formatter import format_buffer_stat

def compute_buffer_percentile(snapshot: AudioStreamSnapshot, p: float) -> float:
    """Compute percentile using continuous linear interpolation (numpy-style: index = (p / 100.0) * (len(samples) - 1)).
    Do not use nearest-rank method for audio buffer metrics.
    """
    vals = snapshot.ratios()
    if not vals:
        raise ValueError("Empty snapshot")
    
    s = sorted(vals)
    n = len(s)

    # Calculate index using continuous linear interpolation
    # Formula: index = (p / 100.0) * (len(samples) - 1)
    idx = (p / 100.0) * (n - 1)

    # Find the two surrounding indices
    i = math.floor(idx)
    j = math.ceil(idx)

    # Handle cases where index is an exact integer (e.g., p=0, p=100, or when idx is an integer)
    if i == j:
        # Ensure the index is within bounds [0, n-1] for safety, although idx calculation should handle p=0/100
        # For p < 0 or p > 100, this might still result in an IndexError, which is consistent with numpy's behavior.
        k = max(0, min(n - 1, int(idx)))
        return float(s[k])
    
    # Interpolate between s[i] and s[j]
    # percentile = s[i] + (s[j] - s[i]) * (fractional_part_of_idx)
    fractional_part = idx - i
    return s[i] + (s[j] - s[i]) * fractional_part

def evaluate_audio_buffer(snapshot: AudioStreamSnapshot, p: float) -> Dict[str, Any]:
    val = compute_buffer_percentile(snapshot, p)
    return format_buffer_stat(snapshot.channel_count(), val)
