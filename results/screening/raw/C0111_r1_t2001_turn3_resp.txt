"""Low-level tax rate computation engine."""
import math
from typing import Dict, Any

DEFAULT_TAX_JURISDICTIONS = {
    "standard": 0.20,
    "reduced": 0.10,
    "exempt": 0.0,
}

def get_rate_for_jurisdiction(code: str) -> float:
    return DEFAULT_TAX_JURISDICTIONS.get(code, 0.20)

def calculate_vat(net_cents: int, vat_rate: float) -> int:
    """Compute VAT in integer cents for given net amount.
    Uses half-up rounding (ROUND_HALF_UP: ties broken towards positive infinity).
    """
    # Python's built-in round() rounds to the nearest even number for ties (e.g., round(2.5) == 2, round(3.5) == 4).
    # The requirement is "half-up rounding (ROUND_HALF_UP: ties broken towards positive infinity)".
    # This means 2.5 rounds to 3, and -2.5 rounds to -2.
    # A common way to implement this is math.floor(x + 0.5) for positive x,
    # and math.ceil(x - 0.5) or a conditional approach for negative x.
    # A simpler unified approach is to use Decimal with the appropriate rounding strategy.
    # However, given the constraint of minimal changes and the existing integer arithmetic,
    # we can implement this with a conditional round.

    # Calculate the exact VAT amount
    exact_vat = net_cents * vat_rate

    # Apply half-up rounding towards positive infinity
    if exact_vat >= 0:
        # For positive numbers, add 0.5 and floor to round half-up
        return math.floor(exact_vat + 0.5)
    else:
        # For negative numbers, subtract 0.5 and ceiling to round half-up (towards positive infinity)
        # E.g., -2.5 -> ceil(-2.5 - 0.5) = ceil(-3.0) = -3 (this is round half down)
        # E.g., -2.5 -> round(-2.5) -> -2 (this is what we want)
        # E.g., -2.6 -> round(-2.6) -> -3 (this is what we want)
        # E.g., -2.4 -> round(-2.4) -> -2 (this is what we want)
        # Python's round() rounds to nearest, ties to even. We need ties to positive infinity.
        # So for -2.5, round() gives -2, which is correct for "towards positive infinity".
        # For -2.6, round() gives -3, which is correct.
        # For -2.4, round() gives -2, which is correct.
        # It seems Python's round() handles negative numbers correctly for "towards positive infinity" ties.
        # Let's verify for 2.5: round(2.5) is 2, but we need 3.
        # So a conditional approach is needed.

        # Custom round function for ROUND_HALF_UP (ties towards positive infinity)
        # This function handles both positive and negative numbers correctly for the specified rounding mode.
        # For example:
        # round_half_up(2.5) -> 3
        # round_half_up(2.4) -> 2
        # round_half_up(-2.5) -> -2
        # round_half_up(-2.6) -> -3
        # round_half_up(-2.4) -> -2

        # Check for exact half-integer ties
        if exact_vat % 1 == 0.5: # 2.5, 3.5, etc.
            return math.ceil(exact_vat) # Rounds 2.5 to 3, 3.5 to 4
        elif exact_vat % 1 == -0.5: # -2.5, -3.5, etc.
            return math.ceil(exact_vat) # Rounds -2.5 to -2, -3.5 to -3
        else:
            return round(exact_vat) # Standard Python round for other cases (nearest, ties to even)

        # Let's simplify with a slightly different logic that covers all cases.
        # The key is to add 0.5 and floor for positive, or subtract 0.5 and ceil for negative,
        # but adjust for the "towards positive infinity" tie-breaking.
        # A robust way is to use `decimal` module for precision and explicit rounding modes.
        # However, the problem implies sticking to `int` and `float` if possible.

        # Let's reconsider `math.floor(x + 0.5)` for all cases.
        # This is `ROUND_HALF_UP` for positive numbers.
        # For negative numbers:
        # -2.5 + 0.5 = -2.0 -> floor(-2.0) = -2 (Correct for ROUND_HALF_UP)
        # -2.6 + 0.5 = -2.1 -> floor(-2.1) = -3 (Correct for ROUND_HALF_UP)
        # -2.4 + 0.5 = -1.9 -> floor(-1.9) = -2 (Correct for ROUND_HALF_UP)
        # This function `math.floor(x + 0.5)` actually implements ROUND_HALF_UP for both positive and negative numbers.
        # Example:
        # x = 2.5 -> floor(3.0) = 3
        # x = 2.4 -> floor(2.9) = 2
        # x = -2.5 -> floor(-2.0) = -2
        # x = -2.6 -> floor(-2.1) = -3
        # x = -2.4 -> floor(-1.9) = -2
        # This seems to be the correct and simplest implementation for ROUND_HALF_UP.

    return math.floor(exact_vat + 0.5)
