"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses.

    This function uses the "nearest rank" method (NIST method) to find the
    percentile value. The rank is calculated as L = (p / 100.0) * (n - 1),
    and the percentile is the value at the integer index closest to L
    (rounding half to even for .5 values).
    """
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)

    # Handle edge cases for p=0 and p=100 explicitly for clarity,
    # though the formula below generally covers them.
    if p == 0:
        return float(s[0])
    if p == 100:
        return float(s[n - 1])

    # Calculate the rank (L) for the percentile.
    # The (n-1) factor is for 0-indexed arrays, where the first element is at rank 0
    # and the last element is at rank n-1.
    rank_L = (p / 100.0) * (n - 1)

    # Round the rank to the nearest integer.
    # Python's round() function rounds to the nearest even number for .5
    # (e.g., round(2.5) is 2, round(3.5) is 4).
    # This is a common and statistically sound rounding method.
    idx = int(round(rank_L))

    # Ensure the index is within bounds [0, n-1].
    # This should be implicitly handled by rank_L calculation for p in [0, 100],
    # but added for robustness.
    idx = max(0, min(idx, n - 1))

    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
