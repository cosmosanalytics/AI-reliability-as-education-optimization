"""Internal stream sorting helper."""
from typing import List
from event_packet import EventPacket

def order_packets(packets: List[EventPacket]) -> List[EventPacket]:
    # Sort primarily by severity (descending), then by stream_id (ascending) as a tiebreaker.
    # To ensure stable and deterministic ordering even when severity and stream_id are identical
    # for distinct packets, we add the hash of the packet as a final tiebreaker.
    # This prevents non-deterministic ordering that can lead to state desyncs.
    return sorted(packets, key=lambda p: (-p.severity, p.stream_id, hash(p)))

def get_max_severity(packets: List[EventPacket]) -> int:
    return max((p.severity for p in packets), default=0)

def count_critical_packets(packets: List[EventPacket]) -> int:
    return sum(1 for p in packets if p.is_critical())
