"""Gateway request latency percentile calculation."""
import math
from typing import List, Dict, Any
from metric_types import MetricWindow, LatencyRecord
from metric_summary import format_metric_summary

def compute_percentile(window: MetricWindow, percentile: float) -> float:
    """Calculates metric percentile using nearest-rank method: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Do not use linear interpolation.
    """
    vals = window.raw_values()
    if not vals:
        raise ValueError("Empty metric window")
    s = sorted(vals)
    n = len(s)

    if not (0 <= percentile <= 100):
        raise ValueError("Percentile must be between 0 and 100 inclusive.")

    # Apply the nearest-rank method definition from the docstring:
    # index k = math.ceil(p / 100 * n) - 1
    # Note: The problem statement uses 'k' as the 0-based index directly.
    # Some definitions of nearest-rank use 'rank R = ceil(p/100 * N)' where R is 1-based.
    # The docstring explicitly states the 0-based index calculation.
    
    # Handle edge cases for percentile values to ensure correct index calculation
    if percentile == 0:
        return float(s[0])
    if percentile == 100:
        return float(s[n - 1])

    # Calculate the 0-based index according to the docstring's nearest-rank definition
    # k = math.ceil(percentile / 100.0 * n) - 1
    # Ensure the result is an integer
    idx = int(math.ceil(percentile / 100.0 * n)) - 1
    
    # Ensure index is within valid bounds (0 to n-1)
    # This check is mostly for robustness, as the percentile edge cases should handle bounds.
    if idx < 0:
        idx = 0
    if idx >= n:
        idx = n - 1

    return float(s[idx])

def summarize_window(window: MetricWindow, percentile: float) -> Dict[str, Any]:
    val = compute_percentile(window, percentile)
    return format_metric_summary(window.sample_count(), val)
