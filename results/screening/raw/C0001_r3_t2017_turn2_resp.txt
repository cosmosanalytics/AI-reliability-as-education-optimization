"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    policy_pattern = policy.allowed_pattern

    policy_segments = policy_pattern.split('::')
    request_segments = clean_req.split('::')

    # Handle empty policy pattern
    if not policy_pattern: # This means policy_segments will be ['']
        return not clean_req # Only an empty request scope matches an empty policy pattern

    if policy_segments[-1] == '**':
        # Double asterisk matches arbitrary multi-segment prefixes.
        # The request scope must start with the policy prefix (excluding '**').
        prefix_len = len(policy_segments) - 1
        if len(request_segments) < prefix_len:
            return False
        for i in range(prefix_len):
            if policy_segments[i] != request_segments[i]:
                return False
        return True
    elif policy_segments[-1] == '*':
        # Single asterisk matches exactly one segment.
        # The request scope must have the same number of segments as the policy pattern.
        if len(request_segments) != len(policy_segments):
            return False
        # All segments except the last must match.
        for i in range(len(policy_segments) - 1):
            if policy_segments[i] != request_segments[i]:
                return False
        return True
    else:
        # No wildcards, exact segment match required.
        return policy_segments == request_segments

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
