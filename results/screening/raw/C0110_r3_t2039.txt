"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# A placeholder service name for the envelope. In a real application, this would likely
# be configured or passed in. For this patch, a constant is sufficient to demonstrate
# the envelope building.
SERVICE_NAME = "telemetry_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This includes normalizing keys and wrapping the payload in a wire envelope."""
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following line was likely intended as a validation or logging step.
    # It's kept for now, but its return value is not used.
    _ = is_valid_envelope(pkt)
    # The following line extracted normalized keys from the *original* data and
    # discarded the result. It's removed as it doesn't contribute to the packet.
    # _ = extract_normalized_keys(data)
    return pkt
