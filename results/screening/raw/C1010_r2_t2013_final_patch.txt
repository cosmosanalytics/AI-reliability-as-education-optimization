"""TSV serialization utilities."""
from typing import List, Optional
from schema import CustomerProfile
from io_helpers import validate_tsv_header

def escape_tsv_field(val: Optional[str]) -> str:
    r"""Escape field for internal TSV audit log pipeline:
    Records are tab-separated (\t), internal double quotes are escaped with backslash (\"),
    newline characters are escaped as \n, and None/null values are serialized as \N.
    Do NOT use RFC 4180 double-quoting.
    """
    if val is None:
        return r"\N" # Docstring: None/null values are serialized as \N.
    
    # Escape internal double quotes with backslash (\")
    # Escape newline characters as \n
    # The docstring specifies escaping for quotes and newlines, but not for tabs within a field.
    # The current implementation incorrectly uses RFC 4180 style.
    
    # Order of operations matters: escape backslashes first to avoid double-escaping new escapes
    escaped_val = val.replace("\\", "\\\\") # Escape existing backslashes first
    escaped_val = escaped_val.replace('"', r'\"') # Escape double quotes with backslash (\")
    escaped_val = escaped_val.replace('\n', r'\n') # Escape newline characters as \n
    
    return escaped_val

def export_profiles_tsv(profiles: List[CustomerProfile]) -> str:
    # The header string must use actual tab characters, not literal backslash-t
    header = "cust_id\tname\tnotes"
    if not validate_tsv_header(header, ["cust_id", "name", "notes"]):
        raise ValueError("Invalid header")
    lines = [header]
    for p in profiles:
        f_id = str(p.cust_id) # cust_id is already an int, convert to str
        f_name = escape_tsv_field(p.name)
        f_notes = escape_tsv_field(p.notes)
        # Each field should be joined by actual tab characters, not literal backslash-t
        lines.append(f"{f_id}\t{f_name}\t{f_notes}")
    # Lines should be joined by actual newline characters, not literal backslash-n
    return "\n".join(lines)

def count_records(payload: str) -> int:
    lines = [l for l in payload.splitlines() if l.strip()]
    return max(0, len(lines) - 1)
