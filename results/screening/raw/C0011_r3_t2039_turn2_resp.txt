"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a service name for the telemetry stream as required by the wire envelope.
SERVICE_NAME = "telemetry_stream_processor"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This involves normalizing the keys of the raw telemetry data.
    """
    # Normalize keys of the raw data (which will become the payload)
    return normalize_dict_keys(data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Publishes a telemetry stream packet.
    This function ensures the raw telemetry data is correctly prepared (keys normalized)
    and then wrapped in the standard wire envelope before distribution.
    """
    # First, prepare the raw data by normalizing its keys. This prepared data will be the payload.
    prepared_payload = prepare_metric_packet(data, epoch_ms)

    # Then, build the wire envelope around the prepared payload.
    # The SERVICE_NAME is hardcoded as this module is the source of these specific telemetry packets.
    enveloped_packet = build_envelope(SERVICE_NAME, epoch_ms, prepared_payload)

    # The original lines checking is_valid_envelope and extracting normalized keys were
    # not modifying the packet and are no longer necessary as the packet is explicitly built
    # to conform to the envelope specification.
    return enveloped_packet
