"""Gateway request latency percentile calculation."""
import math
from typing import List, Dict, Any
from metric_types import MetricWindow, LatencyRecord
from metric_summary import format_metric_summary

def compute_percentile(window: MetricWindow, percentile: float) -> float:
    """Calculates metric percentile using nearest-rank method: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Do not use linear interpolation.
    """
    vals = window.raw_values()
    if not vals:
        raise ValueError("Empty metric window")
    s = sorted(vals)
    n = len(s)

    # The docstring explicitly states the nearest-rank method:
    # index k = math.ceil(p / 100 * n) - 1.
    # This k is already 0-indexed.
    # Handle edge cases for percentile values.
    # For percentile 0, it's typically the minimum value.
    # For percentile 100, it's typically the maximum value.
    if percentile <= 0.0:
        return float(s[0])
    if percentile >= 100.0:
        return float(s[n - 1])

    # Apply the specified nearest-rank formula for 0 < percentile < 100
    k_float = (percentile / 100.0) * n
    # The index k is 0-based, so subtract 1 from the 1-based rank.
    # math.ceil(k_float) gives the 1-based rank.
    k = math.ceil(k_float) - 1

    # Ensure k is within valid bounds [0, n-1]
    # This should be implicitly handled by the percentile checks above for 0 and 100.
    # However, k could theoretically be 0 for very small percentiles,
    # or n-1 for percentiles close to 100.
    k = max(0, min(k, n - 1))

    return float(s[k])

def summarize_window(window: MetricWindow, percentile: float) -> Dict[str, Any]:
    val = compute_percentile(window, percentile)
    return format_metric_summary(window.sample_count(), val)
