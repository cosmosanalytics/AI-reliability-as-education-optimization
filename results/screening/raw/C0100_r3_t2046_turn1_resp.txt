"""Timestamp serialization codec for telemetry records."""
from datetime import datetime, timezone

def encode_iso_timestamp(dt: datetime) -> str:
    """Encode timestamp into standard format (ISO 8601 with Z for UTC, milliseconds)."""
    # Ensure the datetime object is in UTC for the 'Z' suffix.
    if dt.tzinfo is None:
        # Assuming naive datetimes are UTC for existing behavior,
        # but it's safer to make them aware if they are truly UTC.
        # However, the current code doesn't convert, so we'll stick to formatting.
        # If input is naive, this will format it as if it were UTC.
        # If input is aware, it will be converted to UTC.
        dt_utc = dt.replace(tzinfo=timezone.utc)
    else:
        dt_utc = dt.astimezone(timezone.utc)

    # Format to ISO 8601 with milliseconds and 'Z' for UTC.
    # datetime.isoformat(timespec='milliseconds') produces 'YYYY-MM-DDTHH:MM:SS.000+00:00' for UTC.
    # We need to replace '+00:00' with 'Z'.
    return dt_utc.isoformat(timespec='milliseconds').replace('+00:00', 'Z')

def validate_timestamp_range(dt: datetime) -> bool:
    return dt.year >= 2020
