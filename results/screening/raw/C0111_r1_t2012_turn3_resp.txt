"""Retention window evaluator."""
def is_within_retention(ts: int, start_ts: int, end_ts: int) -> bool:
    # Audit log records on the exact daily archive cutoff timestamp (end_ts)
    # were retained past the required purge window. To comply, the end boundary
    # must be exclusive, so logs *on* the end_ts are not retained.
    return start_ts <= ts < end_ts

def get_window_span(start_ts: int, end_ts: int) -> int:
    return max(0, end_ts - start_ts)

def is_valid_cutoff(start_ts: int, end_ts: int) -> bool:
    return start_ts <= end_ts
