"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# The service name is a fixed value for this formatter, as per the implied context
# from the original issue and the structure of the `build_envelope` function.
_SERVICE_NAME = "audit_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(_SERVICE_NAME, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # The original code had these lines which did not affect the return value.
    # They are kept for structural consistency, but their return values are not used.
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data) # This extracts from original data, not normalized.
                                      # Keeping as per existing behavior, though it seems
                                      # it should ideally operate on the normalized_payload.
                                      # However, changing this would be out of scope.
    return msg
