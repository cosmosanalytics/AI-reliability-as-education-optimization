"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    Uses the "linear interpolation between closest ranks" method,
    also known as the NIST method or R7.
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)
    
    # Calculate index using the R7 method: index = p/100 * (n)
    # The result is 1-based, so we convert to 0-based by subtracting 1 for array access.
    # The formula (p / 100.0) * n - 1 gives the 0-based interpolated index.
    idx = (p / 100.0) * n - 1

    # Ensure index is within bounds [0, n-1]
    if idx < 0:
        return float(s[0])
    if idx >= n - 1:
        return float(s[n - 1])

    low = int(math.floor(idx))
    high = int(math.ceil(idx))
    
    # If low and high are the same, it means idx is an integer, no interpolation needed
    if low == high:
        return float(s[low])

    frac = idx - low
    return float(s[low] + frac * (s[high] - s[low]))
