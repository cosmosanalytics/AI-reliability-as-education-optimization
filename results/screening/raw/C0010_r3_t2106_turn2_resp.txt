"""Cache Hit Latency Percentile Monitor percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_cache_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_cache_percentile", samples)
    s = validate_metric_samples(win.values)
    n = len(s)
    if n == 0:
        raise ValueError("Cannot calculate percentile for empty samples list.")
    if not (0 <= p <= 100):
        raise ValueError("Percentile must be between 0 and 100.")

    # Nearest-rank method (N+1) with linear interpolation
    # This aligns with the "nearest-rank house standard" often implying a method
    # where the index is calculated relative to N+1 positions, and then interpolated.
    # The failing test traceback implies that for 50th percentile, with an odd number
    # of elements, the median element is expected.
    # For an even number of elements, the interpolation between the two middle elements.
    # The specific failure (45.0 vs 40.0) suggests that the current (N-1) method
    # is off by one index or calculation for certain datasets.
    #
    # Let's consider the definition for P-th percentile (0 < P <= 100):
    # L = P/100 * N
    # If L is an integer, the P-th percentile is the average of the L-th and (L+1)-th value.
    # If L is not an integer, the P-th percentile is the ceil(L)-th value.
    #
    # However, the common "house standard" or "nearest rank" method in some contexts
    # uses a direct index based on (N-1) or N, and then uses linear interpolation
    # if the index is not an integer.
    #
    # The original code's `idx = (p / 100.0) * (len(s) - 1)` is a common definition (e.g., NumPy's default).
    # The failure `assert 45.0 == 40.0` for `calculate_cache_percentile(d1, 50.0)`
    # suggests `d1` might be `[10, 20, 30, 40, 50, 60]`.
    # Let's test this assumption:
    # `s = [10, 20, 30, 40, 50, 60]`, `n = 6`, `p = 50.0`
    # Original: `idx = (50/100) * (6-1) = 0.5 * 5 = 2.5`
    # `low = 2`, `high = 3`
    # `s[2] + (2.5 - 2) * (s[3] - s[2]) = 30 + 0.5 * (40 - 30) = 30 + 0.5 * 10 = 35.0`.
    # This is not 45.0.

    # Let's re-evaluate the traceback: `assert 45.0 == 40.0`.
    # This means `calculate_cache_percentile(d1, 50.0)` returned `45.0`, but should have been `40.0`.
    # If `d1 = [10.0, 20.0, 30.0, 40.0, 50.0]`, n=5.
    # `idx = (50/100) * (5-1) = 0.5 * 4 = 2.0`. Returns `s[2] = 30.0`. Not 45.0.
    # If `d1 = [10.0, 20.0, 30.0, 40.0, 50.0, 60.0, 70.0, 80.0, 90.0]`, n=9.
    # `idx = (50/100) * (9-1) = 0.5 * 8 = 4.0`. Returns `s[4] = 50.0`. Not 45.0.

    # Let's try to achieve 45.0 from the current formula.
    # `s[low] + (idx - low) * (s[high] - s[low]) = 45.0`
    # This implies `s[low]` and `s[high]` are around 40-50.
    # If `d1 = [10, 20, 30, 40, 50]`, and the expected 50th percentile is 40.0.
    # The current code gives 30.0.
    # A standard definition for the median (50th percentile) of `[10, 20, 30, 40, 50]` is `30.0`.
    # The expected value `40.0` implies a different interpretation of the percentile definition.

    # One common definition, often called "Method R-7" or "NIST method", is:
    # `h = (p / 100.0) * (n + 1)`
    # `j = int(h)`
    # `g = h - j`
    # `result = s[j-1] + g * (s[j] - s[j-1])` (using 0-based indexing)

    # Let's test this method with `d1 = [10, 20, 30, 40, 50]` (n=5) and p=50.0.
    # `h = (50/100) * (5 + 1) = 0.5 * 6 = 3.0`
    # `j = int(3.0) = 3`
    # `g = 3.0 - 3 = 0.0`
    # `result = s[3-1] + 0.0 * (s[3] - s[3-1]) = s[2] + 0.0 * (s[3] - s[2]) = s[2] = 30.0`.
    # This still yields 30.0, not 40.0.

    # What if the "nearest-rank house standard" means for an odd number of items,
    # the median is the middle item, and for an even number of items, the average
    # of the two middle items (standard median definition), and for percentiles
    # other than 50, it uses a rank that is shifted?

    # Let's consider the possibility that `d1` was `[10, 20, 30, 40, 50, 60, 70]`.
    # `n = 7`, `p = 50.0`.
    # Original: `idx = (50/100) * (7-1) = 0.5 * 6 = 3.0`. Returns `s[3] = 40.0`.
    # If the expected was 40.0, and the actual was 45.0, this implies `d1` was not `[10, ..., 70]`.

    # Let's try to work backwards from the actual result 45.0.
    # If `s = [10, 20, 30, 40, 50]`, and the current code returned 45.0,
    # it must have been `s[3] + 0.5 * (s[4] - s[3]) = 40 + 0.5 * (50 - 40) = 40 + 5 = 45.0`.
    # This would happen if `idx` was `3.5`.
    # `idx = (p / 100.0) * (n - 1) = 3.5`.
    # If `p = 50.0`, `n = 8` (e.g., `d1 = [10, 20, 30, 40, 50, 60, 70, 80]`).
    # `idx = (50/100) * (8-1) = 0.5 * 7 = 3.5`.
    # `low = 3`, `high = 4`.
    # `s[3] + (3.5 - 3) * (s[4] - s[3]) = s[3] + 0.5 * (s[4] - s[3]) = 40 + 0.5 * (50 - 40) = 45.0`.
    # So, if `d1 = [10, 20, 30, 40, 50, 60, 70, 80]`, the current code returns 45.0 for p=50.
    # The traceback says `assert 45.0 == 40.0`.
    # This means for `d1 = [10, 20, 30, 40, 50, 60, 70, 80]`, the 50th percentile is expected to be 40.0.

    # This is a strong hint about the "nearest-rank house standard".
    # For an even number of elements (N=8), `[10, 20, 30, 40, 50, 60, 70, 80]`,
    # the 50th percentile (median) is typically `(s[3] + s[4]) / 2 = (40 + 50) / 2 = 45.0`.
    # If the expected value is 40.0, this implies a percentile definition that
    # picks the lower of the two middle values, or is somehow index-shifted.

    # Let's consider if the `idx` calculation should be based on `N` instead of `N-1`.
    # This is method R-6 or "C=1" in some classifications.
    # `idx = (p / 100.0) * n`
    # If `d1 = [10, 20, 30, 40, 50, 60, 70, 80]` (n=8), `p = 50.0`.
    # `idx = (50/100) * 8 = 4.0`.
    # If `idx` is an integer, some definitions take `s[idx-1]`.
    # So `s[4-1] = s[3] = 40.0`.
    # This matches the expected value `40.0` for `d1 = [10, 20, 30, 40, 50, 60, 70, 80]`!
    # And the current code for this `d1` returns `45.0`.

    # So the fix is to change the `idx` calculation.
    # The "nearest-rank house standard" seems to imply this `idx = (p / 100.0) * n` method.
    # And if `idx` is an integer, return `s[idx-1]`. Otherwise, interpolate.
    # Let's verify with an example for non-integer `idx`.
    # `d1 = [10, 20, 30, 40, 50]`, `n = 5`, `p = 25.0`.
    # New `idx = (25/100) * 5 = 1.25`.
    # `low = 1`, `high = 2`.
    # `s[1] + (1.25 - 1) * (s[2] - s[1]) = 20 + 0.25 * (30 - 20) = 20 + 0.25 * 10 = 22.5`.
    #
    # Let's compare with `idx = (p / 100.0) * (n - 1)` (original).
    # `idx = (25/100) * (5-1) = 0.25 * 4 = 1.0`. Returns `s[1] = 20.0`.
    #
    # The "house standard" could also be:
    # `k = math.ceil((p / 100.0) * n)`
    # If `p` is 0, return `s[0]`. If `p` is 100, return `s[n-1]`.
    # This is often called "nearest rank" method.
    #
    # If `d1 = [10, 20, 30, 40, 50, 60, 70, 80]` (n=8), `p=50`.
    # `k = math.ceil(0.5 * 8) = math.ceil(4.0) = 4`. Returns `s[4-1] = s[3] = 40.0`. This matches.
    #
    # If `d1 = [10, 20, 30, 40, 50]` (n=5), `p=50`.
    # `k = math.ceil(0.5 * 5) = math.ceil(2.5) = 3`. Returns `s[3-1] = s[2] = 30.0`.
    #
    # This "nearest rank" method without interpolation seems to match the `40.0` expectation
    # for `d1 = [10, 20, 30, 40, 50, 60, 70, 80]` (n=8, p=50), where `s[3]` is 40.0.
    #
    # The wording "discrepancy against nearest-rank house standard" and the `45.0 == 40.0`
    # error strongly suggest that for `d1 = [10, 20, 30, 40, 50, 60, 70, 80]`,
    # the 50th percentile should be `40.0` (which is `s[3]`), not `45.0` (which is `(s[3]+s[4])/2`).
    # This means a method that picks an element directly, not interpolates for integers.

    # Let's implement the "nearest rank" (rank-based) method.
    # It typically calculates the rank `r = ceil(p/100 * N)`. The value is `s[r-1]`.
    # However, for 0th or 100th percentile, edge cases need to be handled.
    # For `p=0`, it should be `s[0]`. For `p=100`, it should be `s[n-1]`.

    if p == 0.0:
        return s[0]
    if p == 100.0:
        return s[n - 1]

    # Calculate rank using the N-based index for percentile.
    # This is often referred to as the "NIST method" or "R-6" in some literature,
    # or simply "nearest rank" when values are directly picked.
    # The traceback implies this method where for even N, P=50, it picks s[N/2 - 1].
    # For N=8, P=50, 0.5 * 8 = 4. s[4-1] = s[3] = 40.0.
    #
    # The original code's `idx = (p / 100.0) * (len(s) - 1)` is R-7.
    # The "nearest-rank house standard" for `45.0 == 40.0` points to R-6.

    # R-6 method:
    # h = (p / 100.0) * n
    # If h is an integer, result is s[h-1] (0-indexed)
    # If h is not an integer, result is s[floor(h)] (0-indexed)
    # This is equivalent to `s[math.ceil(h) - 1]` for non-integer `h` if we round up,
    # but the interpolation part is tricky.

    # Let's stick to the interpretation that `40.0` is expected when `45.0` is returned.
    # This happens when `s = [10, 20, 30, 40, 50, 60, 70, 80]` and `p = 50`.
    # Current: `idx = (0.5) * (8-1) = 3.5`. `low=3, high=4`. `s[3] + 0.5*(s[4]-s[3]) = 40 + 0.5*(50-40) = 45.0`.
    # Expected: `40.0`. This is `s[3]`.
    # This means when `idx` is `X.5`, we should pick `s[X]`. This is equivalent to `floor(idx)`.

    # This is a specific variant of "nearest rank" where if the rank is exactly halfway,
    # it rounds down to the lower index.
    # If `idx` is `X.0`, it's `s[X]`.
    # If `idx` is `X.1` to `X.9`, it's `s[floor(idx)]` or `s[ceil(idx)]` or interpolated.

    # Let's consider the "NIST method" for percentiles (often called R-6 or R-7 variant).
    # The current implementation is definition R-7 (linear interpolation between `N-1` indices).
    # The error `45.0 != 40.0` where `45.0` is `(s[3]+s[4])/2` for 8 elements,
    # and `40.0` is `s[3]`, implies a definition where for `P=50` and even `N`,
    # the percentile is `s[N*P/100 - 1]`.

    # Let's try to adjust `idx` calculation to match this.
    # The simplest "nearest rank" without interpolation for `p` between 0 and 100:
    # `rank = math.ceil((p / 100.0) * n)`
    # `return s[rank - 1]` (adjust for 0-based indexing)

    # Let's test this "nearest rank" method for the given case:
    # `d1 = [10, 20, 30, 40, 50, 60, 70, 80]`, `n = 8`, `p = 50.0`.
    # `rank = math.ceil((50.0 / 100.0) * 8) = math.ceil(4.0) = 4`.
    # `return s[4 - 1] = s[3] = 40.0`. This matches the expected value!

    # Test with `p = 100.0`:
    # `rank = math.ceil((100.0 / 100.0) * 8) = math.ceil(8.0) = 8`.
    # `return s[8 - 1] = s[7] = 80.0`. This matches `test_perc`'s `30.0` for `[10,20,30], 100.0`.

    # Test with `p = 0.0`:
    # `rank = math.ceil((0.0 / 100.0) * 8) = math.ceil(0.0) = 0`. This would lead to `s[-1]`.
    # So `p=0` and `p=100` need explicit handling.

    # This "nearest-rank house standard" seems to be the one where we calculate the rank
    # and pick the element at that rank directly, without interpolation, for values between 0 and 100.
    # The `idx` as calculated in the original code is for linear interpolation (R-7).
    # The `AssertionError` suggests a different definition.

    # Final logic based on "nearest-rank" interpretation:
    # 0th percentile is the first element.
    # 100th percentile is the last element.
    # For other percentiles, calculate the rank `r = ceil(p/100 * N)` and return `s[r-1]`.

    rank = math.ceil((p / 100.0) * n)
    return s[int(rank) - 1]

def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
