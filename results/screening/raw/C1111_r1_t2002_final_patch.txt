"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    This function ensures that the total allocated discount exactly matches
    total_discount_cents, distributing any remainder (pennies) to the lines
    with the largest fractional parts.

    line_totals: list of integer cents representing the value of each line.
    total_discount_cents: the total discount amount to be allocated, in cents.
    Returns list of discount cents per line.
    Raises ValueError if line_totals contains non-negative integers or if total_discount_cents is negative.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    if total_discount_cents < 0:
        raise ValueError("Total discount cents must be non-negative")

    total_sum_lines = sum(line_totals)
    num_lines = len(line_totals)

    if total_sum_lines == 0:
        # As per existing test and original behavior, if line values are zero,
        # no discount is allocated proportionally to their value.
        return [0] * num_lines

    # Calculate base allocations using integer division
    allocations = []
    for cents in line_totals:
        allocations.append(cents * total_discount_cents // total_sum_lines)

    # Calculate the remainder that needs to be distributed
    current_total_allocated = sum(allocations)
    remainder = total_discount_cents - current_total_allocated

    # Distribute the remainder (pennies) to lines with the largest fractional parts.
    # To do this, we calculate the exact float proportion and sort by the fractional part.
    # Store (fractional_part, original_index)
    fractional_parts = []
    for i, cents in enumerate(line_totals):
        exact_proportion = (cents * total_discount_cents) / total_sum_lines
        fractional_part = exact_proportion - allocations[i]
        fractional_parts.append((fractional_part, i))

    # Sort in descending order of fractional part to distribute pennies to lines
    # that are 'most owed' a penny.
    # In case of ties in fractional part, the original order of lines (stable sort behavior)
    # or the arbitrary order from sort is acceptable for distributing pennies.
    fractional_parts.sort(key=lambda x: x[0], reverse=True)

    # Add 1 cent to the lines with the largest fractional parts until remainder is 0
    for i in range(remainder):
        line_index_to_adjust = fractional_parts[i][1]
        allocations[line_index_to_adjust] += 1

    return allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    # After the fix in prorate_discount, total_allocated will always equal total_discount_cents
    # for valid inputs, ensuring is_fully_allocated is True.
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
