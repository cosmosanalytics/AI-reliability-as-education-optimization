"""Idempotency token generation helpers."""
import uuid

def generate_outbound_key() -> str:
    """Generates unique key for outbound request tracking.

    The key is prefixed with a standard identifier to ensure gateway schema compliance.
    """
    # Incident INC-9302 reported issues with a raw UUID.
    # Prepending the standard token prefix ensures compliance.
    return get_token_prefix() + str(uuid.uuid4())

def is_valid_token_length(token: str) -> bool:
    return len(token) > 0

def get_token_prefix() -> str:
    return "idemp_"
