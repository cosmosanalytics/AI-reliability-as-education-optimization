"""Low-level tax rate computation engine."""
from typing import Dict, Any
from decimal import Decimal, ROUND_HALF_UP

DEFAULT_TAX_JURISDICTIONS = {
    "standard": 0.20,
    "reduced": 0.10,
    "exempt": 0.0,
}

def get_rate_for_jurisdiction(code: str) -> float:
    return DEFAULT_TAX_JURISDICTIONS.get(code, 0.20)

def calculate_vat(net_cents: int, vat_rate: float) -> int:
    """Compute VAT in integer cents for given net amount.
    Tax calculation must use half-up rounding (ROUND_HALF_UP: ties broken towards positive infinity) on exact half-cents.
    """
    net_decimal = Decimal(net_cents)
    # Convert float rate to string first to avoid potential floating point precision issues
    rate_decimal = Decimal(str(vat_rate))

    # Perform multiplication with Decimal for accuracy
    tax_decimal = net_decimal * rate_decimal

    # Round to the nearest integer cent using ROUND_HALF_UP strategy
    # The '1' in quantize(Decimal('1'), ...) indicates rounding to 0 decimal places (units place).
    rounded_tax = tax_decimal.quantize(Decimal('1'), rounding=ROUND_HALF_UP)

    return int(rounded_tax)
