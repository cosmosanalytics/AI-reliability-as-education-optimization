"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    This function uses the "nearest rank" method for percentile calculation,
    which is commonly used in SLA monitoring to ensure that a certain
    percentage of queries fall within a threshold. For a given list of
    `n` sorted values, the index `i` is calculated as `ceil(p * n / 100)`.
    The percentile is then the value at `s[i-1]` (using 0-based indexing).
    If the calculated index `i` is 0, the smallest value is returned.
    If `p` is 100, the largest value is returned.

    Args:
        values: A list of numerical values.
        p: The percentile to calculate (e.g., 90 for 90th percentile).

    Returns:
        The calculated percentile value.

    Raises:
        ValueError: If `values` is empty.
    """
    if not values:
        raise ValueError("Empty values")
    if not (0 <= p <= 100):
        raise ValueError("Percentile p must be between 0 and 100 inclusive")

    s = sorted(values)
    n = len(s)

    if p == 0:
        return float(s[0])
    if p == 100:
        return float(s[n - 1])

    # Using the "nearest rank" method: i = ceil(p * n / 100)
    # The percentile value is then the i-th value in the sorted list (1-indexed),
    # which corresponds to s[i-1] in 0-indexed list.
    rank = math.ceil((p / 100.0) * n)
    # Ensure rank is at least 1 and at most n
    rank = max(1, min(n, int(rank)))
    return float(s[rank - 1])
