"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses.

    This implementation uses a "nearest rank" method. The 0-indexed rank `k` is
    calculated as `k = ceil(p * n / 100) - 1`, where `n` is the number of samples.
    The index `k` is clamped to ensure it falls within the valid range `[0, n-1]`.
    This method ensures that for a value to be at the p-th percentile, at least
    p% of the data is less than or equal to it, addressing inconsistencies
    reported in INC-7103 where values were sometimes rounded down when an
    inclusive percentile definition was expected.
    """
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)

    # Calculate the 1-based rank using the formula k = ceil(p * n / 100).
    # This corresponds to a common definition of percentiles where the p-th
    # percentile is the smallest value in the data set that is greater than
    # or equal to p percent of the data.
    rank_float = (p / 100.0) * n
    # Convert 1-based rank to 0-indexed position.
    idx = int(math.ceil(rank_float)) - 1

    # Ensure the calculated index is within the valid bounds [0, n-1].
    # This handles edge cases for p=0, p<0, p=100, or p>100.
    idx = max(0, min(idx, n - 1))

    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
