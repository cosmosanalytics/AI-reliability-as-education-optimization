"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses."""
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)

    # The original formula `int(round((p / 100.0) * (n - 1)))` uses Python's
    # built-in `round()` function, which for .5 values rounds to the nearest even number.
    # This behavior (round half to even) can lead to inconsistencies compared to
    # statistical definitions that typically round .5 up (e.g., 2.5 becomes 3).
    #
    # The traceback `assert 20.0 == 30.0` suggests an off-by-one error where
    # the calculated index is one less than expected, often observed when
    # `round(X.5)` yields X instead of X+1.
    #
    # To align with a "round half up" behavior for percentile index calculation,
    # we use `math.floor(x + 0.5)` which effectively rounds X.5 up to X+1 for positive X.
    # The index is also clamped to ensure it stays within the valid bounds [0, n-1].
    idx_float = (p / 100.0) * (n - 1)
    idx = max(0, min(n - 1, int(math.floor(idx_float + 0.5))))

    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
