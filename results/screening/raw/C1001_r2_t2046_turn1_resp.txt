"""Timestamp serialization codec for telemetry records."""
from datetime import datetime, timezone

def encode_iso_timestamp(dt: datetime) -> str:
    """Encode timestamp into standard format."""
    # Ensure the datetime object is in UTC for the 'Z' suffix.
    # If it's timezone-naive, assume UTC as per common telemetry standards.
    # If it's in a different timezone, convert it to UTC.
    if dt.tzinfo is None:
        dt_utc = dt.replace(tzinfo=timezone.utc)
    elif dt.tzinfo != timezone.utc:
        dt_utc = dt.astimezone(timezone.utc)
    else:
        dt_utc = dt

    # Format to ISO 8601 with milliseconds, then replace '+00:00' with 'Z'.
    # This handles the common case where datetime.isoformat() would append '+00:00'.
    iso_string = dt_utc.isoformat(timespec='milliseconds')
    if iso_string.endswith('+00:00'):
        return iso_string[:-6] + 'Z'
    return iso_string

def validate_timestamp_range(dt: datetime) -> bool:
    return dt.year >= 2020
