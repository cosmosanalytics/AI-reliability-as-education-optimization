"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    pattern = policy.allowed_pattern

    if pattern == "**":
        return True

    policy_segments = pattern.split('::')
    request_segments = clean_req.split('::')

    p_idx = 0
    r_idx = 0

    while p_idx < len(policy_segments) and r_idx < len(request_segments):
        p_seg = policy_segments[p_idx]
        r_seg = request_segments[r_idx]

        if p_seg == "**":
            # If '**' is the last segment in the policy, it matches any remaining segments in the request.
            # Otherwise, it's an invalid pattern, or a more complex one not covered.
            # Per the problem description, "**" matches arbitrary multi-segment prefixes.
            # This implies it should appear at the end or as the sole pattern.
            # If it's not the last segment, it's ambiguous. Let's assume it only appears at the end.
            # "corp::**" -> matches corp::a, corp::a::b, corp::a::b::c
            # "corp::a::**" -> matches corp::a::b, corp::a::b::c
            return True # Matches any remaining segments

        if p_seg == "*":
            # '*' matches exactly one segment.
            # It matches the current request segment, then moves to the next policy segment.
            p_idx += 1
            r_idx += 1
        elif p_seg == r_seg:
            # Exact segment match.
            p_idx += 1
            r_idx += 1
        else:
            # Mismatch.
            return False

    # After loop, all policy segments must have been matched.
    # If p_idx < len(policy_segments), it means some policy segments were not matched (e.g., policy "a::b::c", request "a::b").
    # The only exception is if the remaining policy segment is "**" (which is handled above).
    # If p_idx == len(policy_segments), all policy segments were matched.
    # If r_idx < len(request_segments), it means the request has extra segments not covered by the policy.
    # E.g., policy "a::b", request "a::b::c". This should not be permitted.
    # So, both indices must reach their respective ends.

    return p_idx == len(policy_segments) and r_idx == len(request_segments)
