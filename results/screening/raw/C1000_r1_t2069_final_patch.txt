"""Gateway request latency percentile calculation."""
import math
from typing import List, Dict, Any
from metric_types import MetricWindow, LatencyRecord
from metric_summary import format_metric_summary

def compute_percentile(window: MetricWindow, percentile: float) -> float:
    """Calculates metric percentile using nearest-rank method: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Do not use linear interpolation.
    """
    vals = window.raw_values()
    if not vals:
        raise ValueError("Empty metric window")
    s = sorted(vals)
    n = len(s)

    # Implement nearest-rank method: index k = math.ceil(p / 100 * n) - 1
    # The percentile value must be between 0 and 100.
    if not (0 <= percentile <= 100):
        raise ValueError("Percentile must be between 0 and 100, inclusive.")

    # Special handling for 0th percentile and 100th percentile
    if percentile == 0:
        return float(s[0])
    if percentile == 100:
        return float(s[n - 1])

    # Calculate index using the nearest-rank method formula: k = ceil(p / 100 * n) - 1
    # The result of ceil(p / 100 * n) will be between 1 and n for 0 < p <= 100.
    # Subtracting 1 gives an index between 0 and n-1, inclusive.
    k = math.ceil(percentile / 100.0 * n) - 1

    # Ensure k is within valid bounds [0, n-1] in case of floating point inaccuracies
    # or edge cases not fully covered by the formula for 0 and 100 percentiles.
    k = max(0, min(k, n - 1))

    return float(s[k])

def summarize_window(window: MetricWindow, percentile: float) -> Dict[str, Any]:
    val = compute_percentile(window, percentile)
    return format_metric_summary(window.sample_count(), val)
