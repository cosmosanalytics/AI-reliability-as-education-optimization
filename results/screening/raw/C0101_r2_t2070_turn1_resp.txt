"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    Uses the "nearest-rank" method (NIST method) where the index is
    calculated as (p / 100.0) * (n - 1), and linear interpolation
    is applied between the two nearest ranks. This is equivalent to
    the R-7 method often described in statistical literature.
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)
    # The index calculation for the R-7 method (linear interpolation of empirical distribution function)
    # This method computes a weighted average between two observations.
    # The index `idx` is 0-based.
    idx = (p / 100.0) * (n - 1)

    low_idx = int(math.floor(idx))
    high_idx = int(math.ceil(idx))

    # Handle edge case where p=100 and idx=n-1, so low_idx=high_idx=n-1
    # or p=0 and idx=0, so low_idx=high_idx=0.
    # Also, if n=1, idx will be 0, low_idx=0, high_idx=0.
    if low_idx == high_idx:
        return float(s[low_idx])

    # Linear interpolation
    frac = idx - low_idx
    return float(s[low_idx] + frac * (s[high_idx] - s[low_idx]))
