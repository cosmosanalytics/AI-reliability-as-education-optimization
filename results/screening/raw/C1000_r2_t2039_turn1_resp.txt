"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int, service_name: str = "telemetry_service") -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # The original implementation returned the raw data, which was not a valid wire envelope.
    # We now use build_envelope to wrap the data with the required metadata.
    # A default service_name is provided to maintain compatibility with existing callers.
    return build_envelope(service_name, epoch_ms, data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following line checks if the packet is a valid envelope.
    # With the fix in prepare_metric_packet, this should now pass for valid inputs.
    _ = is_valid_envelope(pkt)
    # This call extracts normalized keys from the original payload data, not the envelope.
    _ = extract_normalized_keys(data)
    return pkt
