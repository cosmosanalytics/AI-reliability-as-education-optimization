"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Default service name for telemetry packets when not explicitly provided.
# This assumes the telemetry stream itself acts as the source service.
DEFAULT_SERVICE_NAME = "telemetry_stream"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int, service_name: str = DEFAULT_SERVICE_NAME) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    The packet will be wrapped in a wire envelope with required metadata.
    """
    return build_envelope(service_name, epoch_ms, data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    _ = is_valid_envelope(pkt) # This check ensures the envelope is valid, though its result is not used here.
    _ = extract_normalized_keys(data) # Note: this is called on the original data, not the enveloped packet.
    return pkt
