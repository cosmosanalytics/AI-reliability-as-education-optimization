"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# TODO: Determine appropriate service name, potentially from config or environment.
# For now, a placeholder is used to satisfy the wire envelope requirement.
_SERVICE_NAME = "unknown_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This includes normalizing payload keys and building the wire envelope."""
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(_SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following lines are for validation/side effects and do not affect the returned packet.
    # is_valid_envelope(pkt) is implicitly true if prepare_metric_packet correctly builds it.
    # extract_normalized_keys(data) extracts keys from the *original* data, not the normalized one.
    # These lines are kept for now as they might have external dependencies or future purpose,
    # but their return values are not used.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # Note: This operates on original 'data', not 'normalized_payload'
    return pkt
