"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total_line_value = sum(line_totals)
    num_lines = len(line_totals)

    if total_line_value == 0:
        return [0] * num_lines

    # Calculate base allocations using integer division
    allocations = []
    for cents in line_totals:
        allocations.append(cents * total_discount_cents // total_line_value)

    # Distribute remaining cents due to truncation
    current_allocated_sum = sum(allocations)
    remainder = total_discount_cents - current_allocated_sum

    if remainder < 0:
        # This should ideally not happen if total_discount_cents is non-negative
        # and line_totals are non-negative. But as a safeguard, if over-allocated,
        # we could try to remove from largest lines, or just return as is if small.
        # For now, we assume the original formula does not over-allocate.
        pass
    elif remainder > 0:
        # Distribute the remainder one cent at a time, prioritizing lines
        # that would have received a larger fraction.
        # This is done by checking the fractional part.
        # A more robust way is to sort by remaining fractional part of the division.
        # For simplicity, we can distribute to the first 'remainder' lines.
        # A more precise distribution would consider the `cents % total_line_value`
        # or the exact float value `(cents * total_discount_cents) / total_line_value`
        # to identify which lines "lost" the most due to truncation.

        # A common approach is to distribute the remainder to the lines with the largest
        # unallocated fractional parts. We can approximate this by iterating and adding
        # to lines that have a non-zero portion of their `cents * total_discount_cents`
        # when divided by `total_line_value`.

        # Calculate the "ideal" fractional allocations for tie-breaking
        # and sort lines by the amount they "lost" due to integer truncation.
        fractional_parts = [
            (cents * total_discount_cents % total_line_value, i)
            for i, cents in enumerate(line_totals)
        ]
        # Sort in descending order of fractional part to distribute remaining cents
        # to lines that were most "shortchanged" by truncation.
        # If fractional parts are equal, secondary sort by original line value (descending)
        # to avoid arbitrary distribution if all fractions are zero or equal.
        # If that's also equal, then by index to ensure determinism.
        sorted_indices = [
            idx for frac, idx in
            sorted(fractional_parts, key=lambda x: (x[0], line_totals[x[1]]), reverse=True)
        ]

        for i in range(remainder):
            allocations[sorted_indices[i]] += 1

    return allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
