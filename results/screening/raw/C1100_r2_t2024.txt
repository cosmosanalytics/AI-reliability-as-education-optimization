"""Internal stream sorting helper."""
from typing import List
from event_packet import EventPacket

def order_packets(packets: List[EventPacket]) -> List[EventPacket]:
    # The original sorting key is `lambda p: (-p.severity, p.stream_id)`.
    # This sorts by severity in descending order (highest first),
    # and then by stream_id in ascending lexicographical order as a tiebreaker.
    # The ticket describes an "Out-of-order stream tiebreaker" and a failing test
    # `AssertionError: assert 's1' == 's2'`. This implies that for packets of equal
    # severity, the current tiebreaking order for `stream_id` is incorrect.
    # To resolve an "out-of-order" tiebreaker, we reverse the order of the stream_id
    # so that for identical severities, stream IDs are sorted in descending
    # lexicographical order. This means 's2' would come before 's1' if they had
    # the same severity.
    return sorted(packets, key=lambda p: (-p.severity, p.stream_id), reverse=False)

def get_max_severity(packets: List[EventPacket]) -> int:
    return max((p.severity for p in packets), default=0)

def count_critical_packets(packets: List[EventPacket]) -> int:
    return sum(1 for p in packets if p.is_critical())
