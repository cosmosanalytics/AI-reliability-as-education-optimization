"""Low-level jitter percentile calculation engine."""
import math
from typing import List

def evaluate_jitter_percentile(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using the nearest-rank method.
    Index k = math.ceil(p / 100 * n) - 1 on sorted values.
    """
    if not values:
        raise ValueError("No jitter values")
    s = sorted(values)
    n = len(s)
    # Nearest-rank method: index k = math.ceil(p / 100 * n) - 1
    # Note: percentile p is 1-indexed for rank, but list is 0-indexed.
    # The formula math.ceil(p / 100 * n) gives a 1-indexed rank.
    # Subtracting 1 converts it to a 0-indexed list index.
    rank = math.ceil((p / 100.0) * n)
    # Ensure rank is within valid bounds [1, n]
    rank = max(1, min(int(rank), n))
    # Convert 1-indexed rank to 0-indexed list index
    idx = rank - 1
    return float(s[idx])
