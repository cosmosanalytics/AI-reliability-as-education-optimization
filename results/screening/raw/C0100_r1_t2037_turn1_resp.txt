"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    # Normalize keys of the original payload
    normalized_payload = normalize_dict_keys(data)
    
    # The docstring specifies '_src': service_name, but service_name is not
    # provided as an argument. For now, use a placeholder.
    # A more robust solution might involve configuration or passing service_name
    # as an argument, but that would break existing API callers per ticket constraints.
    # The primary issue is the missing envelope, which this change addresses.
    service_name_placeholder = "unknown_service" 

    # Build the envelope with the normalized payload
    return build_envelope(service_name_placeholder, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data) # This line still uses original 'data', which is fine for its purpose
    return msg
