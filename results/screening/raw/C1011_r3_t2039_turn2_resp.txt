"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # The incident indicates that the telemetry payload is missing wire envelope metadata.
    # The `build_envelope` function from `wire_envelope` is designed for this purpose.
    # A placeholder service name "telemetry_service" is used as the issue doesn't
    # specify how to obtain the service name, and this function is responsible for
    # creating the complete packet including the envelope.
    # The original `data` becomes the `payload` within the envelope.
    return build_envelope(service_name="telemetry_service", epoch_ms=epoch_ms, payload=data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # `prepare_metric_packet` now creates the full enveloped packet.
    pkt = prepare_metric_packet(data, epoch_ms)
    # The checks `is_valid_envelope` and `extract_normalized_keys` are informational
    # or for internal validation within this function, but the primary responsibility
    # is to return the correctly prepared packet.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # This operates on the original data, not the enveloped packet.
    return pkt
