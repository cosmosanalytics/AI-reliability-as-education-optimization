"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    
    # Handle the special case where the pattern is the global wildcard "**"
    if policy.allowed_pattern == "**":
        return True

    # Split patterns and requests into segments.
    # An empty string splits into [''] by default, which is not ideal for segment matching.
    # Adjust to treat empty strings as having no segments.
    pattern_segments = policy.allowed_pattern.split('::') if policy.allowed_pattern else []
    request_segments = clean_req.split('::') if clean_req else []

    p_idx = 0
    r_idx = 0

    while p_idx < len(pattern_segments) and r_idx < len(request_segments):
        p_seg = pattern_segments[p_idx]
        r_seg = request_segments[r_idx]

        if p_seg == '*':
            # '*' matches exactly one segment.
            # We already checked r_idx < len(request_segments) in the loop condition, so r_seg exists.
            p_idx += 1
            r_idx += 1
        elif p_seg == '**':
            # '**' matches arbitrary multi-segment prefixes.
            # If '**' is the last pattern segment, it matches any remaining request segments.
            if p_idx == len(pattern_segments) - 1:
                return True
            
            # Otherwise, it matches some segments, and we need to find the next literal pattern segment.
            # Advance r_idx until request_segments[r_idx] matches pattern_segments[p_idx + 1].
            next_p_seg = pattern_segments[p_idx + 1]
            found_next_segment = False
            while r_idx < len(request_segments):
                if request_segments[r_idx] == next_p_seg:
                    found_next_segment = True
                    break
                r_idx += 1
            
            if not found_next_segment:
                return False # The segment after '**' was not found in the remaining request.
            
            p_idx += 2 # Move past '**' and the segment it matched up to (which is next_p_seg).
        elif p_seg == r_seg:
            # Exact segment match.
            p_idx += 1
            r_idx += 1
        else:
            # Mismatch.
            return False

    # After iterating, both pattern and request segments must be fully consumed for a full match.
    # The case where pattern ends with '**' is handled inside the loop.
    return p_idx == len(pattern_segments) and r_idx == len(request_segments)

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
