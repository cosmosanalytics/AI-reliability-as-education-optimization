"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)

    if policy.is_wildcard():  # Handles the "**" pattern
        return True

    policy_segments = policy.allowed_pattern.split("::")
    request_segments = clean_req.split("::")

    if len(request_segments) < len(policy_segments):
        return False

    for i, p_segment in enumerate(policy_segments):
        if p_segment == "**":
            return True  # "**" matches the rest of the request segments
        elif p_segment == "*":
            # "*" matches exactly one segment, so we need to ensure there's a segment
            # at this position in request_segments and it's not empty, which
            # split("::") already ensures.
            # If "*" is the last policy segment, it matches exactly one remaining request segment.
            # If there are more policy segments after "*", this implies a "mid-pattern" wildcard.
            # For this simple implementation, we assume "*" only appears at the end.
            # If it's not the last segment, it's a more complex matching scenario.
            # Given the problem context, we assume simple trailing wildcards.
            if i >= len(request_segments): # No segment in request for the *
                return False
            # If '*' is the last segment in policy, and we have a request segment here, it's a match.
            # If not the last segment, more complex logic is needed, but the current problem doesn't require it.
            # For now, it matches one segment, so we continue.
            pass
        elif p_segment != request_segments[i]:
            return False # Segment mismatch

    # All policy segments (excluding wildcards) matched up to this point.
    # Check if a wildcard allows more segments or if it's an exact segment match.
    if policy_segments[-1] == "*":
        # If the last policy segment is "*", it matches exactly one request segment.
        # So, the request must have exactly the same number of segments as the policy.
        return len(request_segments) == len(policy_segments)
    elif policy_segments[-1] == "**":
        # This case is handled by `policy.is_wildcard()` and the loop's `p_segment == "**"` check.
        # If we reach here, it implies "**" was not the last segment or was not present.
        # This branch should ideally not be reached if the loop handles "**" correctly.
        pass

    # If no wildcards, it must be an exact segment match in length too.
    return len(request_segments) == len(policy_segments)

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
