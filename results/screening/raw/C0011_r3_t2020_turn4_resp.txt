"""Webhook scope dispatch checker."""
from prefix_validator import is_topic_subscribed, clean_topic_segment
from webhook import WebhookEvent

def should_dispatch_webhook(subscribed_topic: str, event: WebhookEvent) -> bool:
    """Determine if webhook event topic is permitted under the subscription filter.

    A subscription to a root topic (e.g., "orders") matches events only if their
    root topic exactly matches the subscription. It does not match similarly
    named but distinct root topics (e.g., "orders_admin").

    A subscription to a specific path (e.g., "orders::created") matches events
    that are sub-topics of that path (e.g., "orders::created::item_added").
    """
    if "::" not in subscribed_topic:
        # If the subscribed topic is a root topic, it must exactly match the event's root topic.
        return subscribed_topic == event.topic_root()
    else:
        # If the subscribed topic specifies a path, use prefix matching.
        return is_topic_subscribed(subscribed_topic, event.topic_path)

def filter_dispatch_queue(subscribed_topic: str, events: list) -> list:
    return [e for e in events if should_dispatch_webhook(subscribed_topic, e)]

def batch_topic_summary(events: list) -> dict:
    summary = {}
    for e in events:
        summary[e.topic_path] = summary.get(e.topic_path, 0) + 1
    return summary
