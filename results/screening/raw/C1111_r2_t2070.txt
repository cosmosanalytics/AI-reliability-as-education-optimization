"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    Uses the "linear interpolation between the closest two ranks" method (e.g., NumPy's
    'linear' method or R-7 in NIST SP 800-22).
    """
    if not values:
        raise ValueError("Empty values")
    if not (0.0 <= p <= 100.0):
        raise ValueError("Percentile p must be between 0 and 100, inclusive.")

    s = sorted(values)
    n = len(s)

    # The index calculation for percentile using the R-7 method.
    # This formula ensures that for p=0, idx=0 (s[0]) and for p=100, idx=n-1 (s[n-1]).
    idx = (p / 100.0) * (n - 1)

    # Handle edge cases for idx to ensure it's within valid array bounds [0, n-1].
    # This is particularly important for p=0 or p=100 with small n, or floating point inaccuracies.
    # If n=1, idx will be 0 regardless of p.
    if n == 1:
        return float(s[0])

    # Ensure idx is clamped within [0, n-1] to prevent index errors for values slightly
    # out of range due to floating point arithmetic (e.g., idx might be -1e-15 or n-1+1e-15).
    idx = max(0.0, min(n - 1.0, idx))

    low = int(math.floor(idx))
    high = int(math.ceil(idx))

    if low == high:
        # If idx is an exact integer, or floor and ceil are the same (e.g., idx=3.0),
        # then the value is directly at s[low].
        return float(s[low])
    else:
        # Linear interpolation between s[low] and s[high].
        # The fractional part represents how far between s[low] and s[high] the percentile lies.
        frac = idx - low
        return float(s[low] + frac * (s[high] - s[low]))
