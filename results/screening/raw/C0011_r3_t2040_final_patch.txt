"""IoT sensor ingest stream demux."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def process_iot_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Process IoT device packet.
    
    For raw IoT ingest pipeline, MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope.
    """
    # Incident IOT-703: The original implementation incorrectly normalized keys
    # and wrapped the payload in a v2 envelope, contradicting the docstring.
    # The requirement is to return the original, unwrapped payload dictionary
    # with its original camelCase keys.
    return data

def dispatch_iot_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = process_iot_packet(data, epoch_ms)
    # This line calculates normalized keys but does not use them for the return value
    # of dispatch_iot_record. It's retained as it doesn't affect the primary fix
    # and its removal might have unforeseen side-effects in other parts of the system
    # not covered by this ticket.
    _ = extract_normalized_keys(data)
    return msg
