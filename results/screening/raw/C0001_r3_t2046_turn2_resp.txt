"""Timestamp serialization codec for telemetry records."""
from datetime import datetime, timezone

def encode_iso_timestamp(dt: datetime) -> str:
    """
    Encode timestamp into standard ISO 8601 format (YYYY-MM-DDTHH:MM:SS.sssZ for UTC).

    This function ensures the timestamp is in UTC and formats it with milliseconds
    and the 'Z' suffix, as required for standard telemetry log exports.
    """
    # Ensure the datetime object is timezone-aware and in UTC.
    # If it's naive, assume UTC as per common telemetry practices and failing test context.
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    else:
        dt = dt.astimezone(timezone.utc)

    # Format as YYYY-MM-DDTHH:MM:SS.sss
    # %f gives microseconds (6 digits), so we slice it to get milliseconds (3 digits).
    iso_format_without_tz = dt.strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3]

    # Append 'Z' to indicate UTC time.
    return iso_format_without_tz + 'Z'

def validate_timestamp_range(dt: datetime) -> bool:
    return dt.year >= 2020
