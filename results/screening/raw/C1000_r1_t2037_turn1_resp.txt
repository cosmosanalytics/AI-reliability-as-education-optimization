"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Internal constant for the service name, as it's not provided externally in the current signature.
# This ensures the envelope can be built as per the docstring.
_DEFAULT_SERVICE_NAME = "unknown_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    # The docstring specifies 'service_name' for '_src'. Since it's not an argument,
    # we use an internal default to satisfy the envelope structure.
    return build_envelope(
        service_name=_DEFAULT_SERVICE_NAME,
        epoch_ms=epoch_ms,
        payload=normalized_payload
    )

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # These calls are for side effects or internal validation within serialize_audit_event,
    # not directly used for the return value of serialize_audit_event itself.
    _ = is_valid_envelope(msg)
    _ = extract_normalized_keys(data) # This extracts keys from the *original* data, not the normalized payload.
    return msg
