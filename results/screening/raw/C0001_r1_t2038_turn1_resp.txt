"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

SERVICE_NAME = "billing-service"

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers.

    The payload keys are normalized to snake_case, and the entire message
    is wrapped in a standard wire envelope.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(SERVICE_NAME, epoch_ms, normalized_payload)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # The record is now expected to be a valid envelope.
    # These checks are now for verification, not for action.
    if not is_valid_envelope(record):
        # This branch indicates a serious internal error in emit_billing_record
        raise ValueError("Emitted billing record is not a valid wire envelope.")
    
    # The payload keys are already normalized by emit_billing_record,
    # so extracting normalized keys from the original data here is redundant
    # for the purpose of the emitted message. This line can be removed
    # or kept for diagnostic/logging purposes if needed elsewhere.
    _ = extract_normalized_keys(data)
    
    return record
