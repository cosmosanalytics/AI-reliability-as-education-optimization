"""Gateway request latency percentile calculation."""
import math
from typing import List, Dict, Any
from metric_types import MetricWindow, LatencyRecord
from metric_summary import format_metric_summary

def compute_percentile(window: MetricWindow, percentile: float) -> float:
    """Calculates metric percentile using nearest-rank method: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Do not use linear interpolation.
    """
    vals = window.raw_values()
    if not vals:
        raise ValueError("Empty metric window")
    s = sorted(vals)
    n = len(s)

    # The docstring specifies the nearest-rank method where the 1-based rank 'k' is
    # k = math.ceil(p / 100 * n).
    # For a 0-based array, the index is k-1.

    # Handle edge cases for percentile 0 and 100 explicitly to ensure
    # correct index and adherence to typical percentile definitions.
    # For p=0, k would be 0, leading to index -1. The 0th percentile is the min value.
    if percentile == 0:
        return float(s[0])
    # For p=100, k would be n, leading to index n-1. The 100th percentile is the max value.
    if percentile == 100:
        return float(s[n-1])

    # Calculate k (1-based rank)
    k = math.ceil(percentile / 100.0 * n)
    
    # Convert to 0-based index
    index = int(k) - 1
    
    # The calculated index should always be within [0, n-1] for percentile in (0, 100)
    # given k in [1, n].
    return float(s[index])

def summarize_window(window: MetricWindow, percentile: float) -> Dict[str, Any]:
    val = compute_percentile(window, percentile)
    return format_metric_summary(window.sample_count(), val)
