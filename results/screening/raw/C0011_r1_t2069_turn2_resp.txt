"""Gateway request latency percentile calculation."""
import math
from typing import List, Dict, Any
from metric_types import MetricWindow, LatencyRecord
from metric_summary import format_metric_summary

def compute_percentile(window: MetricWindow, percentile: float) -> float:
    """Calculates metric percentile using nearest-rank method: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Do not use linear interpolation.
    """
    vals = window.raw_values()
    if not vals:
        raise ValueError("Empty metric window")
    s = sorted(vals)
    n = len(s)

    # Apply the nearest-rank method: k = math.ceil(p / 100 * n) - 1
    # The percentile 'p' is expected as a value like 90.0 for 90th percentile.
    # The index 'k' must be 0-based.
    # math.ceil(percentile / 100.0 * n) gives a 1-based rank.
    # Subtract 1 to get a 0-based index.
    rank_1_based = math.ceil(percentile / 100.0 * n)
    
    # Ensure the index is within bounds [0, n-1]
    # For 0th percentile, rank_1_based can be 0, resulting in index -1.
    # For 100th percentile, rank_1_based is n, resulting in index n.
    # The smallest valid index is 0, the largest is n-1.
    # math.ceil(p/100 * n) can be 0 only if p=0. In this case, rank_1_based will be 0.
    # We want the first element for 0th percentile, which is index 0.
    # For p=0, 0/100*n = 0. ceil(0) = 0. 0-1 = -1. This is incorrect.
    # If percentile is 0, we should return the minimum value (s[0]).
    # If percentile is 100, we should return the maximum value (s[n-1]).

    if percentile <= 0.0:
        return float(s[0])
    if percentile >= 100.0:
        return float(s[n - 1])

    # For percentiles > 0 and < 100, calculate index using the formula.
    # The formula k = math.ceil(p / 100 * n) - 1 naturally handles the bounds correctly
    # when p is strictly between 0 and 100.
    # For example, if p=0.01, n=100, ceil(0.01/100 * 100) - 1 = ceil(0.01) - 1 = 1 - 1 = 0.
    # If p=99.99, n=100, ceil(99.99/100 * 100) - 1 = ceil(99.99) - 1 = 100 - 1 = 99.
    index = int(rank_1_based) - 1
    
    # Due to floating point precision or edge cases with small n, ensure index is within valid range.
    index = max(0, min(index, n - 1))

    return float(s[index])

def summarize_window(window: MetricWindow, percentile: float) -> Dict[str, Any]:
    val = compute_percentile(window, percentile)
    return format_metric_summary(window.sample_count(), val)
