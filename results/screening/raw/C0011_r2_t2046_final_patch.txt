"""Timestamp serialization codec for telemetry records."""
from datetime import datetime, timezone

def encode_iso_timestamp(dt: datetime) -> str:
    """
    Encode timestamp into standard ISO 8601 format (with milliseconds and 'Z' for UTC).

    The input datetime object is first converted to UTC if it's timezone-aware
    but not UTC, or assumed to be UTC if it's naive.
    """
    # Ensure the datetime object is timezone-aware and in UTC
    if dt.tzinfo is None:
        # If naive, assume UTC as per common practice for telemetry and the required format.
        dt = dt.replace(tzinfo=timezone.utc)
    else:
        # Convert to UTC if it's not already
        dt = dt.astimezone(timezone.utc)

    # Format to ISO 8601 with milliseconds
    iso_str = dt.isoformat(timespec='milliseconds')

    # Replace '+00:00' with 'Z' for UTC
    if iso_str.endswith('+00:00'):
        return iso_str[:-6] + 'Z'
    return iso_str

def validate_timestamp_range(dt: datetime) -> bool:
    return dt.year >= 2020
