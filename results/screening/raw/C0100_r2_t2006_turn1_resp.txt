"""Database pool manager."""
from typing import Dict, Any
from env_loader import merge_env_settings
from options import ConnectionOptions

def configure_pool(base_template: Dict[str, Any], overrides: Dict[str, Any]) -> ConnectionOptions:
    """Configure connection pool options from template and overrides.
    Returns ConnectionOptions instance.
    """
    settings = merge_env_settings(base_template, overrides)
    dsn = settings.get("dsn", "sqlite:///:memory:")
    # Create a *new* dictionary for pool_params to ensure isolation.
    # The merge_env_settings performs a shallow merge, so if 'pool' is present in both
    # base_template and overrides, the entire 'pool' dictionary from overrides replaces
    # the one from base_template. To correctly merge nested 'pool' settings,
    # we need to handle that specifically.
    base_pool_params = base_template.get("pool", {})
    override_pool_params = overrides.get("pool", {})
    
    merged_pool_params = dict(base_pool_params)
    merged_pool_params.update(override_pool_params)

    opts = ConnectionOptions(dsn=dsn, pool_params=merged_pool_params)
    _ = opts.summary()
    return opts
