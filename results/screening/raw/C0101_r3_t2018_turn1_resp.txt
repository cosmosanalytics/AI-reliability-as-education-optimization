"""Document scope RBAC filter."""
from path_resolver import split_scope_segments
from audit_logger import format_access_attempt

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern."""
    allowed_parts = split_scope_segments(allowed_scope)
    requested_parts = split_scope_segments(requested_scope)

    # An empty allowed_scope implies no specific permissions, thus no access.
    # An empty requested_scope cannot be accessed.
    # This also implicitly handles cases where split_scope_segments returns empty for invalid syntax.
    if not allowed_parts or not requested_parts:
        allowed = False
    elif len(requested_parts) < len(allowed_parts):
        allowed = False
    else:
        allowed = requested_parts[:len(allowed_parts)] == allowed_parts

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
