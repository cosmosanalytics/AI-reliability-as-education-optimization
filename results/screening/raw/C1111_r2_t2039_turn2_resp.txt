"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # The incident report indicates a missing wire envelope.
    # The prepare_metric_packet function should return the payload,
    # with its keys normalized, to be wrapped by a caller.
    return normalize_dict_keys(data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The original implementation called is_valid_envelope and extract_normalized_keys
    # on the raw data, which was not a valid envelope.
    # The incident indicates that the output of this function should be a valid wire envelope.
    # We use a placeholder service name as it's not provided by the current signature
    # and not relevant to the key normalization aspect.
    envelope = build_envelope(service_name="unknown_service", epoch_ms=epoch_ms, payload=pkt)
    # Perform validation on the *constructed* envelope, not the raw data.
    _ = is_valid_envelope(envelope)
    # Extract normalized keys from the *payload* of the constructed packet, not the raw data.
    # This call is likely for internal logging or validation and does not affect the return value.
    _ = extract_normalized_keys(pkt)
    return envelope
