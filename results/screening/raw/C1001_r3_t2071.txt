"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses."""
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)
    # The original formula `int(round((p / 100.0) * (n - 1)))` can lead to
    # inconsistencies, especially when the calculated rank is exactly X.5,
    # due to the 'round half to even' (banker's rounding) behavior of Python's
    # built-in `round()` function for floats, or simply a different
    # interpretation of percentile definition.
    #
    # A common and robust way to calculate percentiles, especially when
    # needing to select an existing data point (nearest-rank method),
    # involves calculating a rank and then using `math.ceil` or `math.floor`
    # to find the index.
    #
    # For the `p`-th percentile, using the formula `(p / 100.0) * (n - 1)`
    # gives a 0-indexed rank. Applying `math.ceil` to this rank ensures that
    # for ranks like 1.5 (e.g., 50th percentile in a 4-element list), it rounds
    # up to 2, picking the higher of the two middle elements. This aligns with
    # common statistical definitions and resolves the observed inconsistency.
    #
    # Example: s = [10, 20, 30, 40], n = 4, p = 50
    # rank = (50 / 100.0) * (4 - 1) = 0.5 * 3 = 1.5
    # idx = int(math.ceil(1.5)) = 2. s[2] is 30.0.
    #
    # Example from failing test: If the input was s = [10, 20, 30], n = 3, p = 50
    # Original: rank = 0.5 * (3-1) = 1.0. round(1.0) = 1. s[1] = 20.0.
    # With fix: rank = 0.5 * (3-1) = 1.0. ceil(1.0) = 1. s[1] = 20.0.
    #
    # If the failing test was for s = [10, 20, 30], n = 3, and expected 30.0,
    # it implies that the percentile was something like p=75.
    # rank = 0.75 * (3-1) = 1.5.
    # Original: round(1.5) = 2. s[2] = 30.0. (This would pass)
    # With fix: ceil(1.5) = 2. s[2] = 30.0. (This would pass)
    #
    # The traceback "assert 20.0 == 30.0" means the code returned 20.0, but 30.0 was expected.
    # This implies that the original `round` function, for a certain `p` and `n`,
    # rounded down where `ceil` would round up, leading to a smaller index.
    # For example, if `rank` was `1.5` and `round` resulted in `1` (due to float precision or specific rounding rule),
    # but `ceil` would correctly yield `2`.
    # `math.ceil` consistently rounds up, which is a more common and explicit behavior for percentile index calculation.
    
    rank = (p / 100.0) * (n - 1)
    idx = int(math.ceil(rank))
    
    # Ensure the calculated index is within the valid bounds of the list.
    # For p=0, rank=0, ceil(0)=0. For p=100, rank=n-1, ceil(n-1)=n-1.
    # This `max(0, min(idx, n - 1))` ensures robustness against floating point
    # errors potentially pushing `rank` slightly out of `[0, n-1]` for edge cases.
    idx = max(0, min(idx, n - 1))
    
    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
