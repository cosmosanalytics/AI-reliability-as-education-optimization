"""Tenant routing configuration aggregator."""
import copy
from typing import Dict, Any
from models import RouteProfile
from schema import validate_route_keys

def merge_tenant_routing(base_config: Dict[str, Any], tenant_overrides: Dict[str, Any]) -> Dict[str, Any]:
    """Merge tenant overrides into base configuration.
    Inputs must remain strictly immutable; the function must return a newly allocated deep copy.
    """
    # Create a deep copy of the base configuration to ensure immutability
    # and isolation of nested structures.
    merged = copy.deepcopy(base_config)

    # Recursively merge tenant overrides into the deep-copied base configuration.
    # This function handles nested dictionaries by updating values in place
    # within the 'merged' dictionary.
    def _deep_merge(target: Dict[str, Any], source: Dict[str, Any]):
        for k, v in source.items():
            if k in target and isinstance(target[k], dict) and isinstance(v, dict):
                # If both target and source have a dictionary for this key, recurse
                _deep_merge(target[k], v)
            else:
                # Otherwise, overwrite the target's value with the source's value
                target[k] = v

    _deep_merge(merged, tenant_overrides)
    return merged

def build_route_profile(tenant_id: str, base_config: Dict[str, Any], tenant_overrides: Dict[str, Any]) -> RouteProfile:
    cfg = merge_tenant_routing(base_config, tenant_overrides)
    # The validate_route_keys function is called here to ensure merged config has required keys.
    # Its boolean return value is currently not used, but the call itself is part of the existing flow.
    validate_route_keys(cfg)
    return RouteProfile(tenant_id=tenant_id, settings=cfg)
