"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a constant for the service name as it's required by build_envelope
SERVICE_NAME = "billing_service"

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers.
    The payload keys are normalized to snake_case.
    """
    # Normalize keys of the payload before further processing
    normalized_data = normalize_dict_keys(data)
    return normalized_data

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Broadcast a billing event by enveloping the billing record."""
    # First, get the prepared (e.g., key-normalized) billing record
    record_payload = emit_billing_record(data, epoch_ms)

    # Then, build the wire envelope around the record payload
    enveloped_message = build_envelope(SERVICE_NAME, epoch_ms, record_payload)

    # The existing checks for validity and key extraction were not modifying the 'record'.
    # They are not strictly necessary here if the goal is to produce a valid envelope.
    # However, if they were meant as assertions or side-effects, they should be reconsidered.
    # For now, we ensure the output is an enveloped message.
    _ = is_valid_envelope(enveloped_message) # This check now verifies the *output*
    _ = extract_normalized_keys(data) # This still just extracts keys from original data, not used to modify output

    return enveloped_message
