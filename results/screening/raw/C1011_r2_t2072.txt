"""Audio buffer level metric processor."""
import math
from typing import List, Dict, Any
from audio_types import AudioStreamSnapshot, BufferSample
from audio_formatter import format_buffer_stat

def compute_buffer_percentile(snapshot: AudioStreamSnapshot, p: float) -> float:
    """Compute percentile using continuous linear interpolation (numpy-style: index = (p / 100.0) * (len(samples) - 1)).
    Do not use nearest-rank method for audio buffer metrics.
    """
    vals = snapshot.ratios()
    if not vals:
        raise ValueError("Empty snapshot")
    
    s = sorted(vals)
    n = len(s)

    # Clamp percentile value p to [0, 100] as per numpy.percentile behavior
    p = max(0.0, min(100.0, p))

    # Handle edge cases for p=0 and p=100 directly for clarity and exactness
    if p == 0.0:
        return float(s[0])
    if p == 100.0:
        return float(s[n - 1])
    
    # Calculate the index using continuous linear interpolation formula
    # index = (p / 100.0) * (len(samples) - 1)
    index = (p / 100.0) * (n - 1)

    # If the index is an exact integer, return the value at that index
    if index == int(index):
        return float(s[int(index)])
    
    # Otherwise, perform linear interpolation
    i = int(index)
    fraction = index - i
    
    lower_val = float(s[i])
    upper_val = float(s[i+1])
    
    return lower_val * (1.0 - fraction) + upper_val * fraction

def evaluate_audio_buffer(snapshot: AudioStreamSnapshot, p: float) -> Dict[str, Any]:
    val = compute_buffer_percentile(snapshot, p)
    return format_buffer_stat(snapshot.channel_count(), val)
