"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.
    Uses the nearest-rank method (NIST Type 6) suitable for discrete distributions
    where the Pth percentile is the smallest value in the data set that is greater
    than or equal to P percent of the values.
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)
    # The Pth percentile is the smallest value in the data set that is greater
    # than or equal to P percent of the values. This is often used for SLAs.
    # The rank is calculated as ceil((p / 100.0) * n).
    # Since list indices are 0-based, we subtract 1 from the rank.
    # We also clamp the index to ensure it's within [0, n-1] for robustness
    # against p values outside the [0, 100] range.
    rank = math.ceil((p / 100.0) * n)
    idx = int(max(0, min(rank - 1, n - 1)))
    return float(s[idx])
