"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define the service name for this emitter.
# This name identifies the source of the messages in the wire envelope.
SERVICE_NAME = "billing"

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers.

    The emitted message is a wire-enveloped payload, including service metadata
    and the provided transaction data.
    """
    # The incident report indicates that payloads must be enveloped.
    # We use build_envelope to ensure the message conforms to the wire protocol.
    enveloped_payload = build_envelope(SERVICE_NAME, epoch_ms, data)
    return enveloped_payload

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # Validate that the emitted record is a valid envelope.
    _ = is_valid_envelope(record)
    # Extract normalized keys from the original data, likely for logging or internal processing.
    _ = extract_normalized_keys(data)
    return record
