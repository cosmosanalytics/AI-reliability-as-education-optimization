"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses."""
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)
    # The original implementation uses round(), which in Python 3 rounds half to nearest even.
    # This can cause inconsistencies for percentiles where the index falls exactly on X.5,
    # leading to an off-by-one error compared to common 'round half up' behavior.
    # Changing to int(x + 0.5) ensures 'round half up' behavior for positive numbers,
    # which is often expected for percentile index calculations.
    # This addresses cases where the index was rounded down (e.g., 2.5 -> 2) when it should have been rounded up (2.5 -> 3).
    idx = int((p / 100.0) * (n - 1) + 0.5)
    
    # Ensure index is within bounds [0, n-1]
    idx = max(0, min(idx, n - 1))
    
    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)

"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses."""
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)
    
    # The original implementation used round(), which in Python 3 rounds half to nearest even.
    # This can lead to inconsistencies where the calculated index for a percentile
    # (e.g., 2.5) was rounded down (to 2) when a 'round half up' behavior (to 3)
    # was implicitly expected by a test or standard definition, causing an off-by-one error.
    # Changing to `int(x + 0.5)` for positive x ensures 'round half up' behavior,
    # which is a common expectation for percentile index calculations.
    # This addresses cases where the result was 20.0 but 30.0 was expected,
    # implying the index was off by one due to rounding.
    idx = int((p / 100.0) * (n - 1) + 0.5)
    
    # Ensure the calculated index is within the valid bounds [0, n-1].
    # For p in [0, 100] and n > 0, the computed idx will naturally fall within these bounds
    # due to the formula (p/100)*(n-1) ranging from 0 to n-1, and adding 0.5 then int()
    # maintaining this. However, an explicit clamp ensures robustness against edge cases
    # or unexpected inputs for p.
    idx = max(0, min(idx, n - 1))
    
    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
