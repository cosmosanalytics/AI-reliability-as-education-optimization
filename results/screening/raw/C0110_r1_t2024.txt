"""Internal stream sorting helper."""
from typing import List
from event_packet import EventPacket

def order_packets(packets: List[EventPacket]) -> List[EventPacket]:
    # Sort primarily by severity (descending). For tie-breaking, use a stable sort
    # to preserve existing order, which is crucial for consistent state replication
    # across distributed systems when stream_id alone might not be sufficient
    # or the natural order is preferred as a secondary tie-breaker.
    # Python's Timsort (used by sorted()) is stable, so explicitly sorting
    # by stream_id for a secondary key is sufficient and robust.
    return sorted(packets, key=lambda p: (-p.severity, p.stream_id))

def get_max_severity(packets: List[EventPacket]) -> int:
    return max((p.severity for p in packets), default=0)

def count_critical_packets(packets: List[EventPacket]) -> int:
    return sum(1 for p in packets if p.is_critical())
