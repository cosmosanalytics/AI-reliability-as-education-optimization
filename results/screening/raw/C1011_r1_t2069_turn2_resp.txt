"""Gateway request latency percentile calculation."""
import math
from typing import List, Dict, Any
from metric_types import MetricWindow, LatencyRecord
from metric_summary import format_metric_summary

def compute_percentile(window: MetricWindow, percentile: float) -> float:
    """Calculates metric percentile using nearest-rank method: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Do not use linear interpolation.
    """
    vals = window.raw_values()
    if not vals:
        raise ValueError("Empty metric window")
    s = sorted(vals)
    n = len(s)
    
    # Calculate index k using the nearest-rank method as specified in the docstring:
    # k = math.ceil(p / 100 * n) - 1
    # The percentile value must be between 0 and 100.
    if not (0 <= percentile <= 100):
        raise ValueError("Percentile must be between 0 and 100.")

    # For percentile 0, return the minimum value (first element).
    if percentile == 0:
        return float(s[0])
    
    # For percentile 100, return the maximum value (last element).
    if percentile == 100:
        return float(s[n-1])

    # Calculate rank based on the formula: p/100 * n
    rank = (percentile / 100.0) * n
    
    # The nearest rank method uses the ceiling of the rank.
    # The index is then rank_ceiling - 1 because Python lists are 0-indexed.
    k_idx = math.ceil(rank) - 1
    
    # Ensure the index is within bounds [0, n-1].
    # This handles edge cases where ceil(rank) might be 0 or n+1 if percentile is 0 or 100,
    # though 0 and 100 are handled above.
    # For any percentile > 0 and < 100, k_idx will naturally fall within [0, n-1] for n >= 1.
    k_idx = max(0, min(k_idx, n - 1))
    
    return float(s[k_idx])

def summarize_window(window: MetricWindow, percentile: float) -> Dict[str, Any]:
    val = compute_percentile(window, percentile)
    return format_metric_summary(window.sample_count(), val)
