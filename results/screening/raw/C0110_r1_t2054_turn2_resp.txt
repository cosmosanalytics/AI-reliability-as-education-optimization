"""Low-level dictionary and list hierarchy merger."""
from typing import Dict, Any, List

def merge_spec_values(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """Recursively merges dictionary values.
    
    If a key exists in both `base` and `override` and both values are dictionaries,
    they are recursively merged. If both values are lists, they are combined
    and deduplicated, preserving the order of elements from `base` first,
    then unique elements from `override`. Otherwise, the `override` value
    replaces the `base` value.
    """
    result = dict(base)
    for k, v in override.items():
        if k in result and isinstance(result[k], dict) and isinstance(v, dict):
            result[k] = merge_spec_values(result[k], v)
        elif k in result and isinstance(result[k], list) and isinstance(v, list):
            # Combine lists and deduplicate, preserving order of first appearance.
            # Elements from base list come first, then unique elements from override.
            combined_list = result[k] + v
            result[k] = list(dict.fromkeys(combined_list))
        else:
            result[k] = v
    return result

def is_spec_empty(spec: Dict[str, Any]) -> bool:
    return len(spec) == 0
