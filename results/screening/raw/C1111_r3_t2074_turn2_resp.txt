"""Network jitter SLA evaluation service."""
import math
from typing import List, Dict, Any
from jitter_types import JitterStream, JitterSample
import jitter_core

def check_jitter_sla(stream: JitterStream, p: float, max_jitter: float) -> bool:
    """SLA compliance requires nearest-rank percentile estimation: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Returns True if compliance limit is respected.
    """
    vals = stream.raw_jitters()
    if not vals:
        # Handle empty stream case as per jitter_core's ValueError
        raise ValueError("No jitter values for SLA check")

    n = len(vals)
    s = sorted(vals)

    # Nearest-rank percentile estimation as specified in the docstring
    # index k = math.ceil(p / 100 * n) - 1
    # Note: percentile p is 1-indexed, so we subtract 1 for 0-indexed list access.
    # Also, math.ceil(x) for x > 0 is at least 1. If p=0, this would be math.ceil(0)-1 = -1.
    # If p=0, the 0th percentile is usually min. If p=100, 100th percentile is max.
    # The formula implies p > 0. If p=0, the index would be -1, which is invalid.
    # If p=100, index would be n-1 (for the largest element).
    # Assuming p is typically between (0, 100].
    # For p=0, the behavior is undefined by this formula, but often min value.
    # The problem implies p > 0, as it's a percentile.
    
    # Clamp p to a reasonable range for index calculation to avoid math.ceil(0) - 1
    # and to ensure a valid index for n > 0.
    # If p is 0, the nearest rank formula would yield -1.
    # If p is very small, ceil(p/100 * n) might be 0, leading to -1.
    # To correctly handle edge cases where p/100 * n results in a small float close to 0,
    # we should ensure the index is at least 0.
    
    # The formula `k = math.ceil(p / 100 * n) - 1` naturally handles p=100 (k=n-1).
    # For p=0, it would be k=-1. A common interpretation for 0th percentile is min value (index 0).
    # Let's adjust the formula slightly to guarantee a valid index >= 0 when n > 0.
    # The lowest valid index is 0.
    # The highest valid index is n - 1.
    
    # The nearest-rank method typically defines index i as ceil(p/100 * n).
    # The value is then s[i-1] (0-indexed).
    # So k = max(0, int(math.ceil(p / 100.0 * n)) - 1)
    
    # Given the failing test suggests 22.0 should be 20.0, it points to a direct application
    # of the formula `k = math.ceil(p / 100 * n) - 1` on the sorted array.
    # This formula implies p > 0. If p is 0, it would be -1.
    # For practical percentiles, p is usually > 0.
    
    # Let's stick strictly to the stated formula for k, assuming valid p values.
    k_float = (p / 100.0) * n
    # The formula specifically says: index k = math.ceil(p / 100 * n) - 1
    k = int(math.ceil(k_float)) - 1

    # Ensure k is within valid bounds [0, n-1]
    k = max(0, min(k, n - 1))
    
    percentile_val = float(s[k])
    return percentile_val <= max_jitter

def get_stream_jitter_metrics(stream: JitterStream, p: float) -> Dict[str, Any]:
    vals = stream.raw_jitters()
    q = jitter_core.evaluate_jitter_percentile(vals, p)
    return {"count": stream.sample_count(), "jitter_p": q}
