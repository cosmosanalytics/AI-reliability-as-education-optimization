"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int, service_name: str = "unknown_service") -> Dict[str, Any]:
    """
    Prepare telemetry metric packet for distribution to event stream.
    The payload keys are normalized to snake_case and then wrapped in a wire envelope.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int, service_name: str = "unknown_service") -> Dict[str, Any]:
    """
    Publishes a telemetry stream packet.
    The packet is first prepared, normalizing keys and adding wire envelope metadata.
    """
    # The original logic included checks that were not used or were applied to the wrong data.
    # The packet is now guaranteed to be a valid envelope if prepare_metric_packet is correct.
    pkt = prepare_metric_packet(data, epoch_ms, service_name)
    return pkt
