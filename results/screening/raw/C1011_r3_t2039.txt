"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # Incident STREAM-105 indicates that the telemetry payload is missing required
    # wire envelope metadata. The `build_envelope` function is designed to add
    # this metadata.
    # A placeholder service name "telemetry_service" is used as the issue doesn't
    # specify how to obtain this value, making this the most direct fix.
    # The original `data` is now correctly nested as the `payload` within the envelope.
    return build_envelope(service_name="telemetry_service", epoch_ms=epoch_ms, payload=data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # `prepare_metric_packet` now produces the complete, enveloped packet.
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following lines perform checks on the packet/data but do not alter
    # the returned `pkt`, which is already correctly formatted.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # Operates on the original data, not the enveloped packet.
    return pkt
