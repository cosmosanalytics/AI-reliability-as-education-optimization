"""Low-level jitter percentile calculation engine."""
import math
from typing import List

def evaluate_jitter_percentile(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using the nearest-rank method.
    Index k = math.ceil(p / 100 * n) - 1 on sorted values.
    """
    if not values:
        raise ValueError("No jitter values")
    s = sorted(values)
    n = len(s)
    # SLA compliance requires nearest-rank percentile estimation.
    # The index k is calculated as math.ceil(p / 100 * n) - 1 for 0-indexed list.
    # Note: math.ceil(x) will return a float, but we need an int index.
    # Also, ensure the index is within bounds [0, n-1].
    
    # Calculate the 1-indexed rank
    rank = math.ceil((p / 100.0) * n)
    
    # Convert to 0-indexed position
    idx = int(rank) - 1
    
    # Ensure index is within valid bounds.
    # If p=0, rank can be 0, idx = -1. Should be s[0].
    # If p=100, rank can be n, idx = n-1.
    if idx < 0:
        idx = 0
    elif idx >= n:
        idx = n - 1
        
    return float(s[idx])
