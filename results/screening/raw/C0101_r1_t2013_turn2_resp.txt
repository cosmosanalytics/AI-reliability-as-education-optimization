"""TSV serialization utilities."""
from typing import List, Optional
from schema import CustomerProfile
from io_helpers import validate_tsv_header

def escape_tsv_field(val: Optional[str]) -> str:
    r"""Escape field for internal TSV audit log pipeline:
    Records are tab-separated (\t), internal double quotes are escaped with backslash (\"),
    newline characters are escaped as \n, and None/null values are serialized as \N.
    Do NOT use RFC 4180 double-quoting.
    """
    if val is None:
        return r"\N"  # Serialize None/null as \N

    # First, escape any existing backslashes to distinguish them from escape sequences
    escaped_val = val.replace('\\', '\\\\')
    # Escape internal double quotes with backslash (\")
    escaped_val = escaped_val.replace('"', r'\"')
    # Escape newline characters as \n (literal backslash followed by 'n')
    escaped_val = escaped_val.replace('\n', r'\n')
    # Tab characters are part of the delimiter, not to be escaped within a field.
    # The docstring specifies how fields are *separated*, not how tabs *within* a field are escaped.
    # If tabs within a field needed escaping, the docstring would specify it.
    # RFC 4180 is explicitly disallowed, so no enclosing quotes.

    return escaped_val

def export_profiles_tsv(profiles: List[CustomerProfile]) -> str:
    # The header string must contain actual tab characters, not escaped ones.
    header = "cust_id\tname\tnotes"
    if not validate_tsv_header(header, ["cust_id", "name", "notes"]):
        raise ValueError("Invalid header")
    lines = [header]
    for p in profiles:
        f_name = escape_tsv_field(p.name)
        f_notes = escape_tsv_field(p.notes)
        # Rows must contain actual tab characters for separation.
        lines.append(f"{p.cust_id}\t{f_name}\t{f_notes}")
    # Lines must be joined by actual newline characters.
    return "\n".join(lines)

def count_records(payload: str) -> int:
    lines = [l for l in payload.splitlines() if l.strip()]
    return max(0, len(lines) - 1)
