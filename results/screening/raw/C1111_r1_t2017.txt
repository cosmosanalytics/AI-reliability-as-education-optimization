"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    clean_pattern = clean_scope(policy.allowed_pattern)

    pattern_segments = clean_pattern.split("::")
    request_segments = clean_req.split("::")

    # Handle empty strings resulting in [''] from split()
    if pattern_segments == ['']:
        pattern_segments = []
    if request_segments == ['']:
        request_segments = []

    p_idx = 0
    r_idx = 0

    while p_idx < len(pattern_segments) and r_idx < len(request_segments):
        pattern_segment = pattern_segments[p_idx]
        request_segment = request_segments[r_idx]

        if pattern_segment == "**":
            # If '**' is the last pattern segment, it matches everything remaining in the request.
            if p_idx == len(pattern_segments) - 1:
                return True
            
            # If '**' is not the last segment, find the next non-wildcard segment in the pattern
            # and try to match it in the remaining request segments.
            next_pattern_segment = pattern_segments[p_idx + 1]
            
            # If the next segment is also a wildcard, skip it for now and continue
            # looking for the first non-wildcard segment or the end of the pattern.
            # This handles cases like `A::**::*::C` or `A::**::**::C`
            temp_p_idx = p_idx + 1
            while temp_p_idx < len(pattern_segments) and (pattern_segments[temp_p_idx] == "*" or pattern_segments[temp_p_idx] == "**"):
                temp_p_idx += 1
            
            if temp_p_idx == len(pattern_segments):
                # If only wildcards are left after '**', then '**' matches everything remaining.
                return True
            
            next_concrete_pattern_segment = pattern_segments[temp_p_idx]

            # Search for the next_concrete_pattern_segment in the remaining request_segments
            found_match_for_next = False
            for lookahead_r_idx in range(r_idx, len(request_segments)):
                if request_segments[lookahead_r_idx] == next_concrete_pattern_segment:
                    # Found a potential match for the segment after '**'.
                    # Advance pattern index to the segment after '**' and request index to the found segment.
                    p_idx = temp_p_idx
                    r_idx = lookahead_r_idx
                    found_match_for_next = True
                    break
            
            if not found_match_for_next:
                return False # Could not find the segment after '**' in the request.
            
            # Continue the loop from the new p_idx and r_idx.
            continue 

        elif pattern_segment == "*":
            # '*' matches exactly one segment.
            p_idx += 1
            r_idx += 1
        elif pattern_segment == request_segment:
            # Exact segment match
            p_idx += 1
            r_idx += 1
        else:
            # Mismatch
            return False

    # After the loop, check if all pattern segments are consumed.
    # If there are remaining pattern segments, they must all be '**' to be considered a match.
    while p_idx < len(pattern_segments) and pattern_segments[p_idx] == "**":
        p_idx += 1

    # A full match occurs if all pattern segments are consumed AND all request segments are consumed.
    return p_idx == len(pattern_segments) and r_idx == len(request_segments)

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
