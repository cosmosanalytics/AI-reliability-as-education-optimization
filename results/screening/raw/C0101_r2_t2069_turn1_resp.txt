"""Gateway request latency percentile calculation."""
import math
from typing import List, Dict, Any
from metric_types import MetricWindow, LatencyRecord
from metric_summary import format_metric_summary

def compute_percentile(window: MetricWindow, percentile: float) -> float:
    """Calculates metric percentile using nearest-rank method: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Do not use linear interpolation.
    """
    vals = window.raw_values()
    if not vals:
        raise ValueError("Empty metric window")
    s = sorted(vals)
    n = len(s)

    # The docstring explicitly states "nearest-rank method: index k = math.ceil(p / 100 * n) - 1".
    # This implies 1-based indexing for k, then converting to 0-based for lists.
    # The current implementation uses linear interpolation.
    # We must implement the specified nearest-rank method.

    if not (0 <= percentile <= 100):
        raise ValueError("Percentile must be between 0 and 100, inclusive.")

    # Handle edge cases for percentile = 0 and percentile = 100
    if percentile == 0:
        return float(s[0]) # 0th percentile is the minimum value
    if percentile == 100:
        return float(s[n - 1]) # 100th percentile is the maximum value

    # For 0 < percentile < 100, apply the nearest-rank formula.
    # The rank k is 1-based.
    k_rank = math.ceil((percentile / 100.0) * n)
    # Convert to 0-based index for list access.
    idx = int(k_rank - 1)
    return float(s[idx])

def summarize_window(window: MetricWindow, percentile: float) -> Dict[str, Any]:
    val = compute_percentile(window, percentile)
    return format_metric_summary(window.sample_count(), val)
