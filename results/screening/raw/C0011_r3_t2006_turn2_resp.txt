"""Environment settings loader and merger."""
from typing import Dict, Any
import copy

DEFAULT_CONN_TIMEOUT = 30

def load_system_env() -> Dict[str, Any]:
    return {"timeout": DEFAULT_CONN_TIMEOUT}

def merge_env_settings(base_settings: Dict[str, Any], override_settings: Dict[str, Any]) -> Dict[str, Any]:
    """Merges override settings into base settings, performing a deep copy
    to prevent shared mutable state issues for nested dictionaries.
    """
    merged = copy.deepcopy(base_settings)
    
    # Recursively merge dictionaries to handle nested overrides without
    # corrupting default templates.
    for key, value in override_settings.items():
        if isinstance(value, dict) and isinstance(merged.get(key), dict):
            merged[key] = merge_env_settings(merged[key], value)
        else:
            merged[key] = value
    
    return merged
