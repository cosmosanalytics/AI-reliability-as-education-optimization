"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses.
    Uses a "nearest rank" method based on the (P/100)*(N+1) formula,
    rounded half-up to determine the 1-based rank. The resulting 0-based index
    is then clamped to the valid range of samples [0, N-1].
    """
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)

    # Calculate the 1-based rank using the (P/100)*(N+1) formula.
    # This formula is common in statistical software for defining percentiles.
    rank_val = (p / 100.0) * (n + 1)

    # Convert the rank to a 0-based index by rounding half-up and subtracting 1.
    # Python's built-in round() uses "round half to even", which can lead to
    # inconsistencies compared to "round half up" often expected for percentiles.
    # int(x + 0.5) provides "round half up" for non-negative x.
    idx_candidate = int(rank_val + 0.5) - 1

    # Clamp the calculated index to ensure it is within the valid range of
    # 0-based indices for the sorted list (0 to n-1). This handles edge cases
    # where p=0 or p=100 (or outside [0,100]) might otherwise produce an
    # out-of-bounds index.
    idx = max(0, min(n - 1, idx_candidate))

    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
