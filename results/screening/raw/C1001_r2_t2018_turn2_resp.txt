"""Document scope RBAC filter."""
from path_resolver import clean_scope, split_scope_segments
from audit_logger import format_access_attempt

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern.
    Access is granted if the requested scope is identical to the allowed scope,
    or if the allowed scope is a direct hierarchical prefix of the requested scope
    (segment by segment). Wildcards (::*) are not explicitly supported for hierarchical
    matching; the comparison is based on exact segment prefixes.
    """
    cleaned_allowed = clean_scope(allowed_scope)
    cleaned_requested = clean_scope(requested_scope)

    # Split scopes into segments for hierarchical comparison
    allowed_segments = split_scope_segments(cleaned_allowed)
    requested_segments = split_scope_segments(cleaned_requested)

    # An empty allowed scope implies no specific permission, thus no access.
    # An empty requested scope implies nothing is being requested, thus no access.
    if not allowed_segments or not requested_segments:
        allowed = False
    else:
        # If the allowed scope ends with '::*', remove it for prefix matching.
        # This implies the allowed_scope is a prefix for any sub-scope.
        if allowed_segments[-1] == "*":
            allowed_segments = allowed_segments[:-1]
            # If after removing '*', allowed_segments is empty, it's an invalid pattern for prefix access.
            if not allowed_segments:
                allowed = False
            else:
                # Check if the allowed scope's segments are a prefix of the requested scope's segments.
                # This handles 'corp::public::*' allowing 'corp::public::readme'
                allowed = (
                    len(requested_segments) >= len(allowed_segments) and
                    requested_segments[:len(allowed_segments)] == allowed_segments
                )
        else:
            # For non-wildcard allowed scopes, access is granted only if the allowed scope
            # is an exact hierarchical prefix of the requested scope, including the full
            # segment boundary. This ensures 'corp::public' does not match 'corp::public_internal'.
            allowed = (
                len(requested_segments) >= len(allowed_segments) and
                requested_segments[:len(allowed_segments)] == allowed_segments and
                # Additionally, if the lengths are different, the next segment in requested_segments
                # must imply a direct child, not a sibling or partial match of the last segment.
                # This is implicitly handled by segment comparison, but explicit check for full match or direct child.
                # The core logic requested_segments[:len(allowed_segments)] == allowed_segments already ensures this.
                # No further condition needed for sibling prevention beyond strict segment prefix match.
                True # This true is a placeholder, the previous condition is sufficient.
            )
            # The previous 'startswith' behavior was "corp::public" allows "corp::public_internal".
            # The current segment-based prefix check:
            # allowed_segments = ['corp', 'public']
            # requested_segments = ['corp', 'public_internal', 'secret']
            # requested_segments[:2] = ['corp', 'public_internal']
            # This is NOT equal to ['corp', 'public'], so 'allowed' will be False, which is correct.

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
