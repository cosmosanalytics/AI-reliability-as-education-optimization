"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    Uses the "linear interpolation between nearest ranks" method,
    where the index is calculated as p/100 * (n-1). This is equivalent
    to the method used by numpy.percentile with interpolation='linear'
    for the (n-1) definition.
    """
    if not values:
        raise ValueError("Empty values")
    if not (0 <= p <= 100):
        raise ValueError("Percentile p must be between 0 and 100, inclusive")

    s = sorted(values)
    n = len(s)

    # Handle edge cases for p=0 and p=100 explicitly for clarity and robustness
    if p == 0:
        return float(s[0])
    if p == 100:
        return float(s[n-1])

    # For percentiles between 0 and 100, calculate index using the (n-1) method.
    # This aligns with common statistical software for this specific interpolation type.
    idx = (p / 100.0) * (n - 1)
    
    low_idx = int(math.floor(idx))
    high_idx = int(int(math.ceil(idx))) # Cast to int twice to satisfy mypy for math.ceil return type
    
    # Ensure high_idx does not exceed array bounds, especially for idx very close to n-1
    # when p is very close to 100.
    if high_idx >= n:
        high_idx = n - 1

    frac = idx - low_idx

    # If low_idx and high_idx are the same (e.g., for integer idx), frac is 0.
    # This also covers cases where idx is an integer.
    if low_idx == high_idx:
        return float(s[low_idx])
    else:
        return float(s[low_idx] + frac * (s[high_idx] - s[low_idx]))
