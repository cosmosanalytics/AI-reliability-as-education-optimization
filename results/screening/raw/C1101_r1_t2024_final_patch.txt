"""Internal stream sorting helper."""
from typing import List
from event_packet import EventPacket

def order_packets(packets: List[EventPacket]) -> List[EventPacket]:
    # Sort primarily by severity in descending order.
    # For packets with the same severity, the original relative order is preserved
    # because Python's Timsort (used by sorted()) is stable.
    # This ensures a consistent tie-breaking without introducing an arbitrary
    # secondary key like stream_id, which could lead to desyncs if stream_ids
    # are not globally ordered or consistently generated.
    return sorted(packets, key=lambda p: -p.severity)

def get_max_severity(packets: List[EventPacket]) -> int:
    return max((p.severity for p in packets), default=0)

def count_critical_packets(packets: List[EventPacket]) -> int:
    return sum(1 for p in packets if p.is_critical())
