"""Cache Hit Latency Percentile Monitor percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_cache_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_cache_percentile", samples)
    s = validate_metric_samples(win.values)
    # The incident report indicates a discrepancy against a "nearest-rank house standard".
    # The standard "nearest-rank" method often uses an index calculation of
    # (p / 100.0) * N, where N is the number of samples, and then rounds up.
    # The current implementation uses (p / 100.0) * (N - 1) with linear interpolation.
    # To match the "nearest-rank" standard, we adjust the index calculation.
    # A common "nearest rank" method is to find the smallest index i such that the
    # i-th value is greater than or equal to the desired percentile.
    # This can be approximated by ceil(P/100 * N) - 1 for 0-indexed arrays.
    # Another interpretation of "nearest-rank" is simply using the rank directly
    # without interpolation, often calculated as (p / 100.0) * N.
    # The failing test `assert calculate_cache_percentile(d1, 50.0) == 40.0`
    # where `d1` is likely `[10.0, 20.0, 30.0, 40.0, 50.0, 60.0]`, suggests that
    # for N=6, P=50, the 50th percentile should be the 4th element (40.0).
    # This corresponds to an index of 3 (0-indexed).
    # The formula `idx = (p / 100.0) * len(s)` yields:
    # (50.0 / 100.0) * 6 = 3.0.
    # If we take `s[int(idx)]` when `idx` is an integer, this gives `s[3]` which is 40.0.
    # If `idx` is not an integer, we use the `math.ceil(idx)` approach for "nearest-rank".
    # This implies that for the "nearest-rank house standard", linear interpolation
    # for non-integer indices is not desired, but rather selecting the element
    # at the calculated rank (rounded up if necessary).

    # The current code returns 45.0 for the failing test, which implies linear interpolation
    # between 40.0 and 50.0.
    # This suggests that the "nearest-rank house standard" might be a simpler selection
    # based on rank, rather than interpolation.

    # Let's align with the definition of "nearest-rank" that picks the element
    # corresponding to the rank directly.
    # The standard formula for the rank (1-indexed) is R = ceil(P/100 * N).
    # For a 0-indexed list, this means index `int(math.ceil(p / 100.0 * len(s))) - 1`.
    # Let's test this with the expected failure scenario:
    # d1 = [10.0, 20.0, 30.0, 40.0, 50.0, 60.0], p = 50.0
    # N = 6
    # rank_1_indexed = ceil(0.5 * 6) = ceil(3.0) = 3
    # index_0_indexed = 3 - 1 = 2
    # s[2] = 30.0. This is not 40.0.

    # Another common nearest-rank method is method R-7 (Excel's PERCENTILE.EXC):
    # (p / 100.0) * (N + 1)
    # idx = (50.0 / 100.0) * (6 + 1) = 0.5 * 7 = 3.5
    # Interpolation would yield 30.0 + 0.5 * (40.0 - 30.0) = 35.0. Still not 40.0.

    # The problem description and traceback strongly imply that the expected value for
    # the 50th percentile of `[10, 20, 30, 40, 50, 60]` is `40.0`.
    # This specific behavior implies a definition where the index is `p/100 * N`,
    # and if it's an integer, that element is picked. If it's not an integer,
    # the next element (rounding up the index) is chosen.
    # For `p=50, N=6`, `0.5 * 6 = 3.0`. The 0-indexed element `s[3]` is `40.0`.
    # This is equivalent to the "nearest rank" method using `N*p/100` and picking
    # `s[k]` where `k = ceil(N*p/100) - 1` (for 0-indexed) if `N*p/100` is not an integer,
    # and `s[N*p/100 - 1]` if it is.

    # A simpler way to achieve the desired behaviour (40.0 for 50th percentile of 6 elements)
    # is to use the formula `idx = (p / 100.0) * len(s)` and then directly pick `s[int(idx)]`
    # if idx is an integer, or `s[math.ceil(idx) - 1]` for other nearest-rank policies.
    # The most direct interpretation of "nearest-rank" that results in 40.0 from
    # [10, 20, 30, 40, 50, 60] at 50% is to take the element at index `k` where `k = (p/100 * N) - 1`.
    # If `k` is not an integer, then it typically rounds up to the next integer index.

    # Let's use the definition: `index = P/100 * N`.
    # If `index` is an integer, the percentile is `s[index - 1]`.
    # If `index` is not an integer, the percentile is `s[ceil(index) - 1]`.
    # This can be simplified to `s[math.ceil(p / 100.0 * len(s)) - 1]`, but this would give 30.0.

    # The specific test failure `assert 45.0 == 40.0` points to a change from interpolation
    # (which yielded 45.0) to a direct selection (which should yield 40.0).
    # The current formula `idx = (p / 100.0) * (len(s) - 1)` for N=6, p=50:
    # idx = 0.5 * 5 = 2.5
    # low=2, high=3
    # s[2] + (2.5 - 2) * (s[3] - s[2]) = 30.0 + 0.5 * (40.0 - 30.0) = 30.0 + 5.0 = 35.0.
    # My previous manual trace yielded 35.0, not 45.0. Let me re-evaluate the traceback.
    # The traceback indicates `assert 45.0 == 40.0`. This means the *actual* result was 45.0.
    # How could `calculate_cache_percentile` return 45.0?
    # This happens if `s[low]` is 40.0 and `s[high]` is 50.0, and `idx - low` is 0.5.
    # So `40.0 + 0.5 * (50.0 - 40.0) = 40.0 + 0.5 * 10.0 = 45.0`.
    # This implies `low = 3`, `high = 4`, and `idx = 3.5`.
    # For `idx = 3.5`, `(p / 100.0) * (len(s) - 1) = 3.5`.
    # If `p = 50.0`, then `0.5 * (len(s) - 1) = 3.5`.
    # `len(s) - 1 = 7`.
    # `len(s) = 8`.
    # So `d1` must be a list of 8 elements, where `s[3] = 40.0` and `s[4] = 50.0`.
    # Example `d1`: `[10.0, 20.0, 30.0, 40.0, 50.0, 60.0, 70.0, 80.0]`.
    # For this `d1` and `p=50.0`, the current code returns 45.0.
    # The *expected* value is 40.0.
    # This means for `N=8`, `P=50`, the 50th percentile should be `40.0`.
    # If `N=8`, `P=50`, `idx = (50.0 / 100.0) * N = 0.5 * 8 = 4.0`.
    # The 0-indexed element `s[4]` is `50.0`. This is not `40.0`.
    # The 0-indexed element `s[3]` is `40.0`. This is `s[int(idx) - 1]`.

    # This behavior (returning `s[int(idx) - 1]` when `idx` is an integer)
    # is consistent with the "exclusive percentile" method (e.g., Excel's PERCENTILE.EXC).
    # The definition is `(p / 100.0) * (N + 1)`.
    # For `N=8, P=50`: `idx = 0.5 * (8 + 1) = 0.5 * 9 = 4.5`.
    # `low = 4`, `high = 5`. `s[4]` would be 50.0, `s[5]` would be 60.0.
    # `50.0 + 0.5 * (60.0 - 50.0) = 55.0`. Still not 40.0.

    # The most common "nearest-rank" definition (e.g., NIST) is to calculate
    # `k = (N * P / 100)`. If `k` is an integer, use `s[k-1]`. If `k` is not an integer,
    # use `s[floor(k)]`.
    # For `N=8, P=50`: `k = (8 * 50 / 100) = 4.0`.
    # Since `k` is an integer, use `s[k-1] = s[3]`, which is `40.0`. This matches the expected value!

    # So, the fix is to implement the "nearest-rank" method according to NIST standard:
    # 1. Calculate `k = N * P / 100`.
    # 2. If `k` is an integer, the percentile is `s[k-1]` (using 0-indexed list).
    # 3. If `k` is not an integer, the percentile is `s[floor(k)]` (using 0-indexed list).
    # We must ensure `k-1` or `floor(k)` is within bounds.
    # `p` is between 0 and 100.
    # If `p=0`, `k=0`, `s[-1]` is not valid. The docstring doesn't mention `p=0`.
    # If `p=100`, `k=N`. `s[N-1]` is valid.
    # If `p=0`, `k=0`. The index should be 0.
    # If `p=0`, the definition usually implies the minimum value, `s[0]`.
    # If `p=100`, the definition usually implies the maximum value, `s[N-1]`.

    n = len(s)
    if n == 0: # Already handled by validate_metric_samples, but good to be explicit
        raise ValueError("Samples list cannot be empty")

    if p < 0.0 or p > 100.0:
        raise ValueError("Percentile 'p' must be between 0 and 100 inclusive")

    # Handle edge cases for p=0 and p=100 explicitly as some definitions vary
    if p == 0.0:
        return s[0]
    if p == 100.0:
        return s[n - 1]

    # Implement NIST method for nearest-rank percentile
    # k = N * P / 100
    k_float = (n * p) / 100.0

    if k_float == int(k_float):
        # k is an integer. Percentile is the value at index k-1 (0-indexed).
        # Ensure index is within bounds for p=0 (already handled) and p > 0.
        # For k=1, index is 0. For k=N, index is N-1.
        return s[int(k_float) - 1]
    else:
        # k is not an integer. Percentile is the value at index floor(k) (0-indexed).
        # Ensure index is within bounds. floor(k) will be >= 0.
        return s[math.floor(k_float)]


def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
