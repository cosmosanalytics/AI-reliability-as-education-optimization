"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values using the nearest rank method with linear interpolation.

    This method uses the C = 1 (nearest rank) method for index calculation,
    which means the index is (N*P)/100. If the index is not an integer,
    linear interpolation is used between the two nearest ranks.

        raise ValueError("Empty values")

    # Example: for [10, 20, 30, 40, 50], 90th percentile:

    # `return float(s[0]) = 10.0`.
    # Original code would return `10 + 0.5 * (20-10) = 15.0`.
    # The fix makes it 10.0. This seems to be the intended conservative behavior.

    idx = (p / 100.0) * (n - 1)
    low = int(math.floor(idx))
    high = int(math.ceil(idx))
    frac = idx - low

    if frac > 0:
        # For SLA reporting, if the percentile index is not an integer,
        # return the value at the lower rank (floor) without interpolation.
        # This provides a more conservative estimate as per SLA requirements.
        return float(s[low])
    else:
        # If the percentile index is an exact rank (frac == 0),
        # or if there's no higher element to interpolate with (e.g., p=100, low=n-1),
        # the standard interpolation formula correctly yields s[low].
        return float(s[low] + frac * (s[high] - s[low]))
