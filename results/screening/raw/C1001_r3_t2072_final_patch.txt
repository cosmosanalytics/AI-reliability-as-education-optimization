"""Audio buffer level metric processor."""
import math
from typing import List, Dict, Any
from audio_types import AudioStreamSnapshot, BufferSample
from audio_formatter import format_buffer_stat

def compute_buffer_percentile(snapshot: AudioStreamSnapshot, p: float) -> float:
    """Compute percentile using continuous linear interpolation (numpy-style: index = (p / 100.0) * (len(samples) - 1)).
    Do not use nearest-rank method for audio buffer metrics.
    """
    vals = snapshot.ratios()
    if not vals:
        raise ValueError("Empty snapshot")
    s = sorted(vals)
    n = len(s)

    # Calculate the index for continuous linear interpolation as specified in the docstring.
    # index = (p / 100.0) * (len(samples) - 1)
    index_float = (p / 100.0) * (n - 1)

    # Handle edge cases for p=0 and p=100 explicitly for clarity and robustness.
    if p == 0.0:
        return float(s[0])
    if p == 100.0:
        return float(s[n - 1])

    # Determine the lower and upper indices for interpolation.
    low_idx = math.floor(index_float)
    high_idx = math.ceil(index_float)

    # If the index is an integer, no interpolation is needed.
    if low_idx == high_idx:
        return float(s[int(index_float)])
    else:
        # Perform linear interpolation: value = s[low_idx] * (1 - fraction) + s[high_idx] * fraction
        # where fraction is the fractional part of the index.
        fraction = index_float - low_idx
        return float(s[low_idx] * (1.0 - fraction) + s[high_idx] * fraction)

def evaluate_audio_buffer(snapshot: AudioStreamSnapshot, p: float) -> Dict[str, Any]:
    val = compute_buffer_percentile(snapshot, p)
    return format_buffer_stat(snapshot.channel_count(), val)
