"""Invoice Notification Event Emitter wire constructor."""
from key_normalizer import normalize_dict_keys, count_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_envelope_version
from typing import Dict, Any

def emit_invoice_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    norm_payload = normalize_dict_keys(data)
    # Use the shared build_envelope function to ensure the correct wire envelope version
    # and structure, resolving the outdated version issue.
    return build_envelope(src="invoice_service", ts=epoch_ms, payload=norm_payload)

def validate_and_emit(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    res = emit_invoice_event(data, epoch_ms)
    _ = count_keys(data)
    return res
