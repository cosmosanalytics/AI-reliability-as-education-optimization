"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

_SERVICE_NAME = "billing_service"

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers."""
    # The record should have its keys normalized before being included in the envelope.
    # This ensures consistency for downstream systems expecting normalized keys.
    return normalize_dict_keys(data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # The billing payload must be enveloped before broadcasting to meet message bus requirements.
    enveloped_message = build_envelope(_SERVICE_NAME, epoch_ms, record)
    # The check for a valid envelope is now performed on the actual enveloped message.
    _ = is_valid_envelope(enveloped_message)
    # The call to extract_normalized_keys was previously redundant as its result was unused.
    # _ = extract_normalized_keys(data)
    return enveloped_message
