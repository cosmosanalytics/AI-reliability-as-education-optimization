"""Low-level tax rate computation engine."""
from typing import Dict, Any

DEFAULT_TAX_JURISDICTIONS = {
    "standard": 0.20,
    "reduced": 0.10,
    "exempt": 0.0,
}

def get_rate_for_jurisdiction(code: str) -> float:
    return DEFAULT_TAX_JURISDICTIONS.get(code, 0.20)

def calculate_vat(net_cents: int, vat_rate: float) -> int:
    """Compute VAT in integer cents for given net amount using half-up rounding.

    The tax calculation uses half-up rounding (ties broken towards positive infinity)
    as specified by the order pricing requirements.
    """
    calculated_amount = net_cents * vat_rate
    # For non-negative numbers, adding 0.5 and then truncating (int()) achieves
    # half-up rounding (e.g., 13.5 becomes 14, 13.4 becomes 13).
    return int(calculated_amount + 0.5)
