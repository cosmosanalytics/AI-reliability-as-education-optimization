"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    Ensures that the total allocated discount equals total_discount_cents,
    distributing any remainder cents one by one to lines with the largest
    proportional share.

    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    if total_discount_cents < 0:
        raise ValueError("Total discount cents must be non-negative")
    if not line_totals:
        return []

    total_line_value = sum(line_totals)
    if total_line_value == 0:
        # If total is 0, no proportional allocation is possible.
        # The existing behavior of returning all zeros is maintained.
        return [0] * len(line_totals)

    # Calculate initial proportional allocations and their fractional remainders
    # Store (initial_allocation, fractional_remainder, original_index)
    allocations_with_remainder = []
    for i, cents in enumerate(line_totals):
        # Use float division to preserve precision for remainder calculation
        proportional_share = (cents * total_discount_cents) / total_line_value
        initial_allocation = int(proportional_share)
        remainder = proportional_share - initial_allocation
        allocations_with_remainder.append((initial_allocation, remainder, i))

    # Sort by remainder (descending) to distribute extra pennies to those with largest fractions
    allocations_with_remainder.sort(key=lambda x: x[1], reverse=True)

    # Sum initial allocations to find how many cents are still left to distribute
    current_sum_of_initial_allocations = sum(item[0] for item in allocations_with_remainder)
    remaining_cents_to_distribute = total_discount_cents - current_sum_of_initial_allocations

    # Create a list for final allocations, preserving original order
    final_allocations = [0] * len(line_totals)
    for initial_allocation, _, original_index in allocations_with_remainder:
        final_allocations[original_index] = initial_allocation

    # Distribute the remaining cents one by one to the lines that had the largest remainders
    for i in range(remaining_cents_to_distribute):
        # The first 'remaining_cents_to_distribute' items in the sorted list get an extra penny
        # Use modulo to cycle through lines if remaining_cents_to_distribute is larger than num lines
        # (though this typically implies significant rounding issues or very small line_totals)
        _, _, original_index = allocations_with_remainder[i % len(allocations_with_remainder)]
        final_allocations[original_index] += 1

    return final_allocations


def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
