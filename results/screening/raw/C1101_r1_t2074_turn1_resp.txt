"""Low-level jitter percentile calculation engine."""
import math
from typing import List

def evaluate_jitter_percentile(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using linear interpolation."""
    if not values:
        raise ValueError("No jitter values")
    s = sorted(values)
    n = len(s)
    # This is the linear interpolation method, which is distinct from nearest-rank.
    # It calculates the index as p/100 * (n-1) and interpolates if it's not an integer.
    idx = (p / 100.0) * (n - 1)
    low = int(math.floor(idx))
    high = int(math.ceil(idx))
    frac = idx - low
    if low == high: # Handles cases where idx is an integer or n=1
        return float(s[low])
    return float(s[low] + frac * (s[high] - s[low]))

def evaluate_jitter_percentile_nearest_rank(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using the nearest-rank method.
    Index k = math.ceil(p / 100 * n) - 1 on sorted values.
    """
    if not values:
        raise ValueError("No jitter values")
    s = sorted(values)
    n = len(s)
    # The nearest-rank method calculates the index and picks the value at that index.
    # The problem statement specifies: k = math.ceil(p / 100 * n) - 1
    # Note: math.ceil(x) for x=0.0 is 0.
    # For p=0, k = math.ceil(0) - 1 = -1. This means p must be > 0.
    # For p=100, k = math.ceil(n) - 1 = n - 1. This is the last element.
    # For p/100 * n < 1 (i.e., p < 100/n), ceil(p/100*n) will be 1, so k = 0.
    # This means any percentile between 0 and 100/n (exclusive) will return the first element.
    # Example: n=5. p=10. 10/100 * 5 = 0.5. ceil(0.5)-1 = 0. s[0]
    # Example: n=5. p=20. 20/100 * 5 = 1.0. ceil(1.0)-1 = 0. s[0]
    # Example: n=5. p=20.000000000000001. 0.000000000000001 * 5 = 1.00000000000000005. ceil - 1 = 0.
    # This means that for p <= 100/n, the k will be 0.
    # To handle 0-based indexing correctly, the formula is (p/100 * n) if it's an integer, otherwise ceil(p/100*n).
    # The specified formula `k = math.ceil(p / 100 * n) - 1` directly provides the 0-based index.
    
    # Ensure p is within a valid range for the formula to produce meaningful indices
    if p <= 0:
        return s[0] # Smallest value for 0th or negative percentile
    if p >= 100:
        return s[n-1] # Largest value for 100th or greater percentile

    idx = math.ceil((p / 100.0) * n) - 1
    # Ensure index is within bounds [0, n-1]
    k = max(0, min(n - 1, int(idx)))
    return float(s[k])

"""Network jitter SLA evaluation service."""
from typing import List, Dict, Any
from jitter_types import JitterStream, JitterSample
import jitter_core

def check_jitter_sla(stream: JitterStream, p: float, max_jitter: float) -> bool:
    """SLA compliance requires nearest-rank percentile estimation: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Returns True if compliance limit is respected.
    """
    vals = stream.raw_jitters()
    # Use the specific nearest-rank percentile calculation required for SLA compliance.
    percentile_val = jitter_core.evaluate_jitter_percentile_nearest_rank(vals, p)
    return percentile_val <= max_jitter

def get_stream_jitter_metrics(stream: JitterStream, p: float) -> Dict[str, Any]:
    vals = stream.raw_jitters()
    # This function uses the standard linear interpolation percentile calculation.
    q = jitter_core.evaluate_jitter_percentile(vals, p)
    return {"count": stream.sample_count(), "jitter_p": q}
