"""Document scope RBAC filter."""
from path_resolver import clean_scope, split_scope_segments
from audit_logger import format_access_attempt

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern.

    A requested scope is permitted if it is identical to the allowed scope,
    or if it is a direct descendant of the allowed scope in the hierarchy.
    The `*` wildcard in allowed_scope is treated as implicit, meaning
    "this scope and all its descendants".
    """
    cleaned_allowed = clean_scope(allowed_scope)
    cleaned_requested = clean_scope(requested_scope)

    # Handle the implicit wildcard: remove trailing '::*' if present.
    # This makes 'corp::public' and 'corp::public::*' equivalent for policy checking.
    if cleaned_allowed.endswith('::*'):
        cleaned_allowed = cleaned_allowed[:-3] # Remove '::*'

    allowed_segments = split_scope_segments(cleaned_allowed)
    requested_segments = split_scope_segments(cleaned_requested)

    # An empty allowed_scope or requested_scope (after cleaning and splitting)
    # should generally not grant access unless explicitly defined.
    # Here, we assume a non-empty scope is required for access.
    if not allowed_segments or not requested_segments:
        allowed = False
    elif len(requested_segments) < len(allowed_segments):
        # Requested scope is shorter than allowed scope, so it cannot be a descendant.
        allowed = False
    else:
        # Check if the allowed_scope segments are a prefix of the requested_scope segments.
        allowed = requested_segments[:len(allowed_segments)] == allowed_segments

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
