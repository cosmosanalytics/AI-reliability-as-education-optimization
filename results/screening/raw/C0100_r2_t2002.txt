"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total = sum(line_totals)
    if total == 0:
        return [0] * len(line_totals)

    allocations = []
    total_allocated_so_far = 0
    # Calculate initial allocations, rounding down.
    # Store fractional remainders for later distribution of leftover pennies.
    # We use a list of tuples (fractional_remainder, original_index) to sort.
    remainders_and_indices = []

    for i, cents in enumerate(line_totals):
        # Calculate the exact proportional share using float for precision
        exact_share = (cents * total_discount_cents) / total
        # Round down to get the initial integer allocation
        initial_allocation = int(exact_share)
        allocations.append(initial_allocation)
        total_allocated_so_far += initial_allocation
        # Calculate the fractional remainder
        remainder = exact_share - initial_allocation
        remainders_and_indices.append((remainder, i))

    # Calculate how many pennies are left to distribute
    remaining_pennies = total_discount_cents - total_allocated_so_far

    # Distribute the remaining pennies one by one to lines with the largest fractional remainders
    # Sort in descending order of remainder. If remainders are equal, original index can serve as tie-breaker.
    remainders_and_indices.sort(key=lambda x: x[0], reverse=True)

    for i in range(remaining_pennies):
        # Add one penny to the line with the largest remainder
        # We use the original index to modify the correct allocation
        original_index = remainders_and_indices[i][1]
        allocations[original_index] += 1

    return allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
