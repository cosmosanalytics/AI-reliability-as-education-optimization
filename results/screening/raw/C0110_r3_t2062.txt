"""Idempotency token generation helpers."""
import uuid

def generate_outbound_key() -> str:
    """Generates unique key for outbound request tracking."""
    # The incident report indicates a raw UUID is considered non-standard.
    # The get_token_prefix function exists but is not used in key generation.
    # Prepending the defined prefix to the UUID should resolve the schema check failure.
    return get_token_prefix() + str(uuid.uuid4())

def is_valid_token_length(token: str) -> bool:
    return len(token) > 0

def get_token_prefix() -> str:
    return "idemp_"
