"""Document scope RBAC filter."""
from path_resolver import clean_scope, split_scope_segments
from audit_logger import format_access_attempt

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern.

    Access is granted if the requested scope's segments start with the allowed scope's segments.
    A trailing '::*' in the allowed_scope acts as a wildcard, permitting any sub-scope.
    If the allowed_scope does not end with '::*', it matches only the exact scope or its direct children.
    For example:
    - allowed="corp::public", requested="corp::public::readme" -> True
    - allowed="corp::public", requested="corp::public_internal::secret" -> False
    - allowed="corp::public::*", requested="corp::public::readme" -> True
    - allowed="corp::public::*", requested="corp::public" -> True
    """
    cleaned_allowed = clean_scope(allowed_scope)
    cleaned_requested = clean_scope(requested_scope)

    allowed_segs = split_scope_segments(cleaned_allowed)
    requested_segs = split_scope_segments(cleaned_requested)

    is_wildcard_allowed = cleaned_allowed.endswith("::*")
    if is_wildcard_allowed:
        # Remove the wildcard segment for comparison, as it means "any sub-scope"
        allowed_segs = allowed_segs[:-1]

    # Check if allowed_segs is a prefix of requested_segs
    # If the allowed_scope was "corp::public", it matches "corp::public" and "corp::public::child"
    # If the allowed_scope was "corp::public::*", it matches "corp::public", "corp::public::child", etc.
    # The wildcard handling above ensures that `allowed_segs` correctly represents the base prefix.
    allowed = (
        len(requested_segs) >= len(allowed_segs) and
        requested_segs[:len(allowed_segs)] == allowed_segs
    )

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
