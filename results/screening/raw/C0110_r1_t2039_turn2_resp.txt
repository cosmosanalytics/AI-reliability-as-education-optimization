"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Assuming a default service name for telemetry.
# In a real-world scenario, this might come from configuration or a global context.
DEFAULT_SERVICE_NAME = "unknown_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # Normalize keys in the payload before building the envelope
    normalized_payload = normalize_dict_keys(data)
    # Build the wire envelope around the normalized payload
    return build_envelope(DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The original code ignored the return value of is_valid_envelope,
    # but it's good practice to ensure the packet is valid before publishing.
    # For this patch, we maintain the original behavior of not explicitly checking
    # after preparation, assuming prepare_metric_packet correctly builds it.
    _ = is_valid_envelope(pkt) # This check is now expected to pass.
    _ = extract_normalized_keys(data) # This still operates on the original data, which might be an oversight but not related to the envelope issue.
    return pkt
