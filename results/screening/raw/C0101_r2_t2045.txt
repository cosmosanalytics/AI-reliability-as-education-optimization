"""Audit event serialization service."""
from datetime import datetime, timezone
from models import AuditRecord
from formatter import sanitize_action_name, build_audit_tag

def format_audit_event(event_id: str, action: str, dt: datetime) -> AuditRecord:
    """Formats an audit event record.
    Timestamps must be formatted strictly in ISO 8601 UTC with explicit millisecond
    precision and uppercase 'Z' suffix: YYYY-MM-DDTHH:MM:SS.sssZ (e.g. 2026-10-25T12:00:00.000Z).
    """
    clean_action = sanitize_action_name(action)
    tag = build_audit_tag(clean_action)

    # Ensure the datetime object is in UTC for correct 'Z' suffix.
    # If the datetime object is naive, assume it's UTC.
    # If it has a timezone, convert it to UTC.
    if dt.tzinfo is None:
        dt_utc = dt.replace(tzinfo=timezone.utc)
    else:
        dt_utc = dt.astimezone(timezone.utc)

    # Format timestamp to YYYY-MM-DDTHH:MM:SS.sssZ
    # Use isoformat with 'milliseconds' timespec and replace '+00:00' with 'Z'
    ts = dt_utc.isoformat(timespec='milliseconds').replace('+00:00', 'Z')

    record = AuditRecord(
        event_id=event_id,
        action=clean_action,
        timestamp_str=ts,
        metadata={"tag": tag}
    )
    if not record.is_valid():
        raise ValueError("Invalid record generated")
    return record
