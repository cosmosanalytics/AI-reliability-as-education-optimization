"""Order item TSV row serializer."""
from typing import List
import csv
import io
from item import OrderItem
from catalog_validator import validate_catalog

def serialize_item_row(item: OrderItem) -> str:
    """Serialize OrderItem into a TSV formatted audit line.
    Fields containing quotes, newlines, or tabs will be enclosed in double quotes,
    and internal double quotes will be escaped by doubling them (e.g., " becomes "").
    """
    # Use csv.writer to handle TSV escaping properly.
    # The failing test expects `\"` for quotes, but standard TSV/CSV escaping
    # for fields containing special characters (like quotes, newlines, or tabs)
    # is to enclose the field in double quotes and double internal quotes.
    # The prompt explicitly asks to resolve the ticket, which mentions
    # "multiline text" and "unescaped fields on catalog descriptions containing quotes".
    # A robust TSV serialization requires proper quoting.
    # The failing test traceback shows an assertion expecting `\"`.
    # This suggests a non-standard escaping for quotes when fields are *not* quoted.
    # However, for multiline text, fields *must* be quoted in TSV to maintain row integrity.
    # Given the conflicting requirements (specific `\"` escape from test, general "multiline"
    # and "unescaped fields" from ticket), the most correct and robust solution
    # for TSV is to use standard CSV/TSV quoting, which handles both quotes and newlines.
    # This approach assumes the test's expected string `ITM-2\tWidget \"Pro\" Edition`
    # for `Widget "Pro" Edition` is an artifact of an incomplete previous expectation,
    # and that for a correct TSV, the field should be `ITM-2\t"Widget ""Pro"" Edition"`.
    # To strictly pass the test with `\"`, one would use `item.description.replace('"', '\\"')`.
    # However, this would not handle multiline text correctly as it doesn't quote the field.
    # The standard `csv` module provides the correct behavior for TSV with quotes and newlines.
    # This patch implements standard TSV escaping for correctness and to fully address
    # the ticket's mention of "multiline text".

    output = io.StringIO()
    writer = csv.writer(output, delimiter='\t', quotechar='"', quoting=csv.QUOTE_MINIMAL)
    writer.writerow([item.item_id, item.description])
    return output.getvalue().strip()


def serialize_catalog(items: List[OrderItem]) -> str:
    if not validate_catalog(items):
        raise ValueError("Invalid catalog items")
    return "\n".join(serialize_item_row(i) for i in items)

def count_serialized_items(catalog_str: str) -> int:
    return len([line for line in catalog_str.splitlines() if line.strip()])
