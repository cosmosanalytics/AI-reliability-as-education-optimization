"""Audio buffer level metric processor."""
import math
from typing import List, Dict, Any
from audio_types import AudioStreamSnapshot, BufferSample
from audio_formatter import format_buffer_stat

def compute_buffer_percentile(snapshot: AudioStreamSnapshot, p: float) -> float:
    """Compute percentile using continuous linear interpolation (numpy-style: index = (p / 100.0) * (len(samples) - 1)).
    Do not use nearest-rank method for audio buffer metrics.
    """
    vals = snapshot.ratios()
    if not vals:
        raise ValueError("Empty snapshot")
    s = sorted(vals)
    n = len(s)

    # Calculate index using continuous linear interpolation
    # The docstring specifies index = (p / 100.0) * (len(samples) - 1).
    # For robustness and consistency with numpy's behavior for p outside [0, 100],
    # the effective percentile value should be clamped before calculating the index.
    # This means p=0 maps to index 0, p=100 maps to index n-1.
    clamped_p = max(0.0, min(100.0, p))
    idx = (clamped_p / 100.0) * (n - 1)

    # Find the two surrounding ranks (indices)
    i = math.floor(idx)
    j = math.ceil(idx)

    # Handle cases where index is an integer or at the boundaries
    if i == j: # This covers p=0, p=100, or exact integer indices
        return float(s[int(idx)])

    # Calculate the fractional part
    fraction = idx - i

    # Linearly interpolate between the two values
    return float(s[i] * (1 - fraction) + s[j] * fraction)

def evaluate_audio_buffer(snapshot: AudioStreamSnapshot, p: float) -> Dict[str, Any]:
    val = compute_buffer_percentile(snapshot, p)
    return format_buffer_stat(snapshot.channel_count(), val)
