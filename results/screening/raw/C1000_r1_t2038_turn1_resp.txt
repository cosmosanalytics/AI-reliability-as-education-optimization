"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

SERVICE_NAME = "billing_emitter"  # Define a constant for the service name

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers.

    The payload is normalized to snake_case keys and then enveloped.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(SERVICE_NAME, epoch_ms, normalized_payload)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # These checks were previously discarding results; they are now implicitly
    # satisfied by emit_billing_record producing a valid, enveloped record.
    _ = is_valid_envelope(record)
    _ = extract_normalized_keys(data) # This still operates on original data, as per existing test
    return record
