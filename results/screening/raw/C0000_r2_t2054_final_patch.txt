"""Low-level dictionary and list hierarchy merger."""
from typing import Dict, Any, List

def merge_spec_values(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """
    Recursively merges dictionary values.
    - If both base and override values for a key are dictionaries, they are merged recursively.
    - If both base and override values for a key are lists, the override list's unique elements
      are appended to the base list, preserving the original order of base elements.
    - For all other cases (different types, or non-dict/non-list types), the override value replaces the base value.
    """
    result = dict(base)
    for k, v in override.items():
        if k in result and isinstance(result[k], dict) and isinstance(v, dict):
            result[k] = merge_spec_values(result[k], v)
        elif k in result and isinstance(result[k], list) and isinstance(v, list):
            # Extend the base list with new items from the override list, avoiding duplicates.
            # Preserve the order of existing items and append new unique items.
            current_list = result[k]
            for item in v:
                if item not in current_list:
                    current_list.append(item)
            # No need to reassign current_list to result[k] as current_list is a reference
            # to result[k] and modifications are in-place.
        else:
            result[k] = v
    return result

def is_spec_empty(spec: Dict[str, Any]) -> bool:
    return len(spec) == 0
