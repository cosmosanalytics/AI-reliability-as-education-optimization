"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    total = sum(line_totals)
    if total == 0:
        return [0] * len(line_totals)

    # Calculate base allocations using float division to maintain precision
    # and then sum to find the total allocated so far.
    # The `round()` function ensures proper rounding for each individual line
    # before distributing the remainder.
    # Note: `round()` in Python 3.x rounds to the nearest even number for .5
    # (e.g., round(2.5) is 2, round(3.5) is 4). For financial calculations,
    # rounding half up is often desired, but since we are dealing with cents
    # and then distributing a remainder, the precise rounding of individual
    # lines matters less than ensuring the total is correct.
    # We will distribute the remainder to ensure the sum matches.
    
    # Calculate initial allocations using float division
    # and then convert to cents, rounding to nearest integer.
    # This might not sum up to total_discount_cents due to rounding.
    allocs_float = [cents * total_discount_cents / total for cents in line_totals]
    allocs = [round(a) for a in allocs_float]

    # Calculate the remainder and distribute it.
    # The remainder is the difference between the target total_discount_cents
    # and the sum of the rounded allocations.
    remainder = total_discount_cents - sum(allocs)

    # Distribute the remainder one cent at a time to the lines
    # that lost the most from rounding down (or gained least from rounding up).
    # This is done by sorting based on the fractional part of the float allocation.
    
    # Create a list of (original_index, fractional_part, current_allocation)
    # The fractional_part is `allocs_float[i] - floor(allocs_float[i])` for rounding down
    # or `ceil(allocs_float[i]) - allocs_float[i]` for rounding up.
    # To distribute the remainder, we want to prioritize lines that were "cut short"
    # the most. This means lines where `allocs_float[i]` was just above `X.5` but rounded
    # down, or just below `X.5` and rounded down, etc.
    # A simpler approach for distributing a small remainder is to iterate and add 1.
    
    # A robust way to distribute the remainder is to add 1 cent to the lines
    # that currently have the largest positive difference between their float allocation
    # and their integer allocation, if the remainder is positive.
    # Or subtract 1 cent from lines that have the largest negative difference, if remainder is negative.

    # Let's use a simpler, common approach for small remainders:
    # distribute to the first `remainder` lines.
    # This assumes `remainder` is small and distributing it arbitrarily to the first few
    # lines is acceptable, which is common for penny-mismatch corrections.
    
    # If using `round()` for initial allocations, the remainder can be positive or negative.
    # We need to distribute positive remainders and "collect" negative remainders.

    # A more precise way to handle rounding and distribution:
    # Calculate each line's proportional share as a float.
    # Sum these float shares.
    # The difference between total_discount_cents and sum(allocs) is the remainder.
    # The `prorate_discount` function should produce integer cents.
    
    # Let's re-think the initial allocation to avoid `round()` and control the remainder.
    # Calculate initial allocations by truncating (floor division).
    # Then distribute the remainder.
    
    initial_allocations = [cents * total_discount_cents // total for cents in line_totals]
    current_sum = sum(initial_allocations)
    
    remainder_to_distribute = total_discount_cents - current_sum

    # To distribute the remainder, we want to give cents to the lines that
    # had the largest fractional parts when calculating their share.
    # This ensures a fairer distribution of the "leftover" cents.
    
    # Calculate the fractional parts for each line.
    # We need to sort by these fractional parts in descending order.
    fractional_parts = []
    for i, cents in enumerate(line_totals):
        # Calculate the exact float value for precision
        exact_value = (cents * total_discount_cents) / total
        # The fractional part is what was "lost" due to integer truncation
        fractional = exact_value - initial_allocations[i]
        fractional_parts.append((fractional, i))

    # Sort lines by their fractional parts in descending order
    # (largest fractional part gets the next cent first).
    fractional_parts.sort(key=lambda x: x[0], reverse=True)

    # Distribute the remainder (one cent at a time)
    for i in range(remainder_to_distribute):
        # Add 1 cent to the line with the current largest fractional part
        # The index is stored in fractional_parts[i % len(line_totals)][1]
        # Using modulo to cycle through if remainder_to_distribute > len(line_totals)
        # However, remainder should generally be less than len(line_totals)
        # as each line can contribute at most 1 cent to the remainder.
        initial_allocations[fractional_parts[i][1]] += 1

    return initial_allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    # The prorate_discount function now guarantees that tot == total_discount_cents
    # if total_discount_cents >= 0 and total > 0.
    # If total_discount_cents < 0, it might not hold, but discounts are usually positive.
    # The problem description implies positive discounts.
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
