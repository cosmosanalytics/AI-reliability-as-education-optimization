"""Audit event serialization service."""
from datetime import datetime, timezone
from models import AuditRecord
from formatter import sanitize_action_name, build_audit_tag

def format_audit_event(event_id: str, action: str, dt: datetime) -> AuditRecord:
    """Formats an audit event record.
    Timestamps must be formatted strictly in ISO 8601 UTC with explicit millisecond
    precision and uppercase 'Z' suffix: YYYY-MM-DDTHH:MM:SS.sssZ (e.g. 2026-10-25T12:00:00.000Z).
    """
    clean_action = sanitize_action_name(action)
    tag = build_audit_tag(clean_action)

    # Ensure the datetime object is UTC and format it with millisecond precision
    # and a 'Z' suffix as per ISO 8601 UTC standard.
    if dt.tzinfo is None or dt.tzinfo.utcoffset(dt) != dt.utcoffset():
        # This means dt is naive or not UTC, so convert it to UTC.
        # This is a defensive check, assuming input dt should ideally be UTC.
        # If dt is naive, it's treated as local and then converted to UTC.
        # If dt is timezone-aware but not UTC, it's converted to UTC.
        dt = dt.astimezone(timezone.utc)
    
    # Format to ISO 8601 with milliseconds and replace '+00:00' with 'Z'
    ts = dt.isoformat(timespec='milliseconds').replace('+00:00', 'Z')
    
    record = AuditRecord(
        event_id=event_id,
        action=clean_action,
        timestamp_str=ts,
        metadata={"tag": tag}
    )
    if not record.is_valid():
        raise ValueError("Invalid record generated")
    return record
