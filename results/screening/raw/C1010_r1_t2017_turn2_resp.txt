"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    pattern_segments = policy.allowed_pattern.split("::")
    request_segments = clean_req.split("::")

    p_idx = 0
    r_idx = 0
    
    while p_idx < len(pattern_segments) and r_idx < len(request_segments):
        pattern_segment = pattern_segments[p_idx]
        request_segment = request_segments[r_idx]

        if pattern_segment == "**":
            # If '**' is the last segment in the pattern, it matches any remaining segments in the request.
            if p_idx == len(pattern_segments) - 1:
                return True
            else:
                # If there are more segments after '**', we need to find a match for the subsequent pattern.
                # We try to match the rest of the pattern from every possible position in the remaining request segments.
                # This is a greedy match for '**'.
                next_pattern_segment_start_idx = p_idx + 1
                remaining_pattern = pattern_segments[next_pattern_segment_start_idx:]
                
                # Iterate through remaining request segments to find a potential match for the rest of the pattern
                for i in range(r_idx, len(request_segments)):
                    if _match_segments_recursive(remaining_pattern, request_segments[i:]):
                        return True
                return False # No match found for the rest of the pattern
        elif pattern_segment == "*":
            # '*' matches exactly one segment.
            # If the request has no more segments, '*' cannot match.
            if r_idx >= len(request_segments):
                return False
            p_idx += 1
            r_idx += 1
        else:
            # Regular segment match.
            if pattern_segment != request_segment:
                return False
            p_idx += 1
            r_idx += 1
    
    # After loops, both pattern and request segments should be exhausted for a full match.
    # If pattern has remaining segments (e.g., "a::b" vs "a"), it's not a match unless it's "**" at the end.
    # If request has remaining segments (e.g., "a" vs "a::b"), it's not a match.
    return p_idx == len(pattern_segments) and r_idx == len(request_segments)

# Helper function for recursive matching used by '**'
def _match_segments_recursive(pattern_parts: list, request_parts: list) -> bool:
    p_idx = 0
    r_idx = 0

    while p_idx < len(pattern_parts) and r_idx < len(request_parts):
        pattern_segment = pattern_parts[p_idx]
        request_segment = request_parts[r_idx]

        if pattern_segment == "**":
            # If '**' is the last segment, it matches any remaining request parts.
            if p_idx == len(pattern_parts) - 1:
                return True
            else:
                # Recursively try to match the rest of the pattern from every possible position
                next_pattern_segment_start_idx = p_idx + 1
                remaining_pattern = pattern_parts[next_pattern_segment_start_idx:]
                for i in range(r_idx, len(request_parts)):
                    if _match_segments_recursive(remaining_pattern, request_parts[i:]):
                        return True
                return False
        elif pattern_segment == "*":
            # '*' matches exactly one segment.
            if r_idx >= len(request_parts): # No request segment to match
                return False
            p_idx += 1
            r_idx += 1
        else:
            # Regular segment match.
            if pattern_segment != request_segment:
                return False
            p_idx += 1
            r_idx += 1
    
    # Both pattern and request parts should be exhausted for a full match.
    return p_idx == len(pattern_parts) and r_idx == len(request_parts)


def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
