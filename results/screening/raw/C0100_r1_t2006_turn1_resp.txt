"""Database pool manager."""
from typing import Dict, Any
from env_loader import merge_env_settings
from options import ConnectionOptions

def configure_pool(base_template: Dict[str, Any], overrides: Dict[str, Any]) -> ConnectionOptions:
    """Configure connection pool options from template and overrides.
    Returns ConnectionOptions instance.
    """
    settings = merge_env_settings(base_template, overrides)
    dsn = settings.get("dsn", "sqlite:///:memory:")
    
    # Extract specific parameters for ConnectionOptions,
    # falling back to ConnectionOptions' own defaults if not provided in settings.
    # The 'timeout' key from env_loader.py's load_system_env() was intended for timeout_seconds.
    timeout_seconds = settings.get("timeout", ConnectionOptions.timeout_seconds)
    max_overflow = settings.get("max_overflow", ConnectionOptions.max_overflow)

    # All other pool-specific parameters go into pool_params.
    # Create a shallow copy of the 'pool' dictionary to avoid modifying shared references
    # if overrides contained mutable objects directly in 'pool'.
    pool_params = dict(settings.get("pool", {}))

    opts = ConnectionOptions(
        dsn=dsn,
        pool_params=pool_params,
        max_overflow=max_overflow,
        timeout_seconds=timeout_seconds,
    )
    _ = opts.summary()
    return opts
