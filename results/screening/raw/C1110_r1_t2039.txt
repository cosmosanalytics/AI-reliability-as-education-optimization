"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a default service name for the telemetry stream
# This could be made configurable in a real application, but for this patch,
# a sensible default is provided as no specific service_name is given in the ticket.
DEFAULT_SERVICE_NAME = "telemetry_stream_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int, service_name: str = DEFAULT_SERVICE_NAME) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This includes wrapping the payload in the wire envelope metadata.
    """
    # Normalize keys in the actual payload before building the envelope
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following checks are for verification/side effects, but the core
    # task is to ensure `pkt` is a valid envelope.
    _ = is_valid_envelope(pkt) # This call now effectively checks the prepared packet
    # The following line extracts normalized keys from the original data, not the enveloped packet.
    # It might be intended for logging or other purposes, so it's kept as is.
    _ = extract_normalized_keys(data)
    return pkt
