"""Low-level jitter percentile calculation engine."""
import math
from typing import List

def evaluate_jitter_percentile(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using the nearest-rank method.
    The index k is calculated as math.ceil(p / 100 * n) - 1 on sorted values.
    For p=0, returns the minimum value (s[0]). For p=100, returns the maximum value (s[n-1]).
    """
    if not values:
        raise ValueError("No jitter values")
    if not (0 <= p <= 100):
        raise ValueError("Percentile 'p' must be between 0 and 100 inclusive.")

    s = sorted(values)
    n = len(s)

    if p == 0:
        return float(s[0])
    if p == 100:
        return float(s[n - 1])

    # Nearest-rank calculation: k = ceil(p / 100 * n) - 1
    # The index is 0-based.
    # The formula ceil(p*n/100) gives a 1-based rank. Subtract 1 for 0-based index.
    k_1_based = math.ceil((p / 100.0) * n)
    k_0_based = int(k_1_based) - 1

    # Ensure k_0_based is within valid bounds [0, n-1]
    # For very small p, k_1_based could be 0, leading to -1. Clamp to 0.
    # For p=100, k_1_based is n, k_0_based is n-1. Clamp to n-1.
    k_0_based = max(0, min(k_0_based, n - 1))

    return float(s[k_0_based])
