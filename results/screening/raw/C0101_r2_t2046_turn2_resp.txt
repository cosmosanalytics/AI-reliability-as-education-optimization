"""Timestamp serialization codec for telemetry records."""
from datetime import datetime, timezone

def encode_iso_timestamp(dt: datetime) -> str:
    """Encode timestamp into standard format (ISO 8601 with milliseconds and 'Z' for UTC)."""
    # Ensure the datetime object is UTC-aware for the 'Z' suffix.
    # If naive, assume it's UTC for telemetry purposes.
    # If timezone-aware but not UTC, convert it to UTC.
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    elif dt.tzinfo != timezone.utc:
        dt = dt.astimezone(timezone.utc)

    # Use isoformat for standard ISO 8601 format with milliseconds.
    # isoformat() for UTC datetimes might produce '+00:00' or 'Z' depending on Python version.
    iso_str = dt.isoformat(timespec='milliseconds')

    # Replace '+00:00' with 'Z' to ensure consistent 'Z' suffix for UTC.
    if iso_str.endswith('+00:00'):
        iso_str = iso_str[:-6] + 'Z'
    return iso_str

def validate_timestamp_range(dt: datetime) -> bool:
    return dt.year >= 2020
