"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a default service name as required by build_envelope, which is not provided by existing call sites.
# This assumes a reasonable default for audit events originating from this formatter.
DEFAULT_SERVICE_NAME = "audit-service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    # Use the DEFAULT_SERVICE_NAME as the service_name for the envelope.
    # This addresses the missing service_name argument for build_envelope based on existing call patterns.
    return build_envelope(DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # The primary responsibility of enveloping and key normalization lies with format_audit_event.
    # serialize_audit_event can now simply call format_audit_event.
    # The previous calls to is_valid_envelope and extract_normalized_keys were not used to
    # modify the message or assert its validity in a way that would prevent the incident.
    # By ensuring format_audit_event correctly builds the enveloped message, serialize_audit_event
    # can directly return its output.
    msg = format_audit_event(data, epoch_ms)
    return msg
