"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define the service name for this telemetry stream.
# This is a fixed requirement for the `build_envelope` function.
SERVICE_NAME = "telemetry-service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This function now constructs the full wire envelope around the payload,
    normalizing the keys within the payload before enveloping.
    """
    # Normalize the keys of the provided data dictionary.
    normalized_payload = normalize_dict_keys(data)
    # Build the complete wire envelope, including version, timestamp, source, and the normalized payload.
    return build_envelope(SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepares and conceptually publishes a telemetry stream packet.
    The actual stream publishing mechanism is external to this function.
    """
    # Prepare the metric packet, which now includes the full wire envelope.
    pkt = prepare_metric_packet(data, epoch_ms)

    # These lines are for internal validation or side effects within this function.
    # The `is_valid_envelope` check confirms that `prepare_metric_packet` correctly
    # produced a valid envelope.
    _ = is_valid_envelope(pkt)

    # `extract_normalized_keys(data)` was likely a leftover or a check on the *raw* data's keys.
    # Since `prepare_metric_packet` now handles normalization *within* the payload,
    # this specific call on the original `data` argument here might be redundant
    # or intended for a different purpose not directly related to the envelope's validity.
    # For this fix, the primary goal is to ensure the output `pkt` is a valid envelope.
    # We maintain this line as it does not affect the correctness of the envelope.
    _ = extract_normalized_keys(data)

    return pkt
