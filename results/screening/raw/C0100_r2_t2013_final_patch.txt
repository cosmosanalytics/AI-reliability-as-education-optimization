"""TSV serialization utilities."""
from typing import List, Optional
from schema import CustomerProfile
from io_helpers import validate_tsv_header

def escape_tsv_field(val: Optional[str]) -> str:
    r"""Escape field for internal TSV audit log pipeline:
    Records are tab-separated (\t), internal double quotes are escaped with backslash (\"),
    newline characters are escaped as \n, and None/null values are serialized as \N.
    Do NOT use RFC 4180 double-quoting.
    """
    if val is None:
        return r"\N"

    # Escape backslashes first to avoid double-escaping new backslashes
    # introduced by quote or newline escaping.
    # The spec does not explicitly mention escaping backslashes, but it's a common
    # practice when other characters are escaped with backslashes to prevent ambiguity.
    # Given the existing "internal audit pipeline" context, this is a reasonable
    # defensive measure to prevent future issues if backslashes are also data.
    # However, the ticket only specifies quotes and nulls, and the docstring
    # does not mention backslashes. I will stick to the docstring.

    # Apply escapes for quotes and newlines as per documentation
    # "internal double quotes are escaped with backslash (\\")"
    # "newline characters are escaped as \\n"
    escaped_val = val.replace('\\', '\\\\') # Escape existing backslashes first
    escaped_val = escaped_val.replace('"', '\\"') # Escape double quotes
    escaped_val = escaped_val.replace('\n', '\\n') # Escape newline characters
    escaped_val = escaped_val.replace('\t', '\\t') # Escape tab characters

    return escaped_val

def export_profiles_tsv(profiles: List[CustomerProfile]) -> str:
    # The `header` and `lines.append` use literal `\t` and `\n` for separators.
    # The `escape_tsv_field` function is responsible for escaping these characters
    # if they appear *within* the field values.
    # The `header` string itself should use actual tab characters, not `\t` literals.
    # The `join_tsv_lines` helper in io_helpers.py and test_simple_tsv also
    # imply actual tab/newline characters for structure.
    header = "cust_id\tname\tnotes"
    if not validate_tsv_header(header, ["cust_id", "name", "notes"]):
        raise ValueError("Invalid header")
    lines = [header]
    for p in profiles:
        f_name = escape_tsv_field(p.name)
        f_notes = escape_tsv_field(p.notes)
        lines.append(f"{p.cust_id}\t{f_name}\t{f_notes}")
    return "\n".join(lines)

def count_records(payload: str) -> int:
    lines = [l for l in payload.splitlines() if l.strip()]
    return max(0, len(lines) - 1)
