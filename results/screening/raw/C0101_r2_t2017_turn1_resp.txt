"""API gateway authorization guard."""
import re
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    pattern = policy.allowed_pattern

    if pattern == "**":
        return True

    if pattern.endswith("::**"):
        prefix = pattern[:-3] # Remove "::**"
        # Match if request_scope starts with the prefix, respecting segment boundaries
        # e.g., "corp::reports::**" matches "corp::reports", "corp::reports::foo", "corp::reports::foo::bar"
        # It should not match "corp::reports_secret"
        if clean_req == prefix:
            return True
        if clean_req.startswith(prefix + "::"):
            return True
        return False

    if pattern.endswith("::*"):
        prefix = pattern[:-3] # Remove "::*"
        # Match if request_scope starts with the prefix and has exactly one more segment
        # e.g., "corp::reports::*" matches "corp::reports::foo" but not "corp::reports", "corp::reports::foo::bar"
        if not clean_req.startswith(prefix + "::"):
            return False
        remaining = clean_req[len(prefix) + 2:] # +2 for "::"
        # Check if 'remaining' contains exactly one segment (no further '::')
        return "::" not in remaining

    # No wildcards, must be an exact match
    # "Sibling paths must be checked against '::' segment boundaries."
    # implies "corp::reports" should not match "corp::reports_secret"
    return clean_req == pattern

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
