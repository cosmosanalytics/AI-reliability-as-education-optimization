"""Low-level dictionary and list hierarchy merger."""
from typing import Dict, Any, List

def merge_spec_values(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """Recursively merges dictionary values, combining lists and overwriting other types."""
    result = dict(base)
    for k, v in override.items():
        if k in result and isinstance(result[k], dict) and isinstance(v, dict):
            result[k] = merge_spec_values(result[k], v)
        elif k in result and isinstance(result[k], list) and isinstance(v, list):
            # Combine lists by extending the base list with new items from the override.
            # Convert to set for unique elements, then back to list to preserve order if needed,
            # but for zones, uniqueness is typically more important than strict order.
            # Using set union to ensure unique elements from both lists.
            result[k] = list(set(result[k]) | set(v))
        else:
            result[k] = v
    return result

def is_spec_empty(spec: Dict[str, Any]) -> bool:
    return len(spec) == 0
