"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    Any fractional cents are distributed one by one to the lines with the largest
    remaining fractional part, to ensure the total allocated discount matches
    total_discount_cents exactly.

    line_totals: list of integer cents representing the value of each line item.
    total_discount_cents: the total discount amount in cents to be allocated.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")

    if total_discount_cents < 0:
        raise ValueError("Total discount cents must be non-negative")

    total_line_value = sum(line_totals)
    num_lines = len(line_totals)

    if num_lines == 0:
        return []

    if total_line_value == 0:
        # If no line value, distribute 0 to all lines.
        # This implicitly means if total_discount_cents > 0, it won't be fully allocated,
        # as there's no value to prorate against.
        return [0] * num_lines

    # Calculate base allocation for each line using integer division (truncation)
    base_allocations = []
    for cents in line_totals:
        base_allocations.append(cents * total_discount_cents // total_line_value)

    # Calculate the sum of base allocations
    sum_base_allocations = sum(base_allocations)

    # Determine remaining cents to distribute
    remaining_cents = total_discount_cents - sum_base_allocations

    # Distribute remaining cents one by one to the lines with the largest fractional remainders
    if remaining_cents > 0:
        # Calculate the fractional part (numerator of the remainder) for each line
        # This determines which lines 'lost' the most due to truncation
        fractional_remainders = []
        for i, cents in enumerate(line_totals):
            # The remainder of (cents * total_discount_cents) / total_line_value
            remainder = (cents * total_discount_cents) % total_line_value
            fractional_remainders.append((remainder, i)) # Store (remainder, original_index)

        # Sort lines by their fractional remainders in descending order.
        # In case of ties, the original index (smaller index) is preferred due to stable sort,
        # which is a reasonable tie-breaking mechanism.
        fractional_remainders.sort(key=lambda x: x[0], reverse=True)

        # Add 1 cent to the 'remaining_cents' number of lines with the largest remainders
        for i in range(remaining_cents):
            original_index = fractional_remainders[i][1]
            base_allocations[original_index] += 1

    return base_allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    # is_fully_allocated should compare against the requested total_discount_cents
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
