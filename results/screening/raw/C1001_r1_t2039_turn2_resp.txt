"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a default service name for this module's telemetry.
# This could be made configurable in a larger refactor, but for this patch,
# a sensible default is provided to ensure the wire envelope is correctly formed.
_DEFAULT_SERVICE_NAME = "telemetry-stream-processor"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # The packet must be wrapped in a wire envelope with metadata.
    # The original implementation returned raw data, leading to missing metadata.
    return build_envelope(
        service_name=_DEFAULT_SERVICE_NAME,
        epoch_ms=epoch_ms,
        payload=data
    )

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following lines are for demonstrating usage of utility functions,
    # but their return values are not used to modify the packet.
    # They remain as per original implementation.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # Note: this operates on original 'data', not the enveloped 'pkt'
    return pkt
