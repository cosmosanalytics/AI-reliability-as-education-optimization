"""Document scope RBAC filter."""
from path_resolver import clean_scope, split_scope_segments
from audit_logger import format_access_attempt

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern."""
    clean_allowed = clean_scope(allowed_scope)
    clean_requested = clean_scope(requested_scope)

    allowed_segments = split_scope_segments(clean_allowed)
    requested_segments = split_scope_segments(clean_requested)

    allowed = False
    if not allowed_segments:
        # If the allowed scope is empty or resolves to no segments, no access is granted.
        # This includes cases like "", " ", "::", etc.
        allowed = False
    elif clean_allowed.endswith("::*"):
        # Wildcard access: requested scope must be a child or descendant of the base allowed scope.
        # The "::*" part is a special indicator, not an actual segment to match.
        base_allowed_segments = allowed_segments[:-1] # Remove the "*" segment
        
        # If base_allowed_segments is empty (e.g., allowed_scope was just "::*"),
        # it implies no specific base scope, thus no access.
        if not base_allowed_segments:
            allowed = False
        else:
            # Requested segments must start with base_allowed_segments and be at least as long
            allowed = len(requested_segments) >= len(base_allowed_segments) and \
                      requested_segments[:len(base_allowed_segments)] == base_allowed_segments
    else:
        # Exact scope access: requested scope must exactly match the allowed scope's segments.
        # This prevents "corp::public" from matching "corp::public_internal".
        allowed = requested_segments == allowed_segments

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
