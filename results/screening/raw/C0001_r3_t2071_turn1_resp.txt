"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses.

    This implementation uses the nearest-rank method, where the index is
    calculated as (p / 100) * n, rounded to the nearest integer.
    If the calculated index is an integer, the value at that index (0-based) is returned.
    If the index is not an integer, it is rounded to the nearest integer index.
    In cases where an index is exactly halfway between two integers (e.g., 2.5),
    it is rounded up to maintain consistency with common percentile expectations
    (e.g., 50th percentile of [10, 20, 30, 40] is 30, not 20).
    """
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)

    # Handle edge cases for p=0 and p=100 explicitly
    if p < 0.0 or p > 100.0:
        raise ValueError("Percentile must be between 0 and 100.")
    if p == 0.0:
        return float(s[0])
    if p == 100.0:
        return float(s[n - 1])

    # Calculate index using (N*P)/100 method, ensuring 0-based indexing
    # This formula (n * p / 100) gives a rank, not an index directly.
    # To get a 0-based index from this rank, we typically subtract 1 if the rank is 1-based.
    # For a dataset of size N, the ranks are 1 to N.
    # The index for a percentile P is typically floor(P/100 * N) or ceil(P/100 * N),
    # or some form of interpolation.
    # The current issue points to the index calculation.
    # Using the "nearest rank" method (Type 1 in some classifications) where
    # index = ceil(p/100 * n) - 1 for 0-based index.
    # This ensures that for a 50th percentile on [10, 20, 30], it picks 20.
    # For [10, 20, 30, 40], 50th percentile (rank 2.0) should be 20,
    # but often 30 is desired (using interpolation or different rounding).
    # The original was `int(round((p / 100.0) * (n - 1)))`.
    # This is equivalent to Type 3 in R.
    # Let's use the method where the index is calculated as `(p / 100.0) * n`,
    # then rounded to the nearest integer. For ties (e.g., 2.5), round up.
    # This is often referred to as the "NIST method" or similar, where
    # the index is `(p / 100.0) * n`.
    # To get a 0-based index: `idx = math.ceil((p / 100.0) * n) - 1`.
    # This is often called the "inclusive" method where the p-th percentile is the smallest value
    # in the data set that is greater than or equal to p percent of the values.
    # Example: [10, 20, 30, 40], n=4. 50th percentile.
    # `idx = math.ceil(0.5 * 4) - 1 = math.ceil(2.0) - 1 = 2 - 1 = 1`. s[1] is 20.
    # This might not be what the failing test expects (20.0 == 30.0).

    # The issue's traceback "assert 20.0 == 30.0" implies that 30.0 was computed
    # when 20.0 was expected. This typically happens in a small dataset where
    # the index calculation rounds up too aggressively or doesn't interpolate.
    # Let's use a common definition that interpolates or picks a more central value.
    # A robust approach is to calculate the rank as `(n - 1) * p / 100`.
    # If the rank is an integer, use that element.
    # If the rank is not an integer, interpolate linearly between the two surrounding elements.
    # This is similar to numpy.percentile's default (linear interpolation) or Excel's PERCENTILE.INC.

    # Calculate the rank (0-based)
    rank = (p / 100.0) * (n - 1)
    
    # Get the integer part and fractional part
    lower_idx = int(math.floor(rank))
    upper_idx = int(math.ceil(rank))
    
    if lower_idx == upper_idx:
        # Rank is an integer, no interpolation needed
        return float(s[lower_idx])
    
    # Linear interpolation
    lower_val = s[lower_idx]
    upper_val = s[upper_idx]
    
    fraction = rank - lower_idx
    
    return float(lower_val + fraction * (upper_val - lower_val))


def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
