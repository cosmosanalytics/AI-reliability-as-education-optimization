"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    pattern_segments = policy.allowed_pattern.split("::")
    request_segments = clean_req.split("::")

    p_idx = 0
    r_idx = 0

    while p_idx < len(pattern_segments) and r_idx < len(request_segments):
        pattern_segment = pattern_segments[p_idx]
        request_segment = request_segments[r_idx]

        if pattern_segment == "**":
            # '**' matches arbitrary multi-segment prefixes.
            # If it's the last pattern segment, it matches everything remaining in request_segments.
            # Otherwise, it matches 0 or more segments, and we need to find if the rest of the pattern matches
            # any suffix of the remaining request segments.
            if p_idx == len(pattern_segments) - 1:
                return True # Matches everything remaining
            
            # Try to match the rest of the pattern (pattern_segments[p_idx+1:])
            # against suffixes of the remaining request (request_segments[r_idx:]).
            # This is a greedy match attempt.
            next_pattern_segment = pattern_segments[p_idx + 1]
            for i in range(r_idx, len(request_segments)):
                if request_segments[i] == next_pattern_segment:
                    # If we find a potential match for the next segment,
                    # recursively check if the rest of the pattern matches.
                    # Create temporary policy and scope for the recursive call.
                    # This is effectively checking if pattern_segments[p_idx+1:] matches request_segments[i:]
                    temp_policy = RoutePolicy(role="", allowed_pattern="::".join(pattern_segments[p_idx+1:]))
                    temp_scope = "::".join(request_segments[i:])
                    if is_scope_permitted(temp_policy, temp_scope):
                        return True
            return False # No suffix of request_segments matched the rest of the pattern.

        elif pattern_segment == "*":
            # '*' matches exactly one segment.
            p_idx += 1
            r_idx += 1
        elif pattern_segment == request_segment:
            # Exact segment match.
            p_idx += 1
            r_idx += 1
        else:
            # Mismatch.
            return False

    # After loop, check if both pattern and request segments were fully consumed.
    # If pattern_segments has remaining parts, it means request_segments was too short to match.
    # The only exception is if the remaining pattern segment is '**' at the very end.
    if p_idx < len(pattern_segments):
        return pattern_segments[p_idx] == "**" and p_idx == len(pattern_segments) - 1

    # If request_segments has remaining parts, it means pattern_segments was too short.
    # This should only be allowed if a '**' wildcard was present earlier in the pattern
    # and matched multiple segments, which would have been handled by the '**' logic.
    # If we are here, it means no '**' matched, so if r_idx < len(request_segments), it's a mismatch.
    return p_idx == len(pattern_segments) and r_idx == len(request_segments)


def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
