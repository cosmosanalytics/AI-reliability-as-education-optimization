"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Assuming a default service name for telemetry packets, as per common practice
# for internal services where the source is implicit or configured externally.
# If `service_name` were meant to be dynamic, it would be an argument.
# Given the incident, the fix is to ensure the envelope is built.
SERVICE_NAME = "telemetry_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This function is intended to prepare the *payload* part of the envelope.
    """
    # This function currently just copies the input data.
    # Future enhancements might involve additional processing or validation here.
    return dict(data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Publish a telemetry stream packet by encapsulating the payload in a wire envelope.

    The `data` provided is treated as the raw payload and will be normalized
    before being placed into the 'payload' field of the wire envelope.
    """
    # Prepare the payload part of the packet.
    prepared_payload = prepare_metric_packet(data, epoch_ms)

    # Normalize keys of the payload as part of the streaming process.
    # The result of extract_normalized_keys is not directly used for the packet,
    # but it implies that normalization is a step that happens.
    # The actual payload sent should have normalized keys.
    normalized_payload = normalize_dict_keys(prepared_payload)

    # Build the complete wire envelope using the normalized payload.
    # The service name is assumed to be a fixed value for this stream.
    envelope = build_envelope(SERVICE_NAME, epoch_ms, normalized_payload)

    # The original code had a check for is_valid_envelope and extract_normalized_keys
    # but discarded the results. The core issue is that the envelope itself was not built.
    # By building the envelope, we ensure it's valid.
    # We can still assert validity for robustness, though build_envelope guarantees it.
    if not is_valid_envelope(envelope):
        # This case should ideally not be reachable if build_envelope is correct,
        # but it provides a safeguard if the envelope structure changes.
        raise ValueError("Failed to build a valid telemetry wire envelope.")

    return envelope
