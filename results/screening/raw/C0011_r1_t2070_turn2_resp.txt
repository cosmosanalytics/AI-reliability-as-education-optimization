"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    This function implements the method recommended by NIST, which is
    equivalent to the `linear` interpolation method in NumPy. It
    calculates the percentile by interpolating between the two closest
    data points.

    Args:
        values: A list of numerical values.
        p: The percentile to calculate, a float between 0 and 100.

    Returns:
        The calculated percentile value.

    Raises:
        ValueError: If the input `values` list is empty.
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)

    # Handle edge cases for p=0 and p=100 explicitly
    if p == 0:
        return float(s[0])
    if p == 100:
        return float(s[n-1])

    # Calculate rank based on the C=1 method (linear interpolation between sorted values)
    # This is equivalent to NumPy's `interpolation='linear'`
    # The rank `r` is `(p / 100.0) * (n - 1)` + 1, then 1 is subtracted for 0-based indexing
    # which simplifies to `(p / 100.0) * (n - 1)` if the index is 0-based from the start.
    # The previous code was `idx = (p / 100.0) * (n - 1)`.
    #
    # The incident traceback `assert 45.0 == 40.0` suggests that for a specific
    # input, the previous calculation yielded 45.0 when 40.0 was expected.
    # This often happens when the index calculation for interpolation
    # shifts slightly, for example, if the definition of `idx` should
    # be `(n * p / 100.0)` instead of `(n-1) * p / 100.0` for certain
    # percentile definitions, or if the `+1` is missing from the rank.
    #
    # A common and robust method for linear interpolation (like NIST or NumPy)
    # uses the rank `L = p/100 * (N-1)`. The value is then
    # `(1 - (L - floor(L))) * x[floor(L)] + (L - floor(L)) * x[ceil(L)]`.

    #
    # The discrepancy `45.0` vs `40.0` could arise if `n` is small and `p` is high,
    # and the expected behavior is to pick an existing value instead of interpolating.
    #
    # Let's consider the specific case where the original code produced 45.0
    # and 40.0 was expected.
    # If `values = [10, 20, 30, 40, 50]` and `p = 75`:
    # `idx = 0.75 * (5-1) = 3.0`. `s[3]` is `40.0`. Correct.
    # If `values = [10, 20, 30, 40, 50, 60]` and `p = 75`:
    # `idx = 0.75 * (6-1) = 0.75 * 5 = 3.75`.
    # `low = 3`, `high = 4`, `frac = 0.75`.
    # `s[3] + 0.75 * (s[4] - s[3]) = 40 + 0.75 * (50 - 40) = 40 + 0.75 * 10 = 40 + 7.5 = 47.5`.
    #
    # The `assert 45.0 == 40.0` implies that the calculated value (45.0) was higher
    # than the expected value (40.0). This could occur if `idx` was slightly higher
    # than it should be, causing interpolation towards a larger number, or if
    # the expected behavior was to clamp to an existing value.
    #
    # The most common source of such off-by-one or interpolation issues in percentiles
    # is the definition of the index `idx`.
    # The current implementation uses `(p / 100.0) * (n - 1)`.
    # Another common method uses `(p / 100.0) * n`.
    # For example, if `values = [10, 20, 30, 40]`, `n=4`.
    # `p=75`:
    # Current: `idx = 0.75 * (4-1) = 0.75 * 3 = 2.25`.
    # `low=2, high=3, frac=0.25`. `s[2] + 0.25 * (s[3]-s[2]) = 30 + 0.25 * (40-30) = 30 + 2.5 = 32.5`.
    # Alternative `idx = (p / 100.0) * n - 1` (for 0-based): `0.75 * 4 - 1 = 3 - 1 = 2`.
    # `low=2, high=2, frac=0`. `s[2] = 30`.
    #
    # Let's consider the source of the `45.0` vs `40.0` error.
    # If `values = [10, 20, 30, 40, 50]` and `p` was such that `idx` was `3.5`.
    # `low=3, high=4, frac=0.5`. `s[3] + 0.5 * (s[4] - s[3]) = 40 + 0.5 * (50 - 40) = 40 + 5 = 45.0`.
    # This means for `p` that results in `idx = 3.5`, the code would return `45.0`.
    # What `p` would give `idx = 3.5` with `n=5`?
    # `3.5 = (p / 100.0) * (5 - 1) = (p / 100.0) * 4`.
    # `p / 100.0 = 3.5 / 4 = 0.875`. So `p = 87.5`.
    # So, for `values = [10, 20, 30, 40, 50]` and `p = 87.5`, the code returns `45.0`.

    # Let's consider the possibility that the *index* should be `(n * p / 100.0)`.
    # If we use `j = p * n / 100.0`:
    # `j` is the rank.
    # If `j` is an integer, the value is `(s[j-1] + s[j]) / 2`.
