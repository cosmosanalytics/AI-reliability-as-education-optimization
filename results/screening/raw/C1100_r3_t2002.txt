"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    Any fractional cents are distributed to ensure the total allocated
    discount matches total_discount_cents, by allocating remaining cents
    to lines with the largest fractional parts until the remainder is zero.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    
    num_lines = len(line_totals)
    if num_lines == 0:
        return []

    total_line_value = sum(line_totals)
    if total_line_value == 0:
        return [0] * num_lines

    # Calculate base allocations using floating-point to preserve precision
    # Store (floor_value, fractional_part, original_index)
    alloc_details = []
    for i, cents in enumerate(line_totals):
        # Calculate the proportional share
        proportion = cents * total_discount_cents / total_line_value
        floor_value = int(proportion)
        fractional_part = proportion - floor_value
        alloc_details.append((floor_value, fractional_part, i))

    # Sort by fractional part in descending order to distribute remaining cents
    # If fractional parts are equal, the original order (index) serves as a tie-breaker.
    alloc_details.sort(key=lambda x: (-x[1], x[2])) 

    # Initialize the result list with the floor values
    final_allocations = [0] * num_lines
    current_total_allocated = 0
    for floor_value, _, original_index in alloc_details:
        final_allocations[original_index] = floor_value
        current_total_allocated += floor_value

    # Distribute remaining cents
    remaining_cents = total_discount_cents - current_total_allocated
    
    # Iterate through the sorted allocations and add 1 cent until remaining_cents is 0
    for i in range(num_lines):
        if remaining_cents <= 0:
            break
        # The allocations are sorted by fractional part (descending), so pick the current one
        original_index = alloc_details[i][2]
        final_allocations[original_index] += 1
        remaining_cents -= 1

    return final_allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
