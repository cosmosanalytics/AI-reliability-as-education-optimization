"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    This function ensures that the sum of allocated discounts exactly matches
    total_discount_cents, distributing any remainder (pennies) to lines
    with the largest fractional parts after initial proportional division.

    line_totals: list of integer cents representing the value of each line.
    total_discount_cents: the total discount amount to be allocated. Must be non-negative.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    if total_discount_cents < 0:
        raise ValueError("Total discount cents must be non-negative")

    total_value = sum(line_totals)
    num_lines = len(line_totals)

    if total_value == 0:
        return [0] * num_lines

    # Calculate base allocations by truncating (rounding down)
    # Also store the fractional parts for remainder distribution
    exact_allocations = []
    base_allocations = [0] * num_lines
    for i, cents in enumerate(line_totals):
        exact = (cents * total_discount_cents) / total_value
        base_allocations[i] = int(exact) # Truncate towards zero
        exact_allocations.append((exact, i)) # Store exact value and original index

    # Calculate the remaining cents to distribute
    current_total_allocated = sum(base_allocations)
    remainder = total_discount_cents - current_total_allocated

    # Distribute the remainder to lines with the largest fractional parts
    # Sort by the fractional part in descending order.
    # We calculate fractional_part from exact_allocations[i][0] - base_allocations[i]
    # to handle potential floating point inaccuracies by using the original exact value.
    # The sort key takes (fractional_part, original_index) to ensure stable sorting
    # and consistent tie-breaking.
    fractions_and_indices = sorted(
        [
            (exact_allocations[i][0] - base_allocations[i], i)
            for i in range(num_lines)
        ],
        key=lambda x: (-x[0], x[1]) # Sort by fractional part (desc), then by original index (asc)
    )

    for i in range(remainder):
        # Add 1 cent to the line with the largest remaining fractional part
        original_index = fractions_and_indices[i][1]
        base_allocations[original_index] += 1

    return base_allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
