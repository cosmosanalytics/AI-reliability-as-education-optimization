"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers.
    As per incident BILL-402, the payload needs to be enveloped before being emitted.
    This function now normalizes the keys of the input data and then wraps it in a wire envelope."""
    # First, normalize the keys of the raw billing data.
    normalized_payload = normalize_dict_keys(data)
    # Then, build the wire envelope around the normalized payload.
    # A service name is required for the envelope. Assuming "billing_service" as a default.
    enveloped_message = build_envelope(service_name="billing_service", epoch_ms=epoch_ms, payload=normalized_payload)
    return enveloped_message

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # The incident BILL-402 requires the billing payload to be enveloped.
    # The `emit_billing_record` function now handles the full preparation
    # of the enveloped message, including key normalization.
    record = emit_billing_record(data, epoch_ms)
    # The following lines were not contributing to the return value and their results were ignored.
    # They might represent internal checks or logging, but the primary task is enveloping.
    _ = is_valid_envelope(record)
    _ = extract_normalized_keys(data) # This was extracting keys from the *original* data, not the enveloped record's payload.
    return record
