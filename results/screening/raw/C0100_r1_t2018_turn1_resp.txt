"""Document scope RBAC filter."""
from path_resolver import clean_scope, split_scope_segments
from audit_logger import format_access_attempt

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern."""
    allowed_segs = split_scope_segments(allowed_scope)
    requested_segs = split_scope_segments(requested_scope)

    # An empty allowed_scope implies no specific scope is allowed.
    # If allowed_segs is empty, no access is granted unless requested_segs is also empty (which means both are empty string).
    # However, for hierarchical scope access, an empty allowed_scope usually means nothing is accessible.
    # The `startswith` check on segments naturally handles the case where allowed_scope is empty:
    # `allowed_segs` would be `[]`. `requested_segs[:0]` is `[]`, so `[] == []` is true.
    # This would mean an empty `allowed_scope` grants access to everything, which is incorrect for a security context.
    # Therefore, explicitly deny if `allowed_scope` is effectively empty.
    if not allowed_segs:
        allowed = False
    else:
        # Check if the requested scope's segments start with the allowed scope's segments
        # and if the requested scope has at least as many segments as the allowed scope.
        allowed = (len(requested_segs) >= len(allowed_segs) and
                   requested_segs[:len(allowed_segs)] == allowed_segs)

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
