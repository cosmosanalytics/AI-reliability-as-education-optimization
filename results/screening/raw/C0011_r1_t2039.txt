"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Incident STREAM-105: Kafka consumer failed to parse telemetry payload missing required wire envelope metadata.
# This constant defines the service name to be used in the wire envelope.
# In a full system, this would typically be configurable or passed in.
DEFAULT_SERVICE_NAME = "telemetry_stream_producer"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """
    Prepare telemetry metric packet for distribution to event stream.
    The packet's payload keys will be normalized to snake_case, and the entire
    packet will be wrapped in a standard wire envelope as required by consumers.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """
    Prepares and returns a complete, enveloped, and normalized telemetry packet.
    This function now leverages `prepare_metric_packet` to ensure the output
    is a valid wire envelope with normalized payload keys.
    """
    pkt = prepare_metric_packet(data, epoch_ms)
    # The original implementation called is_valid_envelope and extract_normalized_keys
    # but did not utilize their return values to modify the packet.
    # Now, prepare_metric_packet ensures the packet is correctly formed.
    # We can optionally add a validation check here for robustness,
    # but `prepare_metric_packet` should guarantee validity.
    return pkt
