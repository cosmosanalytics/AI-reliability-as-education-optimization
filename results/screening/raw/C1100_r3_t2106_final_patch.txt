"""Cache Hit Latency Percentile Monitor percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_cache_percentile(samples: List[float], p: float) -> float:
    """
    Calculate the p-th percentile of a list of samples using the nearest-rank method.
    The nearest-rank method calculates the rank `k = ceil(p/100 * n)` and returns
    the value at that rank (1-based index) from the sorted samples.
    """
    win = create_sample_window("calculate_cache_percentile", samples)
    s = validate_metric_samples(win.values) # This sorts the samples
    n = len(s)

    if not (0 <= p <= 100):
        raise ValueError("Percentile 'p' must be between 0 and 100 inclusive.")
    if n == 0:
        raise ValueError("Cannot calculate percentile for an empty list of samples.")

    # Calculate the rank using the nearest-rank method (R-1 or NIST method)
    # k = ceil(p/100 * n)
    # The value is s[k-1] (0-based index)
    rank = math.ceil((p / 100.0) * n)

    # Ensure rank is within valid bounds (1 to n)
    if rank < 1:
        # For p=0, rank would be 0, return the smallest value
        return s[0]
    if rank > n:
        # This case should ideally not happen if p <= 100, but as a safeguard
        return s[n - 1]

    return s[int(rank) - 1]

def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
