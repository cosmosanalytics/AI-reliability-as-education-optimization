"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.
    Uses the nearest-rank method (NIST method R-1) if the index is an integer,
    and linear interpolation between the two nearest ranks otherwise.
    This method is also known as the C = 1 method or Method 7 in some literature
    where the index is calculated as (N-1)p/100.
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)

    # For p=0, the 0th percentile is the minimum value.
    if p == 0:
        return s[0]
    # For p=100, the 100th percentile is the maximum value.
    if p == 100:
        return s[n - 1]

    # Calculate the rank (index) using the (N-1)p/100 method.
    # This is consistent with NumPy's default interpolation='linear'
    # when the index is not an integer.
    # The current implementation calculates idx as (p / 100.0) * (n - 1).
    # This formula implies that for integer index, it returns the value at that index.
    # For a non-integer index, it performs linear interpolation.
    # If the rank is an integer, and it corresponds exactly to an element's position,
    # that element is the percentile. For example, for [10, 20, 30, 40, 50],
    # 75th percentile (p=75, n=5) means idx = 0.75 * 4 = 3.0.
    # In such cases, the result should be s[3] = 40.0.
    # The current code would calculate low=3, high=3, frac=0, result=s[3]=40.0.

    # However, for an array like [10, 20, 30, 40], n=4, 75th percentile (p=75)
    # idx = 0.75 * (4-1) = 2.25.
    # low=2, high=3, frac=0.25. result = s[2] + 0.25 * (s[3] - s[2]) = 30 + 0.25*(40-30) = 32.5.
    # This calculation is standard for linear interpolation.

    # The discrepancy `assert 45.0 == 40.0` suggests that for a specific
    # dataset and percentile, the current method yielded 45.0, but 40.0 was expected.
    # This is typically due to different percentile definitions.
    # The ticket describes an anomaly in SLA calculation, which often relies on a
    # specific, strict definition of percentile.
    # If the requirement is that for `p` percent, `p` percent of values are strictly
    # *below or equal* to the percentile value, and the values are discrete,
    # a common interpretation is to pick the value at `ceil((p/100)*n) - 1` index
    # (for 0-based indexing) or `floor((p/100)*n)` for 1-based indexing,
    # if `(p/100)*n` is not an integer.
    # If `(p/100)*n` is an integer `k`, some definitions take the average of `s[k-1]` and `s[k]`,
    # while others just take `s[k]`.

    # A common alternative for discrete data, especially when `p*N/100` is an integer,
    # is to take the value at that exact rank, or the next higher one if not exact.
    # For example, if we have [10, 20, 30, 40, 50] (n=5) and want 80th percentile (p=80):
    # (80/100) * 5 = 4.0. The 4th element (0-indexed s[3]) is 40.
    # The current method (idx = (p/100)*(n-1)) would give idx = 0.8 * 4 = 3.2.
    # low=3, high=4, frac=0.2. result = s[3] + 0.2*(s[4]-s[3]) = 40 + 0.2*(50-40) = 42.0.
    # If the expected value is 40.0, it suggests a definition that finds the smallest
    # value in the dataset that is greater than or equal to `p` percent of the values.
    # This is often achieved by calculating the rank `k = ceil((p/100)*n)`.
    # For `p=80, n=5`, `k = ceil(0.8*5) = ceil(4.0) = 4`. The value would be `s[k-1]` (0-indexed), i.e., `s[3] = 40`.

    # Let's implement this alternative definition if it aligns with the expected behavior.
    # This is method R-1 (Nearest Rank) from NIST, also known as the "definition 1" or "exclusive method".
    # It calculates the rank `k` as `ceil(p * n / 100)`.
    # The percentile is then `s[k-1]` (using 0-based indexing).
    # This method is suitable for discrete data and when the percentile should be an actual data point.

    rank = math.ceil((p / 100.0) * n)
    # Ensure rank is within bounds [1, n]
    rank = max(1, min(int(rank), n))
    return float(s[rank - 1])
