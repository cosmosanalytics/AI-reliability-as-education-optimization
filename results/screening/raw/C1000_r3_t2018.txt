"""Document scope RBAC filter."""
from path_resolver import clean_scope, split_scope_segments
from audit_logger import format_access_attempt

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern.

    Access is granted if the segments of the requested_scope exactly match the segments
    of the allowed_scope, or if the allowed_scope segments are a strict prefix of
    the requested_scope segments. This prevents granting access to sibling scopes
    that merely share a prefix string.
    """
    allowed_segs = split_scope_segments(allowed_scope)
    requested_segs = split_scope_segments(requested_scope)

    # An empty allowed_scope implies no specific access, so deny.
    # An empty requested_scope also implies nothing to access, so deny.
    if not allowed_segs or not requested_segs:
        allowed = False
    elif len(requested_segs) < len(allowed_segs):
        # Requested scope is shorter than allowed scope, so it cannot be a sub-scope.
        allowed = False
    else:
        # Check if the allowed_scope segments are a prefix of the requested_scope segments.
        allowed = requested_segs[:len(allowed_segs)] == allowed_segs

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
