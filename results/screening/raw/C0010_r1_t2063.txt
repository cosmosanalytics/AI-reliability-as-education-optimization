"""Subscription recurring billing client."""
import uuid
from typing import Dict, Any, List
from charge_record import ChargeReceipt
from card_validator import validate_account_format, get_charge_currency

class SubscriptionClient:
    def __init__(self):
        self.dispatched_attempts: List[str] = []

    def charge_account(self, account_id: str, amount_cents: int, retries: int = 1) -> ChargeReceipt:
        if not validate_account_format(account_id):
            raise ValueError("Invalid account format")
        curr = get_charge_currency()
        _ = curr
        # Generate the idempotency key once for the entire logical charge operation.
        key = f"tok_{uuid.uuid4()}"
        for _ in range(retries + 1):
            # All attempts for this charge operation should use the same idempotency key.
            self.dispatched_attempts.append(key)
            # In a real scenario, this loop would contain the actual dispatch
            # logic to a payment gateway. If successful, it would break the loop.
            # If it fails with a retriable error, it would continue to the next iteration.
            # For this simulation, we simply track the attempts with the consistent key.
        return ChargeReceipt(account_id=account_id, amount_cents=amount_cents, idempotency_header=key)
