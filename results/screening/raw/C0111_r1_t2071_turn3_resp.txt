"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses."""
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)
    
    # Ensure p is within [0, 100] range for robust calculation
    p = max(0.0, min(100.0, p))

    # Calculate index using "round half up" logic for consistency.
    # The original `round()` function in Python 3 uses "round half to even",
    # which can lead to unexpected results for values ending in .5,
    # potentially causing off-by-one errors in index selection compared to
    # common percentile definitions (e.g., median of two elements).
    # Adding 0.5 before `int()` effectively implements "round half up" for
    # positive numbers, which is a more common and predictable rounding
    # strategy for percentile index calculation.
    # The (n - 1) factor is standard for 0-indexed arrays where 0th percentile
    # maps to s[0] and 100th percentile maps to s[n-1].
    idx = int((p / 100.0) * (n - 1) + 0.5)
    
    # Ensure the calculated index is within valid bounds [0, n-1].
    # This also implicitly handles edge cases where p might be slightly
    # outside [0, 100] due to floating point inaccuracies, although
    # `p = max(0.0, min(100.0, p))` already mitigates this.
    idx = max(0, min(n - 1, idx))
    
    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
