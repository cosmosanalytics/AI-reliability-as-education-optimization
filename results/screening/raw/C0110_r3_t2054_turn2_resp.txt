"""Low-level dictionary and list hierarchy merger."""
from typing import Dict, Any, List

def merge_spec_values(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """Recursively merges dictionary values and concatenates lists.
    If a key exists in both `base` and `override`, and both values are dictionaries,
    they are recursively merged. If both are lists, they are concatenated.
    Otherwise, the `override` value replaces the `base` value."""
    result = dict(base)
    for k, v in override.items():
        if k in result and isinstance(result[k], dict) and isinstance(v, dict):
            result[k] = merge_spec_values(result[k], v)
        elif k in result and isinstance(result[k], list) and isinstance(v, list):
            # Concatenate lists as per incident description (secondary zone availability lost
            # when tenant spec replaced base zone array).
            result[k] = result[k] + v
        else:
            result[k] = v
    return result

def is_spec_empty(spec: Dict[str, Any]) -> bool:
    return len(spec) == 0
