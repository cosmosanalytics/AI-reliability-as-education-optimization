"""Subscription recurring billing client."""
import uuid
from typing import Dict, Any, List
from charge_record import ChargeReceipt
from card_validator import validate_account_format, get_charge_currency

class SubscriptionClient:
    def __init__(self):
        self.dispatched_attempts: List[str] = []

    def charge_account(self, account_id: str, amount_cents: int, retries: int = 1) -> ChargeReceipt:
        if not validate_account_format(account_id):
            raise ValueError("Invalid account format")
        curr = get_charge_currency()
        _ = curr
        # Generate the idempotency key once for the entire charge operation.
        # This key should remain the same across retries for the same logical charge.
        idempotency_key = f"tok_{uuid.uuid4()}"
        for _ in range(retries + 1):
            # In a real system, the API call would be made here,
            # using idempotency_key as the idempotency header.
            # For tracking, we record each attempt with the consistent key.
            self.dispatched_attempts.append(idempotency_key)
            # If a successful response was received, we would break here.
            # If a timeout or transient error occurred, the loop would continue to retry.
        return ChargeReceipt(account_id=account_id, amount_cents=amount_cents, idempotency_header=idempotency_key)
