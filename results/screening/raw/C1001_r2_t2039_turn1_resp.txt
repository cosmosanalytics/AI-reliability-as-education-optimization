"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Assuming a fixed service name for this telemetry stream as not specified in the ticket.
# In a real system, this would likely be configured or passed in.
SERVICE_NAME = "telemetry_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    This involves wrapping the raw data in the standard wire envelope.
    """
    # The actual payload should be normalized before being put into the envelope,
    # as per typical telemetry requirements, but the ticket only mentions
    # missing wire envelope metadata. The existing `extract_normalized_keys`
    # is called in `publish_stream_packet` on the raw data, implying
    # normalization is a separate step if needed.
    # For now, we only address the missing envelope.
    return build_envelope(SERVICE_NAME, epoch_ms, data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The following lines are for informational purposes or potential future use
    # and do not affect the packet content itself.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # Operates on the original data, not the enveloped packet
    return pkt
