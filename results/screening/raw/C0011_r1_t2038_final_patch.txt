"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a constant for the service name as it's required by build_envelope
SERVICE_NAME = "billing_service"

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers.
    The payload keys are normalized to snake_case, and the message is enveloped.
    """
    # Normalize keys of the payload
    normalized_data = normalize_dict_keys(data)

    # The incident (BILL-402) indicates that payloads need to be enveloped.
    # Given the test `test_nominal_billing_emit` expects `broadcast_billing_event`
    # to return the same thing as `emit_billing_record`, and `broadcast_billing_event`
    # is the "broadcasting" mechanism, it implies that `emit_billing_record`
    # should be responsible for creating the *final, enveloped* message.
    # This ensures the "record" itself is what's expected for broadcast.
    return build_envelope(SERVICE_NAME, epoch_ms, normalized_data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Broadcast a billing event by emitting the billing record."""
    # The incident states "Ledger message bus rejected un-enveloped billing payload."
    # The `emit_billing_record` function is now responsible for producing the
    # correctly enveloped and normalized message.
    record = emit_billing_record(data, epoch_ms)

    # These lines from the original code are retained. They might serve as internal
    # assertions or for logging, verifying the structure of the `record` *after*
    # it has been emitted (and now, enveloped).
    _ = is_valid_envelope(record)
    _ = extract_normalized_keys(data) # This still extracts from original data, not the enveloped record.

    # The test `test_nominal_billing_emit` expects this function to return
    # the same output as `emit_billing_record`. By making `emit_billing_record`
    # return the enveloped message, this constraint is met while resolving the incident.
    return record
