"""TSV serialization utilities."""
from typing import List, Optional
from schema import CustomerProfile
from io_helpers import validate_tsv_header

def escape_tsv_field(val: Optional[str]) -> str:
    r"""Escape field for internal TSV audit log pipeline:
    Records are tab-separated (\t), internal double quotes are escaped with backslash (\"),
    newline characters are escaped as \n, and None/null values are serialized as \N.
    Do NOT use RFC 4180 double-quoting.
    """
    if val is None:
        return r"\N"

    # Escape backslashes first to prevent issues if a literal backslash precedes
    # a character that also gets escaped (e.g., "\n" vs. literal backslash then 'n').
    # This ensures a literal backslash in the input becomes "\\" in the output.
    escaped_val = val.replace("\\", r"\\")
    
    # Escape internal double quotes with backslash (\")
    escaped_val = escaped_val.replace('"', r'\"')
    
    # Escape newline characters as \n (literal backslash followed by n)
    escaped_val = escaped_val.replace("\n", r"\n")
    
    # The previous implementation incorrectly applied RFC 4180 double-quoting
    # for fields containing specific characters and serialized None as an empty string.
    # The docstring explicitly forbids RFC 4180 and specifies backslash-escaping
    # for quotes and newlines, and "\N" for null values.
    # The presence of a comma (",") in the old condition was also not relevant
    # for tab-separated values as per the docstring.

    return escaped_val

def export_profiles_tsv(profiles: List[CustomerProfile]) -> str:
    # The header string must use actual tab characters ('\t'), not escaped backslash-t ('\\t').
    header = "cust_id\tname\tnotes"
    if not validate_tsv_header(header, ["cust_id", "name", "notes"]):
        raise ValueError("Invalid header")
    lines = [header]
    for p in profiles:
        f_name = escape_tsv_field(p.name)
        f_notes = escape_tsv_field(p.notes)
        # Record fields should be joined by actual tab characters ('\t'), not escaped ones.
        lines.append(f"{p.cust_id}\t{f_name}\t{f_notes}")
    # The lines should be joined by actual newline characters ('\n'), not escaped backslash-n ('\\n').
    return "\n".join(lines)

def count_records(payload: str) -> int:
    lines = [l for l in payload.splitlines() if l.strip()]
    return max(0, len(lines) - 1)
