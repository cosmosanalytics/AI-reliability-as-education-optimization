"""Low-level tax rate computation engine."""
import math
from typing import Dict, Any

DEFAULT_TAX_JURISDICTIONS = {
    "standard": 0.20,
    "reduced": 0.10,
    "exempt": 0.0,
}

def get_rate_for_jurisdiction(code: str) -> float:
    return DEFAULT_TAX_JURISDICTIONS.get(code, 0.20)

def calculate_vat(net_cents: int, vat_rate: float) -> int:
    """Compute VAT in integer cents for given net amount.
    Uses half-up rounding (ROUND_HALF_UP: ties broken towards positive infinity) on exact half-cents.
    """
    # Implement "round half up towards positive infinity".
    # For a number X, this means:
    # If fractional part is < 0.5, round down.
    # If fractional part is > 0.5, round up.
    # If fractional part is exactly 0.5, round up (towards positive infinity).
    # This can be achieved for both positive and negative numbers by adding 0.5 and taking the floor.
    # E.g., 13.4 -> 13.9 -> 13
    #       13.5 -> 14.0 -> 14
    #       13.6 -> 14.1 -> 14
    #       -13.4 -> -12.9 -> -13
    #       -13.5 -> -13.0 -> -13
    #       -13.6 -> -13.1 -> -14
    return math.floor(net_cents * vat_rate + 0.5)
