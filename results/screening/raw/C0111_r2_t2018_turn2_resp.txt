"""Document scope RBAC filter."""
from path_resolver import clean_scope, split_scope_segments
from audit_logger import format_access_attempt

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern.

    The `allowed_scope` can specify a hierarchical prefix (e.g., "corp::public")
    or a wildcard for sub-scopes (e.g., "corp::public::*").
    - If `allowed_scope` is "A::B", it permits "A::B", "A::B::C", etc.
    - If `allowed_scope` is "A::B::*", it permits "A::B::C", "A::B::C::D", etc.,
      but not "A::B" itself.
    """
    cleaned_allowed = clean_scope(allowed_scope)
    cleaned_requested = clean_scope(requested_scope)

    is_wildcard_allowed = False
    if cleaned_allowed.endswith("::*"):
        is_wildcard_allowed = True
        # Remove "::*" for segment comparison
        cleaned_allowed_prefix = cleaned_allowed[:-3]
    else:
        cleaned_allowed_prefix = cleaned_allowed

    allowed_segments = split_scope_segments(cleaned_allowed_prefix)
    requested_segments = split_scope_segments(cleaned_requested)

    allowed = False
    if not allowed_segments:
        # An empty allowed scope (after removing "::*") means no access
        # unless it was an empty string and the requested scope is also empty.
        # However, `split_scope_segments("")` returns `[]`, so this path is correct.
        allowed = False
    elif len(requested_segments) < len(allowed_segments):
        # Requested scope is shorter than the allowed prefix, so cannot match
        allowed = False
    else:
        # Check if requested scope starts with the allowed segments
        prefix_match = requested_segments[:len(allowed_segments)] == allowed_segments

        if is_wildcard_allowed:
            # For "A::B::*", requested must be "A::B::C" or deeper
            allowed = prefix_match and len(requested_segments) > len(allowed_segments)
        else:
            # For "A::B", requested can be "A::B" or "A::B::C" or deeper
            allowed = prefix_match

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
