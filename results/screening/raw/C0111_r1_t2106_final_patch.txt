"""Cache Hit Latency Percentile Monitor percentile calculator."""
from typing import List
from percentile_helper import validate_metric_samples, sample_range
from metric_model import create_sample_window
import math

def calculate_cache_percentile(samples: List[float], p: float) -> float:
    win = create_sample_window("calculate_cache_percentile", samples)
    s = validate_metric_samples(win.values)

    if not s:
        # validate_metric_samples already raises ValueError for empty lists,
        # but this provides a more specific message in this context.
        raise ValueError("Samples list cannot be empty for percentile calculation")

    # Ensure p is within valid range [0, 100]
    if not (0 <= p <= 100):
        raise ValueError("Percentile 'p' must be between 0 and 100 inclusive")

    # The "nearest-rank house standard" (often Type 1 or R-1 in statistical software)
    # defines the percentile rank L as (N * P / 100).
    # The percentile value is then the L-th smallest value in the sorted list.
    # For 0-based indexing, this corresponds to index ceil(L) - 1.

    # Handle edge cases for p=0 and p=100 explicitly for clarity and robustness.
    # For p=0, the 0th percentile is typically the minimum value (s[0]).
    # For p=100, the 100th percentile is the maximum value (s[-1]).
    if p == 0:
        return s[0]
    if p == 100:
        return s[-1]

    # Calculate the rank using the nearest-rank method.
    # idx_float represents the 1-based rank.
    idx_float = (p / 100.0) * len(s)

    # Convert to 0-based index by taking the ceiling of the rank and subtracting 1.
    # Use max(0, ...) to ensure the index is not negative, though p=0 is handled above.
    idx = max(0, math.ceil(idx_float) - 1)

    # Ensure the calculated index is within the valid bounds of the list.
    # This might be necessary for edge cases with very small N or specific P values
    # where ceil(idx_float) - 1 could theoretically exceed len(s) - 1 if not for p=100 guard.
    idx = min(idx, len(s) - 1)

    return s[idx]

def compute_summary_range(samples: List[float]) -> float:
    return sample_range(samples)
