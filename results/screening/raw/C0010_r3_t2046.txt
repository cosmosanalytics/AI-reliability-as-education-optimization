"""Timestamp serialization codec for telemetry records."""
from datetime import datetime, timezone

def encode_iso_timestamp(dt: datetime) -> str:
    """Encode timestamp into standard ISO 8601 format (YYYY-MM-DDTHH:MM:SS.sssZ)."""
    # Ensure the datetime object is in UTC
    if dt.tzinfo is None:
        # If naive, assume it's UTC for serialization as per standard practice for Z suffix
        dt_utc = dt.replace(tzinfo=timezone.utc)
    else:
        dt_utc = dt.astimezone(timezone.utc)

    # Format to 'YYYY-MM-DDTHH:MM:SS.sss' and append 'Z'
    # .isoformat(timespec='milliseconds') produces 'YYYY-MM-DDTHH:MM:SS.sss' for naive
    # or 'YYYY-MM-DDTHH:MM:SS.sss+00:00' for UTC aware.
    # We need to strip the timezone offset if present and append 'Z'.
    iso_str = dt_utc.isoformat(timespec='milliseconds')
    if iso_str.endswith('+00:00'):
        iso_str = iso_str[:-6] # Remove '+00:00'
    elif iso_str.endswith('-00:00'): # Handle potential negative offset for UTC
        iso_str = iso_str[:-6] # Remove '-00:00'
    
    return iso_str + "Z"

def validate_timestamp_range(dt: datetime) -> bool:
    return dt.year >= 2020
