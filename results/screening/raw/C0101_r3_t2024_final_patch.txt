"""Internal stream sorting helper."""
from typing import List
from event_packet import EventPacket

def order_packets(packets: List[EventPacket]) -> List[EventPacket]:
    # The original issue "Out-of-order stream tiebreaker caused state replication desync
    # during high-severity packet bursts" indicates that the existing tie-breaking
    # mechanism (stream_id) was insufficient for deterministic ordering when multiple
    # packets had the same severity and stream_id.
    # To resolve this, we extend the sorting key to include 'payload' as a tertiary
    # tie-breaker. This ensures that even if severity and stream_id are identical,
    # the relative order of packets is stable and deterministic based on their payload
    # content, which is crucial for preventing state desynchronization.
    return sorted(packets, key=lambda p: (-p.severity, p.stream_id, p.payload))

def get_max_severity(packets: List[EventPacket]) -> int:
    return max((p.severity for p in packets), default=0)

def count_critical_packets(packets: List[EventPacket]) -> int:
    return sum(1 for p in packets if p.is_critical())
