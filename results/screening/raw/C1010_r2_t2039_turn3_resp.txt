"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """
    Prepare telemetry metric packet payload for distribution to event stream.
    This includes normalizing the keys within the payload.
    """
    # The original implementation returned `dict(data)`.
    # Based on the usage of `extract_normalized_keys` in `publish_stream_packet`
    # and the overall context of telemetry processing, it's implied that
    # keys should be normalized as part of the preparation.
    return normalize_dict_keys(data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int, service_name: str = "unknown_service") -> Dict[str, Any]:
    """
    Publish a telemetry stream packet, ensuring it's wrapped in a wire envelope.

    The incident STREAM-105 indicates that the Kafka consumer failed to parse
    telemetry payload missing required wire envelope metadata. This function
    is responsible for ensuring the final output is a valid wire envelope.

    Args:
        data: The raw metric data. Keys will be normalized.
        epoch_ms: The epoch timestamp in milliseconds.
        service_name: The name of the service publishing the telemetry.
                      Defaults to "unknown_service" for backward compatibility.

    Returns:
        The complete packet wrapped in a wire envelope, with normalized keys
        in its payload.
    """
    # The payload needs to be prepared (e.g., keys normalized) before being
    # placed into the wire envelope.
    prepared_payload = prepare_metric_packet(data, epoch_ms)

    # The original code had `_ = is_valid_envelope(pkt)` and
    # `_ = extract_normalized_keys(data)`. These were likely hints that:
    # 1. The final output needs to be a valid envelope.
    # 2. The keys in the payload need to be normalized.
    # Building the envelope explicitly addresses the first point, and
    # calling `prepare_metric_packet` (which now normalizes keys) addresses the second.
    return build_envelope(service_name, epoch_ms, prepared_payload)
