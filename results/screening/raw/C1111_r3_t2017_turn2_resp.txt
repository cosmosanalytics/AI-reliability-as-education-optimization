"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    policy_segments = policy.allowed_pattern.split('::')
    request_segments = clean_req.split('::')

    i = 0
    j = 0
    while i < len(policy_segments) and j < len(request_segments):
        policy_segment = policy_segments[i]
        request_segment = request_segments[j]

        if policy_segment == '*':
            # '*' matches exactly one segment
            i += 1
            j += 1
        elif policy_segment == '**':
            # '**' matches arbitrary multi-segment prefixes (zero or more segments)
            # If '**' is the last policy segment, it matches the rest of the request scope.
            return True
        elif policy_segment == request_segment:
            # Exact segment match
            i += 1
            j += 1
        else:
            # Mismatch
            return False

    # After loop, check if both patterns are exhausted
    # If policy has more segments, it means the request was too short to match.
    # The only exception is if the remaining policy segment is '**', which means it matched everything.
    if i < len(policy_segments):
        return policy_segments[i] == '**'
    
    # If request has more segments, it means the policy was too short to match.
    if j < len(request_segments):
        return False
    
    # Both exhausted, perfect match
    return True

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
