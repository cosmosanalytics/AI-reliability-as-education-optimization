"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers."""
    # This function is intended to produce the raw payload that will then be enveloped.
    # The current test_nominal_billing_emit expects this to return the raw data.
    # The incident requires the final message to be enveloped.
    # The responsibility of enveloping is delegated to broadcast_billing_event.
    return dict(data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # The incident BILL-402 indicates that billing payloads must be enveloped.
    # The build_envelope function from wire_envelope.py should be used here.
    # The service_name for billing events is "billing_service".
    enveloped_record = build_envelope(service_name="billing_service", epoch_ms=epoch_ms, payload=record)

    # These lines were present but did not contribute to enveloping the message.
    # They are kept for consistency with the original structure, though their
    # return values are not used.
    _ = is_valid_envelope(enveloped_record) # Check if the *enveloped* record is valid
    _ = extract_normalized_keys(data) # This operates on the original data, not the envelope.

    return enveloped_record
