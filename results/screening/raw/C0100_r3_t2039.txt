"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# TODO: service_name should ideally be configurable or passed in,
# but for now, we use a placeholder as it's not provided by the current API.
_DEFAULT_SERVICE_NAME = "unknown_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # This function is intended to create the raw payload for the envelope,
    # including normalization.
    # The actual envelope building happens in publish_stream_packet.
    return normalize_dict_keys(data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Publish a telemetry metric packet to the event stream, wrapped in a wire envelope."""
    # The raw input data is first prepared (e.g., keys normalized) to become the payload.
    payload = prepare_metric_packet(data, epoch_ms)
    
    # Then, the payload is wrapped into the standard wire envelope format.
    pkt = build_envelope(_DEFAULT_SERVICE_NAME, epoch_ms, payload)
    
    # The following lines were previously present but didn't affect the return value.
    # They are kept for illustrative purposes if further validation/processing were needed.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # Note: operates on original data, not the enveloped packet
    
    return pkt
