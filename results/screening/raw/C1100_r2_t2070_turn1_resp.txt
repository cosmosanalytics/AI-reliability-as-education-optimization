"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values using the nearest-rank method with linear interpolation.

    Specifically, it uses the formula:
    index = p/100 * (n + 1)
    where n is the number of values.
    The value is then interpolated between the two nearest ranks.
    This ensures that for p=0, the minimum value is returned, and for p=100, the maximum value is returned.
    """
    if not values:
        raise ValueError("Empty values")
    if not (0 <= p <= 100):
        raise ValueError("Percentile p must be between 0 and 100 inclusive")

    s = sorted(values)
    n = len(s)

    # For p=0, return the minimum value
    if p == 0:
        return float(s[0])
    # For p=100, return the maximum value
    if p == 100:
        return float(s[n-1])

    # Calculate rank using (p/100) * (n + 1) formula.
    # This is a common definition for percentiles where the index is calculated
    # based on the (n+1) positions, and then interpolated.
    # This matches the behavior of some statistical software for percentiles.
    rank = (p / 100.0) * (n + 1)

    # The rank is 1-based, convert to 0-based index for list access
    idx = rank - 1

    low_idx = int(math.floor(idx))
    high_idx = int(math.ceil(idx))

    # Handle edge cases for index bounds
    # If rank is very close to 1, low_idx can be -1.
    # If rank is very close to n+1, high_idx can be n.
    # Adjust to stay within valid list indices [0, n-1]
    low_idx = max(0, low_idx)
    high_idx = min(n - 1, high_idx)

    # If low_idx and high_idx are the same (e.g., integer rank), no interpolation needed
    if low_idx == high_idx:
        return float(s[low_idx])

    # Linear interpolation
    frac = idx - low_idx
    return float(s[low_idx] + frac * (s[high_idx] - s[low_idx]))
