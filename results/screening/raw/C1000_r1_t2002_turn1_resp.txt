"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.

    The total discount is distributed such that the sum of allocated discounts
    equals total_discount_cents. Pennies are distributed one by one to lines
    that received the largest fractional discounts, to ensure full allocation
    and fair distribution.

    line_totals: list of integer cents.
    total_discount_cents: total discount amount in cents.
    Returns list of discount cents per line.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")

    num_lines = len(line_totals)
    if num_lines == 0:
        return []

    total_line_value = sum(line_totals)
    if total_line_value == 0:
        # If line totals are all zero, distribute discount evenly or all to first line.
        # For proportional distribution, 0 total means no basis for proportion.
        # Distribute 0 to all lines and handle remainder if any.
        return [0] * num_lines

    # Calculate base allocation for each line using float division to preserve fractions
    # and then truncate to integer cents.
    # Store the fractional part to determine priority for remainder distribution.
    allocations_with_fractions = []
    for cents in line_totals:
        if total_line_value > 0:
            fractional_alloc = (cents * total_discount_cents) / total_line_value
        else:
            fractional_alloc = 0.0 # Should not happen due to total_line_value check above
        allocations_with_fractions.append(fractional_alloc)

    # Calculate integer part of allocations
    allocated_cents = [int(alloc) for alloc in allocations_with_fractions]

    # Calculate the sum of currently allocated cents
    current_sum = sum(allocated_cents)

    # Calculate the remainder that needs to be distributed
    remainder = total_discount_cents - current_sum

    # Distribute the remainder one cent at a time to lines that had the largest fractional parts
    # To do this, we sort by the descending order of fractional parts (alloc - int(alloc))
    # Create a list of (fractional_part, original_index) tuples
    fractional_parts_with_indices = []
    for i, alloc_float in enumerate(allocations_with_fractions):
        fractional_parts_with_indices.append((alloc_float - allocated_cents[i], i))

    # Sort in descending order of fractional part
    fractional_parts_with_indices.sort(key=lambda x: x[0], reverse=True)

    # Distribute remainder
    for i in range(remainder):
        if i < num_lines: # Ensure we don't try to add to more lines than available
            original_index = fractional_parts_with_indices[i][1]
            allocated_cents[original_index] += 1
        else:
            # If remainder is larger than number of lines, distribute cyclically
            # or add to the first few lines again. For simplicity and to match
            # common rounding rules, we distribute to the lines with largest fractions first.
            # If remainder > num_lines, this implies each line would get at least one extra cent
            # from a previous pass over this loop, and we continue from the beginning of the sorted list.
            # This handles cases like total_discount_cents = 25, line_totals = [100, 100], expected [13, 12]
            # or total_discount_cents = 26, line_totals = [100, 100], expected [13, 13]
            original_index = fractional_parts_with_indices[i % num_lines][1]
            allocated_cents[original_index] += 1


    return allocated_cents

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
