"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # The issue indicates missing wire envelope metadata.
    # The build_envelope function requires a service_name.
    # Assuming a default or placeholder service name for telemetry packets
    # as no specific service name is provided in the issue for this function.
    service_name = "telemetry_service"
    return build_envelope(service_name, epoch_ms, data)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # The original code called is_valid_envelope and extract_normalized_keys
    # but discarded their return values. These calls are kept for consistency
    # with the original intent, even if their results are not explicitly used
    # in the final return value of this function.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # This operates on the original data, not the packet.
    return pkt
