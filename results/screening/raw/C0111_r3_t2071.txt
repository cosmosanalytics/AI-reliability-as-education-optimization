"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses."""
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)

    # Handle edge cases for p=0 and p=100 explicitly to ensure correct min/max behavior.
    if p == 0.0:
        return float(s[0])
    if p == 100.0:
        return float(s[n - 1])

    # Calculate index using the formula for inclusive percentiles (Type 7 or R-7 method),
    # which is common for telemetry reporting and statistical software.
    # This formula finds the smallest value in the dataset that is greater than
    # or equal to p percent of the values.
    # idx = ceil((p / 100.0) * n) - 1 for 0-based indexing.
    idx = math.ceil((p / 100.0) * n) - 1
    
    # Ensure idx is within valid bounds [0, n-1] for robustness,
    # especially for cases where 'p' might be slightly out of the [0, 100] range
    # or due to floating point inaccuracies for intermediate 'p' values.
    idx = max(0, min(idx, n - 1))
    
    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
