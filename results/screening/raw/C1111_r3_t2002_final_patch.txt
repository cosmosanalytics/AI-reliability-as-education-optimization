"""Invoice discount allocation module."""
from typing import List
from models import AllocationResult
from utils import validate_line_totals

def prorate_discount(line_totals: List[int], total_discount_cents: int) -> List[int]:
    """Prorate discount across lines proportionally to their value.
    line_totals: list of integer cents.
    Returns list of discount cents per line.
    The sum of returned discounts must equal total_discount_cents, with any
    remainder distributed to the lines with the largest proportional discount amounts.
    """
    if not validate_line_totals(line_totals):
        raise ValueError("Line totals must be non-negative integers")
    
    num_lines = len(line_totals)
    if num_lines == 0:
        return []

    total_line_value = sum(line_totals)
    if total_line_value == 0:
        return [0] * num_lines

    # Calculate base allocations using integer division, as per the original logic,
    # but store the full fractional values for remainder distribution.
    # To avoid floating point issues, we work with integer numerators and denominators.
    
    base_allocations = []
    # Store (remainder_numerator, original_index) for distributing extra pennies
    # remainder_numerator = (line_total * total_discount_cents) % total_line_value
    remainder_tracking = []

    for i, cents in enumerate(line_totals):
        # Calculate the proportional share using integer arithmetic
        # This is `(cents * total_discount_cents) / total_line_value`
        # We need the floor for the base allocation
        base_share = (cents * total_discount_cents) // total_line_value
        base_allocations.append(base_share)
        
        # The "fractional part" can be represented by the remainder of the integer division.
        # We want to give extra cents to lines with the largest positive remainder.
        remainder_num = (cents * total_discount_cents) % total_line_value
        remainder_tracking.append((remainder_num, i))

    current_total_allocated = sum(base_allocations)
    remainder_to_distribute = total_discount_cents - current_total_allocated

    # Distribute the remainder
    if remainder_to_distribute > 0:
        # Sort by remainder_num in descending order.
        # For ties in remainder_num, sort by original_index in ascending order for deterministic behavior.
        remainder_tracking.sort(key=lambda x: (-x[0], x[1]))

        # Add one cent to the lines with the largest remainders, up to `remainder_to_distribute` times
        for i in range(remainder_to_distribute):
            original_index = remainder_tracking[i][1]
            base_allocations[original_index] += 1

    return base_allocations

def allocate_discounts(line_totals: List[int], total_discount_cents: int) -> AllocationResult:
    """Wrapper returning structured allocation result."""
    allocs = prorate_discount(line_totals, total_discount_cents)
    tot = sum(allocs)
    # The prorate_discount function now guarantees tot == total_discount_cents
    return AllocationResult(discounts=allocs, total_allocated=tot, is_fully_allocated=(tot == total_discount_cents))
