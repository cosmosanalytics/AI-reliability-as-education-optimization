"""Low-level jitter percentile calculation engine."""
import math
from typing import List

def evaluate_jitter_percentile(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using nearest-rank estimation.

    The index k for the sorted list is calculated as:
    k = math.ceil(p / 100 * n) - 1, where n is the number of samples.
    The result is s[k].

    Args:
        values: A list of float jitter values.
        p: The percentile to calculate (e.g., 90 for 90th percentile).

    Returns:
        The calculated percentile value.

    Raises:
        ValueError: If the input 'values' list is empty.
        IndexError: If 'p' is outside the valid range [0, 100].
    """
    if not values:
        raise ValueError("No jitter values")
    if not (0 <= p <= 100):
        # This could also be a ValueError, but IndexError is more specific for array access issues
        raise IndexError("Percentile 'p' must be between 0 and 100 inclusive.")

    s = sorted(values)
    n = len(s)

    # Nearest-rank percentile estimation: index k = math.ceil(p / 100 * n) - 1
    # Handle p=0 explicitly to return the minimum value (s[0])
    if p == 0:
        k = 0
    else:
        # For p > 0, calculate index using the specified formula
        # math.ceil(p / 100 * n) could be 0 if p is very small and n is also small,
        # e.g., p=0.01, n=10 => p/100*n = 0.001 => ceil(0.001) = 1.
        # The result of the formula must be at least 1 to yield index 0 for s[0].
        # The formula as stated in the issue: k = math.ceil(p / 100 * n) - 1
        # This gives a 0-based index.
        k_one_based = math.ceil((p / 100.0) * n)
        k = int(k_one_based) - 1

    # Ensure k is within valid bounds [0, n-1]
    # The formula k = math.ceil(p / 100 * n) - 1 naturally handles 100th percentile to n-1.
    # For very small p, k could be -1 if ceil(p/100*n) was 0.
    # The explicit p=0 check ensures k=0 for 0th percentile.
    # For p > 0, k_one_based will be at least 1, so k will be at least 0.
    # k will be at most n-1 for p=100.
    k = max(0, min(k, n - 1))

    return float(s[k])
