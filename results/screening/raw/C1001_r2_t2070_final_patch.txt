"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    For SLA reporting, if the calculated rank for a percentile falls between
    two values, the value at the lower rank is returned without linear
    interpolation. This ensures a conservative percentile estimate for
    performance metrics, preventing upward interpolation.
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)

    idx = (p / 100.0) * (n - 1)
    low = int(math.floor(idx))
    high = int(math.ceil(idx))
    frac = idx - low

    if frac > 0:
        # As per SLA requirements, if the percentile index is not an integer,
        # return the value at the lower rank (floor) without interpolation.
        # This provides a more conservative estimate for performance metrics.
        return float(s[low])
    else:
        # If the percentile index is an exact rank (frac == 0),
        # or if there's no higher element to interpolate with (e.g., p=100, low=n-1),
        # the standard interpolation formula correctly yields s[low].
        return float(s[low] + frac * (s[high] - s[low]))
