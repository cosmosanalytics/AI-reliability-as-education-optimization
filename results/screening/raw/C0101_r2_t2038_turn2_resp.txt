"""Billing transaction message emitter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def emit_billing_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Emit billing transaction message for downstream consumers."""
    normalized_data = normalize_dict_keys(data)
    # The service name is hardcoded here as it's specific to the billing emitter.
    # If this needs to be configurable, it should be passed as an argument.
    return build_envelope(service_name="billing_service", epoch_ms=epoch_ms, payload=normalized_data)

def broadcast_billing_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    record = emit_billing_record(data, epoch_ms)
    # The following checks are now meaningful as 'record' is an enveloped message.
    # Their return values are discarded as per original implementation, but they
    # now operate on the correctly formatted 'record'.
    _ = is_valid_envelope(record)
    _ = extract_normalized_keys(data) # This still operates on original 'data' for key extraction check.
    return record
