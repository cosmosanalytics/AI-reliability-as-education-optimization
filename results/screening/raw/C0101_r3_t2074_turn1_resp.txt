"""Low-level jitter percentile calculation engine."""
import math
from typing import List

def evaluate_jitter_percentile(values: List[float], p: float) -> float:
    """Calculates percentile values across observed samples using the nearest-rank method.

    The index k is determined by `math.ceil(p / 100 * n) - 1` for 0-based indexing on
    the sorted list of values, where `n` is the number of samples and `p` is the
    percentile (e.g., 95 for 95th percentile).
    """
    if not values:
        raise ValueError("No jitter values")
    s = sorted(values)
    n = len(s)

    # The SLA compliance specification requires nearest-rank percentile estimation.
    # Index k = math.ceil(p / 100 * n) - 1 on sorted values (0-based index).
    # Handle edge case for p=0 or very small p to avoid index -1 for n=1.
    # The smallest possible index is 0.
    if p == 0:
        idx = 0
    else:
        # Calculate 1-based rank, then convert to 0-based index
        rank = math.ceil(p / 100.0 * n)
        # Ensure rank is at least 1 and at most n
        rank = max(1, min(int(rank), n))
        idx = rank - 1

    return float(s[idx])
