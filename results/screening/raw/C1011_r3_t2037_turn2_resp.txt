"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys
from wire_envelope import build_envelope
from typing import Dict, Any

# Define a default service name as it's not provided in the function signature.
# This assumes a system-wide default or that 'audit' is appropriate for audit events.
# If the service name needs to be dynamic, the function signature would need to be updated,
# but the current ticket asks for the smallest change to fix the envelope.
DEFAULT_SERVICE_NAME = "audit"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(DEFAULT_SERVICE_NAME, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # The original implementation called is_valid_envelope and extract_normalized_keys
    # but their results were ignored. These lines are not essential for the fix
    # and removing them makes serialize_audit_event a simple pass-through to format_audit_event,
    # which is consistent with the existing test_nominal_audit_execution.
    return msg
