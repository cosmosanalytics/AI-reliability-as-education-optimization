"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Define a default service name for cases where it's not explicitly provided,
# such as in existing tests that cannot be modified.
DEFAULT_SERVICE_NAME = "unknown_service"

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int, service_name: str = DEFAULT_SERVICE_NAME) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream.
    
    The `service_name` parameter is made optional with a default to maintain
    compatibility with existing tests that do not pass this argument,
    while ensuring the wire envelope is correctly built with a source.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name, epoch_ms, normalized_payload)

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int, service_name: str = DEFAULT_SERVICE_NAME) -> Dict[str, Any]:
    """Publish a telemetry stream packet.
    
    The `service_name` parameter is made optional with a default to maintain
    compatibility with existing tests that do not pass this argument.
    """
    pkt = prepare_metric_packet(data, epoch_ms, service_name)
    # The following lines were likely intended for validation or side effects,
    # but their return values were not used.
    # For now, we ensure the packet is a valid envelope as per the incident.
    _ = is_valid_envelope(pkt)
    _ = extract_normalized_keys(data) # Note: This extracts keys from the *original* data, not the normalized payload within the envelope.
    return pkt
