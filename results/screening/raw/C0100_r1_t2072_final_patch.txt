"""Audio buffer level metric processor."""
import math
from typing import List, Dict, Any
from audio_types import AudioStreamSnapshot, BufferSample
from audio_formatter import format_buffer_stat

def compute_buffer_percentile(snapshot: AudioStreamSnapshot, p: float) -> float:
    """Compute percentile using continuous linear interpolation (numpy-style: index = (p / 100.0) * (len(samples) - 1)).
    Do not use nearest-rank method for audio buffer metrics.
    """
    vals = snapshot.ratios()
    if not vals:
        raise ValueError("Empty snapshot")
    
    if not (0 <= p <= 100):
        raise ValueError("Percentile 'p' must be between 0 and 100, inclusive.")

    s = sorted(vals)
    n = len(s)

    if n == 1:
        return float(s[0])

    # Calculate the index using continuous linear interpolation
    # index = (p / 100.0) * (len(samples) - 1)
    index = (p / 100.0) * (n - 1)

    # The result is a weighted average of the two surrounding values
    # result = s[floor(index)] * (1 - (index % 1)) + s[ceil(index)] * (index % 1)
    
    # floor_idx will be the lower index
    floor_idx = math.floor(index)
    
    # If index is an integer, the percentile is exactly s[floor_idx]
    if index == floor_idx:
        return float(s[int(index)])

    # ceil_idx will be the upper index
    ceil_idx = math.ceil(index)

    # The fractional part of the index is the weight for the upper value
    weight = index - floor_idx

    # Perform linear interpolation
    interpolated_value = s[int(floor_idx)] * (1.0 - weight) + s[int(ceil_idx)] * weight
    
    return float(interpolated_value)

def evaluate_audio_buffer(snapshot: AudioStreamSnapshot, p: float) -> Dict[str, Any]:
    val = compute_buffer_percentile(snapshot, p)
    return format_buffer_stat(snapshot.channel_count(), val)
