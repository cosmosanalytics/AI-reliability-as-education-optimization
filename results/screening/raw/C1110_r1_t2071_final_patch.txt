import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses."""
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)
    
    # Calculate the rank (0-indexed position) for the percentile.
    # The original formula `int(round((p / 100.0) * (n - 1)))` uses "round half to even"
    # which can lead to inconsistencies for values exactly halfway between two integers.
    # For statistical percentiles, it's often desired to round half up (e.g., 0.5 rounds to 1).
    # This change specifically addresses scenarios where the original calculation
    # might round down, leading to a lower-than-expected percentile value,
    # as implied by the "assert 20.0 == 30.0" traceback.
    # We use `math.floor(x + 0.5)` to achieve "round half up" behavior.
    
    # Calculate the fractional index. For p=0, index is 0. For p=100, index is n-1.
    # This means the multiplier should be `(n-1)`.
    fractional_idx = (p / 100.0) * (n - 1)
    
    # Apply "round half up" logic.
    idx = int(math.floor(fractional_idx + 0.5))
    
    # Ensure the index is within valid bounds [0, n-1].
    # This is implicitly handled by the (n-1) multiplier for p in [0, 100].
    # For p=0, fractional_idx=0, idx=0.
    # For p=100, fractional_idx=n-1, idx=n-1.
    # For intermediate values, it will be within [0, n-1].
    
    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)

import math

from rpc_data import RpcSeries, RpcSample

    # index is exactly halfway between two integers (e.g., 0.5, 1.5, 2.5).

    # the index was rounded down when it should have effectively rounded up,
    # leading to a lower percentile value being selected.

    # where half values are typically rounded up (e.g., 0.5 rounds to 1),
    # we replace `round()` with a manual "round half up" implementation: `math.floor(x + 0.5)`.
    # This ensures that `X.5` always rounds to `X+1` (for positive X).
    
    # Calculate the floating-point index based on the (N-1) method, where
    # p=0 maps to index 0 and p=100 maps to index N-1.
    fractional_idx = (p / 100.0) * (n - 1)
    
    # Apply "round half up" logic and convert to integer.
    # For example:
    #   fractional_idx = 0.0 -> floor(0.0 + 0.5) = floor(0.5) = 0
    #   fractional_idx = 0.4 -> floor(0.4 + 0.5) = floor(0.9) = 0
    #   fractional_idx = 0.5 -> floor(0.5 + 0.5) = floor(1.0) = 1 (original round(0.5) is 0)
    #   fractional_idx = 0.6 -> floor(0.6 + 0.5) = floor(1.1) = 1
    #   fractional_idx = 1.5 -> floor(1.5 + 0.5) = floor(2.0) = 2 (original round(1.5) is 2)
    #   fractional_idx = 2.5 -> floor(2.5 + 0.5) = floor(3.0) = 3 (original round(2.5) is 2)
    idx = int(math.floor(fractional_idx + 0.5))
    
    # The index `idx` will naturally be within [0, n-1] for p in [0, 100]
    # because `fractional_idx` is in [0, n-1].
    # For p=0, fractional_idx=0, idx=0.
    # For p=100, fractional_idx=n-1, idx=n-1.
    
    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
