"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    pattern = policy.allowed_pattern

    if pattern == "**":
        return True # Universal wildcard

    pattern_segments = pattern.split('::')
    request_segments = clean_req.split('::')

    if not pattern_segments:
        return False # Empty pattern is invalid

    # Handle the special case of '::' at the end of the pattern
    # e.g., "corp::reports::" should behave like "corp::reports"
    if pattern.endswith('::'):
        pattern_segments = pattern_segments[:-1]

    # Special handling for single-segment patterns like "corp" matching "corp::*"
    # This is implicitly handled by segment-by-segment comparison below,
    # as long as the pattern does not end in a wildcard.

    # Check for exact match or suffix wildcard
    if pattern_segments[-1] == "**":
        # Match if the request starts with the prefix before '**'
        # e.g., "corp::reports::**" matches "corp::reports", "corp::reports::summary", "corp::reports::summary::detail"
        prefix_len = len(pattern_segments) - 1
        if len(request_segments) < prefix_len:
            return False
        for i in range(prefix_len):
            if pattern_segments[i] != request_segments[i]:
                return False
        return True
    elif pattern_segments[-1] == "*":
        # Match if the request has the same number of segments and the last segment is non-empty
        # and matches the prefix, e.g., "corp::reports::*" matches "corp::reports::summary"
        if len(pattern_segments) != len(request_segments):
            return False
        for i in range(len(pattern_segments) - 1): # Compare all but the last segment
            if pattern_segments[i] != request_segments[i]:
                return False
        # The last segment must exist and not be empty for '*' to match
        return bool(request_segments[-1])
    else:
        # No wildcard at the end, requires exact match
        # e.g., "corp::reports" matches only "corp::reports", not "corp::reports_secret" or "corp::reports::summary"
        return clean_req == pattern


def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
