"""Percentile estimation engine for database telemetry."""
import math
from typing import List

def calculate_percentile(values: List[float], p: float) -> float:
    """Calculates percentile across sample values.

    Uses a definition where the index is calculated as `p / 100.0 * n`,
    and values are linearly interpolated. This corresponds to the
    "R-5" method or NIST method for percentile calculation, which is
    common in statistical reporting for SLA.
    """
    if not values:
        raise ValueError("Empty values")
    s = sorted(values)
    n = len(s)

    # The original implementation used (p / 100.0) * (n - 1), which is
    # a different percentile definition (e.g., R-7 in R's quantile).
    # For SLA reporting, a common definition is to use index (p / 100.0) * n,
    # often referred to as the "NIST" method or "R-5" method in R.
    # This change addresses INC-7102 by aligning with the expected SLA calculation.
    idx = (p / 100.0) * n

    if idx <= 0:
        return s[0]
    if idx >= n:
        return s[n - 1]

    # Convert 1-based index to 0-based
    idx -= 1

    low = int(math.floor(idx))
    high = int(math.ceil(idx))
    frac = idx - low

    if low == high:
        return float(s[low])
    
    return float(s[low] + frac * (s[high] - s[low]))
