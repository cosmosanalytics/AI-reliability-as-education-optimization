"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    allowed_pattern = policy.allowed_pattern

    if allowed_pattern == clean_req:
        return True # Exact match for simple cases like 'corp::reports' == 'corp::reports'

    pattern_parts = allowed_pattern.split('::')
    request_parts = clean_req.split('::')

    i = 0 # index for pattern_parts
    j = 0 # index for request_parts

    while i < len(pattern_parts) and j < len(request_parts):
        pattern_part = pattern_parts[i]
        request_part = request_parts[j]

        if pattern_part == "**":
            # '**' matches arbitrary multi-segment prefixes.
            # If it's the last part of the pattern, it matches any remaining request segments (including none).
            if i == len(pattern_parts) - 1:
                return True
            # If not the last part, it means '**' matches some segments, and the next pattern_part must match
            # a segment later in the request. This is a greedy match.
            next_pattern_part = pattern_parts[i + 1]
            found = False
            for k in range(j, len(request_parts)):
                if is_scope_permitted(RoutePolicy("", "::".join(pattern_parts[i+1:])), "::".join(request_parts[k:])):
                    return True
            return False # No match found after '**'
        elif pattern_part == "*":
            # '*' matches exactly one segment.
            # An empty segment doesn't match '*'.
            if not request_part:
                return False
            i += 1
            j += 1
        elif pattern_part == request_part:
            # Exact segment match.
            i += 1
            j += 1
        else:
            # Segments don't match.
            return False

    # After iterating, both indices should be at the end for a full match,
    # unless the pattern ended with a '**'.
    if i == len(pattern_parts) and j == len(request_parts):
        return True
    elif i == len(pattern_parts) - 1 and pattern_parts[i] == "**":
        # Pattern ends with '**', it matches any remaining segments in request (including none).
        return True
    else:
        # Mismatch in length or remaining pattern parts without a wildcard match.
        return False


def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
