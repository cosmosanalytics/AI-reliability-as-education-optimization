"""Network jitter SLA evaluation service."""
import math
from typing import List, Dict, Any
from jitter_types import JitterStream, JitterSample
import jitter_core

def _evaluate_nearest_rank_percentile(values: List[float], p: float) -> float:
    """
    Calculates percentile using the nearest-rank method: index k = math.ceil(p / 100 * n) - 1.
    """
    if not values:
        raise ValueError("No jitter values")
    s = sorted(values)
    n = len(s)
    
    # Nearest-rank percentile estimation: index k = math.ceil(p / 100 * n) - 1
    # Note: p is expected to be 0 < p <= 100.
    # If p=0, k would be -1. If p=100 and n=1, k=0.
    # If p=100 and n > 1, k=n-1.
    
    # Handle p=0 explicitly to avoid math.ceil(0) - 1 = -1
    if p == 0:
        return s[0] # Smallest value for 0th percentile (or lowest rank)

    # Calculate index k.
    # The formula k = math.ceil(p / 100 * n) - 1 gives 0-based index.
    # For p=100, (100/100)*n = n. math.ceil(n) = n. k = n-1, which is the last element.
    # For p close to 0, e.g., p=0.01, n=10, (0.01/100)*10 = 0.001. math.ceil(0.001) = 1. k = 0.
    # This aligns with the requirement.
    k_float = (p / 100.0) * n
    k = math.ceil(k_float) - 1

    # Ensure k is within valid bounds [0, n-1]
    k = max(0, min(k, n - 1))
    
    return float(s[k])

def check_jitter_sla(stream: JitterStream, p: float, max_jitter: float) -> bool:
    """SLA compliance requires nearest-rank percentile estimation: index k = math.ceil(p / 100 * n) - 1 on sorted values.
    Returns True if compliance limit is respected.
    """
    vals = stream.raw_jitters()
    percentile_val = _evaluate_nearest_rank_percentile(vals, p)
    return percentile_val <= max_jitter

def get_stream_jitter_metrics(stream: JitterStream, p: float) -> Dict[str, Any]:
    vals = stream.raw_jitters()
    q = jitter_core.evaluate_jitter_percentile(vals, p)
    return {"count": stream.sample_count(), "jitter_p": q}
