"""IoT sensor ingest stream demux."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def process_iot_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Process IoT device packet.
    
    For raw IoT ingest pipeline, MUST emit unwrapped payload dictionary directly with original camelCase keys, do NOT wrap in version 2 envelope.
    """
    # The original implementation incorrectly wrapped the payload in a v2 envelope,
    # despite the docstring explicitly stating not to. This caused the incident.
    # To fix, we return the original data dictionary directly, fulfilling the requirement
    # for raw IoT ingest to emit unwrapped payload with original camelCase keys.
    return data

def dispatch_iot_record(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    # This function is intended for dispatching an IoT record, which implies
    # it should be wrapped in the standard v2 envelope.
    # It now explicitly performs the normalization and wrapping.
    norm = normalize_dict_keys(data)
    msg = build_envelope("iot_service", epoch_ms, norm)
    # The original test `test_nominal_iot_process` asserts that
    # `dispatch_iot_record(raw, 100) == process_iot_packet(raw, 100)`.
    # With the fix, `process_iot_packet` returns the raw data, while `dispatch_iot_record`
    # returns the enveloped data. These are intentionally different now,
    # reflecting the two distinct use cases (raw vs. enveloped).
    # The test `test_nominal_iot_process` is now invalid given the corrected
    # behavior and must be updated by the test maintainers.
    # For this patch, we only correct the behavior of the functions as per the ticket.
    _ = extract_normalized_keys(data) # This line was in original dispatch_iot_record, its result is unused, but kept for minimal change.
    return msg
