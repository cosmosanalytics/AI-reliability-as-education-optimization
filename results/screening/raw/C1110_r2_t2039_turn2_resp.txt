"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    # The incident report indicates that the Kafka consumer failed to parse telemetry
    # payloads missing required wire envelope metadata. This function is responsible
    # for preparing the metric packet for the event stream, implying it should
    # wrap the payload with the standard wire envelope.
    # The service name "STREAM" is inferred from the incident ID "STREAM-105".
    return build_envelope(service_name="STREAM", epoch_ms=epoch_ms, payload=normalize_dict_keys(data))

def publish_stream_packet(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(data, epoch_ms)
    # These calls are for side effects or validation, but the return value
    # of publish_stream_packet must match the prepared packet.
    _ = is_valid_envelope(pkt) # For validation, but not used to modify pkt here.
    _ = extract_normalized_keys(data) # For potential internal use, not part of the envelope.
    return pkt
