"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses.
    Uses the Nearest Rank method (Type 1) where the index is calculated as
    ceil((p / 100.0) * n) - 1. This aligns with common interpretations for
    discrete data where a specific observation is chosen.
    """
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)
    # Using the "nearest rank" method (Type 1 in R's quantile function)
    # The index is 1-based in the formula, so we subtract 1 for 0-based indexing.
    # index = ceil((p / 100.0) * n)
    # 0-based index = int(math.ceil((p / 100.0) * n)) - 1
    # Ensure index is within bounds [0, n-1]
    idx = max(0, min(n - 1, int(math.ceil((p / 100.0) * n)) - 1))
    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
