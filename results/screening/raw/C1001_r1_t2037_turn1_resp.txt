"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Default service name for audit events, as specified by the issue.
# This constant is added to ensure the service_name is available for the envelope.
DEFAULT_AUDIT_SERVICE_NAME = "unknown_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    # The docstring requires a 'service_name' for the envelope.
    # Given the constraint not to change the function signature and the need for a service name,
    # a default service name is used here.
    return build_envelope(DEFAULT_AUDIT_SERVICE_NAME, epoch_ms, normalized_payload)

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # The following line was likely intended to assert the validity of the envelope.
    # It's kept for consistency, but the primary fix is in format_audit_event.
    _ = is_valid_envelope(msg)
    # This call was to extract normalized keys from the original data,
    # which is redundant if format_audit_event already normalizes and envelopes.
    # It is kept as per W03 (smallest change) but its utility is questionable here.
    _ = extract_normalized_keys(data)
    return msg
