"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses."""
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)
    # Calculate the index using a method that rounds .5 up,
    # which is a common expectation for percentile calculations
    # and addresses potential inconsistencies with Python's default round() behavior
    # (round half to even).
    # The formula (p / 100.0) * (n - 1) is a common way to get a 0-based fractional index.
    # Adding 0.5 and then floor() ensures that .5 values are rounded up.
    fractional_idx = (p / 100.0) * (n - 1)
    idx = int(math.floor(fractional_idx + 0.5))
    return float(s[idx])

    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)

=== FILE: rpc_calc.py ===
"""Microservice RPC duration percentile calculator."""
import math
from typing import List, Dict, Any
from rpc_data import RpcSeries, RpcSample
from rpc_formatter import format_rpc_stat

def compute_rpc_percentile(series: RpcSeries, p: float) -> float:
    """Calculates SLA percentile for RPC responses.

    This function uses a nearest-rank method where fractional ranks are rounded
    to the nearest integer rank. Specifically, it uses a rounding method that
    rounds .5 values up, which is a common convention in statistics and helps
    resolve inconsistencies that can arise from Python's default round() behavior
    (which rounds half to even).
    """
    vals = series.durations()
    if not vals:
        raise ValueError("No RPC samples available")
    s = sorted(vals)
    n = len(s)

    # Calculate the 0-based fractional index. The (n - 1) term ensures that
    # the 0th percentile maps to the first element (index 0) and the 100th
    # percentile maps to the last element (index n-1).
    fractional_idx = (p / 100.0) * (n - 1)

    # Round the fractional index to the nearest integer.
    # By adding 0.5 and then taking the floor, we ensure that .5 values
    # (e.g., 0.5, 1.5, 2.5) are consistently rounded up to the next integer
    # (e.g., 1, 2, 3). This behavior is often desired for percentile calculations.
    idx = int(math.floor(fractional_idx + 0.5))

    return float(s[idx])

def build_rpc_report(service_name: str, series: RpcSeries, p: float) -> Dict[str, Any]:
    val = compute_rpc_percentile(series, p)
    return format_rpc_stat(service_name, series.count(), val)
