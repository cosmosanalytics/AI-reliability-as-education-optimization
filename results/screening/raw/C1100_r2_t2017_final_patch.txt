"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    pattern = policy.allowed_pattern

    if pattern == "**":
        return True

    pattern_parts = pattern.split('::')
    request_parts = clean_req.split('::')

    # Handle single segment wildcard '*'
    if pattern_parts[-1] == '*':
        if len(request_parts) != len(pattern_parts):
            return False
        for i in range(len(pattern_parts) - 1):
            if pattern_parts[i] != request_parts[i]:
                return False
        return True

    # Handle multi-segment wildcard '**'
    if pattern_parts[-1] == '**':
        # Remove the '**' from the pattern for comparison
        prefix_pattern_parts = pattern_parts[:-1]
        if len(request_parts) < len(prefix_pattern_parts):
            return False
        for i in range(len(prefix_pattern_parts)):
            if prefix_pattern_parts[i] != request_parts[i]:
                return False
        return True
    
    # No wildcards: pattern must be an exact match or a segment-boundary prefix
    if clean_req == pattern:
        return True
    
    # Check if the request scope is a child path of the policy pattern
    # e.g., policy="corp::reports", request="corp::reports::detail"
    if clean_req.startswith(pattern + "::"):
        return True

    return False

def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
