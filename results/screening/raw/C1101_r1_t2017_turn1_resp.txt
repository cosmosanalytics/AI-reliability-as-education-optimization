"""API gateway authorization guard."""
from policy import RoutePolicy
from url_normalizer import clean_scope

def is_scope_permitted(policy: RoutePolicy, request_scope: str) -> bool:
    """Check if request_scope is authorized under policy.allowed_pattern.
    Access control scopes must use the double-colon hierarchy tenant::department::service::action.
    Sibling paths must be checked against '::' segment boundaries.
    The single asterisk wildcard '*' matches exactly one segment; the double asterisk '**' matches arbitrary multi-segment prefixes.
    """
    clean_req = clean_scope(request_scope)
    
    pattern_parts = policy.allowed_pattern.split('::')
    request_parts = clean_req.split('::')

    p_idx = 0
    r_idx = 0
    
    while p_idx < len(pattern_parts) and r_idx < len(request_parts):
        pattern_part = pattern_parts[p_idx]
        request_part = request_parts[r_idx]

        if pattern_part == '**':
            # If '**' is the last part of the pattern, it matches everything remaining in the request scope.
            if p_idx == len(pattern_parts) - 1:
                return True
            # Otherwise, find the next pattern part in the request scope.
            # '**' matches zero or more segments.
            next_pattern_part = pattern_parts[p_idx + 1]
            while r_idx < len(request_parts) and request_parts[r_idx] != next_pattern_part:
                r_idx += 1
            # If we didn't find the next pattern part, it's not a match.
            if r_idx == len(request_parts):
                return False
            # Advance pattern index past '**' so it can match the next segment
            p_idx += 1
            # The loop will then compare request_parts[r_idx] with pattern_parts[p_idx] (which is next_pattern_part)
            continue # Continue to the next iteration to compare the found segment

        elif pattern_part == '*':
            # '*' matches exactly one non-empty segment.
            if not request_part: # An empty segment (e.g., "a::") should not be matched by '*'
                return False
            p_idx += 1
            r_idx += 1
        elif pattern_part == request_part:
            # Exact segment match.
            p_idx += 1
            r_idx += 1
        else:
            # Mismatch.
            return False

    # After iterating, both pattern and request must be exhausted for a full match,
    # unless the remaining pattern parts are '**'.
    while p_idx < len(pattern_parts) and pattern_parts[p_idx] == '**':
        p_idx += 1

    return p_idx == len(pattern_parts) and r_idx == len(request_parts)


def evaluate_request(policies: list, role: str, scope: str) -> bool:
    for p in policies:
        if p.matches_role(role) and is_scope_permitted(p, scope):
            return True
    return False

def count_permitted_scopes(policy: RoutePolicy, scopes: list) -> int:
    return sum(1 for s in scopes if is_scope_permitted(policy, s))
