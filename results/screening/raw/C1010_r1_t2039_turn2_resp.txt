"""Telemetry metric streaming normalizer."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

def prepare_metric_packet(service_name: str, data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Prepare telemetry metric packet for distribution to event stream."""
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(service_name, epoch_ms, normalized_payload)

def publish_stream_packet(service_name: str, data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    pkt = prepare_metric_packet(service_name, data, epoch_ms)
    # The original logic included checks that were not used.
    # The packet is now guaranteed to be a valid envelope if prepare_metric_packet is correct.
    # _ = is_valid_envelope(pkt)
    # _ = extract_normalized_keys(data) # This was extracting keys from the raw data, not the packet.
    return pkt
