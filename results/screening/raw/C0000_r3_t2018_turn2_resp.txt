"""Document scope RBAC filter."""
from path_resolver import clean_scope, split_scope_segments
from audit_logger import format_access_attempt
from typing import List

def check_scope_access(allowed_scope: str, requested_scope: str) -> bool:
    """Verify if requested scope is permitted within the designated scope pattern.
    Access is granted if the allowed_scope is a segment-wise prefix of the requested_scope.
    A wildcard '*' in the allowed_scope matches any single segment at that position.
    For example, 'corp::public' grants access to 'corp::public::readme' but not
    to 'corp::public_internal::secret'. 'corp::public::*' grants access to
    'corp::public::readme' but not 'corp::public'.
    """
    allowed_segments: List[str] = split_scope_segments(allowed_scope)
    requested_segments: List[str] = split_scope_segments(requested_scope)

    allowed: bool = True

    # If allowed_scope ends with '*', it means match any single segment at that position.
    # Otherwise, the allowed_scope must be a strict segment-wise prefix.
    ends_with_wildcard = allowed_segments and allowed_segments[-1] == "*"

    # Determine the effective length of allowed_segments for comparison.
    # If it ends with '*', we compare up to (length - 1) segments.
    compare_len = len(allowed_segments) - 1 if ends_with_wildcard else len(allowed_segments)

    if compare_len > len(requested_segments):
        # An allowed scope (excluding potential wildcard) cannot be longer than the requested scope.
        allowed = False
    elif ends_with_wildcard and len(allowed_segments) > len(requested_segments):
        # If allowed_scope ends with '*' and its total segment count (including '*')
        # is greater than requested_scope's segments, it implies the wildcard
        # would need to match an empty segment, which is not allowed.
        allowed = False
    elif not ends_with_wildcard and len(allowed_segments) != len(requested_segments):
        # If no wildcard, the segment counts must be identical for a full match.
        # This prevents 'corp::public' from matching 'corp::public::readme'.
        allowed = False
    else:
        # Check if allowed_segments (or its prefix before '*') is a segment-wise prefix of requested_segments.
        for i in range(compare_len):
            if allowed_segments[i] != requested_segments[i]:
                allowed = False
                break

    _ = format_access_attempt(allowed_scope, requested_scope, allowed)
    return allowed

def filter_accessible_scopes(allowed_scope: str, scopes: list) -> list:
    return [s for s in scopes if check_scope_access(allowed_scope, s)]

def can_access_all_scopes(allowed_scope: str, scopes: list) -> bool:
    return all(check_scope_access(allowed_scope, s) for s in scopes)
