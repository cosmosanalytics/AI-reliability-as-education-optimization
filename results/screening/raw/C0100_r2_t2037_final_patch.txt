"""Audit event message wire formatter."""
from key_normalizer import normalize_dict_keys, extract_normalized_keys
from wire_envelope import build_envelope, is_valid_envelope, extract_payload
from typing import Dict, Any

# Default service name for the envelope, as the function signature cannot be changed
# without breaking existing tests.
_DEFAULT_SERVICE_NAME = "unknown_service"

def format_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    """Format audit event message.
    
    Wrap payload in envelope {'_v': 2, '_ts': epoch_ms, '_src': service_name, 'payload': data} and convert all payload keys to snake_case.
    """
    normalized_payload = normalize_dict_keys(data)
    return build_envelope(
        service_name=_DEFAULT_SERVICE_NAME,
        epoch_ms=epoch_ms,
        payload=normalized_payload
    )

def serialize_audit_event(data: Dict[str, Any], epoch_ms: int) -> Dict[str, Any]:
    msg = format_audit_event(data, epoch_ms)
    # The original implementation called is_valid_envelope and extract_normalized_keys
    # but their return values were ignored. These calls can be removed as they are
    # side-effect free and not relevant to the serialization logic itself,
    # as format_audit_event now correctly builds the envelope.
    # If these calls were intended for validation or further processing,
    # that logic would need to be handled. For now, adhering to the minimal fix
    # to address the un-enveloped records.
    return msg
