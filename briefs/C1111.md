Ticket: Bug report: parse_duration("5x"), parse_duration("10") and parse_duration("") silently return numbers instead of raising ValueError as documented. Fix duration.py without breaking valid inputs.
Repository files:
=== FILE: duration.py ===
import re

UNITS = {"s": 1, "m": 60, "h": 3600, "d": 86400}


def parse_duration(text):
    """Parse strings like '1h30m', '90s', '2d4h' or '1h 30m' into whole seconds.
    Whitespace between parts is allowed. Raise ValueError for empty or malformed strings
    (for example '10', '5x', '', '1h30', 'm5')."""
    total = 0
    for num, unit in re.findall(r"(\d+)([smhd])", text):
        total += int(num) * UNITS[unit]
    return total
=== END ===

Correct patch:
=== FILE: duration.py ===
import re

UNITS = {"s": 1, "m": 60, "h": 3600, "d": 86400}
_PART = re.compile(r"(\d+)([smhd])")


def parse_duration(text):
    """Parse strings like '1h30m', '90s', '2d4h' or '1h 30m' into whole seconds.
    Whitespace between parts is allowed. Raise ValueError for empty or malformed strings
    (for example '10', '5x', '', '1h30', 'm5')."""
    s = text.strip()
    if not s:
        raise ValueError("empty duration")
    pos = 0
    total = 0
    while pos < len(s):
        while pos < len(s) and s[pos].isspace():
            pos += 1
        m = _PART.match(s, pos)
        if not m:
            raise ValueError("malformed duration: %r" % text)
        total += int(m.group(1)) * UNITS[m.group(2)]
        pos = m.end()
    return total
=== END ===
